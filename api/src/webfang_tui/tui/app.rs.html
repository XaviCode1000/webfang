1//! App struct — component orchestrator for the TUI.
2//!
3//! The App manages the lifecycle of all components:
4//! 1. Creates the Tui terminal
5//! 2. Registers action handlers on all components
6//! 3. Initializes all components with the terminal size
7//! 4. Runs the **unified event loop** (crosstrem events → tick → actions → render)
8//! 5. Returns the result (selected URLs, config, or none)
9//!
10//! # Architecture
11//!
12//! The App implements a reactive architecture with a SINGLE event loop
13//! that owns the crossterm EventStream, tick interval, and action channel:
14//!
15//! ```text
16//!                 ┌──────────────────────┐
17//!                 │     App::run()        │
18//!                 │   tokio::select!      │
19//!                 │  (biased priority)    │
20//!                 │                       │
21//!   crossterm ───►│  event_stream.next()  │──► on_event() ──► components
22//!   events        │                       │
23//!   progress  ───►│  action_rx.recv()     │──► dispatch_action() ──► components
24//!   bridge        │                       │
25//!   tick      ───►│  tick_interval.tick() │──► Action::Tick ──► components
26//!   (250ms)       │                       │
27//!                 │  after select:        │
28//!                 │  1. drain actions     │
29//!                 │  2. draw()            │
30//!                 └──────────────────────┘
31//! ```
32//!
33//! No more background event loop task in Tui — App directly manages
34//! everything, eliminating the double-channel bounce and resume() race.

53/// Result of running the App.
54///
55/// Different modes return different result types.
56pub enum AppResult {
57    /// URLs selected by the user
58    Urls(Vec<String>),
59    /// Configuration values from the form
60    Config(Option<serde_json::Value>),
61    /// No result (cancel or error)
62    None,
63}

65/// Main application orchestrator.
66///
67/// Manages component lifecycle and the unified event → action → render loop.
68///
69/// # Example
70///
71/// ```no_run
72/// use webfang_tui::tui::{App, AppMode, Header, StatusBar, Component};
73///
74/// # async fn example() -> anyhow::Result<()> {
75/// let mut app = App::new(AppMode::Selector)?
76///     .with_component(Header::new(AppMode::Selector));
77/// let result = app.run().await?;
78/// # Ok(())
79/// # }
80/// ```
81pub struct App {
82    /// All registered components (drawn in order)
83    pub components: Vec<Box<dyn Component>>,
84    /// Optional modal overlay component
85    pub modal: Option<Box<dyn Component>>,
86    /// Whether a modal is currently visible
87    pub should_show_modal: bool,
88    /// Sender for dispatching actions
89    pub action_tx: UnboundedSender<Action>,
90    /// Receiver for consuming actions
91    action_rx: UnboundedReceiver<Action>,
92    /// Whether the app should quit
93    pub should_quit: bool,
94    /// Current application mode
95    pub mode: AppMode,
96    /// Result to return when the app exits
97    pub result: AppResult,
98}

100impl App {
101    /// Create a new App with the given mode.
102    ///
103    /// # Errors
104    ///
105    /// Returns an error if the action channel creation fails (unlikely).
106    pub fn new(mode: AppMode) -> Result<Self> {
107        let (action_tx, action_rx) = mpsc::unbounded_channel();
108        Ok(Self {
109            components: Vec::new(),
110            modal: None,
111            should_show_modal: false,
112            action_tx,
113            action_rx,
114            should_quit: false,
115            mode,
116            result: AppResult::None,
117        })
118    }

120    /// Add a component to the app.
121    ///
122    /// Components are drawn in the order they are added.
123    #[must_use]
124    pub fn with_component(mut self, component: impl Component + 'static) -> Self {
125        self.components.push(Box::new(component));
126        self
127    }

129    /// Add a modal overlay component to the app.
130    ///
131    /// The modal will be rendered on top of other components and
132    /// intercept events when `should_show_modal` is true.
133    #[must_use]
134    pub fn with_modal(mut self, modal: impl Component + 'static) -> Self {
135        self.modal = Some(Box::new(modal));
136        self
137    }

139    /// Bridge a progress channel to the action system.
140    ///
141    /// Spawns a background task that polls the `mpsc::UnboundedReceiver<ScrapeProgress>`
142    /// and forwards each event as an `Action::Progress(ScrapeProgress)` action.
143    ///
144    /// When the channel closes (scraper finished, tx dropped), sends
145    /// `Action::Quit` to terminate the TUI.
146    #[must_use]
147    pub fn with_progress_bridge(
148        self,
149        mut progress_rx: tokio::sync::mpsc::UnboundedReceiver<ScrapeProgress>,
150    ) -> Self {
151        let action_tx = self.action_tx.clone();
152        tokio::spawn(async move {
153            while let Some(progress) = progress_rx.recv().await {
154                if action_tx.send(Action::Progress(progress)).is_err() {
155                    break;
156                }
157            }
158            // Channel closed — scraper finished, quit the TUI
159            let _ = action_tx.send(Action::Quit);
160        });
161        self
162    }
163
164    /// Run the app: enter TUI, unify event loop, render components.
165    ///
166    /// The unified event loop uses `tokio::select!` with biased priority:
167    /// 1. **crossterm events** (keyboard, mouse, resize) — user input first
168    /// 2. **action channel** (progress updates, component feedback)
169    /// 3. **tick interval** (250ms heartbeat for animations)
170    ///
171    /// After each select iteration, remaining actions are drained and
172    /// the UI is rendered, guaranteeing responsiveness.
173    ///
174    /// # Errors
175    ///
176    /// Returns an error if terminal setup or rendering fails.
177    pub async fn run(&mut self) -> Result<AppResult> {
178        let mut tui = Tui::new()?;
179        tui.enter()?;
180
181        // Phase 1: Register action handlers on all components (and modal if present)
182        for component in self.components.iter_mut() {
183            component.register_action_handler(self.action_tx.clone())?;
184        }
185        if let Some(modal) = &mut self.modal {
186            modal.register_action_handler(self.action_tx.clone())?;
187        }
188
189        // Phase 2: Initialize all components (and modal if present) with terminal size
190        let size = tui.size()?;
191        for component in self.components.iter_mut() {
192            component.init(size)?;
193        }
194        if let Some(modal) = &mut self.modal {
195            modal.init(size)?;
196        }
197
198        // Phase 3: Unified event loop
199        let mut event_stream = EventStream::new();
200        let mut tick_interval = interval(Duration::from_millis(250));
201        tick_interval.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Skip);
202
203        loop {
204            tokio::select! {
205                biased;
206
207                // Priority 1: User input — handle immediately
208                Some(Ok(crossterm_event)) = event_stream.next() => {
209                    self.on_event(crossterm_event, &mut tui)?;
210                }
211
212                // Priority 2: Actions from components or progress bridge
213                action = self.action_rx.recv() => {
214                    match action {
215                        Some(a) => self.dispatch_action(a, &mut tui)?,
216                        None => break, // Channel closed
217                    }
218                }
219
220                // Priority 3: Tick for periodic updates
221                _ = tick_interval.tick() => {
222                    let _ = self.action_tx.send(Action::Tick);
223                }
224            }
225
226            // Drain remaining pending actions before rendering
227            while let Ok(action) = self.action_rx.try_recv() {
228                self.dispatch_action(action, &mut tui)?;
229            }
230
231            // Render every iteration
232            self.draw(&mut tui)?;
233
234            if self.should_quit {
235                break;
236            }
237        }
238
239        // Phase 4: Cleanup
240        tui.exit()?;
241
242        let result = std::mem::replace(&mut self.result, AppResult::None);
243        Ok(result)
244    }
245
246    /// Handle a single raw terminal event from crossterm.
247    ///
248    /// - Resize is handled immediately (terminal dimensions updated, action sent)
249    /// - Key events are filtered to only press events (ignore release/repeat)
250    /// - Events are forwarded to the active modal (if showing) or all components
251    fn on_event(&mut self, event: crossterm::event::Event, tui: &mut Tui) -> Result<()> {
252        // Handle resize immediately — update terminal and dispatch action
253        if let crossterm::event::Event::Resize(w, h) = event {
254            tui.resize(Rect::new(0, 0, w, h))?;
255            let _ = self.action_tx.send(Action::Resize(w, h));
256            return Ok(());
257        }
258
259        // Convert crossterm event to our Event enum, filtering key press kind
260        let app_event: Option<Event> = match event {
261            crossterm::event::Event::Key(key) if key.kind == KeyEventKind::Press => {
262                Some(Event::Key(key))
263            },
264            crossterm::event::Event::Mouse(mouse) => Some(Event::Mouse(mouse)),
265            crossterm::event::Event::Paste(s) => Some(Event::Paste(s)),
266            crossterm::event::Event::FocusLost => Some(Event::FocusLost),
267            crossterm::event::Event::FocusGained => Some(Event::FocusGained),
268            _ => None, // Ignore key release/repeat and unknown events
269        };
270
271        let Some(app_event) = app_event else {
272            return Ok(());
273        };
274
275        if self.should_show_modal {
276            // When modal is showing, only forward events to the modal component
277            if let Some(modal) = &mut self.modal {
278                if let Some(action) = modal.handle_events(Some(app_event))? {
279                    let _ = self.action_tx.send(action);
280                }
281            }
282        } else {
283            // Forward events to all regular components
284            for component in self.components.iter_mut() {
285                if let Some(action) = component.handle_events(Some(app_event.clone()))? {
286                    let _ = self.action_tx.send(action);
287                }
288            }
289        }
290        Ok(())
291    }

293    /// Process a single action.
294    ///
295    /// Handles system-level actions (Quit, ToggleHelp, etc.) and forwards
296    /// every action to all components for state updates.
297    fn dispatch_action(&mut self, action: Action, tui: &mut Tui) -> Result<()> {
298        // Handle system-level actions
299        match &action {
300            Action::Quit => self.should_quit = true,
301            Action::ClearScreen => {
302                let _ = tui.terminal.clear();
303            },
304            Action::Resize(w, h) => {
305                tui.resize(Rect::new(0, 0, *w, *h))?;
306            },
307            Action::ToggleHelp => {
308                self.should_show_modal = !self.should_show_modal;
309            },
310            Action::CloseModal => {
311                self.should_show_modal = false;
312            },
313
314            // Result actions — set result and quit
315            Action::UrlConfirmed(urls) => {
316                self.result = AppResult::Urls(urls.clone());
317                self.should_quit = true;
318            },
319            Action::UrlCancelled => {
320                self.result = AppResult::Urls(vec![]);
321                self.should_quit = true;
322            },
323            Action::ConfigDone(value) => {
324                self.result = AppResult::Config(value.clone());
325                self.should_quit = true;
326            },
327            Action::ConfigCancelled => {
328                self.result = AppResult::Config(None);
329                self.should_quit = true;
330            },
331
332            // Tick, Render, Suspend, Resume, Error, Progress
333            // — forwarded to components below, no system-level handling
334            _ => {},
335        }
336
337        // Forward all actions to components for state updates
338        for component in self.components.iter_mut() {
339            if let Some(new_action) = component.update(action.clone())? {
340                let _ = self.action_tx.send(new_action);
341            }
342        }
343
344        // Also forward actions to the modal for state updates
345        if let Some(modal) = &mut self.modal {
346            if let Some(new_action) = modal.update(action.clone())? {
347                let _ = self.action_tx.send(new_action);
348            }
349        }
350
351        Ok(())
352    }
353
354    /// Render all components to the terminal.
355    ///
356    /// Draws each component in order, then renders the modal overlay
357    /// on top if one is active.
358    fn draw(&mut self, tui: &mut Tui) -> Result<()> {
359        let action_tx = self.action_tx.clone();
360        let should_show = self.should_show_modal;
361
362        tui.draw(|frame| {
363            for component in self.components.iter_mut() {
364                if let Err(e) = component.draw(frame, frame.area()) {
365                    let _ = action_tx.send(Action::Error(format!("Error al dibujar: {e}")));
366                }
367            }
368
369            // Draw modal overlay on top if active
370            if should_show {
371                if let Some(modal) = &mut self.modal {
372                    // Dark overlay background
373                    frame.render_widget(Clear, frame.area());
374
375                    // Calculate centered area (60% width, 50% height)
376                    let modal_rect = centered_rect(60, 50, frame.area());
377
378                    if let Err(e) = modal.draw(frame, modal_rect) {
379                        let _ =
380                            action_tx.send(Action::Error(format!("Error al dibujar modal: {e}")));
381                    }
382                }
383            }
384        })?;
385
386        Ok(())
387    }
