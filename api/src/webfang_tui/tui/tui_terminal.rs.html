1//! Tui struct — terminal management only.
2//!
3//! Wraps a ratatui Terminal with setup, teardown, drawing, and sizing.
4//! Does NOT manage the event loop — that's handled by `App::run()`.
5//!
6//! # Architecture
7//!
8//! Tui is the I/O layer for terminal operations:
9//! - Terminal setup/teardown (alternate screen, raw mode, mouse capture)
10//! - Panic hook to restore terminal on crash
11//! - Drawing and sizing helpers
12//!
13//! The event loop lives in App, which owns the crossterm EventStream
14//! and manages tick/render intervals directly via `tokio::select!`.
15//!
16//! # Usage
17//!
18//! ```no_run
19//! # use anyhow::Result;
20//! # async fn example() -> Result<()> {
21//! let mut tui = webfang_tui::tui::Tui::new()?;
22//! tui.enter()?;
23//! // ... event loop managed by App ...
24//! tui.exit()?;
25//! # Ok(())
26//! # }
27//! ```
28

37use ratatui::prelude::*;

39use crossterm::execute;

41/// Convenience alias for ratatui Frame.
42pub type Frame<'a> = ratatui::Frame<'a>;
43
44/// Terminal manager — setup, teardown, draw, resize.
45///
46/// Does NOT manage the event loop. Use [`super::app::App::run()`] for the
47/// unified event loop that combines crossterm events, tick
48/// intervals, and action processing.
49pub struct Tui {
50    /// The underlying ratatui Terminal
51    pub terminal: Terminal<CrosstermBackend<Stdout>>,
52}
53
54impl Tui {
55    /// Create a new Tui instance.
56    ///
57    /// Does NOT enter the terminal — call [`enter()`](Self::enter) to do that.
58    ///
59    /// # Errors
60    ///
61    /// Returns an error if terminal creation fails.
62    pub fn new() -> Result<Self> {
63        let terminal = Terminal::new(CrosstermBackend::new(stdout()))?;
64        Ok(Self { terminal })
65    }
66
67    /// Enter the terminal: enable raw mode, alternate screen, and mouse capture.
68    ///
69    /// Also sets up a panic hook so the terminal is restored on crash.
70    ///
71    /// # Errors
72    ///
73    /// Returns an error if terminal setup fails.
74    pub fn enter(&mut self) -> Result<()> {
75        enable_raw_mode()?;
76        execute!(stdout(), EnterAlternateScreen, EnableMouseCapture)?;
77        self.terminal.clear()?;
78        self.terminal.hide_cursor()?;
79        setup_panic_hook();
80        Ok(())
81    }
82
83    /// Exit the terminal: restore raw mode, leave alternate screen.
84    ///
85    /// # Errors
86    ///
87    /// Returns an error if terminal restoration fails.
88    pub fn exit(&mut self) -> Result<()> {
89        disable_raw_mode()?;
90        execute!(stdout(), LeaveAlternateScreen, DisableMouseCapture)?;
91        self.terminal.show_cursor()?;
92        Ok(())
93    }
94
95    /// Draw to the terminal using the provided drawing function.
96    ///
97    /// # Errors
98    ///
99    /// Returns an error if the terminal draw fails.
100    pub fn draw(&mut self, draw_fn: impl FnOnce(&mut Frame)) -> Result<()> {
101        self.terminal.draw(draw_fn)?;
102        Ok(())
103    }
104
105    /// Get the current terminal size.
106    ///
107    /// # Errors
108    ///
109    /// Returns an error if the terminal size query fails.
110    pub fn size(&self) -> Result<Size> {
111        Ok(self.terminal.size()?)
112    }
113
114    /// Suspend the TUI (exit terminal, keep state).
115    ///
116    /// Useful for running external commands during the session.
117    ///
118    /// # Errors
119    ///
120    /// Returns an error if terminal restoration fails.
121    pub fn suspend(&mut self) -> Result<()> {
122        self.exit()?;
123        Ok(())
124    }
125
126    /// Resume the TUI (re-enter terminal).
127    ///
128    /// # Errors
129    ///
130    /// Returns an error if terminal setup fails.
131    pub fn resume(&mut self) -> Result<()> {
132        self.enter()?;
133        Ok(())
134    }
135}

137impl Deref for Tui {

139

141        &self.terminal
142    }

146    fn deref_mut(&mut self) -> &mut Self::Target {

148    }
149}

151/// Set up a panic hook that restores the terminal on panic.
152///
153/// Each restoration step runs independently so that a partial
154/// failure (e.g., broken stdout) doesn't prevent other steps.
155fn setup_panic_hook() {
156    let original_hook = std::panic::take_hook();
157    std::panic::set_hook(Box::new(move |panic_info| {
158        // Each step runs independently — failure in one doesn't block others
159        // Following **err-result-over-panic**: ignore errors in cleanup path
160        let _ = disable_raw_mode();
161        let _ = execute!(std::io::stdout(), LeaveAlternateScreen);
162        let _ = execute!(std::io::stdout(), crossterm::cursor::Show);
163        eprintln!("Application panicked. Terminal restored.");
164        original_hook(panic_info);
165    }));
166}

168#[cfg(test)]
169mod tests {
170    use std::io::{stderr, stdin, stdout, IsTerminal};
171
172    use super::*;
173
174    /// Returns true only when terminal setup can run against real TTY streams.
175    /// Some local command runners are not CI, but still do not expose a TTY.
176    fn has_terminal() -> bool {
177        std::env::var("CI").is_err()
178            && stdin().is_terminal()
179            && stdout().is_terminal()
180            && stderr().is_terminal()
181    }
182
183    #[test]
184    fn test_tui_new() {
185        if !has_terminal() {
186            return;
187        }
188        let tui = Tui::new();
189        assert!(tui.is_ok());
190    }
191
192    #[test]
193    fn test_tui_suspend_resume() {
194        if !has_terminal() {
195            return;
196        }
197        let mut tui = Tui::new().unwrap();
198        assert!(tui.enter().is_ok());
199        assert!(tui.suspend().is_ok());
200        assert!(tui.resume().is_ok());
201        assert!(tui.exit().is_ok());
202    }
