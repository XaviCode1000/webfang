1//! Actions for the reactive TUI architecture.
2//!
3//! Actions represent events that have been processed by the event loop
4//! and dispatched to components. They drive state changes in the UI.
5//!
6//! # Architecture
7//!
8//! Actions flow through the system as follows:
9//! 1. Terminal events → crossterm → Event enum
10//! 2. Tui event loop → Event enum → Action enum (via dispatcher)
11//! 3. Components process Actions via `update()` → may produce new Actions
12//! 4. App loop sends Actions back through the component chain
13
14use serde::{Deserialize, Serialize};
15use std::fmt;
16
17use super::ScrapeProgress;

19/// Application-level actions that drive UI state changes.
20///
21/// These are the "verbs" of the application — they represent
22/// meaningful operations rather than raw input events.
23///
24/// Serde derives are provided for serialization (e.g., for
25/// serializing actions in tests or debugging).
26#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
27pub enum Action {
28    /// Internal timer tick for periodic updates
29    Tick,
30    /// Request to render the UI
31    Render,
32    /// Terminal was resized to (width, height)
33    Resize(u16, u16),
34    /// Suspend TUI (e.g., for background operations)
35    Suspend,
36    /// Resume TUI after suspend
37    Resume,
38    /// Exit the application
39    Quit,
40    /// Clear the terminal screen
41    ClearScreen,
42    /// An error occurred with a description
43    Error(String),
44    /// Toggle help overlay
45    ToggleHelp,
46    /// Close the currently open modal
47    #[serde(skip)]
48    CloseModal,
49    /// URLs were confirmed by the user
50    UrlConfirmed(Vec<String>),
51    /// URL selection was cancelled
52    UrlCancelled,
53    /// Config form was submitted (carries optional JSON value)
54    ConfigDone(Option<serde_json::Value>),
55    /// Config form was cancelled
56    ConfigCancelled,
57    /// Progress update from the scraper
58    #[serde(skip)]
59    Progress(ScrapeProgress),
60}
61
62impl fmt::Display for Action {
63    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
64        match self {
65            Self::Tick => write!(f, "Tick"),
66            Self::Render => write!(f, "Render"),
67            Self::Resize(w, h) => write!(f, "Resize({w}, {h})"),
68            Self::Suspend => write!(f, "Suspend"),
69            Self::Resume => write!(f, "Resume"),
70            Self::Quit => write!(f, "Quit"),
71            Self::ClearScreen => write!(f, "ClearScreen"),
72            Self::Error(e) => write!(f, "Error({e})"),
73            Self::ToggleHelp => write!(f, "ToggleHelp"),
74            Self::CloseModal => write!(f, "CloseModal"),
75            Self::UrlConfirmed(urls) => write!(f, "UrlConfirmed({} urls)", urls.len()),
76            Self::UrlCancelled => write!(f, "UrlCancelled"),
77            Self::ConfigDone(_) => write!(f, "ConfigDone"),
78            Self::ConfigCancelled => write!(f, "ConfigCancelled"),
79            Self::Progress(p) => write!(f, "Progress({p:?})"),
80        }
81    }
82}
