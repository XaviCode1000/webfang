1//! Config form schema builders.
2//!
3//! Pure builders that define the TUI configuration form schema — field names,
4//! defaults, and help text — for each collapsible section. Extracted from the
5//! [`CollapsibleConfig`](super::collapsible_config::CollapsibleConfig) view so
6//! the schema lives apart from rendering and navigation logic.
7
8use ratatui_form::Form;
9
10/// Build the "Target" section form (url, selector).
11pub(super) fn build_target_form() -> Form {
12    Form::builder()
13        .text("url", "URL")
14        .placeholder("https://example.com")
15        .done()
16        .text("selector", "CSS Selector")
17        .initial_value("body")
18        .done()
19        .build()
20}
21
22/// Build the "Output" section form (output, format, export_format).
23pub(super) fn build_output_form() -> Form {
24    Form::builder()
25        .text("output", "Output Directory")
26        .initial_value("output")
27        .done()
28        .select("format", "Output Format")
29        .option("markdown", "Markdown")
30        .option("json", "JSON")
31        .option("text", "Plain Text")
32        .initial_value("markdown")
33        .done()
34        .select("export_format", "Export Format")
35        .option("jsonl", "JSONL")
36        .option("vector", "Vector")
37        .option("auto", "Auto")
38        .initial_value("jsonl")
39        .done()
40        .build()
41}
