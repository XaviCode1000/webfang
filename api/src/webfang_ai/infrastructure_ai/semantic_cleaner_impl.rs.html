1//! Semantic Cleaner implementation — Full RAG Pipeline Integration
2//!
3//! This module provides the concrete implementation of the [`SemanticCleaner`]
4//! trait using the complete Phase 2 + Phase 3 pipeline:
5//!
6//! # Architecture
7//!
8//! ```text
9//! HTML Input
10//!     ↓
11//! [Chunker] Split into semantic chunks (arena allocator)
12//!     ↓
13//! [Tokenizer] Convert each chunk to token IDs
14//!     ↓
15//! [InferencePool] Generate embeddings (dedicated worker threads)
16//!     ↓
17//! [RelevanceScorer] Filter by threshold (SIMD cosine similarity)
18//!     ↓
19//! Vec<DocumentChunk> Output
20//! ```
21//!
22//! # Rust-Skills Applied
23//!
24//! - `async-join-parallel`: Use `try_join_all` for concurrent embeddings
25//! - `mem-reuse-collections`: Pre-allocate `Vec::with_capacity`, reuse buffers
26//! - `own-borrow-over-clone`: Borrow `&chunks`, `&embeddings` - don't clone
27//! - `async-spawn-blocking`: InferencePool uses dedicated worker threads
28//! - `err-context-chain`: Add `.context()` to errors
29//! - `anti-unwrap-abuse`: Use `?` operator, NO `.unwrap()` in prod
30//! - `anti-lock-across-await`: Don't hold MutexGuard across `.await`
31//! - `api-builder-pattern`: ModelConfig uses builder pattern
32//! - `type-newtype-ids`: Using `ChunkId` for type-safe IDs
33//! - `opt-simd-portable`: RelevanceScorer uses `wide::f32x8` SIMD
34//!
35//! # Examples
36//!
37//! ```no_run
38//! # #[cfg(feature = "ai")]
39//! # async fn example() -> anyhow::Result<()> {
40//! use webfang_ai::{SemanticCleanerImpl, ModelConfig};
41//! use webfang_ai::SemanticCleaner;
42//!
43//! let config = ModelConfig::default();
44//! let cleaner = SemanticCleanerImpl::new(config).await?;
45//!
46//! let html = "<article><p>Hello world. Test content.</p></article>";
47//! let chunks = cleaner.clean(html).await?;
48//!
49//! println!("Generated {} chunks", chunks.len());
50//! # Ok(())
51//! # }
52//! ```

73/// Model configuration
74///
75/// Controls model loading and inference behavior.
76///
77/// # Builder Pattern
78///
79/// Following `api-builder-pattern`, use builder methods for configuration:
80///
81/// ```
82/// use webfang_ai::infrastructure_ai::ModelConfig;
83/// let config = ModelConfig::new()
84///     .with_repo("ibm-granite/granite-embedding-97m-multilingual-r2")
85///     .with_offline_mode(true)
86///     .with_max_tokens(512);
87/// ```
88#[derive(Debug, Clone)]
89pub struct ModelConfig {
90    /// Model repository on HuggingFace Hub
91    pub repo: String,
92    /// Model filename within repository
93    pub model_file: String,
94    /// Offline mode (fail if not cached)
95    pub offline_mode: bool,
96    /// Maximum tokens per chunk (model-specific)
97    pub max_tokens: usize,
98    /// Relevance threshold for filtering (0.0-1.0)
99    pub relevance_threshold: f32,
100    /// AI model variant to use (Granite-97M or Granite-311M)
101    pub model_variant: AiModel,
102}

114        }

178/// Semantic Cleaner implementation using full RAG pipeline
179///
180/// This is the concrete implementation of the [`SemanticCleaner`] trait.
181/// It integrates all Phase 2 and Phase 3 modules:
182/// - [`HtmlChunker`]: Semantic chunking with arena allocator
183/// - [`MiniLmTokenizer`]: HuggingFace tokenization
184/// - [`InferencePool`]: ONNX model execution with dedicated worker threads
185/// - [`RelevanceScorer`]: SIMD-accelerated cosine similarity filtering
186///
187/// # Thread Safety
188///
189/// This type is `Send + Sync` and can be safely shared across threads.
190/// All components use `Arc` for thread-safe sharing.
191///
192/// # Performance
193///
194/// - **First call**: Model download (~90MB) + load (~100-500ms)
195/// - **Subsequent calls**: ~50-200ms per page (depending on content size)
196/// - **Memory**: Arena allocator reduces allocation overhead
197/// - **Concurrency**: Embeddings generated concurrently with `try_join_all`
198pub struct SemanticCleanerImpl {
199    // Phase 2: Core inference
200    /// ONNX inference pool (dedicated worker threads with persistent sessions)
201    inference_pool: Arc<InferencePool>,
202    /// HuggingFace tokenizer (`Arc`-shared with the embedding adapter via
203    /// [`shared_inference`](Self::shared_inference))
204    tokenizer: Arc<MiniLmTokenizer>,
205
206    // Phase 3: Chunking + scoring
207    /// Semantic HTML chunker with arena allocator
208    chunker: HtmlChunker,
209    /// Relevance scorer with SIMD cosine similarity
210    scorer: RelevanceScorer,
211
212    // Phase 4: Content pruning
213    /// Content pruner (extracts readable content via legible)
214    pruner: LegibleContentPruner,
215
216    // Config
217    /// Model and pipeline configuration
218    config: ModelConfig,
219}

221impl SemanticCleanerImpl {
222    /// Create a new semantic cleaner with full pipeline
223    ///
224    /// This method loads all pipeline components:
225    /// 1. Downloads/loads ONNX model
226    /// 2. Loads tokenizer
227    /// 3. Creates chunker and scorer
228    ///
229    /// # Arguments
230    ///
231    /// * `config` - Model configuration
232    ///
233    /// # Returns
234    ///
235    /// * `Ok(SemanticCleanerImpl)` - Successfully created cleaner
236    /// * `Err(SemanticError)` - Model loading or download failed
237    ///
238    /// # Errors
239    ///
240    /// Returns error if:
241    /// - Model download fails
242    /// - Model file is corrupted (SHA256 mismatch)
243    /// - ONNX model fails to load
244    /// - Tokenizer fails to load
245    /// - Offline mode enabled but model not cached
246    ///
247    /// # Examples
248    ///
249    /// ```no_run
250    /// # async fn example() -> Result<(), Box<dyn std::error::Error>> {
251    /// use webfang_ai::{SemanticCleanerImpl, ModelConfig};
252    ///
253    /// let config = ModelConfig::default();
254    /// let cleaner = SemanticCleanerImpl::new(config).await?;
255    /// # Ok(())
256    /// # }
257    /// ```
258    ///
259    /// # Performance
260    ///
261    /// - **First call**: Model download (~90MB) + load (~100-500ms)
262    /// - **Subsequent calls**: Cache hit, ~10-50ms per page
263    /// - **Memory**: Memory-mapped files, ~90MB virtual memory
264    #[tracing::instrument(skip(config), fields(repo = %config.repo, model_file = %config.model_file, offline_mode = config.offline_mode))]
265    pub async fn new(config: ModelConfig) -> Result<Self, SemanticError> {
266        info!(
267            repo = %config.repo,
268            file = %config.model_file,
269            offline_mode = config.offline_mode,
270            relevance_threshold = config.relevance_threshold,
271            "Initializing semantic cleaner with full RAG pipeline"
272        );
273
274        // Resolve and validate model + tokenizer assets (hf_hub cache-first,
275        // in-memory SHA256 integrity check). Shared with `EmbeddingAdapter::from_config`
276        // so both pipelines resolve and validate models identically.
277        let (model_bytes, tokenizer_path) = resolve_model_assets(&config).await?;
278
279        // Initialize all pipeline components. The tokenizer is `Arc`-wrapped so
280        // `shared_inference` can hand the SAME instance to the embedding adapter.
281        let tokenizer = Arc::new(MiniLmTokenizer::from_file(&tokenizer_path).await?);
282        let inference_pool = Arc::new(InferencePool::new(
283            Arc::clone(&model_bytes),
284            config.model_variant,
285        )?);
286        let chunker = HtmlChunker::new();
287        let scorer = RelevanceScorer::new(config.relevance_threshold);
288
289        info!("Semantic cleaner initialized successfully");
290        debug!(
291            embedding_dim = inference_pool.embedding_dim(),
292            max_tokens = config.max_tokens,
293            relevance_threshold = config.relevance_threshold,
294            "Pipeline components loaded"
295        );
296
297        Ok(Self {
298            inference_pool,
299            tokenizer,
300            chunker,
301            scorer,
302            pruner: LegibleContentPruner::standard(),
303            config,
304        })
305    }
306
307    /// Get the relevance threshold
308    #[must_use]
309    pub fn relevance_threshold(&self) -> f32 {
310        self.config.relevance_threshold
311    }
312
313    /// Share the inference pool and tokenizer with another pipeline.
314    ///
315    /// Returns cheap `Arc` clones of the ONNX [`InferencePool`] and the
316    /// [`MiniLmTokenizer`] this cleaner was built with, so a second consumer
317    /// (e.g. the
318    /// [`EmbeddingAdapter`](crate::infrastructure_ai::embedding_adapter::EmbeddingAdapter))
319    /// can reuse the SAME model + tokenizer instead of resolving and loading a
320    /// second copy. This is what lets the `--ai` path load the ONNX model
321    /// exactly once across the semantic cleaner and the vault-search embedding
322    /// adapter — one `resolve_model_assets` call, one `InferencePool`.
323    #[must_use]
324    pub fn shared_inference(&self) -> (Arc<InferencePool>, Arc<MiniLmTokenizer>) {
325        (
326            Arc::clone(&self.inference_pool),
327            Arc::clone(&self.tokenizer),
328        )
329    }
330
331    /// Set the relevance threshold
332    ///
333    /// # Arguments
334    ///
335    /// * `threshold` - New threshold value (0.0-1.0)
336    ///
337    /// # Panics
338    ///
339    /// Panics if threshold is outside [0.0, 1.0] range
340    pub fn set_relevance_threshold(&mut self, threshold: f32) {
341        assert!(
342            (0.0..=1.0).contains(&threshold),
343            "Relevance threshold must be between 0.0 and 1.0, got {threshold}"
344        );
345        self.config.relevance_threshold = threshold;
346        self.scorer.set_threshold(threshold);
347    }
348}

354impl SemanticCleaner for SemanticCleanerImpl {
355    fn clean<'a>(
356        &'a self,
357        html: &'a str,
358    ) -> Pin<Box<dyn Future<Output = Result<Vec<DocumentChunk>, SemanticError>> + Send + 'a>> {
359        Box::pin(async move {
360            debug!(
361                html_length = html.len(),
362                "Starting full RAG pipeline: prune → chunk → tokenize → embed → score"
363            );
364
365            // Step 0: Content pruning — extract readable content via legible
366            // On failure or empty result, pass through raw HTML unchanged.
367            let pruned = self.pruner.prune(html);
368            let effective_html = if pruned.is_empty() { html } else { &pruned };
369            debug!(
370                pruned_length = effective_html.len(),
371                "Step 0: Content pruning complete"
372            );
373
374            // Step 1: Semantic chunking (uses arena internally)
375            // Following `own-borrow-over-clone`: borrow html, don't clone
376            let chunks = self
377                .chunker
378                .chunk(effective_html)
379                .map_err(|e| SemanticError::Tokenize(format!("Chunking failed: {e}")))?;
380
381            if chunks.is_empty() {
382                debug!("No chunks produced from HTML");
383                return Ok(Vec::new());
384            }
385
386            debug!(chunks_count = chunks.len(), "Step 1: Chunking complete");
387
388            // Step 2: Tokenize all chunks (mem-reuse-collections: reuse buffer)
389            // Pre-allocate with capacity following `mem-with-capacity`
390            let mut token_buffers = Vec::with_capacity(chunks.len());
391            for chunk in &chunks {
392                let input = self.tokenizer.tokenize(&chunk.content).map_err(|e| {
393                    SemanticError::Tokenize(format!("Tokenization failed for chunk: {e}"))
394                })?;
395
396                // Validate token count
397                if input.seq_len() > self.config.max_tokens {
398                    return Err(SemanticError::ChunkTooLarge {
399                        chunk_id: format!("chunk-{}", token_buffers.len()),
400                        tokens: input.seq_len(),
401                        max: self.config.max_tokens,
402                    });
403                }
404
405                token_buffers.push(input);
406            }
407
408            debug!(
409                tokens_generated = token_buffers.len(),
410                "Step 2: Tokenization complete"
411            );
412
413            // Step 3: Generate embeddings CONCURRENTLY (async-join-parallel)
414            // Following `async-join-parallel`: use try_join_all for concurrent independent operations
415            // InferencePool dispatches to dedicated worker threads with persistent sessions
416            // Following `anti-lock-across-await`: No locks held across await points
417            let embeddings = try_join_all(token_buffers.iter().map(|input| {
418                let pool = &self.inference_pool;
419                async move { pool.infer(input).await }
420            }))
421            .await
422            .map_err(|e| {
423                SemanticError::Inference(format!("Concurrent embedding generation failed: {e}"))
424            })?;
425
426            debug!(
427                embeddings_generated = embeddings.len(),
428                embedding_dim = embeddings.first().map(|e| e.len()).unwrap_or(0),
429                "Step 3: Embedding generation complete"
430            );
431
432            // Step 4: Score and filter (own-borrow-over-clone: borrow embeddings)
433            // Following `own-borrow-over-clone`: borrow &chunks and &embeddings, don't clone
434            // Following `opt-simd-portable`: RelevanceScorer uses SIMD cosine similarity
435            let filtered = self.filter_by_relevance(&chunks, &embeddings)?;
436
437            debug!(
438                chunks_before = chunks.len(),
439                chunks_after = filtered.len(),
440                filtered_out = chunks.len() - filtered.len(),
441                "Step 4: Relevance filtering complete"
442            );
443
444            info!(total_chunks = filtered.len(), "Full RAG pipeline complete");
445
446            Ok(filtered)
447        })
448    }
449
450    fn max_tokens(&self) -> usize {
451        self.config.max_tokens
452    }
453
454    fn is_ready(&self) -> bool {
455        self.inference_pool.is_ready()
456    }
457}

459impl SemanticCleanerImpl {
460    /// Filter chunks by relevance score and **preserve embeddings**
461    ///
462    /// Pairs each chunk with its embedding, scores against a reference,
463    /// filters by threshold, and **preserves** the embedding vectors in the output.
464    ///
465    /// **Critical bug fix**: Previously called `scorer.filter()` which discarded
466    /// embeddings via `.map(|(chunk, _)| chunk.clone())`, resulting in:
467    /// - "Generated 0 chunks with embeddings" log messages
468    /// - Empty embeddings fields in JSONL output
469    /// - Loss of 49536 dimensions of embedding data
470    ///
471    /// **Solution**: Uses `scorer.filter_with_embeddings()` to preserve embeddings,
472    /// then restores them to each chunk before returning `Vec<DocumentChunk>`.
473    ///
474    /// # Arguments
475    ///
476    /// * `chunks` - Slice of DocumentChunks (borrowed, following `own-borrow-over-clone`)
477    /// * `embeddings` - Slice of embedding vectors (borrowed)
478    ///
479    /// # Returns
480    ///
481    /// Filtered vector of `DocumentChunk` items meeting relevance threshold.
482    /// **Important**: Each chunk includes its embedding vector (not `None`).
483    ///
484    /// # Errors
485    ///
486    /// Returns `SemanticError::Inference("No embeddings available")` if
487    /// input embeddings slice is empty (no reference vector for scoring).
488    ///
489    /// # Performance
490    ///
491    /// Uses SIMD-accelerated cosine similarity via `RelevanceScorer`.
492    /// Concurrent operations use arena allocator to reduce allocation overhead.
493    ///
494    /// # Examples
495    ///
496    /// ```no_run
497    /// # #[cfg(feature = "ai")]
498    /// # async fn example() -> Result<(), Box<dyn std::error::Error>> {
499    /// use webfang_ai::{SemanticCleaner, SemanticCleanerImpl, ModelConfig, SemanticError};
500    ///
501    /// // Create semantic cleaner (requires --features ai)
502    /// let config = ModelConfig::default();
503    /// let cleaner = SemanticCleanerImpl::new(config).await?;
504    ///
505    /// // Clean HTML content - will generate chunks with embeddings
506    /// let html = "<article><h1>Title</h1><p>Content here.</p></article>";
507    /// let chunks = cleaner.clean(html).await?;
508    ///
509    /// // Verify embeddings are present (bug fix validation)
510    /// let has_embeddings = chunks.first()
511    ///     .map(|c| c.embeddings.is_some())
512    ///     .ok_or_else(|| SemanticError::Inference(
513    ///         "No chunks returned from semantic cleaner. Check HTML content and AI model availability.".to_string()
514    ///     ))?;
515    /// assert!(has_embeddings, "embeddings should not be None after fix");
516    ///
517    /// // Embedding dimension: 384 for all-MiniLM-L6-v2 model
518    /// let dim = chunks.first()
519    ///     .map(|c| c.embeddings.as_ref().map(|e| e.len()))
520    ///     .ok_or(SemanticError::Inference("No chunks or Embeddings returned".to_string()))?;
521    /// assert_eq!(dim, Some(384));
522    /// # Ok(())
523    /// # }
524    /// ```
525    ///
526    /// See also:
527    /// - [`SemanticCleaner::clean()`](SemanticCleaner::clean) - Full pipeline entry point
528    /// - [`RelevanceScorer::filter_with_embeddings()`](RelevanceScorer::filter_with_embeddings)
529    fn filter_by_relevance(
530        &self,
531        chunks: &[DocumentChunk],
532        embeddings: &[Vec<f32>],
533    ) -> Result<Vec<DocumentChunk>, SemanticError> {
534        // Validate that each chunk has a corresponding embedding (mem-prevent-data-loss)
535        if chunks.len() != embeddings.len() {
536            return Err(SemanticError::Inference(format!(
537                "Length mismatch: got {} chunks but {} embedding vectors. \
538                 Each chunk must have exactly one embedding vector.",
539                chunks.len(),
540                embeddings.len()
541            )));
542        }
543
544        // Create (chunk, embedding) pairs
545        // Following `mem-with-capacity`: pre-allocate
546        let mut chunk_embedding_pairs = Vec::with_capacity(chunks.len());
547
548        for (chunk, embedding) in chunks.iter().zip(embeddings.iter()) {
549            chunk_embedding_pairs.push((chunk.clone(), embedding.clone()));
550        }
551
552        // Use first embedding as reference (simple strategy)
553        // In production, this could be a query vector or domain-specific reference
554        let reference = embeddings.first().ok_or_else(|| {
555            SemanticError::Inference("No embeddings available for relevance scoring".to_string())
556        })?;
557
558        // Filter using scorer WITH embeddings preserved
559        let filtered_with_embeddings: Vec<(DocumentChunk, Vec<f32>)> = self
560            .scorer
561            .filter_with_embeddings(&chunk_embedding_pairs, Some(reference));
562
563        // Restore embeddings to chunks following `mem-preserving-embeddings`
564        let mut result = Vec::with_capacity(filtered_with_embeddings.len());
565        for (chunk, embedding) in filtered_with_embeddings {
566            let mut chunk_with_embeddings = chunk.clone();
567            chunk_with_embeddings.embeddings = Some(embedding);
568            result.push(chunk_with_embeddings);
569        }
570
571        Ok(result)
572    }
573}

575/// Resolve and validate model assets from the hf_hub cache.
576///
577/// Resolves the model and tokenizer paths through the hf_hub cache — offline
578/// mode resolves strictly from the local cache (no network) and fails fast with
579/// [`SemanticError::OfflineMode`] when either asset is missing; online mode is
580/// cache-first (hf_hub returns the cached path when present and transparently
581/// downloads missing assets otherwise). Then loads the model bytes once and
582/// validates their SHA256 in memory.
583///
584/// Extracted from [`SemanticCleanerImpl::new`] so
585/// [`crate::infrastructure_ai::embedding_adapter::EmbeddingAdapter::from_config`]
586/// reuses the identical resolution + integrity-validation logic without
587/// duplicating the hf_hub plumbing.
588///
589/// # Returns
590///
591/// `(model_bytes, tokenizer_path)` — SHA256-validated model bytes and the path
592/// to `tokenizer.json`.
593///
594/// # Errors
595///
596/// Returns [`SemanticError::OfflineMode`] when offline and an asset is uncached,
597/// [`SemanticError::Download`] on hf_hub client/API failure,
598/// [`SemanticError::ModelLoad`] on read failure, or
599/// [`SemanticError::CacheValidation`] on SHA256 mismatch.
600#[tracing::instrument(skip(config), fields(repo = %config.repo, model_file = %config.model_file, offline_mode = config.offline_mode))]
601pub(crate) async fn resolve_model_assets(
602    config: &ModelConfig,
603) -> Result<(Arc<Vec<u8>>, std::path::PathBuf), SemanticError> {
604    // Resolve model + tokenizer paths through the hf_hub cache.
605    let (model_path, tokenizer_path) = if config.offline_mode {
606        let cache = HfCache::from_env();
607        let cache_repo = cache.repo(Repo::new(config.repo.clone(), RepoType::Model));
608
609        let model_path =
610            cache_repo
611                .get(&config.model_file)
612                .ok_or_else(|| SemanticError::OfflineMode {
613                    repo: config.repo.clone(),
614                })?;
615        let tokenizer_path =
616            cache_repo
617                .get("tokenizer.json")
618                .ok_or_else(|| SemanticError::OfflineMode {
619                    repo: config.repo.clone(),
620                })?;
621
622        debug!("Resolved model and tokenizer from offline cache");
623        (model_path, tokenizer_path)
624    } else {
625        let api = ApiBuilder::from_env()
626            .with_progress(true)
627            .build()
628            .map_err(|e| SemanticError::Download {
629                repo: config.repo.clone(),
630                cause: format!("Failed to build HuggingFace API client: {e}"),
631            })?;
632
633        let repo = api.model(config.repo.clone());
634
635        // Resolve both assets concurrently (cache-first, downloads if missing).
636        // `with_progress(true)` surfaces hf_hub's built-in progress bar so the
637        // first download (~390MB) is not perceived as a hang; the span makes
638        // the download phase observable in the trace file.
639        let (model_path, tokenizer_path) =
640            try_join(repo.get(&config.model_file), repo.get("tokenizer.json"))
641                .instrument(tracing::info_span!(
642                    "download_model_assets",
643                    repo = %config.repo
644                ))
645                .await
646                .map_err(|e| SemanticError::Download {
647                    repo: config.repo.clone(),
648                    cause: format!("HuggingFace API error: {e}"),
649                })?;
650
651        debug!("Resolved model and tokenizer via hf_hub (cache-first)");
652        (model_path, tokenizer_path)
653    };
654
655    // Load the model bytes once and validate SHA256 on the in-memory buffer
656    // (zero extra I/O — the same bytes feed the caller's inference pool).
657    let model_bytes = Arc::new(
658        tokio::fs::read(&model_path)
659            .await
660            .map_err(SemanticError::ModelLoad)?,
661    );
662
663    validate_model_hash(
664        model_bytes.as_slice(),
665        config.model_variant.sha256(),
666        &config.repo,
667    )?;
668
669    Ok((model_bytes, tokenizer_path))
670}
671
672/// Validate the SHA256 hash of in-memory model bytes against an expected value.
673///
674/// Extracted from [`SemanticCleanerImpl::new`] so the integrity check is
675/// unit-testable without downloading a model or loading the ONNX runtime.
676///
677/// # Errors
678///
679/// Returns [`SemanticError::CacheValidation`] when the computed hash does not
680/// match `expected`.
681#[tracing::instrument(skip(bytes), fields(repo = %repo, expected = %expected))]
682fn validate_model_hash(bytes: &[u8], expected: &str, repo: &str) -> Result<(), SemanticError> {
683    debug!("Validating model integrity...");
684    let actual = format!("{:x}", Sha256::digest(bytes));
685    if actual != expected {
686        return Err(SemanticError::CacheValidation {
687            repo: repo.to_string(),
688            expected: expected.to_string(),
689            actual,
690        });
691    }
692    debug!(sha = %actual, "SHA256 validation passed");
693    Ok(())
694}
