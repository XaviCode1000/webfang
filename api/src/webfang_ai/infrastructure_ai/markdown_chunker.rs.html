1//! Markdown chunker for Obsidian vault notes.
2//!
3//! Heading-aware chunking that respects Obsidian-specific syntax:
4//! - YAML frontmatter is stripped (metadata, not semantic content)
5//! - Headings (`#`, `##`) are hard section boundaries
6//! - Callouts (`> [!note]`) are kept as atomic units
7//! - Wikilinks (`[[...]]`) are preserved in content
8//! - [`SentenceSplitter`] (UAX #29) prevents mid-sentence breaks
9//!
10//! # Thread Safety
11//!
12//! `MarkdownChunker` is `Send + Sync` and can be shared across threads.
13

23/// Markdown chunker for Obsidian vault notes.
24///
25/// Chunks Markdown content into semantic segments using a heading-aware
26/// approach:
27/// 1. **Strip frontmatter**: Remove YAML metadata block
28/// 2. **Split by headings**: `#`, `##`, etc. as hard boundaries
29/// 3. **Preserve callouts**: `> [!type]` blocks stay atomic
30/// 4. **Merge/split by size**: Respect min/max chunk size constraints
31///
32/// # Examples
33///
34/// ```
35/// use webfang_ai::MarkdownChunker;
36///
37/// let chunker = MarkdownChunker::new();
38/// let md = "# Title\n\nThis is the first section with enough content to exceed the minimum chunk size threshold.\n\n## Subtitle\n\nSecond section also has sufficient length to pass the minimum size filter.";
39/// let chunks = chunker.chunk(md).expect("chunking should succeed");
40/// assert!(!chunks.is_empty());
41/// ```
42pub struct MarkdownChunker {
43    /// Minimum chunk size in characters.
44    min_chunk_size: usize,
45    /// Maximum chunk size in characters.
46    max_chunk_size: usize,
47    /// Sentence splitter for sub-section splitting.
48    sentence_splitter: SentenceSplitter,
49}
50
51impl Default for MarkdownChunker {
52    fn default() -> Self {
53        Self::new()
54    }
55}
56
57impl MarkdownChunker {
58    /// Create a new `MarkdownChunker` with default settings.
59    ///
60    /// # Defaults
61    ///
62    /// - `min_chunk_size`: 80 characters
63    /// - `max_chunk_size`: 512 characters (Granite model safe zone)
64    #[must_use]
65    pub fn new() -> Self {
66        Self {
67            min_chunk_size: 80,
68            max_chunk_size: 512,
69            sentence_splitter: SentenceSplitter,
70        }
71    }
72
73    /// Create a new `MarkdownChunker` with custom size constraints.
74    #[must_use]
75    pub fn with_config(min_chunk_size: usize, max_chunk_size: usize) -> Self {
76        Self {
77            min_chunk_size,
78            max_chunk_size,
79            sentence_splitter: SentenceSplitter,
80        }
81    }
82
83    /// Chunk Markdown into semantic segments.
84    ///
85    /// # Process
86    ///
87    /// 1. Strip YAML frontmatter (`---` delimited block)
88    /// 2. Split by heading lines (`# ...`) as hard boundaries
89    /// 3. Keep callout blocks (`> [!type]`) atomic
90    /// 4. Merge small sections, split large ones via [`SentenceSplitter`]
91    ///
92    /// # Errors
93    ///
94    /// Returns [`SemanticError::Tokenize`] if the input is empty after
95    /// frontmatter stripping.
96    pub fn chunk(&self, markdown: &str) -> Result<Vec<DocumentChunk>, SemanticError> {
97        let body = Self::strip_frontmatter(markdown);
98
99        if body.trim().is_empty() {
100            return Err(SemanticError::Tokenize(
101                "contenido vacío después de eliminar frontmatter".to_owned(),
102            ));
103        }
104
105        // Pass 1: Split by headings into sections.
106        let sections = Self::split_by_headings(body);
107
108        // Pass 2: Process each section — protect callouts, merge/split by size.
109        let mut chunks: SmallVec<[DocumentChunk; 16]> = SmallVec::new();
110        for (heading, section_text) in &sections {
111            let blocks = Self::extract_callout_blocks(section_text);
112            for block in blocks {
113                let text = block.trim();
114                if text.is_empty() {
115                    continue;
116                }
117
118                if text.len() <= self.max_chunk_size {
119                    if text.len() >= self.min_chunk_size {
120                        chunks.push(Self::make_chunk(text, heading));
121                    }
122                    // Small blocks are accumulated below.
123                } else {
124                    // Large block: split by sentences.
125                    for sub in self.split_large_text(text) {
126                        chunks.push(Self::make_chunk(&sub, heading));
127                    }
128                }
129            }
130        }
131
132        // Pass 3: Merge remaining small chunks.
133        let merged = self.merge_small_chunks(chunks);
134
135        Ok(merged.into_iter().collect())
136    }
137
138    /// Strip YAML frontmatter (delimited by `---` at start of file).
139    ///
140    /// Returns the body after the closing `---`. If no frontmatter is
141    /// present, returns the input unchanged.
142    fn strip_frontmatter(md: &str) -> &str {
143        let trimmed = md.trim_start();
144        if !trimmed.starts_with("---") {
145            return md;
146        }
147        // Find the closing `---` after the opening one.
148        if let Some(end) = trimmed[3..].find("\n---") {
149            let after = &trimmed[3 + end + 4..]; // skip past "\n---"
150                                                 // Skip the rest of the closing line.
151            after.strip_prefix('\n').unwrap_or(after)
152        } else {
153            md // Malformed frontmatter — treat as regular content.
154        }
155    }
156
157    /// Split Markdown by heading lines into `(heading_text, section_body)` pairs.
158    ///
159    /// Content before the first heading gets an empty heading string.
160    fn split_by_headings(body: &str) -> Vec<(String, String)> {
161        let mut sections: Vec<(String, String)> = Vec::new();
162        let mut current_heading = String::new();
163        let mut current_lines: Vec<&str> = Vec::new();
164
165        for line in body.lines() {
166            if Self::is_heading(line) {
167                // Flush previous section.
168                if !current_lines.is_empty() || !current_heading.is_empty() {
169                    sections.push((current_heading.clone(), current_lines.join("\n")));
170                    current_lines.clear();
171                }
172                current_heading = line.trim_start_matches('#').trim().to_owned();
173            } else {
174                current_lines.push(line);
175            }
176        }
177
178        // Flush last section.
179        if !current_lines.is_empty() || !current_heading.is_empty() {
180            sections.push((current_heading, current_lines.join("\n")));
181        }
182
183        sections
184    }
185
186    /// Check if a line is a Markdown heading (`# ...`).
187    fn is_heading(line: &str) -> bool {
188        let trimmed = line.trim_start();
189        trimmed.starts_with('#') && trimmed.chars().nth(1).is_some_and(|c| c == '#' || c == ' ')
190    }
191
192    /// Extract callout blocks (`> [!type]`) as atomic units.
193    ///
194    /// Non-callout content is returned as separate blocks. Callout blocks
195    /// include all consecutive `>` lines following the `> [!type]` marker.
196    fn extract_callout_blocks(text: &str) -> Vec<String> {
197        let mut blocks: Vec<String> = Vec::new();
198        let mut normal_lines: Vec<&str> = Vec::new();
199        let mut callout_lines: Vec<&str> = Vec::new();
200        let mut in_callout = false;
201
202        for line in text.lines() {
203            let trimmed = line.trim();
204            let is_blockquote = trimmed.starts_with('>');
205            let is_callout_start =
206                is_blockquote && trimmed.len() > 2 && trimmed[2..].trim_start().starts_with("[!");
207
208            if is_callout_start {
209                // Flush accumulated normal lines.
210                if !normal_lines.is_empty() {
211                    blocks.push(normal_lines.join("\n"));
212                    normal_lines.clear();
213                }
214                in_callout = true;
215                callout_lines.push(line);
216            } else if in_callout && is_blockquote {
217                // Continue callout block.
218                callout_lines.push(line);
219            } else {
220                // End callout if active.
221                if in_callout {
222                    blocks.push(callout_lines.join("\n"));
223                    callout_lines.clear();
224                    in_callout = false;
225                }
226                normal_lines.push(line);
227            }
228        }
229
230        // Flush remaining.
231        if in_callout && !callout_lines.is_empty() {
232            blocks.push(callout_lines.join("\n"));
233        }
234        if !normal_lines.is_empty() {
235            blocks.push(normal_lines.join("\n"));
236        }
237
238        blocks
239    }
240
241    /// Split large text by sentences, respecting `max_chunk_size`.
242    fn split_large_text(&self, text: &str) -> Vec<String> {
243        let sentences = self.sentence_splitter.split_trimmed(text);
244        let mut result: Vec<String> = Vec::new();
245        let mut current = String::new();
246
247        for sentence in sentences {
248            if current.len() + sentence.len() + 1 > self.max_chunk_size && !current.is_empty() {
249                result.push(current.trim().to_owned());
250                current.clear();
251            }
252            if !current.is_empty() {
253                current.push(' ');
254            }
255            current.push_str(&sentence);
256        }
257
258        if !current.trim().is_empty() {
259            result.push(current.trim().to_owned());
260        }
261
262        result
263    }
264
265    /// Create a `DocumentChunk` with heading context in metadata.
266    fn make_chunk(content: &str, heading: &str) -> DocumentChunk {
267        let mut chunk = DocumentChunk::new(
268            Uuid::new_v4(),
269            String::new(), // url — filled by caller (vault path)
270            String::new(), // title — filled by caller (note title)
271            content.to_owned(),
272        );
273        if !heading.is_empty() {
274            chunk
275                .metadata
276                .insert("heading".to_owned(), heading.to_owned());
277        }
278        chunk
279    }
280
281    /// Merge chunks below `min_chunk_size` with adjacent chunks.
282    fn merge_small_chunks(&self, chunks: SmallVec<[DocumentChunk; 16]>) -> Vec<DocumentChunk> {
283        if chunks.is_empty() {
284            return Vec::new();
285        }
286
287        let mut result: Vec<DocumentChunk> = Vec::new();
288        let mut pending: Option<DocumentChunk> = None;
289
290        for chunk in chunks {
291            match pending.take() {
292                Some(mut prev) => {
293                    if prev.content.len() < self.min_chunk_size {
294                        // Merge with current chunk.
295                        prev.content.push_str("\n\n");
296                        prev.content.push_str(&chunk.content);
297                        if prev.content.len() <= self.max_chunk_size {
298                            pending = Some(prev);
299                        } else {
300                            // Merged result is too large — split it.
301                            for sub in self.split_large_text(&prev.content) {
302                                result.push(Self::make_chunk(
303                                    &sub,
304                                    prev.metadata.get("heading").map_or("", |s| s),
305                                ));
306                            }
307                        }
308                    } else {
309                        result.push(prev);
310                        pending = Some(chunk);
311                    }
312                },
313                None => {
314                    pending = Some(chunk);
315                },
316            }
317        }
318
319        if let Some(last) = pending {
320            result.push(last);
321        }
322
323        result
324    }
325}

331impl TextChunker for MarkdownChunker {
332    fn chunk_text(&self, text: &str) -> Result<Vec<String>, SemanticError> {
333        let chunks = self.chunk(text)?;
334        Ok(chunks.into_iter().map(|c| c.content).collect())
335    }

371        assert_eq!(sections.len(), 3, "intro + 2 headings");
