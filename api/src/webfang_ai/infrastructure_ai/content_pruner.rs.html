1//! Content pruning module — Extract readable content using `legible` crate.
2//!
3//! Sealed trait pattern: only `LegibleContentPruner` can implement `ContentPruner`.
4//! Feature-gated behind `ai`.
5
6use std::fmt;
7
8/// Aggressiveness level for content pruning.
9///
10/// Controls how aggressively `legible` strips non-content elements.
11#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
12pub enum PruneAggressiveness {
13    /// Minimal pruning — keep most structural elements.
14    Gentle,
15    /// Standard pruning — remove nav, sidebar, ads, footer (default).
16    #[default]
17    Standard,
18    /// Aggressive pruning — extract only the core article text.
19    Aggressive,
20}
21
22/// Sealed trait for content pruning implementations.
23///
24/// Only implementors within this module can implement this trait.
25/// External crates cannot add new pruning strategies.
26pub trait ContentPruner: sealed::Sealed {
27    /// Prune HTML content by extracting the readable main content.
28    ///
29    /// Returns the pruned HTML string. On failure or empty input,
30    /// returns the original HTML unchanged (fallback behavior).
31    fn prune(&self, html: &str) -> String;
32
33    /// Return the aggressiveness level of this pruner.
34    fn aggressiveness(&self) -> PruneAggressiveness;
35}
36
37/// Content pruner backed by the `legible` crate (Mozilla Readability port).
38///
39/// Extracts the main readable content from HTML, stripping navigation,
40/// sidebars, ads, footers, and other boilerplate.
41pub struct LegibleContentPruner {
42    aggressiveness: PruneAggressiveness,
43}
44
45impl LegibleContentPruner {
46    /// Create a new pruner with the given aggressiveness level.
47    #[must_use]
48    pub fn new(aggressiveness: PruneAggressiveness) -> Self {
49        Self { aggressiveness }
50    }
51
52    /// Create a pruner with default (Standard) aggressiveness.
53    #[must_use]
54    pub fn standard() -> Self {
55        Self::new(PruneAggressiveness::Standard)
56    }
57}
58
59impl fmt::Debug for LegibleContentPruner {
60    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
61        f.debug_struct("LegibleContentPruner")
62            .field("aggressiveness", &self.aggressiveness)
63            .finish()
64    }
65}
66
67impl sealed::Sealed for LegibleContentPruner {}
68
69impl ContentPruner for LegibleContentPruner {
70    fn prune(&self, html: &str) -> String {
71        // Fallback: empty input returns empty string
72        if html.trim().is_empty() {
73            return String::new();
74        }
75
76        // Attempt legible parsing
77        match legible::parse(html, None, None) {
78            Ok(article) => {
79                if article.content.trim().is_empty() {
80                    // legible returned empty content — pass through original
81                    html.to_string()
82                } else {
83                    article.content
84                }
85            },
86            Err(_) => {
87                // legible failed — pass through original HTML unchanged
88                html.to_string()
89            },
90        }
91    }
92
93    fn aggressiveness(&self) -> PruneAggressiveness {
94        self.aggressiveness
95    }
96}
