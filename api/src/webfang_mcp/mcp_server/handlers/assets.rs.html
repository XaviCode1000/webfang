14#[tool_router(router = tool_router_assets, vis = "pub")]
15// Allow: the `tool_router_assets` fn generated by rmcp-macros cannot be documented from this crate.
16#[allow(missing_docs)]
17impl McpHandler {
18    /// Download images and documents from HTML into the output directory.
19    ///
20    /// The boolean toggles `images` (default: true) and `documents`
21    /// (default: false) select which asset kinds are downloaded. Files are
22    /// written with SHA-256 hashed filenames; the response reports each
23    /// downloaded asset with its local path.
24    #[tool(
25        description = "Download images (default: true) and/or documents (default: false) referenced in HTML content into the output directory (SHA-256 hashed filenames). 'images' and 'documents' are boolean toggles, not URL lists. Returns the downloaded assets with their local paths. For a full scrape that also downloads assets, use scrape_with_options with download_images/download_documents."
26    )]
27    #[instrument(skip(self), fields(base_url = %params.base_url, images = params.images.unwrap_or(true), documents = params.documents.unwrap_or(false), output_dir = ?params.output_dir))]
28    async fn download_assets(
29        &self,
30        Parameters(params): Parameters<DownloadAssetsParams>,
31    ) -> Result<CallToolResult, McpError> {
32        params.validate()?;
33
34        let _permit = acquire_semaphore!(self, assets);
35
36        let base_url = url::Url::parse(&params.base_url).map_err(|e| {
37            McpError::invalid_params(
38                format!("invalid base URL: {e}"),
39                Some(serde_json::Value::String("base_url".to_string())),
40            )
41        })?;
42
43        let mut config = webfang_core::infrastructure::config::ScraperConfig {
44            download_images: params.images.unwrap_or(true),
45            download_documents: params.documents.unwrap_or(false),
46            ..Default::default()
47        };
48        if let Some(ref output_dir) = params.output_dir {
49            config.output_dir = std::path::PathBuf::from(output_dir);
50        }
51
52        // Shared downloader when injected (connection pooling across calls);
53        // None falls back to a per-call Downloader built from the config.
54        let dl = self
55            .state
56            .downloader
57            .as_deref()
58            .map(|d| d as &dyn webfang_core::domain::ports::AssetDownloaderPort);
59
60        match webfang_core::application::scraper_service::download_assets_if_enabled(
61            &params.html,
62            &base_url,
63            &config,
64            dl,
65        )
66        .await
67        {
68            Ok(assets) => {
69                tracing::info!(
70                    downloaded = assets.len(),
71                    images = config.download_images,
72                    documents = config.download_documents,
73                    "assets downloaded"
74                );
75                let content = serde_json::to_string_pretty(&assets)
76                    .unwrap_or_else(|_| "failed to serialize".into());
77                Ok(CallToolResult::success(vec![Content::text(content)]))
78            },
79            Err(e) => Ok(CallToolResult::error(vec![Content::text(e.to_string())])),
80        }
81    }
82}

121    fn result_text(result: &CallToolResult) -> String {
122        serde_json::to_value(result)
123            .ok()
124            .and_then(|v| v.get("content").and_then(|c| c.as_array()).cloned())
125            .and_then(|arr| arr.first().cloned())
126            .and_then(|first| {
127                first
128                    .get("text")
129                    .and_then(|t| t.as_str())
130                    .map(str::to_owned)
131            })
132            .unwrap_or_default()
133    }

135    #[tokio::test]
136    async fn download_assets_invalid_base_url_is_invalid_params() {
137        let (handler, _tmp) = test_handler().await;
138        let res = handler
139            .download_assets(Parameters(DownloadAssetsParams {
140                html: "<html></html>".to_string(),
141                base_url: "not a url".to_string(),
142                images: None,
143                documents: None,
144                output_dir: None,
145            }))
146            .await;
147        assert!(res.is_err(), "invalid base URL must be a protocol error");
148    }

150    #[tokio::test]
151    async fn download_assets_no_assets_is_empty_success() {
152        let (handler, _tmp) = test_handler().await;
153        // `output_dir` must be a safe relative path (params validation, #512).
154        let out_dir = "test-output/download-assets-empty";
155        let _ = std::fs::remove_dir_all(out_dir);
156        let res = handler
157            .download_assets(Parameters(DownloadAssetsParams {
158                html: "<html><body><p>no images here</p></body></html>".to_string(),
159                base_url: "https://example.com/".to_string(),
160                images: Some(true),
161                documents: None,
162                output_dir: Some(out_dir.to_string()),
163            }))
164            .await
165            .expect("download_assets returns Ok");
166        let text = result_text(&res);
167        assert!(
168            text.contains("[]"),
169            "no assets must yield an empty list: {text}"
170        );
171        let _ = std::fs::remove_dir_all(out_dir);
172    }

174    #[tokio::test]
175    async fn download_assets_downloads_image_from_html() {
176        let (handler, _tmp) = test_handler().await;
177        // `output_dir` must be a safe relative path (params validation, #512).
178        let out_dir = "test-output/download-assets-image";
179        let _ = std::fs::remove_dir_all(out_dir);
180        let server = MockServer::start().await;
181        // The page references an image; wiremock serves it so the download
182        // path executes against a real (ephemeral) HTTP endpoint.
183        Mock::given(method("GET"))
184            .and(path("/img/logo.png"))
185            .respond_with(
186                ResponseTemplate::new(200)
187                    .set_body_bytes(b"fake-png-bytes")
188                    .insert_header("content-type", "image/png"),
189            )
190            .mount(&server)
191            .await;
192
193        let html = format!(
194            r#"<html><body><img src="{}/img/logo.png"></body></html>"#,
195            server.uri()
196        );
197        let res = handler
198            .download_assets(Parameters(DownloadAssetsParams {
199                html,
200                base_url: server.uri(),
201                images: Some(true),
202                documents: None,
203                output_dir: Some(out_dir.to_string()),
204            }))
205            .await
206            .expect("download_assets returns Ok");
207        let text = result_text(&res);
208        // At least one asset downloaded and its local path reported.
209        assert!(
210            text.contains("logo.png") || text.contains("assets") || text.contains(".png"),
211            "downloaded asset must be reported: {text}"
212        );
213        let _ = std::fs::remove_dir_all(out_dir);
214    }
