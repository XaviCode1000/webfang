1//! AI Semantic tools — 2 tools for embeddings and semantic cleaning
2//!
3//! Tools: semantic_cleaner, search_obsidian
4//!
5//! Tool functions are always registered. `semantic_cleaner` runs when a
6//! semantic cleaner is injected into the `Container` (constructed behind the
7//! `ai` feature); without one it returns an honest feature-gated error.
8//! `search_obsidian` requires `embedding_port`, `note_repository`, and
9//! `text_chunker` to be injected (#386); without them it returns an honest
10//! feature-gated error.
11//!
12//! ## Test-coverage exception (issue #516, task 4)
13//! The success paths of both handlers require the ONNX semantic-cleaning
14//! model (`--features ai`) and call ports whose traits (`SemanticCleaner`,
15//! `EmbeddingPort`, `NoteRepository`, `TextChunker`) are sealed to
16//! `webfang_core`, so no stub can be injected from this crate's unit tests.
17//! Only the model-free error/validation branches are therefore unit-tested,
18//! which leaves `ai.rs` at ~65% line coverage. It is **intentionally exempt**
19//! from the ≥70% per-handler line-coverage gate. The model-dependent paths are
20//! exercised by the integration harness under the `ai` feature.

34/// Build an honest tool error (`isError:true`) carrying a Spanish message.
35///
36/// Shared by the AI handlers so feature-gated and operational failures are
37/// reported as `CallToolResult::error` — never an MCP protocol error, never a
38/// false success. Mirrors the slice-1 export handlers (see `export.rs`).
39fn honest_error(message: impl Into<String>) -> CallToolResult {
40    CallToolResult::error(vec![Content::text(message.into())])
41}

43/// Success envelope for `semantic_cleaner` (REQ-01).
44///
45/// Embeddings are inline and full — each chunk carries its vector (truncating
46/// would violate REQ-01). The summary fields give a quick overview without
47/// scanning `documents`.
48#[derive(serde::Serialize)]
49struct SemanticCleanResponse {
50    /// The URL that was fetched and cleaned.
51    url: String,
52    /// Number of semantic chunks produced.
53    chunks: usize,
54    /// Embedding dimensionality (0 when the cleaner produced no embeddings).
55    embedding_dim: usize,
56    /// The cleaned chunks, each with its embedding populated.
57    documents: Vec<DocumentChunk>,
58}
59
60/// Success envelope for `search_obsidian` (#386).
61#[derive(serde::Serialize)]
62struct VaultSearchResponse {
63    /// The search query.
64    query: String,
65    /// Number of results returned.
66    results: usize,
67    /// The matching note chunks, sorted by descending relevance.
68    documents: Vec<VaultSearchResult>,
69}

71#[tool_router(router = tool_router_ai, vis = "pub")]
72// Allow: the `tool_router_ai` fn generated by rmcp-macros cannot be documented from this crate.

75    /// Semantic HTML cleaning with AI embeddings
76    ///
77    /// Fetches a URL, cleans its HTML content, and returns semantically
78    /// chunked content with embeddings. Requires the `ai` feature.
79    #[tool(
80        description = "Fetch a URL, semantically clean its HTML content using AI embeddings, and return chunked content with vectors. Requires --features ai."
81    )]
82    #[instrument(skip(self), fields(url = %params.url))]
83    async fn semantic_cleaner(
84        &self,
85        Parameters(params): Parameters<ScrapeUrlParams>,
86    ) -> Result<CallToolResult, McpError> {
87        params.validate()?;
88
89        let _permit = acquire_semaphore!(self, ai);
90
91        // REQ-03: reject malformed URLs first (invalid-params), regardless of
92        // cleaner presence — no fetch, no cleaning.
93        let _url = url::Url::parse(&params.url).map_err(|e| {
94            McpError::invalid_params(
95                format!("URL inválida: {e}"),
96                Some(serde_json::Value::String("url".to_string())),
97            )
98        })?;
99
100        // REQ-02: absent cleaner (ai feature off) -> honest Spanish error.
101        let Some(cleaner) = self.state.container.cleaner() else {
102            return Ok(honest_error(
103                "funcionalidad no disponible: limpieza semántica con IA. Reconstruye con --features ai para habilitarla.",
104            ));
105        };
106
107        // REQ-01: fetch the page via the existing HTTP port, then clean it.
108        // Operational failures map to honest errors — never a false success.
109        let response = match self.state.container.http_client().get(&params.url).await {
110            Ok(resp) => resp,
111            Err(e) => return Ok(honest_error(format!("no se pudo obtener la página: {e}"))),
112        };
113        let documents = match cleaner.clean(&response.body).await {
114            Ok(docs) => docs,
115            Err(e) => return Ok(honest_error(format!("error al limpiar el contenido: {e}"))),
116        };
117
118        let chunks = documents.len();
119        let embedding_dim = documents
120            .first()
121            .and_then(|doc| doc.embeddings.as_ref())
122            .map(Vec::len)
123            .unwrap_or(0);
124        let envelope = SemanticCleanResponse {
125            url: params.url.clone(),
126            chunks,
127            embedding_dim,
128            documents,
129        };
130        let payload = match serde_json::to_string(&envelope) {
131            Ok(json) => json,
132            Err(e) => {
133                return Ok(honest_error(format!(
134                    "error al serializar la respuesta: {e}"
135                )))
136            },
137        };
138        Ok(CallToolResult::success(vec![Content::text(payload)]))
139    }

141    /// Semantic search over Obsidian vault using embeddings
142    ///
143    /// Searches indexed vault notes by embedding the query and ranking
144    /// against pre-computed chunk embeddings via cosine similarity.
145    /// Requires `embedding_port`, `note_repository`, and `text_chunker`
146    /// to be injected into the Container (#386).
147    #[tool(
148        description = "Semantic search over Obsidian vault using ONNX Runtime embeddings. Returns top matching notes by cosine similarity. Requires --features ai."
149    )]
150    #[instrument(skip(self), fields(query = %params.query))]
151    async fn search_obsidian(
152        &self,
153        Parameters(params): Parameters<SearchObsidianParams>,
154    ) -> Result<CallToolResult, McpError> {
155        params.validate()?;
156
157        let _permit = acquire_semaphore!(self, obsidian);
158
159        // Check all three required ports are available.
160        let Some(embedding) = self.state.container.embedding_port() else {
161            return Ok(honest_error(
162                "funcionalidad no disponible: búsqueda semántica en Obsidian. Reconstruye con --features ai para habilitarla.",
163            ));
164        };
165        let Some(repo) = self.state.container.note_repository() else {
166            return Ok(honest_error(
167                "funcionalidad no disponible: no hay repositorio de notas configurado. Verifica la configuración de persistencia.",
168            ));
169        };
170        let Some(chunker) = self.state.container.text_chunker() else {
171            return Ok(honest_error(
172                "funcionalidad no disponible: no hay chunker de texto configurado. Reconstruye con --features ai.",
173            ));
174        };
175
176        let service = VaultSearchService::new(embedding, repo, chunker);
177        let limit = params.limit.unwrap_or(10);
178
179        // Lazy sync: reconcile the vault filesystem against the index
180        // before searching. Uses the explicit vault_path param or falls
181        // back to auto-detection. Sync failures are non-fatal — we
182        // proceed with whatever is already indexed.
183        let vault_path = params
184            .vault_path
185            .clone()
186            .or_else(|| detect_vault(None, None, None).map(|p| p.to_string_lossy().into_owned()));
187        if let Some(vp) = &vault_path {
188            match service.sync_vault(std::path::Path::new(vp)).await {
189                Ok(SyncSummary {
190                    indexed,
191                    updated,
192                    deleted,
193                    unchanged,
194                }) => {
195                    tracing::info!(
196                        indexed,
197                        updated,
198                        deleted,
199                        unchanged,
200                        vault_path = %vp,
201                        "lazy vault sync completed before search"
202                    );
203                },
204                Err(e) => {
205                    tracing::warn!(
206                        vault_path = %vp,
207                        "vault sync failed, searching existing index: {e}"
208                    );
209                },
210            }
211        }
212
213        match service.search(&params.query, limit).await {
214            Ok(results) => {
215                let envelope = VaultSearchResponse {
216                    query: params.query.clone(),
217                    results: results.len(),
218                    documents: results,
219                };
220                match serde_json::to_string(&envelope) {
221                    Ok(json) => Ok(CallToolResult::success(vec![Content::text(json)])),
222                    Err(e) => Ok(honest_error(format!(
223                        "error al serializar la respuesta: {e}"
224                    ))),
225                }
226            },
227            Err(e) => Ok(honest_error(format!("error en la búsqueda semántica: {e}"))),
228        }
229    }
230}

252    /// REQ-02/04 contract: `honest_error` produces a `CallToolResult` that
253    /// serializes with `isError:true` and carries the exact Spanish text —
254    /// never a protocol error, never a false success. Mirrors the slice-1
255    /// export-handler test (export.rs).
256    #[test]
257    fn honest_error_sets_is_error_and_spanish_text() {
258        let result = honest_error("funcionalidad no disponible: prueba");
259
260        // Serialize exactly as the MCP transport would, then assert the honest
261        // error contract: isError:true plus the Spanish message.
262        let json = serde_json::to_value(&result).expect("CallToolResult must serialize");
263        assert_eq!(
264            json.get("isError").and_then(|v| v.as_bool()),
265            Some(true),
266            "honest_error must set isError:true, got: {json}"
267        );
268        let text = json
269            .get("content")
270            .and_then(|c| c.as_array())
271            .and_then(|arr| arr.first())
272            .and_then(|first| first.get("text"))
273            .and_then(|t| t.as_str())
274            .unwrap_or_default();
275        assert!(
276            text.contains("funcionalidad no disponible"),
277            "honest Spanish error text expected, got: {text}"
278        );
279    }

294            .await
295            .expect("create container");
296        (McpHandler::new(McpState::new(container)), tmp)
297    }

299    fn result_text(result: &CallToolResult) -> String {
300        serde_json::to_value(result)
301            .ok()
302            .and_then(|v| v.get("content").and_then(|c| c.as_array()).cloned())
303            .and_then(|arr| arr.first().cloned())
304            .and_then(|first| {
305                first
306                    .get("text")
307                    .and_then(|t| t.as_str())
308                    .map(str::to_owned)
309            })
310            .unwrap_or_default()
311    }
