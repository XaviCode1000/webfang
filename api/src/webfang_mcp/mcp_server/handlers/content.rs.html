20    /// Remove boilerplate from HTML (scripts, nav, sidebar, footer, SVG)
21    #[tool(
22        description = "Remove boilerplate from HTML including scripts, styles, navigation, sidebar, footer, and SVG elements. Returns cleaned HTML."
23    )]
24    #[instrument(skip(self), fields(html_len = params.html.len()))]
25    async fn clean_html(
26        &self,
27        Parameters(params): Parameters<CleanHtmlParams>,
28    ) -> Result<CallToolResult, McpError> {
29        params.validate()?;
30
31        let _permit = acquire_semaphore!(self, content);
32
33        let cleaned =
34            webfang_core::infrastructure::converter::html_cleaner::clean_html(&params.html);
35        Ok(CallToolResult::success(vec![Content::text(cleaned)]))
36    }
37
38    /// Convert HTML to Markdown
39    #[tool(
40        description = "Convert HTML to Markdown, preserving headings, code blocks, lists, and formatting."
41    )]
42    #[instrument(skip(self), fields(html_len = params.html.len()))]
43    async fn convert_html_to_markdown(
44        &self,
45        Parameters(params): Parameters<HtmlToMarkdownParams>,
46    ) -> Result<CallToolResult, McpError> {
47        params.validate()?;
48
49        let _permit = acquire_semaphore!(self, content);
50
51        let md = webfang_core::infrastructure::converter::html_to_markdown::convert_to_markdown(
52            &params.html,
53        );
54        Ok(CallToolResult::success(vec![Content::text(md)]))
55    }
56
57    /// Extract all crawlable href links from HTML
58    #[tool(
59        description = "Extract href links from HTML content, honoring rel=\"nofollow\" (excluded) and a document <base href>. Returns list of raw href values."
60    )]
61    #[instrument(skip(self), fields(base_url = %params.base_url))]
62    async fn extract_links(
63        &self,
64        Parameters(params): Parameters<ExtractLinksParams>,
65    ) -> Result<CallToolResult, McpError> {
66        params.validate()?;
67
68        let _permit = acquire_semaphore!(self, content);
69
70        match webfang_core::infrastructure::crawler::extract_links(&params.html, &params.base_url) {
71            Ok(links) => {
72                let content = serde_json::to_string_pretty(&links)
73                    .unwrap_or_else(|_| "failed to serialize".into());
74                Ok(CallToolResult::success(vec![Content::text(content)]))
75            },
76            Err(e) => Ok(CallToolResult::error(vec![Content::text(e.to_string())])),
77        }
78    }
79

120    /// Generate YAML frontmatter for a scraped document
121    #[tool(
122        description = "Generate YAML frontmatter with title, URL, date, author, excerpt, and optional rich metadata."
123    )]
124    #[instrument(skip(self), fields(params = ?params))]
125    async fn generate_frontmatter(
126        &self,
127        Parameters(params): Parameters<GenerateFrontmatterParams>,
128    ) -> Result<CallToolResult, McpError> {
129        params.validate()?;
130
131        let _permit = acquire_semaphore!(self, content);
132
133        let title = params.title.as_deref().unwrap_or("Untitled");
134        let url = params.url.as_deref().unwrap_or("");
135        let fm = webfang_core::infrastructure::output::frontmatter::generate_with_metadata(
136            title,
137            url,
138            None,
139            None,
140            None,
141            &[],
142            None,
143        );
144        Ok(CallToolResult::success(vec![Content::text(fm)]))
145    }

147    /// Generate rich metadata from scraped content (word count, reading time, language, content type)
148    #[tool(
149        description = "Generate rich metadata from scraped content including word count, reading time (200 WPM), language detection, and content type classification."
150    )]
151    #[instrument(skip(self), fields(params = ?params))]
152    // serde_json::to_string cannot fail for a serde_json::Value.
153    #[allow(clippy::expect_used)]
154    async fn generate_rich_metadata(
155        &self,
156        Parameters(params): Parameters<GenerateRichMetadataParams>,
157    ) -> Result<CallToolResult, McpError> {
158        params.validate()?;
159
160        let _permit = acquire_semaphore!(self, content);
161
162        let content = params.content.as_deref().unwrap_or("");
163
164        let word_count =
165            webfang_core::infrastructure::obsidian::metadata::compute_word_count(content);
166        let reading_time =
167            webfang_core::infrastructure::obsidian::metadata::compute_reading_time(word_count);
168
169        let meta = serde_json::json!({
170            "word_count": word_count,
171            "reading_time_minutes": reading_time,
172        });
173        Ok(CallToolResult::success(vec![Content::text(
174            serde_json::to_string_pretty(&meta).expect("serializing JSON to a string cannot fail"),
175        )]))
176    }
177}

217    /// Extract the first `content[0].text` from a `CallToolResult` (mirrors the
218    /// MCP transport serialization used by the integration harness).
219    fn result_text(result: &CallToolResult) -> String {
220        serde_json::to_value(result)
221            .ok()
222            .and_then(|v| v.get("content").and_then(|c| c.as_array()).cloned())
223            .and_then(|arr| arr.first().cloned())
224            .and_then(|first| {
225                first
226                    .get("text")
227                    .and_then(|t| t.as_str())
228                    .map(str::to_owned)
229            })
230            .unwrap_or_default()
231    }

235        let (handler, _tmp) = test_handler().await;
236        let html = "<html><head><script>evil()</script></head><body><p>Hello</p></body></html>";

241            .await

244        assert!(

246            "cleaned text must keep body: {text}"
247        );

249    }
