21    /// Detect WAF/CAPTCHA challenge in HTML body
22    #[tool(
23        description = "Scan HTML body for WAF/CAPTCHA signatures (Cloudflare, reCAPTCHA, hCaptcha, DataDome, PerimeterX, Akamai, etc.). Runs in degraded mode (no HTTP context): reports only unambiguous challenge markers — vendor fingerprints need status via verify_waf_integrity. Returns provider name if detected."
24    )]
25    #[instrument(skip(self), fields(html_len = params.html.len()))]
26    async fn detect_waf(
27        &self,
28        Parameters(params): Parameters<DetectWafParams>,
29    ) -> Result<CallToolResult, McpError> {
30        params.validate()?;
31
32        let _permit = acquire_semaphore!(self, security);
33
34        match detect_waf_provider(&params.html) {
35            Some(provider) => Ok(CallToolResult::success(vec![Content::text(format!(
36                "WAF detected: {provider}"
37            ))])),
38            None => Ok(CallToolResult::success(vec![Content::text(
39                "no WAF detected",
40            )])),
41        }
42    }
43
44    /// Multi-layer WAF inspection (headers + body + entropy analysis)
45    #[tool(
46        description = "Multi-layer WAF inspection: checks control headers, body signatures via Aho-Corasick, and entropy analysis for silent challenges. Optionally pass status and content_type for context-aware detection (fingerprint evidence then blocks only on correlated WAF statuses 403/429/503/520-529); without them, runs degraded mode where only unambiguous challenge markers block and fingerprint/control-header evidence never blocks on mere presence. The status/content_type params are additive (tool signature backward compatible); control-header verdict semantics intentionally changed per issue #346 — mere-presence blocking was the bug."
47    )]
48    #[instrument(skip(self), fields(params = ?params))]
49    async fn verify_waf_integrity(
50        &self,
51        Parameters(params): Parameters<VerifyWafIntegrityParams>,
52    ) -> Result<CallToolResult, McpError> {
53        params.validate()?;
54
55        let _permit = acquire_semaphore!(self, security);
56
57        let html = params.html.as_deref().unwrap_or("");
58        let mut header_map = wreq::header::HeaderMap::new();
59        if let Some(ref hdrs) = params.headers {
60            for (key, value) in hdrs {
61                if let (Ok(name), Ok(val)) = (
62                    wreq::header::HeaderName::from_bytes(key.as_bytes()),
63                    wreq::header::HeaderValue::from_str(value),
64                ) {
65                    header_map.insert(name, val);
66                }
67            }
68        }
69        // Additive optional context (REQ-WAF-09): the status/content_type params
70        // are backward compatible (tool signature unchanged). The verdict is NOT
71        // unchanged, though — without them this is degraded mode, where control
72        // header (Fingerprint) evidence never blocks on mere presence. That is the
73        // intentional #346 / REQ-WAF-05 fix (mere-presence blocking was the bug),
74        // so degraded verdicts deliberately differ from the pre-#346 verify_integrity.
75        let verdict = verify_waf_verdict(html, params.status, params.content_type, header_map);
76        if verdict.is_blocked {
77            Ok(CallToolResult::success(vec![Content::text(format!(
78                "WAF blocked: {}",
79                verdict.evidence_chain()
80            ))]))
81        } else {
82            Ok(CallToolResult::success(vec![Content::text(
83                "WAF integrity check passed",
84            )]))
85        }
86    }

124/// Inspect a body in degraded mode for the MCP `detect_waf` tool (REQ-WAF-09).
125///
126/// Returns the first evidence's provider when the degraded verdict blocks.
127/// Without HTTP context only [`WafTier::Challenge`](WafInspector) markers block;
128/// fingerprint evidence is collected but never blocks here — this is the
129/// false-positive fix (#346).
130fn detect_waf_provider(html: &str) -> Option<&'static str> {
131    let verdict = WafInspector::inspect(html, &InspectionContext::default());
132    if verdict.is_blocked {
133        verdict.evidences.first().map(|e| e.provider)
134    } else {
135        None
136    }
137}

139/// Run the `verify_waf_integrity` inspection (REQ-WAF-09).
140///
141/// `status` and `content_type` are additive optional context, so the API shape
142/// (tool signature) is backward compatible. The verdict semantics are NOT
143/// unchanged, though: when both are absent the inspection runs degraded mode,
144/// where only Challenge-tier (T1) markers block and control-header / fingerprint
145/// evidence never blocks on mere presence. That is the intentional fix for
146/// issue #346 / REQ-WAF-05 — mere-presence blocking of control headers was the
147/// bug — so degraded-mode verdicts deliberately differ from the pre-#346
148/// `verify_integrity`.
149fn verify_waf_verdict(
150    html: &str,
151    status: Option<u16>,
152    content_type: Option<String>,
153    headers: wreq::header::HeaderMap,
154) -> WafVerdict {
155    let ctx = InspectionContext {
156        status,
157        content_type,
158        headers,
159        ignore_waf: false,
160    };
161    WafInspector::inspect(html, &ctx)
162}

164/// Render a metrics snapshot as the `get_scrape_metrics` tool result.
165///
166/// Empty accumulator (`total_events == 0`, DD-12) → honest Spanish
167/// pre-condition error (`isError:true`, REQ-05). Populated → the snapshot as
168/// pretty JSON via `CallToolResult::success` (REQ-04). A serialization failure
169/// maps to an honest Spanish error, never a panic (REQ-10). Mirrors the
170/// `load_results_from` honest-error pattern in `export.rs`.
171fn render_metrics(snapshot: &MetricsSnapshot) -> CallToolResult {
172    if snapshot.total_events == 0 {
173        return CallToolResult::error(vec![Content::text(
174            "no hay métricas disponibles: todavía no se registró ninguna operación de scraping",
175        )]);
176    }
177    match serde_json::to_string_pretty(snapshot) {
178        Ok(json) => CallToolResult::success(vec![Content::text(json)]),
179        Err(e) => CallToolResult::error(vec![Content::text(format!(
180            "no se pudieron serializar las métricas: {e}"
181        ))]),
182    }

278    /// REQ-05/REQ-10: an empty accumulator renders an honest Spanish
279    /// pre-condition error (isError:true) — NOT the legacy canned success JSON.
280    #[test]
281    fn render_metrics_empty_is_honest_spanish_error() {
282        use crate::mcp_server::metrics::ScrapeMetrics;
283
284        let snapshot = ScrapeMetrics::default().snapshot();
285        assert_eq!(snapshot.total_events, 0, "fresh accumulator is empty");
286        let result = render_metrics(&snapshot);
287
288        // Serialize exactly as the MCP transport would, then assert the honest
289        // error contract (mirrors export.rs load_results_from_none test).
290        let json = serde_json::to_value(&result).expect("CallToolResult must serialize");
291        assert_eq!(
292            json.get("isError").and_then(|v| v.as_bool()),
293            Some(true),
294            "empty metrics must set isError:true, got: {json}"
295        );
296        let text = json
297            .get("content")
298            .and_then(|c| c.as_array())
299            .and_then(|arr| arr.first())
300            .and_then(|first| first.get("text"))
301            .and_then(|t| t.as_str())
302            .unwrap_or_default();
303        assert!(
304            text.contains("no hay métricas disponibles"),
305            "honest Spanish empty-state error expected, got: {text}"
306        );
307        assert!(
308            !text.contains("Metrics collection requires active scraping session"),
309            "legacy canned text must be gone, got: {text}"
310        );

388    fn result_text(result: &CallToolResult) -> String {
389        serde_json::to_value(result)
390            .ok()
391            .and_then(|v| v.get("content").and_then(|c| c.as_array()).cloned())
392            .and_then(|arr| arr.first().cloned())
393            .and_then(|first| {
394                first
395                    .get("text")
396                    .and_then(|t| t.as_str())
397                    .map(str::to_owned)
398            })
399            .unwrap_or_default()
400    }
