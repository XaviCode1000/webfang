1//! Export tools — 4 tools for real output format conversion
2//!
3//! Tools: export_file, export_jsonl, export_vector,
4//! process_export_pipeline
5//!
6//! Every tool performs a REAL export via the existing `webfang_core`
7//! `export_factory` surface (jsonl/vector/auto) and reports honest
8//! success/error. Operational failures (no repository, empty results,
9//! missing content, I/O errors) map to `CallToolResult::error`
10//! (isError:true, Spanish). Invalid parameters (bad format) map to a
11//! protocol-level `McpError::invalid_params` — never a silent fallback.

29/// Run a real export of persisted results to the given format.
30///
31/// Maps operational [`ExporterError`](webfang_core::domain::exporter::ExporterError)s
32/// to an honest `CallToolResult::error` (isError:true, Spanish) and reports the
33/// real written path on success.
34fn export_results(
35    results: &[ScrapedContent],
36    output_dir: PathBuf,
37    format: ExportFormat,
38    filename: &str,
39) -> Result<CallToolResult, McpError> {
40    let count = results.len();
41    match process_results(results, output_dir.clone(), format, filename, None, false) {
42        Ok(_) => {
43            let path = resolve_export_path(&output_dir, filename, format);
44            tracing::info!(documents = count, path = %path.display(), "export completed");
45            Ok(CallToolResult::success(vec![Content::text(format!(
46                "Exportación completada: {count} documentos → {}",
47                path.display()
48            ))]))
49        },
50        Err(e) => Ok(CallToolResult::error(vec![Content::text(format!(
51            "error al exportar: {e}"
52        ))])),
53    }
54}

56/// Compute the real on-disk path an export wrote to.
57///
58/// For concrete formats this is `{output_dir}/{filename}.{ext}`. For `Auto`
59/// the concrete extension is resolved from what actually exists (jsonl
60/// preferred, then json), so the reported path is always truthful.
61fn resolve_export_path(output_dir: &Path, filename: &str, format: ExportFormat) -> PathBuf {
62    match format {
63        ExportFormat::Auto => {
64            let jsonl = output_dir.join(format!("{filename}.jsonl"));
65            if jsonl.exists() {
66                jsonl
67            } else {
68                output_dir.join(format!("{filename}.json"))
69            }
70        },
71        concrete => output_dir.join(format!("{filename}.{}", concrete.extension())),
72    }
73}

75/// Resolve persisted crawl results from an optional repository, mapping every
76/// operational failure to an honest `CallToolResult::error` (isError:true,
77/// Spanish).
78///
79/// Extracted from [`McpHandler::load_results`] as a free function so the
80/// `None`-repository branch — unreachable through `Container::new`, which
81/// always attempts to wire a repository and tolerates log corruption — can be
82/// unit-tested directly (REQ-MCP-EXPORT-05).
83fn load_results_from(
84    repo: Option<Arc<dyn CrawlResultRepository>>,
85) -> Result<Vec<ScrapedContent>, CallToolResult> {
86    let repo = match repo {
87        Some(repo) => repo,
88        None => {
89            return Err(CallToolResult::error(vec![Content::text(
90                "no hay repositorio de resultados disponible",
91            )]))
92        },
93    };
94    let results = repo.load_all().map_err(|e| {
95        CallToolResult::error(vec![Content::text(format!(
96            "no se pudieron cargar los resultados: {e}"
97        ))])
98    })?;
99    if results.is_empty() {
100        return Err(CallToolResult::error(vec![Content::text(
101            "no hay resultados disponibles para exportar",
102        )]));
103    }
104    Ok(results)
105}

107impl McpHandler {
108    /// Load all persisted crawl results, mapping operational failures to an
109    /// honest `CallToolResult::error` (isError:true, Spanish).
110    ///
111    /// Returns `Err(CallToolResult)` when no repository is wired, the bulk
112    /// load fails, or the repository holds zero results; callers propagate
113    /// this directly.
114    async fn load_results(&self) -> Result<Vec<ScrapedContent>, CallToolResult> {
115        load_results_from(self.state.container.crawl_result_repository())
116    }
117}

119#[tool_router(router = tool_router_export, vis = "pub")]
120// Allow: the `tool_router_export` fn generated by rmcp-macros cannot be documented from this crate.
121#[allow(missing_docs)]
122impl McpHandler {
123    /// Save caller-provided content as a structured export file (jsonl/vector)
124    #[tool(
125        description = "Save caller-provided content to a structured export file. Supported formats: jsonl, vector, auto. Reports the real written path."
126    )]
127    #[instrument(skip(self), fields(filename = %params.filename, format = %params.format))]
128    async fn export_file(
129        &self,
130        Parameters(params): Parameters<ExportFileParams>,
131    ) -> Result<CallToolResult, McpError> {
132        params.validate()?;
133
134        let _permit = acquire_semaphore!(self, export);
135
136        // Honest error on empty content (REQ-MCP-EXPORT-05).
137        if params.content.trim().is_empty() {
138            return Ok(CallToolResult::error(vec![Content::text(
139                "el contenido no puede estar vacío",
140            )]));
141        }
142
143        // Invalid format is a protocol-level invalid-params error, never a
144        // silent fallback (REQ-MCP-EXPORT-07).
145        let format = ExportFormat::parse_str(&params.format).map_err(|e| {
146            McpError::invalid_params(
147                format!("formato inválido: {e}"),
148                Some(serde_json::Value::String("format".to_string())),
149            )
150        })?;
151
152        let output_dir = PathBuf::from(&params.output_dir);
153        let filename = params.filename.clone();
154
155        // Build a validated document chunk from the caller content. The chunk
156        // id/timestamp are generated internally; the synthetic URL satisfies
157        // validation (any parseable scheme) and scopes the doc to this tool.
158        let url = url::Url::parse(&format!("https://webfang.local/{filename}")).map_err(|e| {
159            McpError::invalid_params(
160                format!("nombre de archivo inválido: {e}"),
161                Some(serde_json::Value::String("filename".to_string())),
162            )
163        })?;
164        let scraped = ScrapedContent {
165            title: filename.clone(),
166            content: params.content.clone(),
167            url: webfang_core::domain::ValidUrl::new(url),
168            excerpt: None,
169            author: None,
170            date: None,
171            html: None,
172            assets: vec![],
173            correlation_id: None,
174        };
175        let validated = match DocumentChunkUnvalidated::from_scraped_content(&scraped).validate() {
176            Ok(v) => v,
177            Err(e) => {
178                return Ok(CallToolResult::error(vec![Content::text(format!(
179                    "contenido inválido: {e}"
180                ))]))
181            },
182        };
183
184        let exporter = match create_exporter(output_dir.clone(), &filename, format) {
185            Ok(exporter) => exporter,
186            Err(e) => {
187                return Ok(CallToolResult::error(vec![Content::text(format!(
188                    "no se pudo crear el exportador: {e}"
189                ))]))
190            },
191        };
192
193        match exporter.export(validated) {
194            Ok(()) => {
195                let path = resolve_export_path(&output_dir, &filename, format);
196                tracing::info!(documents = 1, path = %path.display(), "export completed");
197                Ok(CallToolResult::success(vec![Content::text(format!(
198                    "Exportación completada: 1 documentos → {}",
199                    path.display()
200                ))]))
201            },
202            Err(e) => Ok(CallToolResult::error(vec![Content::text(format!(
203                "error al exportar: {e}"
204            ))])),
205        }
206    }
207
208    /// Export persisted crawl results to JSONL format (one JSON object per line)
209    #[tool(
210        description = "Export persisted crawl results to JSONL format (one JSON object per line). Optimal for RAG pipeline ingestion. Reports the real written path."
211    )]
212    #[instrument(skip(self), fields(filename, format = "jsonl", results))]
213    async fn export_jsonl(
214        &self,
215        Parameters(params): Parameters<ExportJsonlParams>,
216    ) -> Result<CallToolResult, McpError> {
217        params.validate()?;
218
219        let _permit = acquire_semaphore!(self, export);
220
221        let output_dir = PathBuf::from(params.output_dir.as_deref().unwrap_or("./output"));
222        let filename = params.filename.as_deref().unwrap_or("export").to_string();
223
224        let results = match self.load_results().await {
225            Ok(results) => results,
226            Err(err) => return Ok(err),
227        };
228        let span = tracing::Span::current();
229        span.record("filename", filename.as_str());
230        span.record("results", results.len());
231
232        export_results(&results, output_dir, ExportFormat::Jsonl, &filename)
233    }

240    async fn export_vector(

249        let filename = params.filename.as_deref().unwrap_or("export").to_string();

261
262    /// Full export pipeline: load persisted results → export synchronously
263    #[tool(
264        description = "Run the export pipeline synchronously: load persisted crawl results and export them to the specified format (jsonl, vector, or auto; default jsonl). Reports the real written path; never queues."
265    )]
266    #[instrument(skip(self), fields(format, results))]
267    async fn process_export_pipeline(
268        &self,
269        Parameters(params): Parameters<ProcessExportPipelineParams>,
270    ) -> Result<CallToolResult, McpError> {
271        params.validate()?;
272
273        let _permit = acquire_semaphore!(self, export);
274
275        let format_str = params.format.as_deref().unwrap_or("jsonl");
276        let format = ExportFormat::parse_str(format_str).map_err(|e| {
277            McpError::invalid_params(
278                format!("formato inválido: {e}"),
279                Some(serde_json::Value::String("format".to_string())),
280            )
281        })?;
282
283        let results = match self.load_results().await {
284            Ok(results) => results,
285            Err(err) => return Ok(err),
286        };
287        let span = tracing::Span::current();
288        span.record("format", format_str);
289        span.record("results", results.len());
290
291        // The pipeline exports to the container's configured output directory.
292        let output_dir = self.state.container.scraper_config.output_dir.clone();
293        export_results(&results, output_dir, format, "export")
294    }
295}

311    /// REQ-MCP-EXPORT-05 (repository unavailable): when no repository is wired,
312    /// the loader must return an honest `CallToolResult::error` (isError:true)
313    /// carrying the Spanish "no hay repositorio" message — never a fake success.
314    ///
315    /// This branch is unreachable through `Container::new` (which always
316    /// attempts to wire a repository and tolerates log corruption), so it is
317    /// exercised directly through the extracted `load_results_from` free
318    /// function by passing `None`.
319    #[test]
320    fn load_results_from_none_repository_is_honest_error() {
321        let err = load_results_from(None).expect_err("None repository must be an error");
322
323        // Serialize exactly as the MCP transport would, then assert the honest
324        // error contract: isError:true plus the Spanish message.
325        let json = serde_json::to_value(&err).expect("CallToolResult must serialize");
326        assert_eq!(
327            json.get("isError").and_then(|v| v.as_bool()),
328            Some(true),
329            "None-repository path must set isError:true, got: {json}"
330        );
331        let text = json
332            .get("content")
333            .and_then(|c| c.as_array())
334            .and_then(|arr| arr.first())
335            .and_then(|first| first.get("text"))
336            .and_then(|t| t.as_str())
337            .unwrap_or_default();
338        assert!(
339            text.contains("no hay repositorio de resultados disponible"),
340            "honest Spanish no-repository error expected, got: {text}"
341        );
342    }
343}

405    fn result_text(result: &CallToolResult) -> String {
406        serde_json::to_value(result)
407            .ok()
408            .and_then(|v| v.get("content").and_then(|c| c.as_array()).cloned())
409            .and_then(|arr| arr.first().cloned())
410            .and_then(|first| {
411                first
412                    .get("text")
413                    .and_then(|t| t.as_str())
414                    .map(str::to_owned)
415            })
416            .unwrap_or_default()
417    }
