1//! AI port wiring for the MCP server (#433).
2//!
3//! Constructs the concrete vault-search ports — the ONNX embedding adapter and
4//! the Markdown chunker — behind the `ai` feature and injects them into the
5//! [`Container`](webfang_core::application::container::Container). The SQLite
6//! note repository is wired ONLY when the consumer
7//! explicitly enables the `persistence` feature (`--features ai,persistence`),
8//! keeping the `ai` feature free of database dependencies.
9//!
10//! Mirrors the semantic-cleaner wiring precedent in `examples/mcp_server.rs`:
11//! construction failures degrade gracefully (a warning is logged and the server
12//! keeps running) so the affected tools answer with honest feature-gated errors
13//! instead of the server failing to boot.
14//!
15//! When `persistence` is enabled, the note repository reuses the elastic
16//! pipeline's SQLite database (resolved via
17//! [`resolve_db_path`](webfang_core::infrastructure::autotuning::resolve_db_path)
18//! / [`env_db_path`](webfang_core::infrastructure::autotuning::env_db_path),
19//! default `~/.webfang/crawl.db`, overridable with `WEBFANG_DB_PATH`). The frozen
20//! schema already carries the `notes` and `note_chunks` tables alongside the
21//! elastic `chunks` table, so one database serves both pipelines.
22
23use std::sync::Arc;
24
25use webfang_core::application::container::Container;
26use webfang_core::domain::embedding_port::EmbeddingPort;
27
28/// Wire the vault-search AI ports into `container`.
29///
30/// Injects, in order: the ONNX [`EmbeddingAdapter`](webfang_ai::EmbeddingAdapter)
31/// — assembled from the semantic cleaner's shared inference pool + tokenizer, so
32/// the ONNX model is loaded exactly once — a
33/// [`MarkdownChunker`](webfang_ai::MarkdownChunker), and a SQLite-backed
34/// [`NoteRepository`](webfang_core::domain::note_repository::NoteRepository).
35/// Embedding + chunker assembly is infallible (the components
36/// are already valid); only the note repository can fail, in which case the
37/// embedding port and chunker remain wired and vault search degrades to an honest
38/// "not available" error at call time.
39///
40/// # Errors
41///
42/// Never fails — the note-repository construction error is logged and degraded
43/// around, so the caller always receives a usable [`Container`].
44#[must_use]
45pub async fn wire_ai_ports(
46    mut container: Container,
47    pool: Arc<webfang_ai::InferencePool>,
48    tokenizer: Arc<webfang_ai::MiniLmTokenizer>,
49) -> Container {
50    // 1. Embedding port (ONNX adapter) — shares the cleaner's pool + tokenizer,
51    //    so this is infallible (no model resolution happens here).
52    let adapter = webfang_ai::EmbeddingAdapter::new(pool, tokenizer);
53    let dim = adapter.embedding_dim();
54    container = container.with_embedding_port(Arc::new(adapter));
55
56    // 2. Text chunker (Markdown segmentation).
57    container = container.with_text_chunker(Arc::new(webfang_ai::MarkdownChunker::new()));
58
59    // 3. Note repository (SQLite persistence) — only when the consumer
60    //    explicitly enables the `persistence` feature. Without it, vault
61    //    search degrades to an honest "not available" error at call time.
62    #[cfg(feature = "persistence")]
63    {
64        match build_note_repository().await {
65            Ok(repo) => {
66                container = container.with_note_repository(repo);
67                tracing::info!(
68                    dim,
69                    "vault-search AI ports wired (embedding + chunker + notes)"
70                );
71            },
72            Err(e) => {
73                tracing::warn!(
74                    error = %e,
75                    "note repository unavailable, vault search persistence disabled"
76                );
77            },
78        }
79    }
80    #[cfg(not(feature = "persistence"))]
81    {
82        tracing::info!(
83            dim,
84            "vault-search AI ports wired (embedding + chunker); enable `persistence` for note storage"
85        );
86    }
87
88    container
89}

91/// Build the SQLite-backed note repository on the elastic pipeline's database.
92///
93/// Resolves the DB path via the hardware-autotuning convention (CLI > env >
94/// `~/.webfang/crawl.db`), opens a WAL-mode pool, and runs the idempotent schema
95/// setup (creating the `notes`/`note_chunks` tables if missing).
96///
97/// Only compiled when the `persistence` feature is explicitly enabled — the
98/// consumer opts in with `--features ai,persistence`.
99#[cfg(feature = "persistence")]
100async fn build_note_repository() -> Result<
101    Arc<dyn webfang_core::domain::note_repository::NoteRepository>,
102    Box<dyn std::error::Error + Send + Sync>,
103> {
104    use webfang_core::infrastructure::autotuning::{env_db_path, resolve_db_path};
105    use webfang_core::infrastructure::persistence::{
106        create_pool, setup_schema, SqliteVectorRepository,
107    };
108
109    let db_path = resolve_db_path(None, env_db_path());
110    let pool = create_pool(&db_path, 4)?;
111    setup_schema(&pool).await?;
112    Ok(Arc::new(SqliteVectorRepository::new(pool)))
113}
