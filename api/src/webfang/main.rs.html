3//! WebFang - Production-ready web scraper with Clean Architecture
4//!
5//! Extracts clean, structured content from web pages using readability algorithm.
6//!
7//! # Architecture
8//!
9//! Following Clean Architecture with TUI support:
10//!
11//! ```text
12//! main.rs (thin entry point) -> orchestrator::run()
13//!     |
14//!     ├─→ Args::try_parse()           ← CLI parsing
15//!     ├─→ handle_completions()        ← Subcommand handling
16//!     ├─→ run_config_tui()             ← Config TUI (if --config-tui)
17//!     ├─→ ConfigDefaults::load()      ← TOML config
18//!     ├─→ preflight::apply_config_defaults() ← Config merge
19//!     ├─→ init_logging_dual()         ← stderr-only tracing
20//!     └─→ orchestrator::run()         ← Full pipeline
21//! ```
22//!
23//! **Golden Rule:** Application layer NEVER imports ratatui/crossterm/indicatif.

64/// Run the unified TUI with collapsible config sections.
65///
66/// Phase 1: Config form with 8 collapsible sections (39 fields)
67/// Phase 2: URL selector (after config submitted)
68///
69/// Returns `Ok(Some(values))` if both phases completed,
70/// `Ok(None)` if cancelled at any point, or `Err` if TTY not available.
71#[cfg(feature = "ui")]
72async fn run_unified_tui() -> Result<Option<serde_json::Value>, CliExit> {
73    // Check if stdout is a TTY
74    if !io::stdout().is_terminal() {
75        tracing::error!("--tui requiere un terminal interactivo");
76        return Err(CliExit::UsageError(
77            "--tui requiere un terminal interactivo".into(),
78        ));
79    }
80
81    // =========================================================================
82    // Phase 1: Configuration Form
83    // =========================================================================
84    let mut config_app = match App::new(AppMode::Config) {
85        Ok(app) => app,
86        Err(e) => {
87            tracing::error!(error = %e, "Error al crear la aplicación TUI");
88            return Err(CliExit::UsageError(format!(
89                "Error creando la aplicación: {e}"
90            )));
91        },
92    }
93    .with_component(Header::new(AppMode::Config))
94    .with_component(CollapsibleConfig::new())
95    .with_component(StatusBar::new().with_items(vec![
96        ("↑↓", "Navegar"),
97        ("Enter", "Expandir"),
98        ("←", "Colapsar"),
99        ("Ctrl+S", "Enviar"),
100        ("q", "Salir"),
101    ]))
102    .with_modal(HelpModal::new(
103        "Ayuda — Configuración".into(),
104        vec![
105            ("↑↓".into(), "Navegar secciones".into()),
106            ("Enter/→".into(), "Expandir sección".into()),
107            ("←".into(), "Colapsar sección".into()),
108            ("Space".into(), "Toggle expand/collapse".into()),
109            ("Tab".into(), "Mover a campos".into()),
110            ("Ctrl+S".into(), "Enviar formulario".into()),
111            ("?".into(), "Mostrar ayuda".into()),
112            ("q".into(), "Salir".into()),
113        ],
114    ));
115
116    let config_values = match config_app.run().await {
117        Ok(AppResult::Config(values)) => values,
118        Ok(AppResult::None) => return Ok(None), // User cancelled
119        Ok(_) => return Ok(None),
120        Err(e) => {
121            tracing::error!(error = %e, "Error en TUI de configuración");
122            return Ok(None);
123        },
124    };
125
126    // If config was cancelled or empty, return None
127    let config_values = match config_values {
128        Some(v) => v,
129        None => return Ok(None),
130    };
131
132    // =========================================================================
133    // Phase 2: URL Selection (using config values)
134    // =========================================================================
135    // The URL will be extracted from config and used for discovery.
136    // For now, return the config values. The orchestrator will handle
137    // URL discovery and selection based on the config.
138    Ok(Some(config_values))
139}

141/// Prompt for URL using inquire (interactive mode).
142#[cfg(feature = "ui")]
143fn prompt_for_url() -> Result<String, CliExit> {
144    use inquire::validator::Validation;
145
146    Text::new("Enter the URL to scrape:")
147        .with_help_message("Example: https://example.com")
148        .with_validator(|input: &str| {
149            if input.is_empty() {
150                Err("URL cannot be empty".into())
151            } else if !input.starts_with("http://") && !input.starts_with("https://") {
152                Err("URL must start with http:// or https://".into())
153            } else {
154                Ok(Validation::Valid)
155            }
156        })
157        .prompt()
158        .map_err(|e| {
159            tracing::error!(error = %e, "Error prompting for URL");
160            CliExit::UsageError("interactive prompt failed".into())
161        })
162}

164#[tokio::main]
165pub async fn main() -> CliExit {
166    // Suppress OTel background thread panics during Tokio runtime shutdown.
167    // The BatchSpanProcessor and PeriodicReader threads panic when the reactor
168    // drops before they finish — this is a known SDK limitation, not our bug.
169    let default_hook = panic::take_hook();
170    panic::set_hook(Box::new(move |info| {
171        let thread_name = std::thread::current()
172            .name()
173            .unwrap_or("unknown")
174            .to_string();
175        if thread_name.starts_with("OpenTelemetry.") {
176            eprintln!("Warning: OTel background thread '{thread_name}' panicked during shutdown (safe to ignore)");
177            return;
178        }
179        default_hook(info);
180    }));
181
182    // tokio-console: usa 'cargo install tokio-console' y corre en otra terminal
183    // El runtime con tokio[unstable] ya expone el endpoint automaticamente
184    __main().await
185}

187async fn __main() -> CliExit {
188    // 1. Parse CLI arguments
189    let mut args = match parse_args() {
190        Ok(args) => args,
191        Err(exit) => return exit,
192    };
193
194    // 1b. Merge positional URL into --url (positional is syntactic sugar)
195    if let Some(pos_url) = args.positional_url.take() {
196        args.crawler.url = Some(pos_url);
197    }
198
199    // 2. Handle subcommands (completions)
200    if let Some(Commands::Completions { shell }) = args.subcommand {
201        return orchestrator::handle_completions(shell);
202    }
203
204    // 3. Unified TUI mode (if --tui flag is set)
205    match handle_tui_mode(args).await {
206        Ok(updated) => args = updated,
207        Err(exit) => return exit,
208    }
209
210    // 4. URL handling with interactive wizard
211    if let Err(exit) = resolve_url(&mut args).await {
212        return exit;
213    }
214
215    // 5. Load config file (graceful: missing file = defaults)
216    let config_path = resolve_config_path();
217    let config_defaults = ConfigDefaults::load(&config_path);
218
219    // 5b. Validate URL before conversion (CrawlOptions::from panics on invalid URL)
220    if let Some(ref url_str) = args.crawler.url {
221        if url::Url::parse(url_str).is_err() {
222            return CliExit::UsageError(format!("Invalid URL: {url_str}"));
223        }
224    }
225
226    // 6. Extract trace_file before args is moved into CrawlOptions
227    let trace_file = args.crawler.trace_file.take();
228
229    // 6b. Convert Args → CrawlOptions and apply config file defaults
230    let opts = CrawlOptions::from(args);
231    let opts = preflight::apply_config_defaults(opts, &config_defaults);
232
233    // 7. Initialize logging (stderr-only, respects quiet + NO_COLOR)
234    let no_color = is_no_color();
235    let log_level = resolve_log_level(opts.verbosity);
236    let file_trace_layer = build_file_trace_layer(trace_file);
237
238    // Initialize logging (stderr + optional JSONL file trace layer)
239    #[allow(clippy::let_unit_value)]
240    let _guard = init_logging_dual(log_level, opts.export.quiet, no_color, file_trace_layer);
241
242    // 8. Build optional engines and delegate to orchestrator
243    build_and_run(opts).await
244}

246/// Parse CLI arguments, translating clap's help/version/error cases into exits.
247fn parse_args() -> Result<Args, CliExit> {
248    match Args::try_parse() {
249        Ok(args) => Ok(args),
250        Err(e) => {
251            // clap returns DisplayHelp/DisplayVersion for --help/--version
252            // These are NOT errors — print and exit 0
253            if e.kind() == clap::error::ErrorKind::DisplayHelp
254                || e.kind() == clap::error::ErrorKind::DisplayVersion
255            {
256                e.print().ok();
257                return Err(CliExit::Success);
258            }
259            eprintln!("{e}");
260            Err(CliExit::UsageError("invalid arguments".into()))
261        },
262    }
263}

265/// Run the unified TUI (or reject it when the `ui` feature is off).
266///
267/// Returns the (possibly TUI-modified) args, or the `CliExit` to propagate —
268/// including `CliExit::Success` when the user cancels the TUI.
269#[cfg(feature = "ui")]
270async fn handle_tui_mode(mut args: Args) -> Result<Args, CliExit> {
271    if args.tui.tui {
272        // Run unified TUI: config form → URL selector → scraping
273        let tui_result = run_unified_tui().await;
274        match tui_result {
275            Ok(Some(config_values)) => {
276                // Apply TUI config values to args (overrides CLI values)
277                args = preflight::apply_tui_config_args(args, &config_values);
278                println!("Config applied from TUI.");
279            },
280            Ok(None) => {
281                println!("TUI cancelled.");
282                return Err(CliExit::Success);
283            },
284            Err(e) => return Err(e),
285        }
286    } else if args.tui.config_tui || args.tui.interactive {
287        // [DEPRECATED] Legacy flags — redirect to unified TUI
288        if args.tui.config_tui {
289            eprintln!(
290                "Warning: --config-tui is deprecated, use --tui instead. Will be removed in v0.6.0"
291            );
292        } else {
293            eprintln!("Warning: --interactive is deprecated, use --tui instead. Will be removed in v0.6.0");
294        }
295        let tui_result = run_unified_tui().await;
296        match tui_result {
297            Ok(Some(config_values)) => {
298                args = preflight::apply_tui_config_args(args, &config_values);
299            },
300            Ok(None) => return Err(CliExit::Success),
301            Err(e) => return Err(e),
302        }
303    }
304    Ok(args)
305}

319/// Ensure a URL is present, prompting interactively when stdin is a TTY.
320async fn resolve_url(args: &mut Args) -> Result<(), CliExit> {
321    // Batch mode reads URLs from stdin/file — --url is not required
322    let is_batch = args.export.batch || args.export.batch_file.is_some();
323
324    // If a URL is provided (or batch mode), nothing to resolve.
325    if args.crawler.url.is_some() || is_batch {
326        return Ok(());
327    }
328
329    // CI environment always requires --url
330    if is_ci() {
331        eprintln!("Error: --url is required for scraping (CI mode)");
332        return Err(CliExit::UsageError("--url is required".into()));
333    }
334
335    // Try interactive prompt only if stdin is a TTY
336    if stdin_is_tty() {
337        prompt_for_url_interactive(args)
338    } else {
339        // Not a TTY and no URL provided
340        eprintln!("Error: --url is required for scraping");
341        Err(CliExit::UsageError("--url is required".into()))
342    }
343}

345/// Prompt for a URL when stdin is a TTY and the `ui` feature is enabled.
346#[cfg(feature = "ui")]
347fn prompt_for_url_interactive(args: &mut Args) -> Result<(), CliExit> {
348    match prompt_for_url() {
349        Ok(url) => {
350            args.crawler.url = Some(url);
351            Ok(())
352        },
353        Err(_e) => {
354            // Prompt failed (e.g., non-interactive), fall through to error
355            eprintln!("Error: --url is required for scraping");
356            Err(CliExit::UsageError("--url is required".into()))
357        },
358    }
359}

376/// Map verbosity count to tracing levels (0=WARN, 1=INFO, 2=DEBUG, 3+=TRACE).
377fn resolve_log_level(verbosity: u8) -> &'static str {

418/// Build the AI semantic cleaner and its vault-search ports.
419#[cfg(feature = "ai")]
420async fn build_ai_cleaner(
421    opts: &CrawlOptions,
422) -> Result<
423    (
424        Option<Arc<dyn SemanticCleaner>>,
425        webfang_core::application::container::VaultAiPorts,
426    ),
427    CliExit,
428> {
429    if !opts.ai || opts.export.dry_run {
430        return Ok((
431            None,
432            webfang_core::application::container::VaultAiPorts::default(),
433        ));
434    }
435
436    // Resolve model variant: CLI flag takes precedence over AI_MODEL_ID env var
437    let model_variant = if opts.ai_config.model.is_empty() {
438        webfang_ai::AiModel::from_env_or_default()
439    } else {
440        match opts.ai_config.model.parse::<webfang_ai::AiModel>() {
441            Ok(variant) => variant,
442            Err(e) => {
443                return Err(CliExit::UsageError(format!(
444                    "Modelo AI inválido para --ai-model: {e}"
445                )));
446            },
447        }
448    };
449
450    let model_config = ModelConfig::default()
451        .with_model_variant(model_variant)
452        .with_relevance_threshold(opts.ai_config.threshold)
453        .map(|c| {
454            c.with_max_tokens(opts.ai_config.max_tokens)
455                .with_offline_mode(opts.ai_config.offline)
456        });
457
458    match model_config {
459        Ok(config) => match SemanticCleanerImpl::new(config).await {
460            Ok(cleaner) => {
461                // Share the cleaner's ONNX pool + tokenizer (#433) so the
462                // vault-search embedding adapter reuses the SAME model — one
463                // `resolve_model_assets` call, one `InferencePool` on `--ai`.
464                // Extracted before type-erasing the cleaner behind the trait.
465                let (pool, tokenizer) = cleaner.shared_inference();
466                let cleaner: Option<Arc<dyn SemanticCleaner>> = Some(Arc::new(cleaner));
467                let ports = build_vault_ports(pool, tokenizer).await;
468                Ok((cleaner, ports))
469            },
470            Err(e) => {
471                let msg = format!("No se pudo inicializar el limpiador semántico AI: {e}");
472                Err(match e {
473                    SemanticError::Download { .. } => CliExit::NetworkError(msg),
474                    _ => CliExit::ConfigError(msg),
475                })
476            },
477        },
478        Err(e) => Err(CliExit::ConfigError(format!(
479            "Configuración de umbral AI inválida: {e}"
480        ))),
481    }
482}

484/// Build optional engines and dispatch to the orchestrator.
485///
486/// The argument list depends on which optional features are compiled in, so each
487/// combination is spelled out explicitly.
488async fn build_and_run(opts: CrawlOptions) -> CliExit {
489    #[cfg(feature = "adaptive-selectors")]
490    let adaptive_engine = build_adaptive_engine(&opts);
491
492    #[cfg(feature = "ai")]
493    let (ai_cleaner, vault_ports) = match build_ai_cleaner(&opts).await {
494        Ok(v) => v,
495        Err(e) => return e,
496    };
497    #[cfg(not(feature = "ai"))]
498    let vault_ports = webfang_core::application::container::VaultAiPorts::default();
499
500    #[cfg(all(feature = "ai", feature = "adaptive-selectors"))]
501    {
502        orchestrator::run(opts, ai_cleaner, adaptive_engine, vault_ports).await
503    }
504    #[cfg(all(feature = "ai", not(feature = "adaptive-selectors")))]
505    {
506        orchestrator::run(opts, ai_cleaner, vault_ports).await
507    }
508    #[cfg(all(not(feature = "ai"), feature = "adaptive-selectors"))]
509    {
510        orchestrator::run(opts, adaptive_engine, vault_ports).await
511    }
512    #[cfg(all(not(feature = "ai"), not(feature = "adaptive-selectors")))]
513    {
514        orchestrator::run(opts, vault_ports).await
515    }
516}

518/// Assemble the vault-search AI ports (#433) from the cleaner's shared model.
519///
520/// Builds the ONNX embedding adapter + Markdown chunker (always under `ai`) from
521/// the semantic cleaner's shared inference pool + tokenizer — so the `--ai` path
522/// loads the ONNX model exactly once — plus the SQLite note repository (under
523/// `persistence`). Embedding + chunker assembly is infallible (the components are
524/// already valid); only the note repository can fail, and it degrades gracefully
525/// — the port stays `None` and the capability answers with an honest error rather
526/// than aborting the run.
527#[cfg(feature = "ai")]
528async fn build_vault_ports(
529    pool: Arc<webfang_ai::InferencePool>,
530    tokenizer: Arc<webfang_ai::MiniLmTokenizer>,
531) -> webfang_core::application::container::VaultAiPorts {
532    use webfang_core::application::container::VaultAiPorts;
533
534    let mut ports = VaultAiPorts::default();
535
536    // Assemble the embedding adapter from the cleaner's shared pool + tokenizer.
537    // Infallible — no model resolution happens here (that already happened once
538    // inside `SemanticCleanerImpl::new`), so there is nothing to degrade around.
539    let adapter = webfang_ai::EmbeddingAdapter::new(pool, tokenizer);
540    ports.embedding_port = Some(Arc::new(adapter));
541    ports.text_chunker = Some(Arc::new(webfang_ai::MarkdownChunker::new()));
542
543    #[cfg(feature = "persistence")]
544    {
545        use webfang_core::infrastructure::autotuning::{env_db_path, resolve_db_path};
546        use webfang_core::infrastructure::persistence::{
547            create_pool, setup_schema, SqliteVectorRepository,
548        };
549
550        let db_path = resolve_db_path(None, env_db_path());
551        match create_pool(&db_path, 4) {
552            Ok(db_pool) => match setup_schema(&db_pool).await {
553                Ok(()) => {
554                    ports.note_repository = Some(Arc::new(SqliteVectorRepository::new(db_pool)));
555                },
556                Err(e) => {
557                    tracing::warn!(
558                        error = %e,
559                        "note repository schema init failed, persistence disabled"
560                    );
561                },
562            },
563            Err(e) => {
564                tracing::warn!(
565                    error = %e,
566                    "note repository pool creation failed, persistence disabled"
567                );
568            },
569        }
570    }
571
572    ports
573}
