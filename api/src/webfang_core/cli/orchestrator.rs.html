38/// Create an unbounded channel and a `LiveProgressObserver` wired to it.
39///
40/// Returns `(observer, rx)` where:
41/// - `observer` wraps the `tx` side and can be passed to `scrape_urls`
42/// - `rx` is the receiving end for the TUI progress view
43///
44/// The caller must pass `rx` to `run_progress_view` on the main thread
45/// (crossterm TTY ownership requirement).
46pub fn prepare_progress_channel(
47    quiet: bool,
48) -> (
49    Box<dyn crate::domain::ports::ProgressObserver>,
50    tokio::sync::mpsc::UnboundedReceiver<crate::domain::entities::progress::ScrapeProgress>,
51) {
52    let (tx, rx) = tokio::sync::mpsc::unbounded_channel();
53    let observer =
54        Box::new(crate::application::progress_observer::LiveProgressObserver::new(Some(tx), quiet));
55    (observer, rx)
56}

58/// Main orchestration entry point.
59///
60/// Coordinates the full scraping pipeline:
61/// 1. URL discovery + config preparation
62/// 2. Scraping with progress
63/// 3. Export results
64/// 4. Report failures + exit code
65#[instrument(level = "info", skip(opts, ai_cleaner, adaptive_engine, vault_ports), fields(url = %opts.url))]
66pub async fn run(
67    opts: CrawlOptions,
68    #[cfg(feature = "ai")] ai_cleaner: Option<std::sync::Arc<dyn SemanticCleaner>>,
69    #[cfg(feature = "adaptive-selectors")] adaptive_engine: Option<
70        std::sync::Arc<AdaptiveSelectorEngine>,
71    >,
72    vault_ports: crate::application::container::VaultAiPorts,
73) -> CliExit {
74    if opts.export.dry_run {
75        println!("Dry-run: 1 URL(s) would be scraped:");
76        println!("  {}", opts.url);
77        return CliExit::Success;
78    }
79    if opts.batch.enabled {
80        // Batch mode uses the crawl Engine, which mints its own run-root
81        // identity per crawl — do not mint one here.
82        return run_batch(opts).await;
83    }
84
85    // Run-root correlation identity (#501): the whole operation owns ONE
86    // root; every page derives `.child()` from it. `#[instrument]` spans
87    // cannot see locals at creation, so declare it offline-visible via a
88    // structured event (lands in the JSONL `.fields`).
89    let root_correlation = domain::CorrelationId::new();
90    info!(
91        correlation_id = %root_correlation,
92        trace_id = %root_correlation.trace_id(),
93        "run identity"
94    );
95
96    let prepare = match prepare_phase(&opts).await {
97        Err(e) => return e,
98        Ok(p) => p,
99    };
100
101    let (urls_to_scrape, state_store) =
102        match apply_resume_mode(prepare.urls_to_scrape, &opts, opts.url.as_str()).await {
103            Ok(v) => v,
104            Err(e) => return e,
105        };
106
107    let elastic_ingestion = match build_elastic_ingestion(&opts, vault_ports).await {
108        Ok(v) => v,
109        Err(e) => return e,
110    };
111
112    #[cfg(not(feature = "ai"))]
113    if opts.elastic.output_vectors.is_some() && opts.ai {
114        warn!(
115            "--output-vectors specified with --clean-ai but AI feature is not compiled in. \
116             The output file will be created but no embedding vectors will be written. \
117             Rebuild with --features ai to generate embeddings."
118        );
119    }
120
121    // Create observer with stderr fallback (no channel) for non-TUI mode.
122    // For TUI mode, call `prepare_progress_channel()` from main.rs and pass
123    // the observer here instead.
124    let observer = Box::new(
125        crate::application::progress_observer::LiveProgressObserver::new(None, opts.export.quiet),
126    );
127    // Bridge the cfg-gated engine into an always-present reference option for
128    // the scrape phase: `None` when the feature is compiled out.
129    #[cfg(feature = "adaptive-selectors")]
130    let engine_ref = adaptive_engine.as_deref();
131    #[cfg(not(feature = "adaptive-selectors"))]
132    let engine_ref: Option<&AdaptiveSelectorEngine> = None;
133
134    let (results, failures) = match scrape_phase(
135        &urls_to_scrape,
136        &prepare.scraper_config,
137        &opts,
138        observer.as_ref(),
139        prepare
140            .shared_downloader
141            .as_deref()
142            .map(|d| d as &dyn crate::domain::ports::AssetDownloaderPort),
143        engine_ref,
144        &root_correlation,
145    )
146    .await
147    {
148        Ok(pair) => pair,
149        // Setup failures (unknown TLS profile, HTTP client build error) are
150        // config errors: surface the message and exit 78 rather than silently
151        // scraping with a wrong fingerprint.
152        Err(e) => return CliExit::ConfigError(e.to_string()),
153    };
154
155    if let Some(ref ingestion) = elastic_ingestion {
156        if let Err(e) = run_elastic_ingestion(ingestion, &results).await {
157            return CliExit::IoError(format!("Falló la ingesta de vectores: {e}"));
158        }
159    }
160
161    // Release the ingestion pipeline (wreq connection pool + Rayon threads)
162    // while the runtime is still active. Without this, the hyper pool
163    // background task or Rayon thread join can block runtime shutdown.
164    // See issue #335.
165    drop(elastic_ingestion);
166    tokio::task::yield_now().await;
167
168    if let Some(exit) = report_phase(&results, &failures, opts.verbosity) {
169        return exit;
170    }
171
172    #[cfg(feature = "ai")]
173    {
174        export_phase(&results, &opts, state_store.as_ref(), ai_cleaner).await
175    }
176    #[cfg(not(feature = "ai"))]
177    {
178        export_phase(&results, &opts, state_store.as_ref()).await
179    }
180}

182/// Export scraped results to files and run AI cleaning if requested.
183async fn export_phase(
184    results: &[domain::ScrapedContent],
185    opts: &CrawlOptions,
186    state_store: Option<&StateStore>,
187    #[cfg(feature = "ai")] ai_cleaner: Option<std::sync::Arc<dyn SemanticCleaner>>,
188) -> CliExit {
189    let output_dir = opts.export.output_dir.clone();
190
191    let obsidian_options = ObsidianOptions {
192        wiki_links: opts.export.obsidian_wiki_links,
193        relative_assets: opts.export.obsidian_relative_assets,
194        tags: opts.export.obsidian_tags.clone(),
195        rich_metadata: opts.export.obsidian_rich_metadata,
196        quick_save: opts.export.quick_save,
197        vault_path: opts.export.obsidian_vault.clone(),
198    };
199
200    let file_output_dir = if opts.export.quick_save {
201        let base = opts.export.obsidian_vault.as_deref().unwrap_or(&output_dir);
202        let inbox = base.join("_inbox");
203        if !inbox.exists() {
204            let _ = std::fs::create_dir_all(&inbox);
205        }
206        inbox
207    } else {
208        output_dir.clone()
209    };
210
211    save_files(
212        results,
213        &file_output_dir,
214        &opts.export.output_format,
215        &obsidian_options,
216    );
217
218    let export_config = ExportConfig {
219        results,
220        output_dir,
221        format: opts.export.output_format,
222        export_format: opts.export.export_format,
223        clean_ai: opts.ai,
224        quick_save: opts.export.quick_save,
225        vault_path: opts.export.obsidian_vault.as_ref(),
226        obsidian_options,
227        state_store,
228        resume: opts.crawl.resume,
229        ai_threshold: opts.ai_config.threshold,
230        ai_max_tokens: opts.ai_config.max_tokens,
231        ai_offline: opts.ai_config.offline,
232        ai_model: opts.ai_config.model.clone(),
233    };
234
235    #[cfg(feature = "ai")]
236    let export_result = run_export(export_config, ai_cleaner).await;
237    #[cfg(not(feature = "ai"))]
238    let export_result = run_export(export_config).await;
239
240    match export_result {
241        Ok(processed_urls) => {
242            info!("Export completed for {} URLs", processed_urls.len());
243            CliExit::Success
244        },
245        Err(e) => {
246            error!(error = ?e, "Export failed");
247            e
248        },
249    }
250}

252/// Prepare scraper config and discover URLs.
253///
254/// Returns the initial `ScraperConfig` (before asset/download wiring) and
255/// the list of URLs to scrape.  On discovery failure, returns the
256/// appropriate `CliExit` error.
257async fn prepare_phase(opts: &CrawlOptions) -> Result<PrepareResult, CliExit> {
258    let urls_to_scrape = if opts.crawl.single_page {
259        plan_urls(true, false, opts.url.clone(), Vec::new())
260    } else {
261        // Honor `--h2-profile` for URL discovery (#312): an unknown profile is a
262        // config error (exit 78), consistent with the scrape and batch phases.
263        let tls_emulation = HttpClientConfig::profile_from_name(&opts.network.h2_profile)
264            .map_err(|e| CliExit::ConfigError(e.to_string()))?;
265
266        let mut crawler_config = CrawlerConfig::builder(opts.url.clone())
267            .max_pages(opts.crawl.max_pages)
268            .max_depth(opts.crawl.max_depth)
269            .include_patterns(opts.crawl.include_patterns.clone())
270            .exclude_patterns(opts.crawl.exclude_patterns.clone())
271            .ignore_robots(opts.crawl.ignore_robots)
272            .use_sitemap(opts.crawl.use_sitemap)
273            .timeout_secs(opts.network.timeout_secs)
274            .tls_emulation(tls_emulation);
275        if let Some(ref sitemap_url) = opts.crawl.sitemap_url {
276            crawler_config = crawler_config.sitemap_url(sitemap_url);
277        }
278        let crawler_config = crawler_config.build();
279
280        let discovered_urls = match discover_urls(&crawler_config, opts).await {
281            Err(e) => {
282                return Err(CliExit::NetworkError(format!("URL discovery failed: {e}")));
283            },
284            // Exit 2 only when the sitemap is the source of truth. In DOM mode an
285            // empty discovery is not fatal: `plan_urls` injects the seed URL so
286            // the site itself is still scraped (#488). The message is always the
287            // sitemap one: this guard no longer fires in DOM mode (#495 made the
288            // message context-aware; the link-extraction branch is now unreachable
289            // here because an empty DOM discovery flows to `plan_urls`).
290            Ok(urls) if urls.is_empty() && opts.crawl.use_sitemap => {
291                return Err(CliExit::EmptyDiscovery(
292                    "No URLs discovered from sitemaps".into(),
293                ));
294            },
295            Ok(urls) => urls,
296        };
297
298        plan_urls(
299            false,
300            opts.crawl.use_sitemap,
301            opts.url.clone(),
302            discovered_urls,
303        )
304    };
305
306    let mut scraper_config = ScraperConfig::default()
307        .with_output_dir(opts.export.output_dir.clone())
308        .with_scraper_concurrency(opts.network.concurrency.resolve())
309        .with_max_pages(opts.crawl.max_pages)
310        .with_selector(opts.crawl.selector.clone())
311        .with_ignore_waf(opts.crawl.ignore_waf);
312
313    if opts.network.download_images {
314        scraper_config = scraper_config.with_images();
315    }
316    if opts.network.download_documents {
317        scraper_config = scraper_config.with_documents();
318    }
319
320    // Wire asset download config from CLI args
321    scraper_config =
322        scraper_config.with_asset_h2_profile(parse_asset_h2_profile(&opts.network.h2_profile));
323    scraper_config =
324        scraper_config.with_asset_include_patterns(opts.crawl.include_patterns.clone());
325    scraper_config =
326        scraper_config.with_asset_exclude_patterns(opts.crawl.exclude_patterns.clone());
327    scraper_config = scraper_config.with_asset_naming(parse_asset_naming(&opts.asset_naming));
328    scraper_config = scraper_config.with_download_concurrency(opts.download_concurrency);
329
330    // Create shared Downloader once for connection pooling across all page scrapes.
331    let shared_downloader = if scraper_config.has_downloads() {
332        match crate::adapters::downloader::Downloader::new(scraper_config.to_download_config()) {
333            Ok(dl) => Some(std::sync::Arc::new(dl)),
334            Err(e) => {
335                return Err(CliExit::IoError(format!(
336                    "No se pudo crear el descargador de assets: {e}"
337                )));
338            },
339        }
340    } else {
341        None
342    };
343
344    Ok(PrepareResult {
345        urls_to_scrape,
346        scraper_config,
347        shared_downloader,
348    })
349}

354    shared_downloader: Option<std::sync::Arc<crate::adapters::downloader::Downloader>>,
355}

391/// Build the user-facing failure line for a single URL.
392///
393/// At `verbosity` 0 only the top-level `Display` message is shown, which keeps
394/// network errors (DNS, connect) to a single readable line. At `verbosity` 1+
395/// the full root-cause chain is preserved via `Error::source()` (D4), appending
396/// each cause as `← cause`.
397fn format_failure(url: &str, error: &crate::error::ScraperError, verbosity: u8) -> String {
398    let mut chain = error.to_string();
399    if verbosity > 0 {
400        let mut src = std::error::Error::source(error);
401        while let Some(cause) = src {
402            chain.push_str(&format!("  ← {cause}"));
403            src = cause.source();
404        }
405    }
406    format!("Failed to scrape {url}: {chain}")
407}

409/// Report failures and determine the exit code.
410///
411/// Returns `None` if all pages scraped successfully (caller proceeds to export).
412/// At `verbosity` 0 only the top-level error message is shown; at 1+ the full
413/// root-cause chain is appended (see [`format_failure`]).
414///
415/// Severity routing (#537): when every URL failed AND at least one failure
416/// classifies as [`crate::error::ErrorClass::InternalFatal`], the exit code is
417/// `CliExit::ScraperFailure` (3) rather than `NetworkError` (69) — an internal
418/// bug must not masquerade as a transient network outage. Purely
419/// transient/permanent all-fail runs keep exit 69. The partial-success case
420/// always reports `PartialSuccess` (69) regardless of failure severity: some
421/// content was scraped, which is the dominant signal.
422fn report_phase(
423    results: &[domain::ScrapedContent],
424    failures: &[(String, crate::error::ScraperError)],
425    verbosity: u8,
426) -> Option<CliExit> {
427    for (url, error) in failures {
428        eprintln!("{}", format_failure(url, error, verbosity));
429    }
430
431    if !failures.is_empty() && !results.is_empty() {
432        return Some(CliExit::PartialSuccess {
433            success: results.len(),
434            failed: failures.len(),
435        });
436    }
437
438    if results.is_empty() {
439        if let Some(exit) = scraper_failure_for_internal_fatal(failures) {
440            return Some(exit);
441        }
442        eprintln!("No pages were successfully scraped");
443        return Some(CliExit::NetworkError(
444            "No pages were successfully scraped".into(),
445        ));
446    }
447
448    info!("Successfully scraped {} pages", results.len());
449    None
450}

452/// Fold an all-failed error set into a severity-aware exit (#537).
453///
454/// Returns `Some(CliExit::ScraperFailure(..))` when at least one failure
455/// classifies as [`crate::error::ErrorClass::InternalFatal`], with a message
456/// that calls out the internal-fatal count. Returns `None` when every failure
457/// is transient/permanent — the caller then keeps its historical
458/// `CliExit::NetworkError` arm.
459fn scraper_failure_for_internal_fatal(
460    failures: &[(String, crate::error::ScraperError)],
461) -> Option<CliExit> {
462    let internal_fatal = failures
463        .iter()
464        .filter(|(_, e)| e.classify() == crate::error::ErrorClass::InternalFatal)
465        .count();
466    (internal_fatal > 0).then(|| {
467        CliExit::ScraperFailure(format!(
468            "Scraper failure: {internal_fatal} internal error(s) out of {} URLs",
469            failures.len()
470        ))
471    })
472}

474/// Run batch processing mode: crawl multiple URLs from stdin or file
475async fn run_batch(opts: CrawlOptions) -> CliExit {
476    // Resolve the TLS/H2 fingerprint once so the batch crawl engine honors
477    // `--h2-profile` (#312). An unknown profile is a config error (exit 78),
478    // matching the scrape phase — never silently crawl with a wrong fingerprint.
479    let tls_emulation = match resolve_batch_tls_emulation(&opts) {
480        Ok(profile) => profile,
481        Err(e) => return e,
482    };
483
484    let crawler_config = build_batch_crawler_config(&opts, tls_emulation);
485
486    let manager = match load_batch_manager(&opts, crawler_config).await {
487        Ok(m) => m,
488        Err(e) => return e,
489    };
490
491    if manager.job_count() == 0 {
492        error!("No URLs provided for batch processing");
493        return CliExit::UsageError("No URLs provided".into());
494    }
495
496    info!(
497        "Starting batch processing: {} jobs, concurrency={}",
498        manager.job_count(),
499        opts.batch.concurrency
500    );
501
502    let summary = manager.process_all_summary().await;
503
504    log_batch_summary(&summary);
505
506    batch_exit_code(summary.succeeded, summary.failed, &summary.errors)
507}
508
509/// Print the batch completion summary and log each failed URL.
510fn log_batch_summary(summary: &BatchManagerSummary) {
511    println!(
512        "Batch complete: {}/{} succeeded, {} failed",
513        summary.succeeded, summary.total_urls, summary.failed
514    );
515
516    for (url, err) in &summary.errors {
517        error!(%url, error = %err, "Batch URL failed");
518    }
519}

521/// Resolve the TLS/H2 fingerprint for the batch crawl engine.
522///
523/// An unknown profile is a config error (exit 78), matching the scrape phase —
524/// never silently crawl with a wrong fingerprint (#312).
525fn resolve_batch_tls_emulation(opts: &CrawlOptions) -> Result<wreq_util::Profile, CliExit> {
526    HttpClientConfig::profile_from_name(&opts.network.h2_profile)
527        .map_err(|e| CliExit::ConfigError(e.to_string()))
528}

530/// Build the crawler config for the batch engine, honoring `--h2-profile`.
531fn build_batch_crawler_config(
532    opts: &CrawlOptions,
533    tls_emulation: wreq_util::Profile,
534) -> CrawlerConfig {
535    let mut crawler_config = CrawlerConfig::builder(opts.url.clone())
536        .max_pages(opts.crawl.max_pages)
537        .max_depth(opts.crawl.max_depth)
538        .include_patterns(opts.crawl.include_patterns.clone())
539        .exclude_patterns(opts.crawl.exclude_patterns.clone())
540        .ignore_robots(opts.crawl.ignore_robots)
541        .use_sitemap(opts.crawl.use_sitemap)
542        .timeout_secs(opts.network.timeout_secs)
543        .tls_emulation(tls_emulation);
544    if let Some(ref sitemap_url) = opts.crawl.sitemap_url {
545        crawler_config = crawler_config.sitemap_url(sitemap_url);
546    }
547    crawler_config.build()
548}

550/// Load the batch manager from a file or stdin.
551async fn load_batch_manager(
552    opts: &CrawlOptions,
553    crawler_config: CrawlerConfig,
554) -> Result<BatchManager, CliExit> {
555    if let Some(ref path) = opts.batch.batch_file {
556        info!("Reading URLs from file: {}", path.display());
557        BatchManager::from_file(path, crawler_config, opts.batch.concurrency).map_err(|e| {
558            error!(error = %e, "Failed to read URLs");
559            CliExit::NetworkError(format!("Failed to read URLs: {e}"))
560        })
561    } else {
562        info!("Reading URLs from stdin");
563        load_batch_manager_from_stdin(crawler_config, opts.batch.concurrency).await
564    }
565}

567/// Read batch URLs from stdin on a blocking thread.
568async fn load_batch_manager_from_stdin(
569    crawler_config: CrawlerConfig,
570    concurrency: usize,
571) -> Result<BatchManager, CliExit> {
572    // spawn_blocking: stdin read is blocking I/O that must not run on the
573    // Tokio async runtime thread pool — it would block other tasks.
574    match tokio::task::spawn_blocking(move || BatchManager::from_stdin(crawler_config, concurrency))
575        .await
576    {
577        Ok(result) => result.map_err(|e| {
578            error!(error = %e, "Failed to read URLs");
579            CliExit::NetworkError(format!("Failed to read URLs: {e}"))
580        }),
581        Err(join_err) => {
582            error!(error = %join_err, "stdin read task panicked");
583            Err(CliExit::NetworkError(format!(
584                "Failed to read URLs: {join_err}"
585            )))
586        },
587    }
588}

590/// Determine the CLI exit code from batch scrape results.
591///
592/// Severity routing (#537): an all-failed batch containing at least one
593/// [`crate::error::ErrorClass::InternalFatal`] error yields
594/// `CliExit::ScraperFailure` (3); purely transient/permanent all-fail runs
595/// keep `CliExit::NetworkError` (69). Partial success remains
596/// `CliExit::PartialSuccess` (69) regardless of failure severity — some
597/// content was scraped, which is the dominant signal.
598fn batch_exit_code(
599    succeeded: usize,
600    failed: usize,
601    errors: &[(String, crate::error::ScraperError)],
602) -> CliExit {
603    if failed > 0 && succeeded == 0 {
604        if let Some(exit) = scraper_failure_for_internal_fatal(errors) {
605            return exit;
606        }
607        CliExit::NetworkError("All batch URLs failed".into())
608    } else if failed > 0 {
609        CliExit::PartialSuccess {
610            success: succeeded,
611            failed,
612        }
613    } else {
614        CliExit::Success
615    }
616}

618fn plan_urls(
619    single_page: bool,
620    use_sitemap: bool,
621    seed_url: url::Url,
622    discovered_urls: Vec<url::Url>,
623) -> Vec<url::Url> {
624    if single_page {
625        vec![seed_url]
626    } else if use_sitemap {
627        // Sitemap is the source of truth — do not inject the seed URL.
628        discovered_urls
629    } else {
630        // DOM discovery: always include the seed URL so it gets crawled
631        // even when link extraction only returns child URLs.
632        let mut urls = discovered_urls;
633        if !urls.contains(&seed_url) {
634            urls.insert(0, seed_url);
635        }
636        urls
637    }
638}

640/// Parse H2/TLS profile from CLI string for the asset download path.
641///
642/// Delegates to the domain resolver
643/// [`crate::domain::profile::profile_from_name`], which accepts the full
644/// [`wreq_util::Profile`] catalog. Unlike the strict page-fetch path, the asset
645/// path is best-effort: an unknown name logs a warning and falls back to
646/// `Chrome145` rather than failing the run.
647fn parse_asset_h2_profile(s: &str) -> wreq_util::Profile {
648    crate::domain::profile::profile_from_name(s).unwrap_or_else(|| {
649        tracing::warn!(
650            "Unknown asset H2 profile '{s}', falling back to Chrome145. \
651             Run `cargo doc -p wreq-util` to see all available profiles."
652        );
653        wreq_util::Profile::Chrome145
654    })
655}

947    fn orchestrator_no_hardcoded_ai_literals() {
948        // Verify orchestrator source does not contain hardcoded AI config literals
949        // at the ExportConfig construction site (the `run` function, NOT test code).
950        let src = include_str!("orchestrator.rs");
951        let lines: Vec<&str> = src.lines().collect();
952        // Find the ExportConfig construction block — it starts with "let export_config = ExportConfig"
953        let mut in_export_config = false;
954        for (i, line) in lines.iter().enumerate() {
955            let line_num = i + 1;
956            if line.contains("let export_config = ExportConfig") {
957                in_export_config = true;
958            }
959            if in_export_config && line.contains('}') && !line.contains("//") {
960                break; // end of ExportConfig struct literal
961            }
962            if in_export_config {
963                // Inside ExportConfig literal — no hardcoded AI values allowed
964                if line.contains("ai_threshold:") && line.contains("0.3") {
965                    panic!(
966                        "Line {line_num}: hardcoded literal 0.3 found — should use opts.ai_config.threshold"
967                    );
968                }
969                if line.contains("ai_max_tokens:") && line.contains("32768") {
970                    panic!(
971                        "Line {line_num}: hardcoded literal 32768 found — should use opts.ai_config.max_tokens"
972                    );
973                }
974                if line.contains("ai_offline:") && line.contains("false") {
975                    panic!(
976                        "Line {line_num}: hardcoded literal false found — should use opts.ai_config.offline"
977                    );
978                }
979            }
980        }
981        assert!(
982            in_export_config,
983            "ExportConfig construction not found in source"
984        );
