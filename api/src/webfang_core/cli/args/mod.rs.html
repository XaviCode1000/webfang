20/// CLI Arguments for the webfang binary.
21///
22/// Parsed using `clap` with derive macros.
23///
24/// # Examples
25///
26/// ```no_run
27/// use webfang_core::Args;
28/// use clap::Parser;
29///
30/// let args = Args::parse_from([
31///     "webfang",
32///     "--url", "https://example.com",
33///     "--output", "./output",
34///     "--export-format", "jsonl",
35///     "--resume",
36/// ]);
37///
38/// assert_eq!(args.crawler.url.as_deref(), Some("https://example.com"));
39/// ```
40#[derive(Parser, Debug, Default)]
41#[command(name = "webfang", version)]
42#[command(
43    about = "High-performance web scraper with WAF evasion and AI-powered content cleaning",
44    after_help = "EXIT CODES:\n  0    Success\n  2    No URLs discovered\n  3    All scrapers failed\n  64   Bad CLI arguments (usage error)\n  69   WAF block or network error\n  74   I/O error\n  76   Protocol error\n  78   Configuration error\n\nEXAMPLES:\n  webfang https://example.com\n  webfang -u https://example.com\n  webfang -u https://example.com --ai\n  webfang -u https://example.com -f jsonl\n  webfang -u https://example.com -v\n  webfang -u https://example.com -vv  # DEBUG\n  webfang --url-list urls.txt --resume"
45)]
46#[command(subcommand_negates_reqs = true)]
47pub struct Args {
48    /// Subcommands
49    #[command(subcommand)]
50    pub subcommand: Option<Commands>,
51
52    /// URL to scrape (positional shorthand — equivalent to --url)

56    /// Crawler and discovery configuration.

59
60    /// Export format and output configuration.

64    /// Obsidian vault integration settings.

68    /// AI-powered semantic cleaning settings.

77/// Subcommands.
78#[derive(Debug, clap::Subcommand)]
79pub enum Commands {
80    /// Generate shell completion scripts
81    Completions {
82        /// Shell to generate completions for
83        #[arg(value_enum)]

85    },

88/// Shell type for completions.
89#[derive(Debug, Clone, Copy, PartialEq, Eq, clap::ValueEnum)]
90pub enum Shell {
91    /// Bash shell completions.
92    Bash,
93    /// Elvish shell completions.
94    Elvish,
95    /// Fish shell completions.
96    Fish,
97    /// PowerShell completions.
98    PowerShell,
99    /// Zsh shell completions.
100    Zsh,
101}
102
103impl From<Shell> for clap_complete::Shell {
104    fn from(s: Shell) -> Self {
105        match s {
106            Shell::Bash => clap_complete::Shell::Bash,
107            Shell::Elvish => clap_complete::Shell::Elvish,
108            Shell::Fish => clap_complete::Shell::Fish,
109            Shell::PowerShell => clap_complete::Shell::PowerShell,
110            Shell::Zsh => clap_complete::Shell::Zsh,
111        }
112    }
113}

115impl Args {
116    /// Build [`ElasticOverrides`] (PR5) from the elastic-ingestion CLI flags.
117    ///
118    /// `--ram-budget` is parsed via [`parse_ram_bytes`] so it accepts suffixed
119    /// values (`8GB`, `2048MB`, plain bytes). The result feeds
120    /// [`ElasticConfig::resolve`] → Rayon pool size, byte-weighted semaphore,
121    /// and SQLite path.
122    ///
123    /// [`ElasticConfig::resolve`]: crate::infrastructure::autotuning::ElasticConfig::resolve
124    /// [`parse_ram_bytes`]: crate::infrastructure::autotuning::parse_ram_bytes
125    /// [`ElasticOverrides`]: crate::infrastructure::autotuning::ElasticOverrides
126    #[must_use]
127    pub fn elastic_overrides(&self) -> crate::infrastructure::autotuning::ElasticOverrides {
128        use crate::infrastructure::autotuning::{parse_ram_bytes, ElasticOverrides};
129        ElasticOverrides {
130            cpu_cores: self.export.cpu_cores,
131            ram_budget_bytes: self.export.ram_budget.as_deref().and_then(parse_ram_bytes),
132            max_resource_bytes: None,
133            db_path: self.export.db_path.clone(),
134        }
135    }
136}

142impl From<Args> for crate::application::crawl_options::CrawlOptions {
143    /// Convert CLI arguments into structured [`crate::application::crawl_options::CrawlOptions`].
144    ///
145    /// This is an owned, lossless conversion — every field in `Args` maps
146    /// to exactly one field in `CrawlOptions`. The `url` field is parsed
147    /// from `Option<String>` into `Url` (panics if invalid; CLI validation
148    /// guarantees validity before this point).
149    fn from(args: Args) -> Self {
150        use crate::application::crawl_options::{
151            CrawlLimits, ExportOptions, IngestionTuning, NetworkOptions,
152        };
153
154        // `From::from` returns `Self`, so the parse error cannot be propagated;
155        // CLI validation guarantees the URL is valid before this point.
156        #[allow(clippy::expect_used)]
157        let url = url::Url::parse(args.crawler.url.as_deref().unwrap_or("https://example.com"))
158            .expect("URL must be valid — CLI validation ensures this");
159
160        let overrides = args.elastic_overrides();
161        let ai_config = build_ai_config(&args);
162
163        Self {
164            url,
165            verbosity: args.crawler.verbose,
166            quiet: args.crawler.quiet,
167            ai: args.crawler.clean_ai,
168            adaptive_selectors: args.crawler.adaptive_selectors,
169            crawl: CrawlLimits {
170                selector: args.crawler.selector,
171                max_depth: args.crawler.max_depth,
172                max_pages: args.crawler.max_pages,
173                single_page: args.crawler.single_page,
174                include_patterns: args.crawler.include_patterns,
175                exclude_patterns: args.crawler.exclude_patterns,
176                interactive: args.tui.interactive,
177                resume: args.crawler.resume,
178                state_dir: args.crawler.state_dir,
179                use_sitemap: args.crawler.use_sitemap,
180                sitemap_url: args.crawler.sitemap_url,
181                checkpoint_interval: args.crawler.checkpoint_interval,
182                no_checkpoint: args.crawler.no_checkpoint,
183                ignore_robots: args.crawler.ignore_robots,
184                ignore_waf: args.crawler.ignore_waf,
185                no_session_health: args.crawler.no_session_health,
186                autoscale_enabled: args.crawler.autoscale,
187            },
188            network: NetworkOptions {
189                user_agent: args.crawler.user_agent,
190                accept_language: args.crawler.accept_language,
191                concurrency: args.crawler.concurrency,
192                delay_ms: args.crawler.delay_ms,
193                timeout_secs: args.crawler.timeout_secs,
194                max_retries: args.crawler.max_retries,
195                backoff_base_ms: args.crawler.backoff_base_ms,
196                backoff_max_ms: args.crawler.backoff_max_ms,
197                download_images: args.crawler.download_images || args.crawler.download_assets,
198                download_documents: args.crawler.download_documents || args.crawler.download_assets,
199                h2_profile: args.crawler.h2_profile,
200                js_strategy: args.crawler.js_strategy,
201                obscura_binary: args.crawler.obscura_binary,
202            },
203            export: ExportOptions {
204                output_format: args.export.format,
205                export_format: args.export.export_format,
206                output_dir: args.export.output,
207                dry_run: args.crawler.dry_run,
208                quiet: args.crawler.quiet,
209                obsidian_vault: args.obsidian.vault,
210                obsidian_rich_metadata: args.obsidian.obsidian_rich_metadata,
211                obsidian_tags: args.obsidian.obsidian_tags.unwrap_or_default(),
212                obsidian_wiki_links: args.obsidian.obsidian_wiki_links,
213                obsidian_relative_assets: args.obsidian.obsidian_relative_assets,
214                quick_save: args.obsidian.quick_save,
215            },
216            elastic: IngestionTuning {
217                enabled: args.export.elastic,
218                cpu_cores: overrides.cpu_cores,
219                ram_budget_bytes: overrides.ram_budget_bytes,
220                db_path: overrides.db_path,
221                max_resource_bytes: overrides.max_resource_bytes,
222                output_vectors: args.export.output_vectors.clone(),
223            },
224            pipeline_enabled: args.export.pipeline,
225            pipeline_output_format: args.export.pipeline_output,
226            batch: crate::application::crawl_options::BatchOptions {
227                enabled: args.export.batch || args.export.batch_file.is_some(),
228                batch_file: args.export.batch_file,
229                concurrency: args.export.batch_concurrency,
230            },
231            asset_naming: args.crawler.asset_naming,
232            download_concurrency: args.crawler.download_concurrency,
233            ai_config,
234        }
235    }
236}

258    /// Remove poisoned env vars once before any arg-parsing test runs.
259    /// CI bug-discovery sets WEBFANG_*=POISON / AI_MODEL_ID=POISON which
260    /// clap reads via `env = "..."` attributes, breaking hermeticity.
261    static ENV_GUARD: std::sync::Once = std::sync::Once::new();
262    fn clean_env() {
263        ENV_GUARD.call_once(|| {
264            let poisoned: Vec<String> = std::env::vars()
265                .filter(|(k, _)| k.starts_with("WEBFANG_") || k == "AI_MODEL_ID")
266                .map(|(k, _)| k)
267                .collect();
268            for key in poisoned {
269                std::env::remove_var(&key);
270            }
271        });
272    }
