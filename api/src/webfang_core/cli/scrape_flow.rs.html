29/// Apply resume mode filtering.
30///
31/// # Errors
32///
33/// Returns `CliExit::IoError` when `--resume` is active and the state store
34/// cannot be created. Without a working store the crawl would silently
35/// re-scrape every URL, defeating the purpose of resume mode.
36pub async fn apply_resume_mode(
37    urls_to_scrape: Vec<Url>,
38    opts: &CrawlOptions,
39    target_url: &str,
40) -> Result<(Vec<Url>, Option<StateStore>), CliExit> {
41    let state_store: Option<StateStore> = if opts.crawl.resume {
42        info!("Resume mode enabled - tracking processed URLs");
43        let state_dir = resolve_state_dir(opts);
44
45        let domain = export_factory::domain_from_url(target_url);
46        info!("State store domain: {}", domain);
47        match export_factory::create_state_store(state_dir, &domain) {
48            Ok(store) => Some(store),
49            // LCOV_EXCL_START defensive: state-store-creation — store creation failure with --resume is an environment invariant break
50            Err(e) => {
51                tracing::error!(error = %e, "state store creation failed with --resume active");
52                return Err(CliExit::IoError(format!(
53                    "No se pudo crear el almacén de estado para --resume: {e}"
54                )));
55            },
56            // LCOV_EXCL_STOP
57        }
58    } else {
59        None
60    };
61
62    let filtered = if opts.crawl.resume {
63        match state_store.as_ref() {
64            Some(store) => filter_processed_urls(urls_to_scrape, store),
65            None => urls_to_scrape,
66        }
67    } else {
68        urls_to_scrape
69    };
70
71    Ok((filtered, state_store))
72}

74/// Resolve the resume state directory, defaulting to the XDG cache path.
75fn resolve_state_dir(opts: &CrawlOptions) -> PathBuf {
76    opts.crawl.state_dir.clone().unwrap_or_else(|| {
77        let cache_base = std::env::var("XDG_CACHE_HOME")
78            .map(PathBuf::from)
79            .unwrap_or_else(|_| {
80                dirs::home_dir()
81                    .unwrap_or_else(|| PathBuf::from("."))
82                    .join(".cache")
83            });
84        cache_base.join("webfang").join("state")
85    })
86}

88/// Drop URLs already processed by a prior run, degrading gracefully on error.
89fn filter_processed_urls(urls_to_scrape: Vec<Url>, store: &StateStore) -> Vec<Url> {
90    match store.load_or_default() {
91        Ok(state) => {
92            let original_count = urls_to_scrape.len();
93            let filtered: Vec<_> = urls_to_scrape
94                .into_iter()
95                .filter(|url| {
96                    let should_skip = store.is_processed(&state, url.as_str());
97                    if should_skip {
98                        info!("Skipping already processed: {}", url);
99                    }
100                    !should_skip
101                })
102                .collect();
103
104            let skipped_count = original_count - filtered.len();
105            info!(
106                "Resume mode: {} URLs already processed, {} new URLs to scrape",
107                skipped_count,
108                filtered.len()
109            );
110
111            filtered
112        },
113        Err(e) => {
114            warn!("Failed to load state: {}", e);
115            urls_to_scrape
116        },
117    }
118}

120/// Scrape all URLs, reporting progress via the provided observer.
121///
122/// Correlation contract (#501): `root_correlation` is the run-root identity
123/// owned by the orchestrator; each page derives `.child()` from it — one
124/// shared `trace_id` for the whole run, a fresh `span_id` per page.
125///
126/// The observer handles quiet/channel logic internally — callers pass
127/// `&NoopObserver` for dry-run or `&LiveProgressObserver` for live output.
128///
129/// # Errors
130///
131/// Returns [`crate::error::ScraperError`] if the configured H2/TLS profile name
132/// (`opts.network.h2_profile`) is not recognized, or if the fetch router's HTTP
133/// client cannot be built. Both are setup failures that abort the whole batch
134/// before any URL is scraped.
135pub async fn scrape_urls(
136    urls: &[Url],
137    scraper_config: &ScraperConfig,
138    opts: &CrawlOptions,
139    observer: &dyn ProgressObserver,
140    downloader: Option<&dyn crate::domain::ports::AssetDownloaderPort>,
141    engine: Option<&AdaptiveSelectorEngine>,
142    root_correlation: &CorrelationId,
143) -> Result<
144    (
145        Vec<ScrapedContent>,
146        Vec<(String, crate::error::ScraperError)>,
147    ),
148    crate::error::ScraperError,
149> {
150    // Build the fetch router from the configured JS strategy.
151    let http_config = build_http_client_config(opts)?;
152    let cookie_bridge = std::sync::Arc::new(std::sync::RwLock::new(CookieBridge::new()));
153    let router = build_fetch_router(
154        &opts.network.js_strategy,
155        http_config.timeout_secs,
156        http_config.tls_emulation,
157        cookie_bridge,
158        opts.crawl.ignore_waf,
159        // #503: move the operator's --user-agent into the wreq layer instead
160        // of dropping it on the floor. `http_config` stays usable below —
161        // only this field moves out.
162        http_config.user_agent,
163        // #509: the scrape flow has no shutdown policy yet — pass an inert
164        // token so Full-strategy governor waits keep pre-#509 behavior.
165        tokio_util::sync::CancellationToken::new(),
166    )?;
167
168    let _total_urls = urls.len();
169
170    // Robots.txt fetcher — shares the batch's TLS fingerprint so the robots.txt
171    // request is indistinguishable from a page fetch (#337). Shared across all
172    // URLs in this batch.
173    let robots_fetcher = RobotsFetcher::new(http_config.tls_emulation, http_config.timeout_secs)?;
174
175    // Apply max_pages limit if configured
176    let urls_to_process = apply_max_pages_limit(urls, scraper_config);
177
178    let processing_count = urls_to_process.len();
179    let mut results = Vec::with_capacity(processing_count);
180    let mut failures: Vec<(String, crate::error::ScraperError)> = Vec::new();
181
182    let ctx = ScrapeContext {
183        router: &router,
184        scraper_config,
185        downloader,
186        engine,
187        robots_fetcher: &robots_fetcher,
188    };
189
190    for url in urls_to_process {
191        // Per-page identity: child of the run root — shared trace_id, fresh
192        // span_id (#501).
193        let page_correlation = root_correlation.child();
194
195        match scrape_one_url(&url, &ctx, opts, observer, &page_correlation).await {
196            Ok(Some(content)) => results.push(content),
197            // Robots.txt blocked — skipped, not a failure.
198            Ok(None) => {},
199            Err(e) => {
200                let url_str = url.as_str().to_string();
201                failures.push((url_str, e));
202            },
203        }
204    }
205
206    let total_successful = results.len();
207    let total_failed = failures.len();
208    observer
209        .on_finished(processing_count, total_successful, total_failed)
210        .await;
211
212    Ok((results, failures))
213}

215/// Shared per-URL dependencies for a single scrape, bundled to keep the
216/// per-page helper's signature small.
217struct ScrapeContext<'a> {
218    router: &'a dyn Downloader,
219    scraper_config: &'a ScraperConfig,
220    downloader: Option<&'a dyn crate::domain::ports::AssetDownloaderPort>,
221    engine: Option<&'a AdaptiveSelectorEngine>,
222    robots_fetcher: &'a RobotsFetcher,
223}
224
225/// Apply the `max_pages` cap to the URL list when configured.
226fn apply_max_pages_limit(urls: &[Url], scraper_config: &ScraperConfig) -> Vec<Url> {
227    if let Some(max_pages) = scraper_config.max_pages {
228        let limited: Vec<_> = urls.iter().take(max_pages).cloned().collect();
229        if limited.len() < urls.len() {
230            tracing::info!(
231                "Limiting to {} pages (max_pages={}), skipping {} URLs",
232                limited.len(),
233                max_pages,
234                urls.len() - limited.len()
235            );
236        }
237        limited
238    } else {
239        urls.to_vec()
240    }
241}

243/// Scrape a single URL, reporting progress and enforcing robots.txt.
244///
245/// Returns `Ok(Some(content))` on success, `Ok(None)` when the URL is blocked
246/// by robots.txt (skipped, not a failure), and `Err(e)` on a scrape failure.
247async fn scrape_one_url(
248    url: &Url,
249    ctx: &ScrapeContext<'_>,
250    opts: &CrawlOptions,
251    observer: &dyn ProgressObserver,
252    page_correlation: &CorrelationId,
253) -> Result<Option<ScrapedContent>, crate::error::ScraperError> {
254    let url_str = url.as_str();
255    let _url_host = url.host_str().unwrap_or("unknown").to_string();
256
257    observer.on_page_started(url_str).await;
258
259    // Robots.txt enforcement — skip disallowed URLs unless --ignore-robots
260    if !opts.crawl.ignore_robots {
261        let domain = url.host_str().unwrap_or("unknown");
262        if !ctx.robots_fetcher.is_allowed(url_str, domain).await {
263            info!("Blocked by robots.txt: {}", url_str);
264            observer.on_robots_blocked(url_str).await;
265            return Ok(None);
266        }
267    }
268
269    observer
270        .on_status_changed(url_str, ScrapeStatus::Fetching)
271        .await;
272
273    match scrape_single_url_for_tui(
274        ctx.router,
275        url,
276        ctx.scraper_config,
277        ctx.downloader,
278        ctx.engine,
279        None,
280        page_correlation,
281    )
282    .await
283    {
284        Ok(content) => {
285            observer
286                .on_status_changed(url_str, ScrapeStatus::Extracting)
287                .await;
288            let chars = content.content.chars().count();
289            observer.on_page_completed(url_str, chars).await;
290            Ok(Some(content))
291        },
292        Err(e) => {
293            let url_str = url.as_str().to_string();
294            warn!("Failed to scrape {}: {}", url_str, e);
295            // ScraperError doesn't impl Clone, so we format for the observer
296            // and keep the original for the failures vec (needed for error chain display).
297            let scrape_err = ScrapeError::Other(format!("{e}"));
298            observer.on_page_failed(&url_str, &scrape_err).await;
299            Err(e)
300        },
301    }
302}

304fn build_http_client_config(
305    opts: &CrawlOptions,
306) -> Result<HttpClientConfig, crate::domain::UnknownProfileError> {
307    Ok(HttpClientConfig {
308        max_retries: opts.network.max_retries,
309        backoff_base_ms: opts.network.backoff_base_ms,
310        backoff_max_ms: opts.network.backoff_max_ms,
311        accept_language: opts.network.accept_language.clone(),
312        user_agent: opts.network.user_agent.clone(),
313        timeout_secs: opts.network.timeout_secs,
314        tls_emulation: HttpClientConfig::profile_from_name(&opts.network.h2_profile)?,
315        ignore_waf: opts.crawl.ignore_waf,
316        ..HttpClientConfig::default()
317    })
318}
