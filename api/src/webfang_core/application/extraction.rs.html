26/// Extract HTML content using a CSS selector.
27///
28/// When `selector` is not "body", parses the HTML and extracts all elements
29/// matching the selector. Returns the outer HTML of matched elements wrapped
30/// in a `<div>` for Readability processing. If no elements match or the
31/// selector is invalid, returns [`ExtractResult::Fallback`] with the full
32/// HTML and an optional diagnostic (when an inspector is provided).
33///
34/// # Arguments
35/// * `html` - The HTML document to extract from
36/// * `selector` - CSS selector string (use `"body"` to skip extraction)
37/// * `inspector` - Optional DOM inspector for diagnostics on failure paths
38pub fn extract_with_selector(
39    html: &str,
40    selector: &str,
41    inspector: Option<&dyn DomInspectorPort>,
42) -> ExtractResult {
43    if selector == "body" {
44        return ExtractResult::Matched(html.to_owned());
45    }
46
47    // Early check: empty or whitespace-only HTML. `scraper::Html::parse_document("")`
48    // creates 3 implicit elements (html, head, body), so without this check the
49    // selector matching would fall through to ZeroMatches instead of
50    // EmptyDocument — leaving SelectorErrorKind::EmptyDocument as dead code.
51    if html.trim().is_empty() {
52        let document = scraper::Html::parse_document(html);
53        return fallback(
54            html,
55            inspector,
56            &document,
57            SelectorErrorKind::EmptyDocument,
58            selector,
59            "HTML document is empty or whitespace-only, falling back with EmptyDocument diagnostic",
60        );
61    }
62
63    let document = scraper::Html::parse_document(html);
64    let sel = match scraper::Selector::parse(selector) {
65        Ok(s) => s,
66        Err(e) => {
67            return fallback(
68                html,
69                inspector,
70                &document,
71                SelectorErrorKind::InvalidSelector(e.to_string()),
72                selector,
73                &format!("Invalid CSS selector '{selector}': {e}, falling back to full HTML"),
74            );
75        },
76    };
77
78    let matched: Vec<String> = document.select(&sel).map(|el| el.html()).collect();
79
80    if matched.is_empty() {
81        return fallback(
82            html,
83            inspector,
84            &document,
85            SelectorErrorKind::ZeroMatches,
86            selector,
87            &format!("CSS selector '{selector}' matched 0 elements, falling back to full HTML"),
88        );
89    }
90
91    debug!(
92        "CSS selector '{}' matched {} elements",
93        selector,
94        matched.len()
95    );
96
97    ExtractResult::Matched(format!(
98        "<div id=\"selector-extracted\">{}</div>",
99        matched.join("\n")
100    ))
101}

103/// Emit the fallback warning and build a [`ExtractResult::Fallback`] carrying
104/// the full HTML and a diagnostic for the given [`SelectorErrorKind`], when an
105/// inspector is available.
106fn fallback(
107    html: &str,
108    inspector: Option<&dyn DomInspectorPort>,
109    document: &scraper::Html,
110    kind: SelectorErrorKind,
111    selector: &str,
112    message: &str,
113) -> ExtractResult {
114    warn!("{}", message);
115    ExtractResult::Fallback {
116        html: html.to_owned(),
117        diagnostic: build_diagnostic(inspector, document, kind, selector),
118    }
119}

121/// Scrape a URL using Readability algorithm for clean content extraction
122///
123/// This is the 2026 best practice approach — uses the same algorithm as
124/// Firefox Reader View to extract only meaningful content.
125///
126/// # Examples
127///
128/// ```no_run
129/// use webfang_core::application::{create_http_client, scrape_with_readability};
130///
131/// # #[tokio::main]
132/// # async fn main() -> Result<(), Box<dyn std::error::Error>> {
133/// let client = create_http_client()?;
134/// let url = url::Url::parse("https://example.com")?;
135/// let results = scrape_with_readability(&client, &url).await?;
136/// # Ok(())
137/// # }
138/// ```
139pub async fn scrape_with_readability(
140    client: &dyn HttpClientPort,
141    url: &url::Url,
142) -> Result<Vec<ScrapedContent>> {
143    // Standalone convenience entry: this call IS the operation, so it mints
144    // its own run-root identity (#501).
145    let root_correlation = CorrelationId::new();
146    let outcome = scrape_with_config(
147        client,
148        url,
149        &ScraperConfig::default(),
150        None,
151        None,
152        None,
153        &root_correlation,
154    )
155    .await?;
156    Ok(outcome.results)
157}

159/// Canonical adaptive selector repair (Tier 1 lexical cascade).
160///
161/// Single home for the "extraction fell back → ask the adaptive engine for a
162/// repaired selector → re-extract" use case, shared by [`extract_content`] and
163/// `scrape_with_config` (issue #442 — previously duplicated in both). When the
164/// engine yields a selector that matches, the repaired [`ExtractResult`] is
165/// returned; otherwise the original fallback is preserved unchanged.
166///
167/// `inspector` is threaded through to [`extract_with_selector`] for diagnostics
168/// (`scrape_with_config` passes its inspector; [`extract_content`] passes
169/// `None`).
170#[cfg(feature = "adaptive-selectors")]
171pub(crate) async fn adaptive_selector_repair(
172    extract_result: ExtractResult,
173    engine: Option<&AdaptiveSelectorEngine>,
174    selector: &str,
175    host: Option<&str>,
176    inspector: Option<&dyn DomInspectorPort>,
177) -> ExtractResult {
178    if let ExtractResult::Fallback { html, diagnostic } = extract_result {
179        if let Some(engine) = engine {
180            match engine
181                .select_sync_aware(
182                    html.clone(),
183                    selector.to_owned(),
184                    host.map(|s| s.to_owned()),
185                )
186                .await
187            {
188                Ok(outcome) => {
189                    let repaired =
190                        extract_with_selector(&html, &outcome.suggestion.selector, inspector);
191                    if repaired.is_matched() {
192                        tracing::info!(
193                            repaired_selector = %outcome.suggestion.selector,
194                            method = ?outcome.status,
195                            "adaptive_repair_resolved"
196                        );
197                        repaired
198                    } else {
199                        ExtractResult::Fallback { html, diagnostic }
200                    }
201                },
202                Err(_) => ExtractResult::Fallback { html, diagnostic },
203            }
204        } else {
205            ExtractResult::Fallback { html, diagnostic }
206        }
207    } else {
208        extract_result
209    }
210}

212/// Pipeline de extracción de contenido: clean → selector → adaptive → readability/fallback.
213///
214/// Recibe HTML ya fetchado y validado (post-WAF). No conoce el transporte.
215///
216/// Correlation identity (#501): the per-page identity is a REQUIRED input —
217/// callers own it (e.g. `scrape_single_url_for_tui` declares it on its trace
218/// span) and inject it here so the exported content shares the same identity
219/// as the page's `span_fields` in the `--trace-file` JSONL. Standalone
220/// callers mint their own root at entry. Identity enters through the type
221/// system or not at all — there is no ad-hoc fallback.
222///
223/// # Errors
224///
225/// Returns [`ScraperError::ExtractionFailed`] when fallback content is below
226/// `MIN_FALLBACK_CONTENT` bytes.
227pub async fn extract_content(
228    html: &str,
229    url: &url::Url,
230    config: &ScraperConfig,
231    asset_downloader: Option<&dyn crate::domain::ports::AssetDownloaderPort>,
232    #[allow(unused_variables)] engine: Option<&AdaptiveSelectorEngine>,
233    correlation_id: &CorrelationId,
234) -> Result<ScrapedContent> {
235    // Clean HTML boilerplate (scripts, styles, nav, sidebar, footer) BEFORE
236    // Readability. This helps legible find the main content without being
237    // confused by navigation elements, JavaScript bundles, and CSS.
238    let cleaned_html = crate::infrastructure::converter::html_cleaner::clean_html(html);
239
240    // Apply CSS selector extraction if a non-default selector is configured.
241    let extract_result = extract_with_selector(&cleaned_html, &config.selector, None);
242    // Adaptive selector repair (Tier 1 lexical): delegate to the canonical
243    // shared helper (#442). This TUI path passes no inspector and keeps its own
244    // binary/metrics handling instead of delegating to `scrape_with_config`.
245    #[cfg(feature = "adaptive-selectors")]
246    let extract_result = adaptive_selector_repair(
247        extract_result,
248        engine,
249        &config.selector,
250        url.host_str(),
251        None,
252    )
253    .await;
254
255    let extraction_html = extract_result.as_html().to_owned();
256
257    // Try Readability first, fallback to plain text extraction
258    match readability::parse(&extraction_html, Some(url.as_str())) {
259        Ok(article) => {
260            let assets = crate::application::asset_download::download_assets_if_enabled(
261                html,
262                url,
263                config,
264                asset_downloader,
265            )
266            .await?;
267
268            let author = crate::infrastructure::scraper::author_extractor::extract_author(
269                html,
270                article.byline.as_deref(),
271            );
272
273            Ok(ScrapedContent {
274                title: crate::application::resolve_title(&article.title, url),
275                content: article.text_content,
276                url: ValidUrl::new(url.clone()),
277                excerpt: article.excerpt,
278                author,
279                date: article.published_time,
280                // Store CLEAN HTML from Readability (not raw HTML with nav/ads/footer)
281                html: Some(article.content),
282                assets,
283                correlation_id: Some(correlation_id.clone()),
284            })
285        },
286        Err(e) => {
287            warn!("Readability failed for {}: {}", url, e);
288            let fallback_content = fallback::extract_text(&extraction_html);
289
290            // Check if fallback produced poor content (likely extraction failure)
291            const MIN_FALLBACK_CONTENT: usize = 100;
292            if fallback_content.len() < MIN_FALLBACK_CONTENT {
293                let msg = format!(
294                    "contenido pobre del fallback: {} bytes (mín {} bytes). Readability: {}",
295                    fallback_content.len(),
296                    MIN_FALLBACK_CONTENT,
297                    e
298                );
299                log_scrape_error(
300                    &msg,
301                    url.as_str(),
302                    "extract",
303                    Some(correlation_id),
304                    "content extraction failed",
305                );
306                return Err(ScraperError::ExtractionFailed {
307                    url: url.to_string(),
308                    reason: msg,
309                });
310            }
311
312            let assets = crate::application::asset_download::download_assets_if_enabled(
313                html,
314                url,
315                config,
316                asset_downloader,
317            )
318            .await?;
319
320            Ok(ScrapedContent {
321                title: url
322                    .host_str()
323                    .ok_or_else(|| ScraperError::invalid_url(format!("URL missing host: {url}")))?
324                    .to_string(),
325                content: fallback_content,
326                url: ValidUrl::new(url.clone()),
327                excerpt: None,
328                author: None,
329                date: None,
330                html: Some(html.to_owned()),
331                assets,
332                correlation_id: Some(correlation_id.clone()),
333            })
334        },
335    }
336}

342    // --- extract_with_selector: pure, no network (scraper::Html from a string) ---
343
344    #[test]
345    fn test_body_selector_is_passthrough() {
346        let html = "<html><body><p>keep me</p></body></html>";
347        let result = extract_with_selector(html, "body", None);
348        assert!(result.is_matched());
349        assert_eq!(
350            result.as_html(),
351            html,
352            "body selector must return HTML verbatim"
353        );
354    }
355
356    #[test]
357    fn test_matching_selector_wraps_elements() {
358        let html = "<html><body><div class=\"main\">Hello</div><aside>noise</aside></body></html>";
359        let result = extract_with_selector(html, "div.main", None);
360        assert!(result.is_matched());
361        let extracted = result.as_html();
362        assert!(
363            extracted.starts_with("<div id=\"selector-extracted\">"),
364            "matched output must be wrapped for Readability, got: {extracted}"
365        );
366        assert!(extracted.contains("Hello"));
367        assert!(
368            !extracted.contains("noise"),
369            "non-matching elements must be excluded"
370        );
371    }
372
373    #[test]
374    fn test_zero_matches_falls_back_without_inspector() {
375        let html = "<html><body><p>content</p></body></html>";
376        let result = extract_with_selector(html, "div.does-not-exist", None);
377        assert!(!result.is_matched());
378        assert_eq!(result.as_html(), html, "fallback must carry the full HTML");
379        match result {
380            ExtractResult::Fallback { diagnostic, .. } => {
381                assert!(diagnostic.is_none(), "no inspector means no diagnostic");
382            },
383            ExtractResult::Matched(_) => panic!("expected Fallback"),
384        }
385    }
386
387    #[test]
388    fn test_invalid_selector_falls_back() {
389        let html = "<html><body><p>content</p></body></html>";
390        let result = extract_with_selector(html, ">>>not-a-selector", None);
391        assert!(!result.is_matched());
392        assert_eq!(result.as_html(), html);
393    }
394
395    #[test]
396    fn test_empty_html_falls_back() {
397        let result = extract_with_selector("   ", "div.main", None);
398        assert!(!result.is_matched(), "whitespace-only HTML must fall back");
399    }
