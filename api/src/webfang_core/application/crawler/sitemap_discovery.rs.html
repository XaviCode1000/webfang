1//! Sitemap discovery — sitemap-based URL discovery use cases.
2//!
3//! Hosts the sitemap crawling orchestration: explicit/auto-discovered sitemap
4//! fetching, sub-path sitemap fallback, and robots.txt sitemap discovery. These
5//! were extracted from `discovery.rs` (issue #442) to separate the sitemap
6//! infrastructure concern from DOM-link discovery.

16/// Crawl site using sitemap (preferred method - FASE 3)
17///
18/// Following **own-borrow-over-clone**: Accepts `&str` not `&String`.
19/// Following **api-builder-pattern**: Uses SitemapConfig builder.
20///
21/// # Arguments
22///
23/// * `base_url` - Base URL of the website
24/// * `sitemap_url` - Optional explicit sitemap URL (auto-discovers if None)
25/// * `config` - Crawler configuration
26///
27/// # Returns
28///
29/// * `Ok(Vec<DiscoveredUrl>)` - URLs discovered from sitemap
30/// * `Err(CrawlError)` - Error during sitemap fetch or parse
31///
32/// # Examples
33///
34/// ```no_run
35/// use webfang_core::application::crawl_with_sitemap;
36/// use webfang_core::domain::CrawlerConfig;
37/// use url::Url;
38///
39/// # #[tokio::main]
40/// # async fn main() -> Result<(), Box<dyn std::error::Error>> {
41/// let seed = Url::parse("https://example.com")?;
42/// let config = CrawlerConfig::new(seed);
43///
44/// let urls = crawl_with_sitemap("https://example.com", None, &config).await?;
45/// println!("Found {} URLs from sitemap", urls.len());
46/// # Ok(())
47/// # }
48/// ```
49#[instrument(
50    name = "crawl_with_sitemap",
51    skip(config),
52    fields(
53        base_url,
54        sitemap_url = ?sitemap_url
55    )
56)]
57pub async fn crawl_with_sitemap(
58    base_url: &str,
59    sitemap_url: Option<&str>,
60    config: &CrawlerConfig,
61) -> Result<Vec<DiscoveredUrl>, CrawlError> {
62    crawl_with_sitemap_internal(base_url, sitemap_url, config).await
63}

65/// Crawl with sitemap (internal version with progress tracking)
66///
67/// This is the internal implementation that supports optional progress tracking.
68/// The public `crawl_with_sitemap` function calls this one.
69///
70/// Following **own-borrow-over-clone**: Accepts `&str` not `&String`.
71#[allow(unused_variables)]
72async fn crawl_with_sitemap_internal(
73    base_url: &str,
74    sitemap_url: Option<&str>,
75    config: &CrawlerConfig,
76) -> Result<Vec<DiscoveredUrl>, CrawlError> {
77    info!("Crawling with sitemap for {}", base_url);
78
79    // Build the discovery client through the shared factory so sitemap probes
80    // carry the same Chrome Client Hints, pooled user-agent, pool tuning, and
81    // gzip/brotli as the DOM path (#298).
82    let discovery_client = build_discovery_client(config)?;
83
84    // Use default batch size (10,000) - SitemapConfig handles pagination
85    const DEFAULT_BATCH_SIZE: usize = 10_000;
86
87    // Auto-discover sitemap URL if not provided
88    let sitemap_url = resolve_sitemap_url(base_url, sitemap_url, &discovery_client).await?;
89
90    tracing::info!("Using sitemap: {}", sitemap_url);
91
92    // Create sitemap parser with config (including pagination settings).
93    // #323: thread the configured TLS/H2 profile so sitemap XML fetches honor
94    // the user's --h2-profile selection instead of a hardcoded Chrome145.
95    let parser = build_sitemap_parser(config, DEFAULT_BATCH_SIZE)?;
96
97    // Parse sitemap
98    let urls = parse_sitemap(&parser, &sitemap_url).await?;
99
100    // Validate sitemap relevance: check if any URLs share a path prefix
101    // with the target URL. This handles cases where robots.txt points to
102    // an unrelated sitemap (e.g. blog sitemap for a docs site).
103    let base = Url::parse(base_url).map_err(|e| CrawlError::InvalidUrl(e.to_string()))?;
104    let target_path = base.path().to_string();
105    let relevant_urls = filter_relevant_urls(urls, &target_path);
106
107    // If no relevant URLs found, try sub-path sitemaps as fallback
108    if relevant_urls.is_empty() {
109        tracing::warn!(
110            "sitemap {} no tiene URLs que coincidan con la ruta objetivo {}, intentando sitemaps de subruta",
111            sitemap_url,
112            target_path
113        );
114        return crawl_with_subpath_sitemaps(
115            base_url,
116            &base,
117            &parser,
118            3,
119            0,
120            config.max_depth,
121            &discovery_client,
122        )
123        .await;
124    }
125
126    Ok(build_discovered_urls(relevant_urls, &base, config))
127}

129/// Build the shared discovery HTTP client with the #281/#312 timeout and TLS
130/// profile policy.
131fn build_discovery_client(config: &CrawlerConfig) -> Result<wreq::Client, CrawlError> {
132    // Timeouts replicate the #281 policy: request timeout as configured,
133    // connect timeout capped at 10s.
134    let http_config = HttpClientConfig {
135        timeout_secs: config.timeout_secs,
136        connect_timeout_secs: config.timeout_secs.min(10), // #281 policy
137        tls_emulation: config.tls_emulation,               // #312 honor configured profile
138        ..Default::default()
139    };
140    super::super::create_http_client_with_config(&http_config)
141        // LCOV_EXCL_LINE defensive: wreq-client-build — client construction fails only on invalid TLS profile, an invariant
142        .map_err(|e| CrawlError::Internal(format!("failed to build discovery client: {e}")))
143}

145/// Resolve the sitemap URL: use the provided one, or auto-discover it.
146async fn resolve_sitemap_url(
147    base_url: &str,
148    sitemap_url: Option<&str>,
149    client: &wreq::Client,
150) -> Result<String, CrawlError> {
151    match sitemap_url {
152        Some(url) if !url.is_empty() => {
153            tracing::info!("Sitemap URL provided: {}", url);
154            Ok(url.to_string())
155        },
156        _ => discover_sitemap_url_for(base_url, client).await,
157    }
158}

160/// Auto-discover a sitemap URL, logging the discovery outcome.
161async fn discover_sitemap_url_for(
162    base_url: &str,
163    client: &wreq::Client,
164) -> Result<String, CrawlError> {
165    tracing::info!("Auto-discovering sitemap URL for {}", base_url);
166    match discover_sitemap_url(base_url, client).await {
167        Ok(url) => {
168            tracing::info!("Discovered sitemap URL: {}", url);
169            Ok(url)
170        },
171        Err(CrawlError::SitemapNotFound(url)) => Err(CrawlError::SitemapNotFound(url)),
172        Err(e) => Err(e),
173    }
174}

176/// Build the sitemap parser with pagination settings and the configured TLS
177/// profile.
178fn build_sitemap_parser(
179    config: &CrawlerConfig,
180    batch_size: usize,
181) -> Result<SitemapParser, CrawlError> {
182    SitemapParser::with_config_and_profile(
183        SitemapConfig::builder()
184            .gzip_enabled(true)
185            .max_depth(3)
186            .concurrency(5)
187            .batch_size(batch_size)
188            .pagination_enabled(true)
189            .build(),
190        config.tls_emulation,
191    )
192}

194/// Parse a sitemap URL, mapping parse failures to a [`CrawlError::Parse`].
195async fn parse_sitemap(parser: &SitemapParser, sitemap_url: &str) -> Result<Vec<Url>, CrawlError> {
196    let urls = parser.parse_from_url(sitemap_url).await.map_err(|e| {
197        tracing::error!("Failed to parse sitemap {}: {}", sitemap_url, e);
198        CrawlError::Parse(e.to_string())
199    })?;
200    tracing::info!("Parsed {} total URLs from sitemap", urls.len());
201    Ok(urls)
202}

204/// Keep only sitemap URLs whose path shares the target's path prefix.
205fn filter_relevant_urls(urls: Vec<Url>, target_path: &str) -> Vec<Url> {
206    urls.into_iter()
207        .filter(|url| url.path().starts_with(target_path))
208        .collect()
209}

211/// Map relevant sitemap URLs to discovered URLs, applying include/exclude
212/// patterns and the depth gate.
213///
214/// Depth assignment: the seed itself is depth 0; every other sitemap URL is one
215/// hop from the seed, hence depth 1. Filtering by `depth <= max_depth` enforces
216/// the CLI contract "0 = only seed URL" here, because the CLI scrape flow scrapes
217/// whatever discovery returns verbatim — there is no later depth gate (the Engine's
218/// `run_crawl_task` check is a separate, non-CLI code path).
219fn build_discovered_urls(
220    relevant_urls: Vec<Url>,
221    base: &Url,
222    config: &CrawlerConfig,
223) -> Vec<DiscoveredUrl> {
224    let max_depth = config.max_depth;
225    relevant_urls
226        .into_iter()
227        .filter(|url| is_allowed(url.as_str(), config))
228        .filter_map(|url| {
229            let depth = if url == *base { 0 } else { 1 };
230            (depth <= max_depth).then(|| DiscoveredUrl::html(url, depth, base.clone()))
231        })
232        .collect()
233}

235/// Try sub-path sitemaps when the discovered sitemap has no relevant URLs
236///
237/// For nested sites like `https://example.com/docs/en/`, this tries
238/// `/docs/sitemap.xml`, `/docs/en/sitemap.xml`, etc.
239/// Follows nested sitemaps recursively up to `sitemap_max_depth` levels.
240///
241/// `crawl_max_depth` is the crawl's configured max depth (CLI `--max-depth`),
242/// distinct from the sitemap-index recursion depth above. Sub-path sitemap URLs
243/// are one hop from the seed (depth 1), so when `crawl_max_depth` is 0 ("only
244/// the seed URL") none of them qualify and an empty list is returned — mirroring
245/// the `depth <= max_depth` gate in `crawl_with_sitemap_internal`.
246///
247/// Following **own-borrow-over-clone**: Accepts `&Url` not `&String`.
248/// Following **err-no-unwrap-prod**: Proper error handling throughout.
249async fn crawl_with_subpath_sitemaps(
250    base_url: &str,
251    base: &Url,
252    parser: &SitemapParser,
253    sitemap_max_depth: usize,
254    sitemap_current_depth: usize,
255    crawl_max_depth: u8,
256    client: &wreq::Client,
257) -> Result<Vec<DiscoveredUrl>, CrawlError> {
258    if sitemap_current_depth >= sitemap_max_depth {
259        tracing::warn!(
260            "sitemap recursion depth {} reached max {}, stopping",
261            sitemap_current_depth,
262            sitemap_max_depth
263        );
264        return Ok(Vec::new());
265    }
266
267    // Sub-path sitemap URLs are depth 1; with a crawl max_depth of 0 none pass
268    // the gate, so short-circuit before probing the network.
269    if crawl_max_depth == 0 {
270        tracing::info!(
271            "crawl max_depth is 0, skipping sub-path sitemap URLs for {}",
272            base_url
273        );
274        return Ok(Vec::new());
275    }
276
277    let all_urls = probe_subpath_sitemaps_for_crawl(base, client, parser).await;
278
279    if all_urls.is_empty() {
280        tracing::warn!("no se encontraron sitemaps de subruta para {}", base_url);
281        Ok(Vec::new())
282    } else {
283        // Sub-path sitemap URLs are at depth 1 (one hop from seed)
284        Ok(all_urls
285            .into_iter()
286            .map(|url| DiscoveredUrl::html(url, 1, base.clone()))
287            .collect())
288    }
289}

291/// Probe up to 3 sub-path sitemap levels (`/docs`, `/docs/en`, ...) for both
292/// `sitemap.xml` and `sitemap_index.xml`, returning every parsed URL.
293async fn probe_subpath_sitemaps_for_crawl(
294    base: &Url,
295    client: &wreq::Client,
296    parser: &SitemapParser,
297) -> Vec<Url> {
298    let segments: Vec<_> = base.path().split('/').filter(|s| !s.is_empty()).collect();
299    let mut all_urls = Vec::new();
300
301    // Try up to 3 path levels: /docs, /docs/en, /docs/en/quickstart
302    for i in 1..=segments.len().min(3) {
303        let sub_path = segments[..i].join("/");
304        for sitemap_name in &["sitemap.xml", "sitemap_index.xml"] {
305            let candidate = format!("/{sub_path}/{sitemap_name}");
306            if let Some(urls) = try_subpath_sitemap(base, client, parser, &candidate).await {
307                all_urls.extend(urls);
308            }
309        }
310    }
311    all_urls
312}

314/// Probe a single sub-path sitemap candidate; returns its parsed URLs when the
315/// server answers 2xx, otherwise `None`.
316async fn try_subpath_sitemap(
317    base: &Url,
318    client: &wreq::Client,
319    parser: &SitemapParser,
320    candidate: &str,
321) -> Option<Vec<Url>> {
322    let sitemap_url = base.join(candidate).ok()?;
323    let sitemap_str = sitemap_url.as_str();
324    tracing::debug!("Trying sub-path sitemap: {}", sitemap_str);
325    let response = client.head(sitemap_str).send().await.ok()?;
326    if !response.status().is_success() {
327        return None;
328    }
329    tracing::info!("Found sub-path sitemap: {}", sitemap_str);
330    parse_subpath_sitemap(parser, sitemap_str).await
331}

333/// Parse a discovered sub-path sitemap, logging the URL count on success.
334async fn parse_subpath_sitemap(parser: &SitemapParser, sitemap_str: &str) -> Option<Vec<Url>> {
335    match parser.parse_from_url(sitemap_str).await {
336        Ok(urls) => {
337            tracing::info!(
338                "Parsed {} URLs from sub-path sitemap {}",
339                urls.len(),
340                sitemap_str
341            );
342            Some(urls)
343        },
344        Err(_) => None,
345    }
346}

348/// Auto-discover sitemap URL from robots.txt or fallback locations.
349///
350/// Resolution order (each step short-circuits on first hit):
351/// 1. Parse the `Sitemap:` directive from `robots.txt` (resolved against `base`).
352/// 2. HEAD-probe common fallback paths (`/sitemap.xml`, ...).
353/// 3. HEAD-probe nested sub-path sitemaps for sectioned sites (GAP 5 / Bug #30).
354///
355/// Following **own-borrow-over-clone**: Accepts `&str`.
356/// Following **security-no-unwrap-in-prod**: Proper error handling.
357///
358/// # Arguments
359///
360/// * `base_url` - Base URL of the website
361///
362/// # Returns
363///
364/// * `Ok(String)` - Discovered sitemap URL
365/// * `Err(CrawlError)` - Error during discovery
366async fn discover_sitemap_url(base_url: &str, client: &wreq::Client) -> Result<String, CrawlError> {
367    let base = Url::parse(base_url).map_err(|e| CrawlError::InvalidUrl(e.to_string()))?;
368
369    // 1. robots.txt `Sitemap:` directive
370    if let Some(url) = fetch_robots_sitemap(&base, client).await {
371        return Ok(url);
372    }
373    tracing::debug!("No sitemap found in robots.txt, trying fallback locations");
374
375    // 2. Common fallback locations
376    const FALLBACK_PATHS: &[&str] = &[
377        "/sitemap.xml",
378        "/sitemap_index.xml",
379        "/sitemap.xml.gz",
380        "/sitemap/sitemap.xml",
381    ];
382    if let Some(url) = probe_sitemap_paths(&base, client, FALLBACK_PATHS).await {
383        return Ok(url);
384    }
385
386    // 3. Nested sub-path sitemaps for sectioned sites
387    if let Some(url) = probe_subpath_sitemaps(&base, client).await {
388        return Ok(url);
389    }
390
391    tracing::warn!("no sitemap found for {}", base_url);
392    Err(CrawlError::SitemapNotFound(base_url.to_string()))
393}

395/// Fetch `robots.txt` and extract its `Sitemap:` directive, if present.
396async fn fetch_robots_sitemap(base: &Url, client: &wreq::Client) -> Option<String> {
397    let robots_url = base.join("/robots.txt").ok()?;
398    tracing::info!("Checking robots.txt: {}", robots_url);
399
400    let response = client.get(robots_url.as_str()).send().await.ok()?;
401    tracing::info!("robots.txt status: {}", response.status());
402    if !response.status().is_success() {
403        return None;
404    }
405
406    let content = response.text().await.ok()?;
407    log_robots_content(&content);
408    extract_robots_sitemap_directive(&content, base)
409}

411/// Log the robots.txt body (capped to the first 500 chars for instrumentation).
412fn log_robots_content(content: &str) {
413    tracing::info!(
414        "robots.txt content (first 500 chars):\n{}",
415        &content[..content.len().min(500)]
416    );
417}

419/// Parse the first `Sitemap:` directive from a `robots.txt` body and resolve it
420/// against `base` (absolute `http(s)` URLs are parsed directly).
421fn extract_robots_sitemap_directive(content: &str, base: &Url) -> Option<String> {
422    for line in content.lines() {
423        if !line.to_lowercase().starts_with("sitemap:") {
424            continue;
425        }
426        let Some(directive) = line
427            .strip_prefix("Sitemap:")
428            .or_else(|| line.strip_prefix("sitemap:"))
429        else {
430            continue;
431        };
432        let sitemap = directive.trim();
433
434        let resolved = if sitemap.starts_with("http://") || sitemap.starts_with("https://") {
435            Url::parse(sitemap).ok()
436        } else {
437            base.join(sitemap).ok()
438        };
439
440        match resolved {
441            Some(url) => {
442                tracing::debug!("Found sitemap in robots.txt: {}", url);
443                return Some(url.to_string());
444            },
445            None => tracing::warn!("Invalid sitemap URL in robots.txt: {}", sitemap),
446        }
447    }
448    None
449}

451/// HEAD-probe each candidate path and return the first that responds 2xx.
452async fn probe_sitemap_paths(base: &Url, client: &wreq::Client, paths: &[&str]) -> Option<String> {
453    for path in paths {
454        if let Some(found) = probe_single_path(base, client, path).await {
455            return Some(found);
456        }
457    }
458    None
459}

461/// HEAD-probe a single candidate path; returns its URL when the server answers
462/// 2xx, otherwise `None`.
463async fn probe_single_path(base: &Url, client: &wreq::Client, path: &str) -> Option<String> {
464    let Ok(sitemap_url) = base.join(path) else {
465        tracing::warn!("Invalid sitemap candidate path: {path}");
466        return None;
467    };
468    probe_head_success(client, sitemap_url.as_str()).await
469}
470
471/// HEAD-probe a resolved sitemap URL, returning it when the server answers 2xx.
472async fn probe_head_success(client: &wreq::Client, sitemap_str: &str) -> Option<String> {
473    tracing::info!("Trying fallback sitemap: {}", sitemap_str);
474
475    let response = client.head(sitemap_str).send().await.ok()?;
476    tracing::info!("  Status: {}", response.status());
477    if response.status().is_success() {
478        tracing::debug!("Found sitemap at fallback location: {}", sitemap_str);
479        return Some(sitemap_str.to_string());
480    }
481    None
482}

484/// Build nested sub-path sitemap candidates (e.g. `/docs/sitemap.xml`) and probe
485/// them, honoring the same 3-level depth cap as `crawl_with_subpath_sitemaps`.
486async fn probe_subpath_sitemaps(base: &Url, client: &wreq::Client) -> Option<String> {
487    let segments: Vec<_> = base.path().split('/').filter(|s| !s.is_empty()).collect();
488    let mut candidates = Vec::with_capacity(segments.len().min(3) * 2);
489    for i in 1..=segments.len().min(3) {
490        let sub_path = segments[..i].join("/");
491        for name in ["sitemap.xml", "sitemap_index.xml"] {
492            candidates.push(format!("/{sub_path}/{name}"));
493        }
494    }
495    let refs: Vec<&str> = candidates.iter().map(String::as_str).collect();
496    probe_sitemap_paths(base, client, &refs).await
497}
