1//! Shared context for all crawl tasks spawned by the engine.
2//!
3//! Consolidates the 18 `Arc::clone()` calls that were previously done
4//! per-`tokio::spawn()` into a single `Arc<CrawlTaskCtx>`, reducing
5//! atomic contention, cache pollution, and the cost of adding new
6//! shared resources.
7
8use std::sync::atomic::{AtomicU64, AtomicUsize};
9use std::sync::{Arc, RwLock};
10
11use tokio_util::sync::CancellationToken;
12
13use crate::application::crawler::checkpoint::BannedDomain;
14use crate::application::crawler::ports::{
15    ContentPipeline, CrawlResultCollector, LinkExtractorPort, PageFetcher, RobotsChecker,
16};
17use crate::application::pipeline::OutputStage;
18use crate::application::rate_limiter::SharedRateLimiter;
19use crate::domain::session_port::SessionPort;
20use crate::domain::{CorrelationId, CrawlerConfig};
21use crate::infrastructure::crawler::UrlQueue;
22use crate::infrastructure::downloader::cookie_bridge::CookieBridge;
23
24/// Shared context for all crawl tasks spawned by the engine.
25///
26/// Instead of cloning 18 individual `Arc`s per `tokio::spawn()`,
27/// we construct one `Arc<CrawlTaskCtx>` and clone only the `Arc` wrapper.
28pub struct CrawlTaskCtx {
29    // --- Shared config (read-only) ---
30    pub(crate) config: Arc<CrawlerConfig>,
31    /// Root correlation ID for the crawl — every task derives a child from it
32    /// so all pages share one `trace_id` (issue #356).
33    pub(crate) correlation_id: CorrelationId,
34    pub(crate) queue: Arc<UrlQueue>,
35    pub(crate) rate_limiter: SharedRateLimiter,
36    /// Engine-wide cancellation token (#509) — fired on shutdown so tasks
37    /// blocked on rate-limit or resource waits abort promptly.
38    pub(crate) cancel_token: CancellationToken,
39    pub(crate) session_pool: Option<Arc<dyn SessionPort>>,
40    pub(crate) ignore_robots: bool,
41    pub(crate) robots_checker: Arc<dyn RobotsChecker>,
42
43    // --- Per-task mutable (atomics) ---
44    pub(crate) error_count: Arc<AtomicUsize>,
45    /// Per-category error counters indexed by `CrawlErrorCategory::index()` (issue #374).
46    pub(crate) error_breakdown: Arc<[AtomicUsize; 8]>,
47    pub(crate) pages_crawled: Arc<AtomicU64>,
48
49    // --- Infrastructure (port-based) ---
50    pub(crate) collector: Arc<dyn CrawlResultCollector>,
51    pub(crate) cookie_bridge: Arc<RwLock<CookieBridge>>,
52    pub(crate) banned_domains: Arc<RwLock<Vec<BannedDomain>>>,
53    pub(crate) fetcher: Arc<dyn PageFetcher>,
54    pub(crate) link_extractor: Arc<dyn LinkExtractorPort>,
55
56    // --- Pipeline ---
57    pub(crate) pipeline: Option<Arc<dyn ContentPipeline>>,
58    pub(crate) output_stages: Vec<Arc<Box<dyn OutputStage>>>,
59}
