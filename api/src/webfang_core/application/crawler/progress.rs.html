1//! Crawl progress metrics (issue #356, Fase 4).
2//!
3//! Pure computation helper for periodic progress logging during a crawl.
4//! Kept free of I/O and tracing so the metrics are unit-testable in isolation.
5
6use std::time::Duration;
7
8/// Computes crawl progress metrics for periodic logging.
9#[derive(Debug, Clone, Copy)]
10pub struct CrawlProgress {
11    /// Pages crawled so far.
12    pub pages_crawled: u64,
13    /// Target page limit (`max_pages`).
14    pub max_pages: usize,
15    /// Time elapsed since the crawl started.
16    pub elapsed: Duration,
17}
18
19impl CrawlProgress {
20    /// Create a progress snapshot.
21    pub fn new(pages_crawled: u64, max_pages: usize, elapsed: Duration) -> Self {
22        Self {
23            pages_crawled,
24            max_pages,
25            elapsed,
26        }
27    }
28
29    /// Completion percentage (0.0–100.0) relative to `max_pages`.
30    ///
31    /// Returns 0.0 when `max_pages` is 0 (no target).
32    pub fn progress_pct(&self) -> f64 {
33        if self.max_pages == 0 {
34            return 0.0;
35        }
36        (self.pages_crawled as f64 / self.max_pages as f64) * 100.0
37    }
38
39    /// Average pages per second over the elapsed time.
40    ///
41    /// Returns 0.0 when no time has elapsed.
42    pub fn pages_per_sec(&self) -> f64 {
43        let secs = self.elapsed.as_secs_f64();
44        if secs <= 0.0 {
45            return 0.0;
46        }
47        self.pages_crawled as f64 / secs
48    }
49
50    /// Estimated seconds remaining to reach `max_pages` at the current rate.
51    ///
52    /// Returns 0 when the rate is zero or there is no target.
53    pub fn eta_secs(&self) -> u64 {
54        let rate = self.pages_per_sec();
55        if rate <= 0.0 || self.max_pages == 0 {
56            return 0;
57        }
58        let remaining = (self.max_pages as f64 - self.pages_crawled as f64).max(0.0);
59        (remaining / rate) as u64
60    }
61}
62
63#[cfg(test)]
64mod tests {
65    use super::*;
66
67    #[test]
68    fn test_progress_pct() {
69        let p = CrawlProgress::new(50, 200, Duration::from_secs(10));
70        assert!((p.progress_pct() - 25.0).abs() < 0.01);
71    }
72
73    #[test]
74    fn test_pages_per_sec() {
75        let p = CrawlProgress::new(100, 200, Duration::from_secs(10));
76        assert!((p.pages_per_sec() - 10.0).abs() < 0.01);
77    }
78
79    #[test]
80    fn test_eta_secs() {
81        // 10 pages/s, 100 pages remaining → 10s ETA
82        let p = CrawlProgress::new(100, 200, Duration::from_secs(10));
83        assert_eq!(p.eta_secs(), 10);
84    }
85
86    #[test]
87    fn test_eta_never_negative_past_target() {
88        // Already past max_pages → ETA clamps to 0
89        let p = CrawlProgress::new(250, 200, Duration::from_secs(10));
90        assert_eq!(p.eta_secs(), 0);
91    }
92
93    #[test]
94    fn test_zero_guards() {
95        let p = CrawlProgress::new(0, 0, Duration::from_secs(0));
96        assert_eq!(p.progress_pct(), 0.0);
97        assert_eq!(p.pages_per_sec(), 0.0);
98        assert_eq!(p.eta_secs(), 0);
99    }
