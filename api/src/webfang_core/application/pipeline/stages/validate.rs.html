1//! Validation pipeline stage.
2//!
3//! Checks incoming [`ScrapedItem`]s for basic validity before processing.
4//! Rejects items with invalid URLs, empty content, or error status codes.
5
6use std::future::Future;
7use std::pin::Pin;
8
9use crate::domain::pipeline_item::{PipelineStage, ScrapedItem, StageOutcome};
10
11/// Patterns in URLs that indicate non-content pages.
12const SKIP_PATHS: &[&str] = &[
13    "/robots.txt",
14    "/sitemap.xml",
15    "/favicon.ico",
16    "/.well-known/",
17    "/ads.txt",
18];
19
20/// Pipeline stage that validates scraped items for basic correctness.
21///
22/// # Validation rules
23///
24/// - **URL validity**: must be non-empty, parseable, with http(s) scheme
25/// - **Status code**: must be in 200..=399
26/// - **Content length**: raw_html must be non-empty
27/// - **Skip patterns**: robots.txt, sitemap.xml, and similar non-content pages
28///   are silently skipped
29pub struct ValidateStage;
30
31impl PipelineStage for ValidateStage {
32    fn name(&self) -> &str {
33        "validate"
34    }
35
36    fn process(
37        &self,
38        item: ScrapedItem,
39    ) -> Pin<Box<dyn Future<Output = StageOutcome> + Send + '_>> {
40        Box::pin(async move { validate(item) })
41    }
42}
43
44fn validate(item: ScrapedItem) -> StageOutcome {
45    // 1. URL must be non-empty
46    if item.url.is_empty() {
47        return StageOutcome::Reject("URL is empty".into());
48    }
49
50    // 2. URL must be parseable with http(s) scheme
51    let parsed = match url::Url::parse(&item.url) {
52        Ok(u) => u,
53        Err(_) => {
54            return StageOutcome::Reject(format!("URL is not valid: {}", item.url));
55        },
56    };
57
58    match parsed.scheme() {
59        "http" | "https" => {},
60        scheme => {
61            return StageOutcome::Reject(format!(
62                "URL scheme '{scheme}' is not supported (requires http or https)"
63            ));
64        },
65    }
66
67    if parsed.host_str().is_none() {
68        return StageOutcome::Reject("URL has no host".into());
69    }
70
71    // 3. Skip non-content pages (robots.txt, sitemap, etc.)
72    let path = parsed.path().to_lowercase();
73    if SKIP_PATHS.iter().any(|p| path == *p || path.starts_with(p)) {
74        return StageOutcome::Skip;
75    }
76
77    // 4. Status code must be in 200..=399
78    if !(200..=399).contains(&item.status_code) {
79        return StageOutcome::Reject(format!(
80            "HTTP status {} indicates an error",
81            item.status_code
82        ));
83    }
84
85    // 5. raw_html must not be empty
86    if item.raw_html.trim().is_empty() {
87        return StageOutcome::Reject("Content is empty".into());
88    }
89
90    StageOutcome::Continue(item)
91}

97    fn make_item(url: &str, status: u16, html: &str) -> ScrapedItem {
98        ScrapedItem {
99            url: url.into(),
100            raw_html: html.into(),
101            status_code: status,
102            ..Default::default()
103        }
104    }
105
106    #[tokio::test]
107    async fn test_valid_item_passes() {
108        let item = make_item("https://example.com", 200, "<p>hi</p>");
109        let stage = ValidateStage;
110        let result = stage.process(item).await;
111        assert!(matches!(result, StageOutcome::Continue(_)));
112    }
