1//! Batch processing module — Planner-Worker pattern for parallel crawl execution
2//!
3//! This module provides batch processing capabilities for crawling multiple URLs
4//! concurrently with progress tracking and graceful shutdown.
5//!
6//! # Architecture
7//!
8//! ```text
9//! BatchJob (input)
10//!     ↓
11//! BatchProcessor (worker pool with semaphore)
12//!     ↓
13//! BatchResult (output)
14//! ```
15//!
16//! The [`BatchProcessor`] wraps the existing `crawl_site()` function to process
17//! multiple URLs concurrently while respecting concurrency limits via
18//! [`tokio::sync::Semaphore`].
19

21pub mod processor;

30/// Status of a batch job
31#[derive(Debug, Clone, PartialEq, Eq)]
32pub enum BatchJobStatus {
33    /// Job is waiting to be processed
34    Pending,
35    /// Job is currently being processed
36    Running,
37    /// Job completed successfully
38    Completed,
39    /// Job failed with an error message
40    Failed(String),
41}
42
43impl fmt::Display for BatchJobStatus {
44    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
45        match self {
46            Self::Pending => write!(f, "Pending"),
47            Self::Running => write!(f, "Running"),
48            Self::Completed => write!(f, "Completed"),
49            Self::Failed(msg) => write!(f, "Failed: {msg}"),
50        }
51    }
52}

54/// Thread-safe progress tracking for batch operations
55///
56/// Uses atomic counters for lock-free updates from concurrent tasks.
57#[derive(Debug, Clone)]
58pub struct BatchProgress {
59    inner: Arc<BatchProgressInner>,
60}
61
62#[derive(Debug)]
63struct BatchProgressInner {
64    total: AtomicUsize,
65    completed: AtomicUsize,
66    failed: AtomicUsize,
67    in_progress: AtomicUsize,
68}
69
70impl BatchProgress {
71    /// Create a new progress tracker with the given total
72    pub fn new(total: usize) -> Self {
73        Self {
74            inner: Arc::new(BatchProgressInner {
75                total: AtomicUsize::new(total),
76                completed: AtomicUsize::new(0),
77                failed: AtomicUsize::new(0),
78                in_progress: AtomicUsize::new(0),
79            }),
80        }
81    }
82
83    /// Mark one task as started
84    pub fn start_one(&self) {
85        self.inner.in_progress.fetch_add(1, Ordering::Relaxed);
86    }
87
88    /// Mark one task as completed
89    pub fn complete_one(&self) {
90        self.inner.in_progress.fetch_sub(1, Ordering::Relaxed);
91        self.inner.completed.fetch_add(1, Ordering::Relaxed);
92    }
93
94    /// Mark one task as failed
95    pub fn fail_one(&self) {
96        self.inner.in_progress.fetch_sub(1, Ordering::Relaxed);
97        self.inner.failed.fetch_add(1, Ordering::Relaxed);
98    }
99
100    /// Get the total number of tasks
101    #[must_use]
102    pub fn total(&self) -> usize {
103        self.inner.total.load(Ordering::Relaxed)
104    }
105
106    /// Get the number of completed tasks
107    #[must_use]
108    pub fn completed(&self) -> usize {
109        self.inner.completed.load(Ordering::Relaxed)
110    }
111
112    /// Get the number of failed tasks
113    #[must_use]
114    pub fn failed(&self) -> usize {
115        self.inner.failed.load(Ordering::Relaxed)
116    }
117
118    /// Get the number of tasks currently in progress
119    #[must_use]
120    pub fn in_progress(&self) -> usize {
121        self.inner.in_progress.load(Ordering::Relaxed)
122    }
123
124    /// Check if all tasks are complete
125    #[must_use]
126    pub fn is_complete(&self) -> bool {
127        self.completed() + self.failed() >= self.total()
128    }
129
130    /// Get completion percentage (0.0 - 100.0)
131    #[must_use]
132    pub fn percent(&self) -> f64 {
133        let total = self.total();
134        if total == 0 {
135            return 100.0;
136        }
137        let done = self.completed() + self.failed();
138        (done as f64 / total as f64) * 100.0
139    }
140}
141
142impl Default for BatchProgress {
143    fn default() -> Self {
144        Self::new(0)
145    }
146}

148/// A batch job containing URLs to crawl
149#[derive(Debug, Clone)]
150pub struct BatchJob {
151    /// Unique identifier for this batch job
152    pub id: String,
153    /// URLs to crawl
154    pub urls: Vec<String>,
155    /// Shared configuration for all URLs in the batch
156    pub config: crate::domain::CrawlerConfig,
157    /// Current status of the job
158    pub status: BatchJobStatus,
159    /// Thread-safe progress tracking
160    pub progress: BatchProgress,
161}
162
163impl BatchJob {
164    /// Create a new batch job
165    pub fn new(id: String, urls: Vec<String>, config: crate::domain::CrawlerConfig) -> Self {
166        let total = urls.len();
167        Self {
168            id,
169            urls,
170            config,
171            status: BatchJobStatus::Pending,
172            progress: BatchProgress::new(total),
173        }
174    }
175
176    /// Mark the job as running
177    pub fn start(&mut self) {
178        self.status = BatchJobStatus::Running;
179    }
180
181    /// Mark the job as completed
182    pub fn complete(&mut self) {
183        self.status = BatchJobStatus::Completed;
184    }
185
186    /// Mark the job as failed
187    pub fn fail(&mut self, reason: String) {
188        self.status = BatchJobStatus::Failed(reason);
189    }
190}

197    #[test]
198    fn test_batch_job_creation() {
199        let config = crate::domain::CrawlerConfig::new(Url::parse("https://example.com").unwrap());
200        let job = BatchJob::new(
201            "test-1".to_string(),

206            config,

210        assert_eq!(job.urls.len(), 2);

241    #[test]
242    fn test_batch_progress_tracking() {
243        let progress = BatchProgress::new(3);
244
245        assert_eq!(progress.total(), 3);
246        assert_eq!(progress.completed(), 0);
247        assert_eq!(progress.failed(), 0);
248        assert_eq!(progress.in_progress(), 0);
249        assert!(!progress.is_complete());
250
251        progress.start_one();
252        assert_eq!(progress.in_progress(), 1);
253
254        progress.complete_one();
255        assert_eq!(progress.completed(), 1);
256        assert_eq!(progress.in_progress(), 0);
257
258        progress.start_one();
259        progress.fail_one();
260        assert_eq!(progress.failed(), 1);
261        assert_eq!(progress.in_progress(), 0);
262
263        progress.start_one();
264        progress.complete_one();
265        assert_eq!(progress.completed(), 2);
266        assert!(progress.is_complete());
267        assert!((progress.percent() - 100.0).abs() < f64::EPSILON);
268    }
