47/// Outcome of a scrape operation, including selector extraction metadata.
48///
49/// Contains both the scraped content results and the [`ExtractResult`] from
50/// CSS selector extraction, allowing callers (e.g. the MCP handler) to
51/// inspect whether the selector matched and access diagnostics.
52#[derive(Debug)]
53pub struct ScrapeOutcome {
54    /// Scraped content results.
55    pub results: Vec<ScrapedContent>,
56    /// CSS selector extraction result (`Matched` or `Fallback` with optional diagnostic).
57    pub extract_result: ExtractResult,
58}
59
60impl ScrapeOutcome {
61    /// Get the scraped content results as a slice.
62    #[must_use]
63    pub fn as_results(&self) -> &[ScrapedContent] {
64        &self.results
65    }
66}

68/// Scrape a URL with asset downloading configuration
69///
70/// Correlation contract (#501): the caller owns the run-root identity
71/// (`root_correlation`); this use case derives one page child from it
72/// (same `trace_id`, fresh `span_id`) so every page of the operation stays
73/// reconstructable under a single trace while each page is distinguishable.
74///
75/// # Arguments
76/// * `client` - HTTP client
77/// * `url` - URL to scrape
78/// * `config` - Scraper configuration with download options
79/// * `downloader` - Optional asset downloader
80/// * `inspector` - Optional DOM inspector for selector diagnostics (None for non-MCP paths)
81/// * `root_correlation` - Run-root correlation identity of the enclosing operation
82///
83/// # Returns
84/// * `ScrapeOutcome` - Scraped content results + CSS selector extraction result
85///
86/// # Errors
87/// Returns `ScraperError::Http` for HTTP errors, `ScraperError::Network` for
88/// connection errors.
89pub async fn scrape_with_config(
90    client: &dyn HttpClientPort,
91    url: &url::Url,
92    config: &ScraperConfig,
93    downloader: Option<&dyn crate::domain::ports::AssetDownloaderPort>,
94    inspector: Option<&dyn DomInspectorPort>,
95    engine: Option<&AdaptiveSelectorEngine>,
96    root_correlation: &CorrelationId,
97) -> Result<ScrapeOutcome> {
98    // Per-page identity derived from the run root BEFORE the span so the span
99    // can declare it (`#[instrument]` only sees function parameters — #501).
100    let page_correlation = root_correlation.child();
101    scrape_with_config_inner(
102        client,
103        url,
104        config,
105        downloader,
106        inspector,
107        engine,
108        page_correlation,
109    )
110    .await
111}
112
113#[instrument(
114    name = "scrape_with_config",
115    skip(client, config, downloader, inspector, engine, correlation),
116    fields(
117        url = %url,
118        correlation_id = %correlation,
119        trace_id = %correlation.trace_id(),
120        has_downloads = config.has_downloads()
121    )
122)]
123async fn scrape_with_config_inner(
124    client: &dyn HttpClientPort,
125    url: &url::Url,
126    config: &ScraperConfig,
127    downloader: Option<&dyn crate::domain::ports::AssetDownloaderPort>,
128    inspector: Option<&dyn DomInspectorPort>,
129    #[allow(unused_variables)] engine: Option<&AdaptiveSelectorEngine>,
130    correlation: CorrelationId,
131) -> Result<ScrapeOutcome> {
132    let mut results = Vec::new();
133
134    info!("🌐 Fetching: {}", url);
135
136    let response = fetch_html(client, url, &correlation).await?;
137
138    let html: &str = &response.body;
139    log_html_size(html, url);
140
141    detect_waf(html, &response, config, url, &correlation)?;
142
143    // H1 FIX: Extract title from original DOM BEFORE any transformation.
144    // This preserves the <title> tag even when --selector filters it out.
145    let original_title = extract_original_title(html);
146
147    // M7 FIX: Log selector feedback when --selector is active
148    if config.selector != "body" {
149        info!(
150            target: "scraper",
151            selector = %config.selector,
152            "Aplicando selector CSS manual"
153        );
154    }
155
156    // Clean HTML boilerplate (scripts, styles, nav, sidebar, footer) BEFORE
157    // Readability. This helps legible find the main content without being
158    // confused by navigation elements, JavaScript bundles, and CSS.
159    let cleaned_html = clean_html_for_scrape(html);
160
161    // Apply CSS selector extraction if a non-default selector is configured.
162    let extract_result = extract_with_selector(&cleaned_html, &config.selector, inspector);
163
164    // Adaptive selector repair: delegate to the canonical shared helper (#442),
165    // threading this use case's inspector through for diagnostics.
166    #[cfg(feature = "adaptive-selectors")]
167    let extract_result = adaptive_selector_repair(
168        extract_result,
169        engine,
170        &config.selector,
171        url.host_str(),
172        inspector,
173    )
174    .await;
175
176    let extraction_html = extract_result.as_html().to_owned();
177
178    // Try Readability first, fallback to plain text extraction
179    let content = build_scraped_content(
180        html,
181        &extraction_html,
182        url,
183        config,
184        downloader,
185        &original_title,
186        &correlation,
187    )
188    .await?;
189    results.push(content);
190
191    info!(
192        "✅ Extracted: {} ({} chars, {} assets)",
193        results
194            .first()
195            .map(|r| r.title.as_str())
196            .unwrap_or("unknown"),
197        results.first().map(|r| r.content.len()).unwrap_or(0),
198        results.first().map(|r| r.assets.len()).unwrap_or(0)
199    );
200
201    Ok(ScrapeOutcome {
202        results,
203        extract_result,
204    })
205}

207/// Fetch the page HTML, logging failures and rejecting non-2xx responses.
208async fn fetch_html(
209    client: &dyn HttpClientPort,
210    url: &url::Url,
211    correlation: &CorrelationId,
212) -> Result<HttpResponse> {
213    let response = match client.get(url.as_str()).await {
214        Ok(resp) => resp,
215        Err(e) => {
216            log_scrape_error(
217                &e,
218                url.as_str(),
219                "fetch",
220                Some(correlation),
221                "HTTP request failed",
222            );
223            return Err(scraper_error_from_http(e, url.as_str()));
224        },
225    };
226
227    if !(200..300).contains(&response.status) {
228        return Err(ScraperError::http(response.status, url.as_str()));
229    }
230
231    Ok(response)
232}

234/// Log the HTML size and record it on the current span, skipping detailed
235/// instrumentation for bodies larger than 1MB.
236fn log_html_size(html: &str, url: &url::Url) {
237    let html_size = html.len();
238    let html_truncated = html_size > MAX_INSTRUMENTED_BODY_SIZE;
239    if html_truncated {
240        tracing::debug!(
241            html_size_bytes = html_size,
242            html_size_skipped = true,
243            "HTML body exceeds 1MB, skipping detailed instrumentation"
244        );
245    } else {
246        tracing::debug!("📄 Downloaded {} bytes from {}", html.len(), url);
247    }
248
249    // Add span field for html size (truncated)
250    let span = tracing::Span::current();
251    span.record("html_size_bytes", html_size.min(MAX_INSTRUMENTED_BODY_SIZE));
252    span.record("html_size_skipped", html_truncated);
253}

255/// Detect WAF/CAPTCHA challenges disguised as HTTP 200 (REQ-WAF-05).
256///
257/// Context-aware inspection: status + content-type + headers drive the tiered
258/// verdict, and a block carries the full Spanish evidence chain (REQ-WAF-08)
259/// instead of a bare first-hit provider. `config.ignore_waf` short-circuits to
260/// a clean verdict (REQ-WAF-07).
261fn detect_waf(
262    html: &str,
263    response: &HttpResponse,
264    config: &ScraperConfig,
265    url: &url::Url,
266    correlation: &CorrelationId,
267) -> Result<()> {
268    let ctx = InspectionContext::from_lowercase_headers(
269        response.status,
270        &response.headers,
271        config.ignore_waf,
272    );
273    let verdict = WafInspector::inspect(html, &ctx);
274    if verdict.is_blocked {
275        let chain = verdict.evidence_chain();
276        log_scrape_error(
277            &chain,
278            url.as_str(),
279            "fetch",
280            Some(correlation),
281            "WAF challenge detected",
282        );
283        return Err(ScraperError::waf_blocked(url.to_string(), chain));
284    }
285    Ok(())
286}

288/// Extract the `<title>` from the original DOM before any transformation.
289///
290/// 'title' is a compile-time-constant selector; `Selector::parse` cannot fail.
291fn extract_original_title(html: &str) -> String {
292    let doc = scraper::Html::parse_document(html);
293    #[allow(clippy::expect_used)]
294    let selector = scraper::Selector::parse("title")
295        // LCOV_EXCL_LINE defensive: compile-time-selector — 'title' is a constant valid selector
296        .expect("invariant: 'title' is a valid CSS selector — this cannot fail");
297    doc.select(&selector)
298        .next()
299        .map(|el| el.text().collect::<String>())
300        .unwrap_or_default()
301}

303/// Clean HTML boilerplate (scripts, styles, nav, sidebar, footer) BEFORE
304/// Readability, logging the reduction.
305fn clean_html_for_scrape(html: &str) -> String {
306    let cleaned_html = crate::infrastructure::converter::html_cleaner::clean_html(html);
307    debug!(
308        "🧹 Cleaned HTML: {} → {} bytes ({}% reduction)",
309        html.len(),
310        cleaned_html.len(),
311        ((html.len() - cleaned_html.len()) as f64 / html.len() as f64 * 100.0).round()
312    );
313    cleaned_html
314}

316/// Build a [`ScrapedContent`] from the fetched HTML, trying Readability first
317/// and falling back to plain-text extraction.
318async fn build_scraped_content(
319    html: &str,
320    extraction_html: &str,
321    url: &url::Url,
322    config: &ScraperConfig,
323    downloader: Option<&dyn crate::domain::ports::AssetDownloaderPort>,
324    original_title: &str,
325    correlation: &CorrelationId,
326) -> Result<ScrapedContent> {
327    match crate::infrastructure::scraper::readability::parse(extraction_html, Some(url.as_str())) {
328        Ok(article) => {
329            let assets = download_assets_if_enabled(
330                html,
331                url,
332                config,
333                downloader.map(|d| d as &dyn crate::domain::ports::AssetDownloaderPort),
334            )
335            .await?;
336
337            log_spa_warning(url.as_str(), &article.text_content, html);
338
339            let author = crate::infrastructure::scraper::author_extractor::extract_author(
340                html,
341                article.byline.as_deref(),
342            );
343
344            Ok(ScrapedContent {
345                // H1 FIX: Use title from original DOM, falling back to Readability's title
346                title: crate::application::resolve_title(
347                    if original_title.is_empty() {
348                        &article.title
349                    } else {
350                        original_title
351                    },
352                    url,
353                ),
354                content: article.text_content,
355                url: ValidUrl::new(url.clone()),
356                excerpt: article.excerpt,
357                author,
358                date: article.published_time,
359                // Store CLEAN HTML from Readability (not raw HTML with nav/ads/footer)
360                // This is what downstream Markdown converters receive.
361                html: Some(article.content),
362                assets,
363                correlation_id: Some(correlation.clone()),
364            })
365        },
366        Err(e) => {
367            warn!("⚠️  Readability failed for {}: {}", url, e);
368            // H2 FIX: Apply clean_html to fallback content to prevent JS/CSS leakage
369            let raw_fallback =
370                crate::infrastructure::scraper::fallback::extract_text(extraction_html);
371            let fallback_content =
372                crate::infrastructure::converter::html_cleaner::clean_html(&raw_fallback);
373            let assets = download_assets_if_enabled(
374                html,
375                url,
376                config,
377                downloader.map(|d| d as &dyn crate::domain::ports::AssetDownloaderPort),
378            )
379            .await?;
380
381            log_spa_warning(url.as_str(), &fallback_content, html);
382
383            let fallback_title = url.host_str().unwrap_or("unknown_host").to_string();
384            Ok(ScrapedContent {
385                // H1 FIX: Use title from original DOM, falling back to host-based fallback
386                title: crate::application::resolve_title(
387                    if original_title.is_empty() {
388                        &fallback_title
389                    } else {
390                        original_title
391                    },
392                    url,
393                ),
394                content: fallback_content,
395                url: ValidUrl::new(url.clone()),
396                excerpt: None,
397                author: None,
398                date: None,
399                html: Some(html.to_string()),
400                assets,
401                correlation_id: Some(correlation.clone()),
402            })
403        },
404    }
405}

407/// Log a warning when extracted content is minimal (possible SPA requiring JS).
408fn log_spa_warning(url: &str, content: &str, html: &str) {
409    // SPA detection: check if extracted content is minimal
410    if let Some(spa_info) = detect_spa_content(url, content, html) {
411        if spa_info.has_spa_markers {
412            warn!(
413                "{} returned minimal content ({} chars) with SPA markers detected. This site may require JavaScript rendering. This feature is not yet implemented. Track: https://github.com/XaviCode1000/webfang/issues/16",
414                spa_info.url, spa_info.char_count
415            );
416        } else {
417            warn!(
418                "{} returned minimal content ({} chars). This site may require JavaScript rendering. This feature is not yet implemented. Track: https://github.com/XaviCode1000/webfang/issues/16",
419                spa_info.url, spa_info.char_count
420            );
421        }
422    }
423}

425/// Scrape multiple URLs with concurrency control
426///
427/// Uses `buffer_unordered` to limit concurrent requests, preventing:
428/// - File descriptor exhaustion
429/// - HDD thrashing (for systems with mechanical drives)
430/// - Anti-bot detection (DDoS-like patterns)
431///
432/// Following **config-externalize**: Concurrency is configurable via ScraperConfig.
433/// Following **async-concurrency-limit**: Uses buffer_unordered for concurrency control.
434///
435/// # Arguments
436/// * `client` - HTTP client
437/// * `urls` - URLs to scrape
438/// * `config` - Scraper configuration
439///
440/// # Returns
441/// * `Vec<ScrapedContent>` - All successfully scraped content
442///
443/// # Note
444/// Failed URLs are logged but don't stop the entire batch.
445#[instrument(
446    name = "scrape_multiple_with_limit",
447    skip(client, urls, config, downloader),
448    fields(
449        urls = urls.len(),
450        concurrency = config.scraper_concurrency
451    )
452)]
453pub async fn scrape_multiple_with_limit(
454    client: &dyn HttpClientPort,
455    urls: &[url::Url],
456    config: &ScraperConfig,
457    downloader: Option<&dyn crate::domain::ports::AssetDownloaderPort>,
458) -> Result<Vec<ScrapedContent>> {
459    if urls.is_empty() {
460        return Ok(Vec::new());
461    }
462
463    // One run-root identity for the whole batch (#501): every
464    // `scrape_with_config` call derives its own child from it, so the trace
465    // is shared across the batch while each page span stays unique.
466    let root_correlation = CorrelationId::new();
467    info!(
468        correlation_id = %root_correlation,
469        trace_id = %root_correlation.trace_id(),
470        "scrape_multiple identity"
471    );
472
473    info!(
474        "🌐 Scraping {} URLs with concurrency limit {}",
475        urls.len(),
476        config.scraper_concurrency
477    );
478
479    let results: Vec<Result<ScrapeOutcome>> =
480        stream::iter(urls.to_vec())
481            .map(|url| {
482                let config = config.clone();
483                let root = root_correlation.clone();
484                async move {
485                    scrape_with_config(client, &url, &config, downloader, None, None, &root).await
486                }
487            })
488            .buffer_unordered(config.scraper_concurrency)
489            .collect()
490            .await;
491
492    let mut all_content = Vec::new();
493    for result in results {
494        match result {
495            Ok(outcome) => all_content.extend(outcome.results),
496            Err(e) => warn!("⚠️  Failed to scrape URL: {}", e),
497        }
498    }
499
500    info!(
501        "✅ Scraped {} pages from {} URLs",
502        all_content.len(),
503        urls.len()
504    );
505    Ok(all_content)
506}
