27/// HTTP client wrapper with configurable retry behavior
28///
29/// Wraps `wreq::Client` and adds:
30/// - Custom headers from config
31/// - Status-specific retry logic
32/// - User-agent rotation on 403
33/// - Exponential backoff on 429 and 5xx
34/// - Request timeout configuration
35/// - Rate limiting
36/// - URL validation
37/// - TLS fingerprint rotation
38/// - Request timeout configuration
39/// - Rate limiting
40/// - URL validation
41/// - TLS fingerprint rotation
42pub struct HttpClient {
43    /// Internal wreq client
44    client: Client,
45    /// Configuration for headers and retry
46    config: HttpClientConfig,
47    /// Pool of user agents for rotation
48    user_agents: Vec<String>,
49    /// Rate limiter for requests per minute
50    rate_limiter: Option<RateLimiter<NotKeyed, InMemoryState, DefaultClock>>,
51    /// Per-domain session health pool (optional)
52    session_pool: Option<Arc<dyn SessionPort>>,
53}
54
55impl HttpClient {
56    /// Create a new HTTP client with the given configuration
57    ///
58    /// # Errors
59    ///
60    /// Returns `ScraperError::Config` if client creation fails
61    pub fn new(config: HttpClientConfig) -> Result<Self, ScraperError> {
62        let client = build_wreq_client(&config, None)?;
63
64        let mut user_agents = UserAgentCache::fallback_agents();
65        if let Some(ref ua) = config.user_agent {
66            user_agents.insert(0, ua.clone());
67        }
68
69        // Create rate limiter if configured
70        let rate_limiter = if let Some(rpm) = config.rate_limit_rpm {
71            if rpm == 0 {
72                return Err(ScraperError::Config(
73                    "rate_limit_rpm must be greater than 0".into(),
74                ));
75            }
76            // rpm > 0 is guaranteed by the early-return check above, so
77            // `NonZeroU32::new` cannot fail — this is a proven invariant.
78            #[allow(clippy::expect_used)]
79            let quota =
80                // LCOV_EXCL_LINE defensive: nonzero-invariant — rpm > 0 is proven by the early-return check above
81                Quota::per_minute(NonZeroU32::new(rpm).expect(
82                    "invariant: rpm > 0 was already checked above — NonZeroU32 cannot fail",
83                ));
84            Some(RateLimiter::direct(quota))
85        } else {
86            None
87        };
88
89        debug!("HttpClient created with {} user agents", user_agents.len());
90        if let Some(rpm) = config.rate_limit_rpm {
91            debug!("Rate limiter configured: {} requests per minute", rpm);
92        }
93
94        Ok(Self {
95            client,
96            config,
97            user_agents,
98            rate_limiter,
99            session_pool: None,
100        })
101    }

118    /// Perform GET request with retry logic
119    ///
120    /// Handles status codes as follows:
121    /// - 200-299: Returns body as String
122    /// - 403: Logs + retries once with rotated user-agent
123    /// - 429: Exponential backoff respecting Retry-After header
124    /// - 500-599: Exponential backoff with automatic retry
125    ///
126    /// # Errors
127    ///
128    /// Returns `HttpError` for failed requests or invalid URLs
129    pub async fn get(&self, url: &str) -> HttpResult<String> {
130        // Validate URL first
131        let parsed_url =
132            Url::parse(url).map_err(|e| HttpError::Request(format!("Invalid URL: {e}")))?;
133
134        // Ensure URL has http or https scheme
135        if !matches!(parsed_url.scheme(), "http" | "https") {
136            return Err(HttpError::Request(
137                "URL must use http or https scheme".into(),
138            ));
139        }
140
141        // Session pool gating: acquire before network request
142        let (session_id, domain) = if let Some(ref pool) = self.session_pool {
143            let domain = crate::application::url_filter::extract_domain(url)
144                .unwrap_or_else(|| parsed_url.host_str().unwrap_or("unknown").to_string());
145            match pool.acquire(&domain) {
146                Some(id) => (Some(id), domain),
147                None => return Err(HttpError::DomainBanned(domain)),
148            }
149        } else {
150            (None, String::new())
151        };
152        // Apply rate limiting if configured
153        if let Some(ref limiter) = self.rate_limiter {
154            limiter.until_ready().await;
155        }
156
157        // Track in-flight requests
158
159        let result = self.get_inner(url).await;
160
161        // Report outcome to session pool
162        self.report_outcome(&domain, session_id, &result);
163
164        result
165    }
166
167    /// Report request outcome to the session pool for health tracking.
168    fn report_outcome(
169        &self,
170        domain: &str,
171        session_id: Option<SessionId>,
172        result: &HttpResult<String>,
173    ) {
174        let (Some(pool), Some(id)) = (self.session_pool.as_ref(), session_id) else {
175            return;
176        };
177        match result {
178            Ok(_) => pool.report_success(domain, id),
179            Err(e) => {
180                let status = match e {
181                    HttpError::WafChallenge(_) | HttpError::Forbidden => 403,
182                    HttpError::RateLimited(_) => 429,
183                    HttpError::Timeout => 504,
184                    HttpError::Connection(_) => 503,
185                    HttpError::ClientError(404) => return, // don't report 404
186                    HttpError::ClientError(code) => *code,
187                    HttpError::ServerError(code) => *code,
188                    HttpError::Request(_) => return,
189                    HttpError::DomainBanned(_) => return,
190                };
191                pool.report_failure(domain, id, status);
192            },
193        }
194    }

196    async fn get_inner(&self, url: &str) -> HttpResult<String> {
197        let mut ua_index = 0;
198
199        loop {
200            if ua_index >= self.user_agents.len() && ua_index > 0 {
201                return Err(HttpError::Forbidden);
202            }
203
204            let ua = self.select_user_agent(ua_index);
205            let request = self.build_request(url, &ua, ua_index);
206
207            let response = request.send().await.map_err(map_send_error)?;
208
209            let status = response.status();
210
211            match status.as_u16() {
212                200..=299 => return self.handle_success(url, response, status.as_u16()).await,
213                403 => {
214                    warn!("403 Forbidden from {}", url);
215                    if ua_index == 0 {
216                        ua_index += 1;
217                        continue;
218                    }
219                    return Err(HttpError::Forbidden);
220                },
221                429 => return self.handle_rate_limited(url, response, &ua).await,
222                500..=599 => {
223                    return self
224                        .handle_server_error(url, response, status.as_u16(), &ua)
225                        .await
226                },
227                code if (400..=499).contains(&code) => {
228                    return Err(HttpError::ClientError(code));
229                },
230                code => {
231                    return Err(HttpError::ServerError(code));
232                },
233            }
234        }
235    }

237    /// Select the user-agent to use for a given rotation index.
238    fn select_user_agent(&self, ua_index: usize) -> String {
239        self.user_agents
240            .get(ua_index % self.user_agents.len())
241            .cloned()
242            .unwrap_or_else(|| {
243                "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36".into()
244            })
245    }

247    /// Build the GET request for `url`, applying the configured headers.
248    ///
249    /// Uses minimal headers for WAF-bypass attempts (`ua_index >= 4`).
250    fn build_request(&self, url: &str, ua: &str, ua_index: usize) -> wreq::RequestBuilder {
251        let mut request = self
252            .client
253            .get(url)
254            .header("Accept-Language", &self.config.accept_language)
255            .header("Accept", &self.config.accept)
256            .header("User-Agent", ua);
257
258        if ua_index < 4 {
259            request = request
260                .header("Referer", &self.config.referer)
261                .header("Cache-Control", &self.config.cache_control);
262        }
263
264        request
265    }

267    /// Handle a 2xx response: read the body and run context-aware WAF inspection.
268    async fn handle_success(
269        &self,
270        url: &str,
271        response: wreq::Response,
272        status: u16,
273    ) -> HttpResult<String> {
274        // Build the inspection context BEFORE consuming the body
275        // (headers borrow must end before `.text()`).
276        let ctx = inspection_context(status, response.headers(), self.config.ignore_waf);
277
278        let body = response
279            .text()
280            .await
281            .map_err(|e| HttpError::Request(e.to_string()))?;
282
283        // Context-aware WAF inspection (REQ-WAF-05). A classified block returns
284        // immediately — the old fallback ladder (~4 requests / ~5s of UA rotation)
285        // is gone: with tiered, evidence-based detection a block is a genuine
286        // challenge that UA rotation cannot bypass, and false positives no longer
287        // reach this branch.
288        if let Some(err) = waf_block_error(url, status, &body, &ctx) {
289            return Err(err);
290        }
291
292        Ok(body)
293    }

295    /// Handle a 429 response: honor `Retry-After` and retry with backoff.
296    async fn handle_rate_limited(
297        &self,
298        url: &str,
299        response: wreq::Response,
300        ua: &str,
301    ) -> HttpResult<String> {
302        let retry_after = response
303            .headers()
304            .get("retry-after")
305            .and_then(|v| v.to_str().ok())
306            .and_then(|s| s.parse().ok())
307            .unwrap_or(1);
308
309        debug!("429 Rate Limited, retry after {}s", retry_after);
310
311        retry_with_backoff(
312            &self.client,
313            url,
314            &self.config,
315            ua,
316            RetryPolicy {
317                retry_after_secs: Some(retry_after),
318                retryable: |code: u16| code == 429 || (500..=599).contains(&code),
319                exhausted: HttpError::RateLimited(retry_after),
320                label: "429",
321            },
322        )
323        .await
324    }
325
326    /// Handle a 5xx response: inspect for WAF challenges, then retry with backoff.
327    async fn handle_server_error(
328        &self,
329        url: &str,
330        response: wreq::Response,
331        status: u16,
332        ua: &str,
333    ) -> HttpResult<String> {
334        debug!("{} from {}", status, url);
335
336        // Inspect the initial response BEFORE retrying (REQ-WAF-05).
337        // Approved behavior change: a 503 Cloudflare challenge is classified as
338        // a WAF block instead of dying as a generic ServerError after exhausting
339        // retries.
340        let ctx = inspection_context(status, response.headers(), self.config.ignore_waf);
341
342        // Non-critical enrichment (FIX C): a body-transfer failure (connection
343        // reset mid-body) must not bypass the retry loop with a fatal Request
344        // error — the base never read the 5xx body. Fall back to an empty body so
345        // inspection proceeds on headers only (cf-mitigated detection still
346        // works) and the retry loop runs. The 2xx branch keeps its fatal body
347        // read: there the body IS the scrape content.
348        let body = match response.text().await {
349            Ok(b) => b,
350            Err(e) => {
351                debug!(url = %url, error = %e, "5xx body read failed; inspecting headers only");
352                String::new()
353            },
354        };
355
356        if let Some(err) = waf_block_error(url, status, &body, &ctx) {
357            return Err(err);
358        }
359
360        retry_with_backoff(
361            &self.client,
362            url,
363            &self.config,
364            ua,
365            RetryPolicy {
366                retry_after_secs: None,
367                retryable: |code: u16| (500..=599).contains(&code),
368                exhausted: HttpError::ServerError(status),
369                label: "5xx",
370            },
371        )
372        .await
373    }
374}

376/// Map a transport error to the corresponding [`HttpError`].
377fn map_send_error(e: wreq::Error) -> HttpError {
378    if e.is_timeout() {
379        HttpError::Timeout
380    } else if e.is_connect() {
381        HttpError::Connection(e.to_string())
382    } else {
383        HttpError::Request(e.to_string())
384    }
385}

387/// Return a WAF-challenge error when the body is classified as blocked.
388fn waf_block_error(
389    url: &str,
390    status: u16,
391    body: &str,
392    ctx: &InspectionContext,
393) -> Option<HttpError> {
394    let verdict = WafInspector::inspect(body, ctx);
395    if verdict.is_blocked {
396        warn!(
397            url = %url,
398            status = %status,
399            evidences = verdict.evidences.len(),
400            "WAF/CAPTCHA challenge detected; blocking"
401        );
402        Some(HttpError::WafChallenge(verdict.evidence_chain()))
403    } else {
404        None
405    }
406}

408/// Build a WAF [`InspectionContext`] from an HTTP response's status and headers.
409///
410/// Extracts the content-type for the REQ-WAF-02 gate and clones the headers for
411/// control-header evidence (REQ-WAF-03). `ignore_waf` short-circuits inspection
412/// to a clean verdict (REQ-WAF-07). The header borrow ends when this returns
413/// (the map is cloned), so the caller can still consume the response body.
414fn inspection_context(status: u16, headers: &HeaderMap, ignore_waf: bool) -> InspectionContext {
415    InspectionContext {
416        status: Some(status),
417        content_type: headers
418            .get("content-type")
419            .and_then(|v| v.to_str().ok())
420            .map(String::from),
421        headers: headers.clone(),
422        ignore_waf,
423    }
424}

430/// Raw, single-shot [`HttpClientPort`] implementation for [`HttpClient`].
431///
432/// # Why this path is deliberately raw
433///
434/// [`HttpClient`] exposes two request paths with different contracts, and they
435/// are intentionally **not** unified:
436///
437/// - The hardened path — `HttpClient::get` / `get_inner` — returns only the
438///   response body as a `String`, after applying rate limiting, per-domain
439///   session gating, status-specific retries (429 / 5xx), user-agent rotation
440///   on 403, and WAF/CAPTCHA inspection. Non-2xx outcomes are surfaced as
441///   [`HttpError`] variants.
442/// - This port implementation returns a full [`HttpResponse`]
443///   (`status` + `headers` + `body`) for a **single** GET with no retry, no
444///   user-agent rotation and no WAF inspection, reporting non-2xx as data in
445///   `status` rather than as an error. It exists for callers that depend on the
446///   domain port and manage their own request lifecycle.
447///
448/// Routing this impl through the hardened path would change its observable
449/// behavior — it would start retrying, rotating user agents, and turning
450/// 5xx/429 responses into [`HttpError`]s instead of returning the status — so
451/// the divergence is preserved and documented here rather than collapsed
452/// (see #444). The bare-client factories (e.g. [`create_http_client`]) are raw
453/// for the same reason.
454///
455/// [`HttpClientPort`]: crate::domain::http_port::HttpClientPort
456/// [`HttpClient`]: crate::application::http_client::HttpClient
457/// [`HttpResponse`]: crate::domain::http_port::HttpResponse
458/// [`HttpError`]: crate::domain::http_error::HttpError
459/// [`create_http_client`]: crate::application::http_client::create_http_client
460impl crate::domain::http_port::HttpClientPort for HttpClient {
461    fn get(
462        &self,
463        url: &str,
464    ) -> std::pin::Pin<
465        Box<
466            dyn std::future::Future<
467                    Output = crate::domain::http_error::HttpResult<
468                        crate::domain::http_port::HttpResponse,
469                    >,
470                > + Send
471                + '_,
472        >,
473    > {
474        let url = url.to_owned();
475        Box::pin(async move {
476            // Use the inner wreq client for the raw request, then build
477            // a full HttpResponse (status + body + headers).
478            let resp = self.client.get(url.as_str()).send().await.map_err(|e| {
479                if e.is_timeout() {
480                    crate::domain::http_error::HttpError::Timeout
481                } else if e.is_connect() {
482                    crate::domain::http_error::HttpError::Connection(e.to_string())
483                } else {
484                    crate::domain::http_error::HttpError::Request(e.to_string())
485                }
486            })?;
487
488            let status = resp.status().as_u16();
489            let mut headers = std::collections::HashMap::new();
490            for (key, value) in resp.headers() {
491                if let Ok(v) = value.to_str() {
492                    headers.insert(key.as_str().to_lowercase(), v.to_string());
493                }
494            }
495            let body = resp
496                .text()
497                .await
498                .map_err(|e| crate::domain::http_error::HttpError::Request(e.to_string()))?;
499
500            Ok(crate::domain::http_port::HttpResponse {
501                status,
502                body,
503                headers,
504            })
505        })
506    }
507}

1113    /// Mock SessionPort for testing.
1114    struct MockSessionPort {
1115        /// Controls what acquire returns: Some(SessionId(0)) or None (banned).
1116        banned: std::sync::atomic::AtomicBool,
1117        /// Counts calls to report_success and report_failure.
1118        success_count: AtomicUsize,
1119        failure_count: AtomicUsize,
1120        last_failure_status: std::sync::atomic::AtomicU16,
1121    }
1122
1123    impl MockSessionPort {
1124        fn new(banned: bool) -> Arc<Self> {
1125            Arc::new(Self {
1126                banned: std::sync::atomic::AtomicBool::new(banned),
1127                success_count: AtomicUsize::new(0),
1128                failure_count: AtomicUsize::new(0),
1129                last_failure_status: std::sync::atomic::AtomicU16::new(0),
1130            })
1131        }
1132    }
1133
1134    impl SessionPort for MockSessionPort {
1135        fn acquire(&self, _domain: &str) -> Option<SessionId> {
1136            if self.banned.load(Ordering::SeqCst) {
1137                None
1138            } else {
1139                Some(SessionId(0))
1140            }
1141        }
1142
1143        fn report_success(&self, _domain: &str, _session: SessionId) {
1144            self.success_count.fetch_add(1, Ordering::SeqCst);
1145        }
1146
1147        fn report_failure(&self, _domain: &str, _session: SessionId, status: u16) {
1148            self.failure_count.fetch_add(1, Ordering::SeqCst);
1149            self.last_failure_status.store(status, Ordering::SeqCst);
1150        }
1151    }
