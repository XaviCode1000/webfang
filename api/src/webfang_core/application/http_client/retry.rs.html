1//! Status-specific retry loops for the hardened HTTP client.
2//!
3//! The hardened client's `get_inner` (see [`HttpClient`]) handles a request's
4//! first response inline; when that response is a `429` or a `5xx`, it
5//! delegates the bounded retry loop to [`retry_with_backoff`] in this module.
6//!
7//! The two historical loops (one for `429`, one for `5xx`) were near-identical
8//! but differed in four load-bearing ways, captured by [`RetryPolicy`].
9//! [`retry_with_backoff`] reproduces both exactly by varying only that policy:
10//!
11//! | Difference | `429` path | `5xx` path |
12//! |---|---|---|
13//! | Delay strategy | honors `Retry-After` (constant `secs * 1000` ms) when `> 0`, else exponential | always exponential backoff |
14//! | Retryable status | `429` **or** any `5xx` | any `5xx` only |
15//! | Exhaustion error | [`HttpError::RateLimited`] | [`HttpError::ServerError`] |
16//! | Tracing label | `"429"` | `"5xx"` |
17//!
18//! Everything else — the attempt counter, request headers, success
19//! short-circuit, non-retryable `ClientError`, and timeout handling — is
20//! identical and lives here once. The 5xx path's pre-loop WAF inspection stays
21//! in `get_inner` because it runs on the initial response, before any retry.
22//!
23//! [`HttpClient`]: crate::application::http_client::HttpClient
24//! [`HttpError::RateLimited`]: crate::domain::http_error::HttpError::RateLimited
25//! [`HttpError::ServerError`]: crate::domain::http_error::HttpError::ServerError
26
27use crate::domain::http_config::HttpClientConfig;
28use crate::domain::http_error::{HttpError, HttpResult};
29use std::time::Duration;
30use tracing::debug;
31use wreq::Client;

33/// The four load-bearing differences between the `429` and `5xx` retry loops.
34///
35/// Every other aspect of the two loops is identical and lives in
36/// [`retry_with_backoff`]; this policy carries only what varies between them so
37/// the shared loop can reproduce both behaviors exactly.
38pub(super) struct RetryPolicy {
39    /// `Retry-After` seconds honored by the `429` path (`Some`); `None` for the
40    /// `5xx` path, which always uses exponential backoff.
41    pub(super) retry_after_secs: Option<u64>,
42    /// Predicate deciding whether a received status code triggers another
43    /// retry: the `429` path retries on `429` **or** any `5xx`, the `5xx` path
44    /// retries on any `5xx` only.
45    pub(super) retryable: fn(u16) -> bool,
46    /// Error returned once `max_retries` attempts are exhausted:
47    /// [`HttpError::RateLimited`] for the `429` path, [`HttpError::ServerError`]
48    /// for the `5xx` path.
49    ///
50    /// [`HttpError::RateLimited`]: crate::domain::http_error::HttpError::RateLimited
51    /// [`HttpError::ServerError`]: crate::domain::http_error::HttpError::ServerError
52    pub(super) exhausted: HttpError,
53    /// Label tagging the tracing debug messages (`"429"` / `"5xx"`).
54    pub(super) label: &'static str,
55}
56
57/// Run the bounded retry loop shared by the `429` and `5xx` handling paths.
58///
59/// Performs up to `config.max_retries` attempts. Each attempt sleeps for
60/// [`compute_backoff_delay`], re-sends the GET with the same headers the
61/// hardened client uses (Accept-Language, Accept, Referer, Cache-Control and
62/// the pinned `user_agent`), then inspects the outcome:
63///
64/// - a success status returns the body immediately;
65/// - a status satisfying `policy.retryable` continues to the next attempt;
66/// - any other status returns [`HttpError::ClientError`];
67/// - a transport error returns [`HttpError::Timeout`] when it is a timeout and
68///   otherwise continues to the next attempt.
69///
70/// When every attempt is exhausted, `policy.exhausted` is returned. The `429`
71/// caller passes a policy with `Some(retry_after)` and
72/// [`HttpError::RateLimited`]; the `5xx` caller passes `None` (pure exponential
73/// backoff) and [`HttpError::ServerError`].
74///
75/// # Errors
76///
77/// Returns the body on success, or the first terminal [`HttpError`] (a client
78/// error, a timeout, or `policy.exhausted` once retries run out).
79///
80/// [`HttpError::ClientError`]: crate::domain::http_error::HttpError::ClientError
81/// [`HttpError::Timeout`]: crate::domain::http_error::HttpError::Timeout
82/// [`HttpError::RateLimited`]: crate::domain::http_error::HttpError::RateLimited
83/// [`HttpError::ServerError`]: crate::domain::http_error::HttpError::ServerError
84pub(super) async fn retry_with_backoff(
85    client: &Client,
86    url: &str,
87    config: &HttpClientConfig,
88    user_agent: &str,
89    policy: RetryPolicy,
90) -> HttpResult<String> {
91    let mut attempt = 0;
92    while attempt < config.max_retries {
93        attempt += 1;
94
95        let delay_ms = compute_backoff_delay(
96            attempt,
97            policy.retry_after_secs,
98            config.backoff_base_ms,
99            config.backoff_max_ms,
100        );
101
102        debug!(
103            "{} retry attempt {} after {}ms",
104            policy.label, attempt, delay_ms
105        );
106        tokio::time::sleep(Duration::from_millis(delay_ms)).await;
107
108        let request = client
109            .get(url)
110            .header("Accept-Language", &config.accept_language)
111            .header("Accept", &config.accept)
112            .header("Referer", &config.referer)
113            .header("Cache-Control", &config.cache_control)
114            .header("User-Agent", user_agent);
115
116        match request.send().await {
117            Ok(resp) => {
118                if resp.status().is_success() {
119                    return resp
120                        .text()
121                        .await
122                        .map_err(|e| HttpError::Request(e.to_string()));
123                } else if (policy.retryable)(resp.status().as_u16()) {
124                    continue;
125                } else {
126                    return Err(HttpError::ClientError(resp.status().as_u16()));
127                }
128            },
129            Err(e) => {
130                if e.is_timeout() {
131                    return Err(HttpError::Timeout);
132                }
133                continue;
134            },
135        }
136    }
137    Err(policy.exhausted)
138}

140/// Compute the delay in milliseconds before retry attempt `attempt` (1-based).
141///
142/// When `retry_after_secs` is `Some(secs)` with `secs > 0` — the `429` path
143/// honoring a `Retry-After` header — the delay is the constant
144/// `secs * 1000` ms. In every other case (the `5xx` path's `None`, or a `429`
145/// carrying `Retry-After: 0`) the delay is exponential backoff
146/// `base_ms * 2^(attempt - 1)` capped at `max_ms`.
147fn compute_backoff_delay(
148    attempt: u32,
149    retry_after_secs: Option<u64>,
150    base_ms: u64,
151    max_ms: u64,
152) -> u64 {
153    match retry_after_secs {
154        Some(secs) if secs > 0 => secs * 1000,
155        _ => {
156            let exponent = attempt.saturating_sub(1);
157            let delay = base_ms * 2_u64.pow(exponent);
158            delay.min(max_ms)
159        },
160    }
161}

163#[cfg(test)]
164mod tests {
165    use super::*;
166
167    #[test]
168    fn retry_after_honored_as_constant_delay() {
169        // 429 path with Retry-After: 2 → constant 2000ms regardless of attempt.
170        assert_eq!(compute_backoff_delay(1, Some(2), 10, 50), 2000);
171        assert_eq!(compute_backoff_delay(2, Some(2), 10, 50), 2000);
172        assert_eq!(compute_backoff_delay(3, Some(2), 10, 50), 2000);
173    }
174
175    #[test]
176    fn retry_after_zero_falls_back_to_exponential() {
177        // 429 path with Retry-After: 0 → exponential backoff like the 5xx path.
178        assert_eq!(compute_backoff_delay(1, Some(0), 10, 50), 10);
179        assert_eq!(compute_backoff_delay(2, Some(0), 10, 50), 20);
180        assert_eq!(compute_backoff_delay(3, Some(0), 10, 50), 40);
181    }
182
183    #[test]
184    fn no_retry_after_uses_exponential_backoff() {
185        // 5xx path (None) → base * 2^(attempt-1).
186        assert_eq!(compute_backoff_delay(1, None, 10, 10_000), 10);
187        assert_eq!(compute_backoff_delay(2, None, 10, 10_000), 20);
188        assert_eq!(compute_backoff_delay(3, None, 10, 10_000), 40);
189        assert_eq!(compute_backoff_delay(4, None, 10, 10_000), 80);
190    }
191
192    #[test]
193    fn exponential_backoff_is_capped_at_max() {
194        // base 1000, max 10000 (production defaults): 1s, 2s, 4s, 8s, then capped.
195        assert_eq!(compute_backoff_delay(1, None, 1000, 10_000), 1000);
196        assert_eq!(compute_backoff_delay(2, None, 1000, 10_000), 2000);
197        assert_eq!(compute_backoff_delay(3, None, 1000, 10_000), 4000);
198        assert_eq!(compute_backoff_delay(4, None, 1000, 10_000), 8000);
199        assert_eq!(compute_backoff_delay(5, None, 1000, 10_000), 10_000);
200        assert_eq!(compute_backoff_delay(6, None, 1000, 10_000), 10_000);
201    }
202}
