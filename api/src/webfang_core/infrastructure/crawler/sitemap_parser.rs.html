1//! Sitemap Parser Module
2//!
3//! Zero-allocation streaming parser for XML sitemaps.
4//! Supports gzip compression and sitemap index recursion.
5//!
6//! # Examples
7//!
8//! ```no_run
9//! use webfang_core::infrastructure::crawler::SitemapParser;
10//!
11//! # #[tokio::main]
12//! # async fn main() -> Result<(), Box<dyn std::error::Error>> {
13//! let parser = SitemapParser::new()?;
14//! let urls = parser.parse_from_url("https://example.com/sitemap.xml").await?;
15//! println!("Found {} URLs", urls.len());
16//! # Ok(())
17//! # }
18//! ```
19//!
20//! # Errors
21//!
22//! Returns `SitemapError` if:
23//! - URL is invalid
24//! - HTTP request fails
25//! - XML parsing fails
26//! - No `<loc>` elements found

43/// Sitemap parser errors
44///
45/// Following err-thiserror-for-libraries: typed, matchable errors
46#[derive(Debug, Error)]
47pub enum SitemapError {
48    /// URL could not be parsed
49    #[error("invalid URL: {0}")]
50    InvalidUrl(#[from] url::ParseError),
51
52    /// HTTP request to fetch the sitemap failed
53    #[error("http request failed: {0}")]
54    HttpError(String),
55
56    /// XML parsing of sitemap content failed
57    #[error("XML parsing failed: {0}")]
58    XmlError(#[from] quick_xml::Error),
59
60    /// I/O error reading or writing sitemap data
61    #[error("IO error: {0}")]
62    IoError(#[from] std::io::Error),
63
64    /// Sitemap contained no URL entries
65    #[error("no URLs found in sitemap")]
66    NoUrlsFound,
67
68    /// Sitemap XML structure does not match expected format
69    #[error("invalid sitemap structure")]
70    InvalidStructure,
71
72    /// Sitemap index depth exceeded the configured maximum
73    #[error("maximum recursion depth exceeded")]
74    MaxDepthExceeded,
75
76    /// URL scheme is not http or https
77    #[error("invalid scheme: {0} (only http/https allowed)")]
78    InvalidScheme(String),
79
80    /// HTTP response body exceeds the size limit
81    #[error("response too large: exceeds {0} bytes")]
82    ResponseTooLarge(usize),
83
84    /// Decompressed sitemap data exceeds the size limit
85    #[error("decompressed data too large: exceeds {0} bytes")]
86    DecompressedTooLarge(usize),
87
88    /// No sitemap found at the expected URL
89    #[error("no sitemap found at {0}")]
90    SitemapNotFound(String),
91
92    /// Response Content-Type is not XML
93    #[error("invalid content type: expected XML, got {0}")]
94    InvalidContentType(String),
95}

100/// Safely resolve a potentially-relative URL against a base URL.
101///
102/// Handles all RFC 3986 reference types: absolute, scheme-relative,
103/// absolute-path, relative-path. Returns `None` for empty inputs.
104///
105/// Following **err-result-over-panic**: returns Option, never panics.
106/// Following **perf-iter-over-index**: uses early returns, no loops.
107///
108/// # Arguments
109///
110/// * `base` - The base URL for resolution context
111/// * `input` - The URL or path to resolve
112///
113/// # Returns
114///
115/// * `Some(Url)` - Successfully resolved URL
116/// * `None` - Empty input or resolution failure
117///
118/// # Examples
119///
120/// ```
121/// use url::Url;
122/// use webfang_core::infrastructure::crawler::sitemap_parser::resolve_url;
123///
124/// let base = Url::parse("https://example.com/sitemap.xml").unwrap();
125/// let resolved = resolve_url(&base, "/page.html").unwrap();
126/// assert_eq!(resolved.as_str(), "https://example.com/page.html");
127/// ```
128#[must_use]
129pub fn resolve_url(base: &Url, input: &str) -> Option<Url> {
130    let input = input.trim();
131    if input.is_empty() {
132        return None;
133    }
134
135    // Fast path: already absolute
136    if input.starts_with("http://") || input.starts_with("https://") {
137        return Url::parse(input).ok();
138    }
139
140    // Use RFC 3986 resolution via url::Url::join
141    base.join(input).ok()
142}

146/// Following mem-streaming-large-data: streaming parser, no buffer accumulation
147pub struct SitemapParser {

149    compression_handler: CompressionHandler,
150    url_validator: UrlValidator,
151    retry_policy: RetryPolicy,
152    memory_manager: MemoryManager,

155    ///
156    /// Threaded from the caller (ultimately the CLI `--h2-profile` value).

225    /// Build the sitemap fetch HTTP client for a given TLS/H2 profile.
226    ///
227    /// The request and connect timeouts are pinned to 10s to preserve the
228    /// historical behavior of the previous hardcoded client (which set
229    /// `.timeout(10s)`); only the TLS/H2 profile is now configurable (#323).
230    /// The remaining settings (Chrome Client Hints, pool tuning, compression)
231    /// come from the shared factory defaults (#299).

237        tls_emulation: wreq_util::Profile,
238    ) -> std::result::Result<wreq::Client, CrawlError> {
239        let http_config = crate::domain::http_config::HttpClientConfig {
240            tls_emulation,
241            timeout_secs: 10,
242            connect_timeout_secs: 10,
243            ..Default::default()
244        };
245        crate::infrastructure::http::create_http_client_with_config(&http_config)
246            // LCOV_EXCL_LINE defensive: wreq-client-build — client construction fails only on invalid TLS profile, an invariant
247            .map_err(|e| CrawlError::Internal(format!("failed to build sitemap client: {e}")))
248    }

267    /// Internal recursive parser with depth tracking
268    async fn parse_with_depth(&self, url: &str, depth: u8) -> Result<Vec<Url>> {
269        // Base case: max depth reached
270        if depth == 0 {
271            return Err(SitemapError::MaxDepthExceeded);
272        }
273
274        let base_url = Url::parse(url)?;
275
276        // [3.6] RetryPolicy: wrap HTTP request with retry logic.
277        // The client is built once in the constructor (honoring tls_emulation)
278        // and cloned cheaply (Arc-backed) into each attempt — no per-retry rebuild (#323).
279        let response = self
280            .retry_policy
281            .execute_with_retry(|| {
282                let url = url.to_string();
283                let client = self.http_client.clone();
284                async move { client.get(&url).send().await }
285            })
286            .await
287            .map_err(|e| SitemapError::HttpError(e.to_string()))?;
288
289        // Validate content type
290        let content_type = response
291            .headers()
292            .get("content-type")
293            .map(|v| v.to_str().unwrap_or(""))
294            .unwrap_or("");
295
296        let is_xml = content_type.is_empty()
297            || content_type.contains("application/xml")
298            || content_type.contains("text/xml")
299            || content_type.contains("application/xhtml+xml")
300            || url.ends_with(".xml")
301            || url.ends_with(".xml.gz");
302
303        if !is_xml {
304            tracing::warn!(
305                "Sitemap URL returned non-XML content type: {} from {}",
306                content_type,
307                url
308            );
309            return Err(SitemapError::InvalidContentType(content_type.to_string()));
310        }
311
312        // Stream response with size limit
313        use futures::StreamExt;
314        let mut stream = response.bytes_stream();
315        let mut raw_bytes = Vec::with_capacity(8192);
316        let mut total_bytes = 0usize;
317        while let Some(chunk) = stream.next().await {
318            let chunk = chunk.map_err(|e| SitemapError::HttpError(e.to_string()))?;
319            total_bytes += chunk.len();
320            if total_bytes > self.config.max_response_size {
321                tracing::warn!(
322                    "Sitemap response too large: {} bytes from {}",
323                    total_bytes,
324                    url
325                );
326                return Err(SitemapError::ResponseTooLarge(
327                    self.config.max_response_size,
328                ));
329            }
330            raw_bytes.extend_from_slice(&chunk);
331        }
332
333        // [3.4] CompressionHandler integration: detect and decompress content
334        let decompressed = self
335            .compression_handler
336            .detect_and_decompress(&raw_bytes, url)
337            .await
338            .map_err(|e| SitemapError::HttpError(e.to_string()))?;
339
340        // Parse using unified decompression handle
341        let urls = if decompressed.is_empty() {
342            return Err(SitemapError::NoUrlsFound);
343        } else {
344            self.parse_xml_sitemap(&decompressed, &base_url).await?
345        };
346
347        // Check if sitemap index (recursive)
348        if self.is_sitemap_index(&urls) {
349            tracing::debug!("Detected sitemap index, recursing (depth: {})", depth);
350
351            // [3.7] MemoryManager: handle disk swapping for large index
352            self.memory_manager
353                .handle_disk_swapping(&urls)
354                .map_err(|e| SitemapError::HttpError(e.to_string()))?;
355
356            self.parse_sitemap_index(&urls, depth - 1).await
357        } else {
358            // [3.7] MemoryManager: check memory limits before returning
359            self.memory_manager
360                .handle_disk_swapping(&urls)
361                .map_err(|e| SitemapError::HttpError(e.to_string()))?;
362
363            // [3.8] BatchProcessor: apply crawl budget optimization
364            let optimized_urls = self.batch_processor.apply_crawl_budget(urls, &self.config);
365
366            Ok(optimized_urls)
367        }
368    }

370    /// Parse gzip-compressed sitemap
371    #[allow(dead_code)]
372    async fn parse_gzip_sitemap(&self, bytes: &[u8], base_url: &Url) -> Result<Vec<Url>> {
373        use tokio::io::{AsyncReadExt, BufReader};
374        let reader = BufReader::new(bytes);
375        let mut decoder = GzipDecoder::new(reader);
376
377        let mut limited =
378            AsyncReadExt::take(&mut decoder, self.config.max_decompressed_size as u64);
379        let mut decompressed = Vec::new();
380        AsyncReadExt::read_to_end(&mut limited, &mut decompressed).await?;
381
382        if decompressed.len() >= self.config.max_decompressed_size {
383            tracing::warn!(
384                "Gzip decompression hit size limit ({} bytes) — possible decompression bomb",
385                decompressed.len()
386            );
387            return Err(SitemapError::DecompressedTooLarge(
388                self.config.max_decompressed_size,
389            ));
390        }
391
392        self.parse_xml_sitemap(&decompressed, base_url).await
393    }

395    /// Parse XML sitemap (zero-allocation streaming)
396    async fn parse_xml_sitemap(&self, bytes: &[u8], base_url: &Url) -> Result<Vec<Url>> {
397        let mut reader = Reader::from_reader(bytes);
398
399        let mut urls = HashSet::new();
400        let mut buf = Vec::new();
401        let mut in_loc = false;
402
403        loop {
404            match reader.read_event_into(&mut buf) {
405                Ok(Event::Start(ref e)) if e.name().as_ref() == b"loc" => {
406                    in_loc = true;
407                },
408                Ok(Event::Text(ref e)) if in_loc => {
409                    if let Ok(text) = e.decode() {
410                        if let Some(url) = resolve_url(base_url, &text) {
411                            // [3.5] UrlValidator integration: filter invalid patterns
412                            let validation = self.url_validator.filter_invalid_patterns(&url);
413                            match validation {
414                                crate::domain::ValidationResult::Valid => {
415                                    urls.insert(url);
416                                },
417                                crate::domain::ValidationResult::Invalid(reason) => {
418                                    tracing::debug!("Filtered invalid URL: {} — {}", url, reason);
419                                },
420                                crate::domain::ValidationResult::NeedsRedirect(new_url) => {
421                                    // Follow redirect by replacing URL
422                                    urls.insert(new_url);
423                                },
424                            }
425                        }
426                    }
427                },
428                Ok(Event::End(ref e)) if e.name().as_ref() == b"loc" => {
429                    in_loc = false;
430                },
431                Ok(Event::Eof) => break,
432                Err(e) => return Err(SitemapError::XmlError(e)),
433                _ => {},
434            }
435            buf.clear();
436        }
437
438        if urls.is_empty() {
439            Err(SitemapError::NoUrlsFound)
440        } else {
441            Ok(urls.into_iter().collect())
442        }
443    }

451    /// Parse sitemap index recursively
452    async fn parse_sitemap_index(&self, sitemap_urls: &[Url], depth: u8) -> Result<Vec<Url>> {
453        use futures::stream::{self, StreamExt};
454
455        let mut all_urls = HashSet::new();
456
457        let results = stream::iter(sitemap_urls.iter().cloned())
458            .map(|url| async move { self.parse_with_depth(url.as_str(), depth).await })
459            .buffered(self.config.concurrency)
460            .collect::<Vec<_>>()
461            .await;
462
463        for result in results {
464            match result {
465                Ok(urls) => all_urls.extend(urls),
466                Err(e) => tracing::warn!("Failed to parse sitemap: {}", e),
467            }
468        }
469
470        if all_urls.is_empty() {
471            Err(SitemapError::NoUrlsFound)
472        } else {
473            Ok(all_urls.into_iter().collect())
474        }
475    }

496/// Parse sitemap XML content using quick-xml (streaming parser)
497///
498/// Standalone streaming parser for sitemap `<loc>` entries, extracted from the
499/// application layer (issue #442): XML parsing is an infrastructure concern and
500/// lives here alongside [`SitemapParser`]. Relative URLs are resolved against
501/// `base_url`.
502///
503/// Following **xml-no-regex**: Uses quick-xml instead of regex for XML parsing.
504/// Following **mem-stream-processing**: Streaming approach avoids loading entire DOM.
505///
506/// # Arguments
507///
508/// * `xml_content` - XML content of the sitemap
509/// * `base_url` - Base URL to resolve relative `<loc>` entries against
510///
511/// # Returns
512///
513/// * `Ok(Vec<String>)` - List of URLs
514/// * `Err(CrawlError)` - Parse error
515pub fn parse_sitemap(
516    xml_content: &str,
517    base_url: &Url,
518) -> std::result::Result<Vec<String>, CrawlError> {
519    let mut reader = Reader::from_str(xml_content);
520    let mut buf = Vec::new();
521    let mut urls = Vec::new();
522    let mut in_loc = false;
523
524    loop {
525        buf.clear();
526        match reader.read_event_into(&mut buf) {
527            Ok(Event::Start(ref e) | Event::Empty(ref e)) if e.name().as_ref() == b"loc" => {
528                in_loc = true;
529            },
530            Ok(Event::End(ref e)) if e.name().as_ref() == b"loc" => {
531                in_loc = false;
532            },
533            Ok(Event::Text(ref e)) if in_loc => {
534                let text = e.decode().map_err(|e| CrawlError::Parse(e.to_string()))?;
535                let url_str = text.trim();
536                if !url_str.is_empty() {
537                    // Resolve relative URLs against base_url
538                    // Following url-join-relative: use base_url.join() for relative paths
539                    let resolved =
540                        if url_str.starts_with("http://") || url_str.starts_with("https://") {
541                            Url::parse(url_str).ok()
542                        } else {
543                            base_url.join(url_str).ok()
544                        };
545                    if let Some(url) = resolved {
546                        urls.push(url.to_string());
547                    }
548                }
549            },
550            Ok(Event::CData(ref e)) if in_loc => {
551                // Handle CDATA sections - BytesCData derefs to [u8]
552                let url_str = String::from_utf8_lossy(e).trim().to_string();
553                if !url_str.is_empty() {
554                    // Resolve relative URLs against base_url
555                    let resolved =
556                        if url_str.starts_with("http://") || url_str.starts_with("https://") {
557                            Url::parse(&url_str).ok()
558                        } else {
559                            base_url.join(&url_str).ok()
560                        };
561                    if let Some(url) = resolved {
562                        urls.push(url.to_string());
563                    }
564                }
565            },
566            Ok(Event::Eof) => break,
567            Err(e) => return Err(CrawlError::Parse(e.to_string())),
568            _ => {},
569        }
570    }
571
572    Ok(urls)
573}
