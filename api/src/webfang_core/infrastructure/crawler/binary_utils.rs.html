12/// Simple percent-decoding for filenames (handles common cases).
13///
14/// Decodes percent-encoded characters in filenames, e.g. `%20` → space.
15///
16/// # Arguments
17///
18/// * `input` - String to decode
19///
20/// # Returns
21///
22/// Decoded string
23///
24/// # Examples
25///
26/// ```
27/// use webfang_core::infrastructure::crawler::binary_utils::percent_decode;
28///
29/// assert_eq!(percent_decode("file%20name.pdf"), "file name.pdf");
30/// assert_eq!(percent_decode("no-encoding"), "no-encoding");
31/// ```
32#[inline]
33#[must_use]
34pub fn percent_decode(input: &str) -> String {
35    let mut result = String::with_capacity(input.len());
36    let mut chars = input.chars();
37    while let Some(c) = chars.next() {
38        if c == '%' {
39            let hex: String = chars.by_ref().take(2).collect();
40            if let Ok(byte) = u8::from_str_radix(&hex, 16) {
41                result.push(byte as char);
42            } else {
43                result.push('%');
44                result.push_str(&hex);
45            }
46        } else {
47            result.push(c);
48        }
49    }
50    result
51}

53/// Derive a filename from Content-Disposition header or URL path.
54///
55/// Priority: Content-Disposition `filename` > URL path basename > fallback.
56///
57/// # Arguments
58///
59/// * `headers` - HTTP response headers
60/// * `url` - URL of the resource
61/// * `content_type` - Content-Type header value
62///
63/// # Returns
64///
65/// Derived filename
66///
67/// # Examples
68///
69/// ```
70/// use webfang_core::infrastructure::crawler::binary_utils::derive_filename_from_response;
71/// use url::Url;
72///
73/// let headers = wreq::header::HeaderMap::new();
74/// let url = Url::parse("https://example.com/docs/report.pdf").unwrap();
75/// let result = derive_filename_from_response(&headers, &url, "application/pdf");
76/// assert_eq!(result, "report.pdf");
77/// ```
78pub fn derive_filename_from_response(
79    headers: &wreq::header::HeaderMap,
80    url: &Url,
81    content_type: &str,
82) -> String {
83    // Try Content-Disposition header first
84    if let Some(disposition) = headers.get(wreq::header::CONTENT_DISPOSITION) {
85        if let Ok(val) = disposition.to_str() {
86            // Parse filename*=UTF-8''encoded or filename="name"
87            if let Some(name) = parse_content_disposition(val) {
88                return name;
89            }
90        }
91    }
92
93    // Derive from URL path
94    let path = url.path();
95    let basename = path.rsplit('/').next().unwrap_or("");
96    if !basename.is_empty() && basename != "/" {
97        // Clean up the basename — remove query params that may be appended
98        let clean = basename.split('?').next().unwrap_or(basename);
99        if !clean.is_empty() {
100            return clean.to_string();
101        }
102    }
103
104    // Fallback: generate filename from content type
105    let ext = match content_type {
106        ct if ct.contains("application/pdf") => "pdf",
107        ct if ct.contains("application/zip") => "zip",
108        ct if ct.contains("application/x-tar") => "tar",
109        ct if ct.contains("image/png") => "png",
110        ct if ct.contains("image/jpeg") => "jpg",
111        ct if ct.contains("image/gif") => "gif",
112        ct if ct.contains("image/webp") => "webp",
113        ct if ct.contains("image/svg") => "svg",
114        ct if ct.contains("audio/mpeg") => "mp3",
115        ct if ct.contains("video/mp4") => "mp4",
116        _ => "bin",
117    };
118
119    // Use URL host + path hash for uniqueness
120    let host = url.host_str().unwrap_or("unknown");
121    let path_hash = {
122        use std::collections::hash_map::DefaultHasher;
123        use std::hash::{Hash, Hasher};
124        let mut hasher = DefaultHasher::new();
125        path.hash(&mut hasher);
126        format!("{:x}", hasher.finish())
127    };
128    format!("{}_{}.{ext}", host.replace('.', "_"), &path_hash[..8])
129}

131/// Parse Content-Disposition header value to extract filename.
132///
133/// Supports:
134/// - `filename="report.pdf"`
135/// - `filename=report.pdf`
136/// - `filename*=UTF-8''encoded-name.pdf`
137///
138/// # Arguments
139///
140/// * `value` - Content-Disposition header value
141///
142/// # Returns
143///
144/// Parsed filename or None
145///
146/// # Examples
147///
148/// ```
149/// use webfang_core::infrastructure::crawler::binary_utils::parse_content_disposition;
150///
151/// assert_eq!(
152///     parse_content_disposition("attachment; filename=\"report.pdf\""),
153///     Some("report.pdf".to_string())
154/// );
155/// assert_eq!(
156///     parse_content_disposition("attachment; filename*=UTF-8''encoded.pdf"),
157///     Some("encoded.pdf".to_string())
158/// );
159/// ```
160pub fn parse_content_disposition(value: &str) -> Option<String> {
161    // Try filename*= first (RFC 5987 encoding)
162    for part in value.split(';') {
163        let part = part.trim();
164        if let Some(rest) = part.strip_prefix("filename*=") {
165            // Format: UTF-8''encoded_name
166            if let Some(name) = rest.strip_prefix("UTF-8''") {
167                // Simple percent-decoding for common cases
168                let decoded = percent_decode(name);
169                if !decoded.is_empty() {
170                    return Some(decoded);
171                }
172            }
173        }
174    }
175
176    // Try filename= (standard)
177    for part in value.split(';') {
178        let part = part.trim();
179        if let Some(rest) = part.strip_prefix("filename=") {
180            let name = rest.trim_matches(|c| c == '"' || c == '\'');
181            if !name.is_empty() {
182                return Some(name.to_string());
183            }
184        }
185    }
186
187    None
188}

194    #[test]
195    fn test_percent_decode_basic() {
196        assert_eq!(percent_decode("file%20name.pdf"), "file name.pdf");
197        assert_eq!(percent_decode("no-encoding"), "no-encoding");
198        assert_eq!(percent_decode("%41%42%43"), "ABC");
199    }

201    #[test]
202    fn test_percent_decode_invalid() {
203        // Invalid hex should keep the original characters
204        assert_eq!(percent_decode("test%ZZ"), "test%ZZ");
205        assert_eq!(percent_decode("%"), "%");
206        // Single hex digit after % is valid (e.g., %2 = char 2)
207        assert_eq!(percent_decode("%2"), "\u{2}");
208    }
