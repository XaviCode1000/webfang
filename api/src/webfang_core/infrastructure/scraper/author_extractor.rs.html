1//! Multi-strategy author extraction
2//!
3//! Extracts the author of a page by cascading through independent
4//! strategies, ordered from most to least reliable:
5//!
6//! 1. JSON-LD structured data (`application/ld+json`)
7//! 2. `<meta>` tags (`name="author"`, `property="article:author"`)
8//! 3. Microdata (`[itemprop="author"]`, `[rel="author"]`)
9//! 4. Common CSS classes (`.author`, `.byline`, ...)
10//! 5. Legible byline heuristic (fallback, supplied by the caller)
11//!
12//! The legible byline alone misses most real-world authors because it only
13//! looks for `<address>` tags and narrow byline patterns. This cascade fills
14//! the gap without touching readability's content-body extraction.
15//!
16//! # Rules Applied
17//!
18//! - **own-borrow-over-clone**: Accept `&str` not `&String`
19//! - **err-no-unwrap-prod**: Static selectors use `expect("BUG: ...")` only for
20//!   compile-time-known-valid selectors (matches `extractor/mod.rs`)
21//! - **opt-inline**: CSS selectors compiled once via `LazyLock`
22
23use std::sync::LazyLock;
24
25use scraper::{Html, Selector};
26use serde_json::Value;
27
28/// Maximum length of a plausible author name.
29///
30/// CSS-class candidates longer than this are almost certainly containers
31/// (a whole paragraph or a footer) rather than a name, so they are skipped.
32const MAX_AUTHOR_LEN: usize = 100;
33
34/// Leading byline prefixes stripped from CSS-class candidates.
35///
36/// Ordered longest-first so "Written by" wins over "by".
37const BYLINE_PREFIXES: &[&str] = &["written by", "por", "by"];
38
39// All selectors below are compile-time-constant strings, so `Selector::parse`
40// cannot fail at runtime; each `expect` documents a true invariant.
41#[allow(clippy::expect_used)]
42static SEL_JSONLD: LazyLock<Selector> = LazyLock::new(|| {
43    Selector::parse("script[type=\"application/ld+json\"]")
44        // LCOV_EXCL_LINE defensive: compile-time-selector — constant selector string, parse cannot fail
45        .expect("BUG: invalid CSS selector application/ld+json")
46});
47#[allow(clippy::expect_used)]
48static SEL_META_AUTHOR: LazyLock<Selector> = LazyLock::new(|| {
49    Selector::parse("meta[name=\"author\"], meta[property=\"article:author\"]")
50        // LCOV_EXCL_LINE defensive: compile-time-selector — constant selector string, parse cannot fail
51        .expect("BUG: invalid CSS selector meta author")
52});
53#[allow(clippy::expect_used)]
54static SEL_ITEMPROP_AUTHOR: LazyLock<Selector> = LazyLock::new(|| {
55    // LCOV_EXCL_LINE defensive: compile-time-selector — constant selector string, parse cannot fail
56    Selector::parse("[itemprop=\"author\"]").expect("BUG: invalid CSS selector itemprop=author")
57});
58#[allow(clippy::expect_used)]
59static SEL_ITEMPROP_NAME: LazyLock<Selector> = LazyLock::new(|| {
60    // LCOV_EXCL_LINE defensive: compile-time-selector — constant selector string, parse cannot fail
61    Selector::parse("[itemprop=\"name\"]").expect("BUG: invalid CSS selector itemprop=name")
62});
63#[allow(clippy::expect_used)]
64static SEL_REL_AUTHOR: LazyLock<Selector> = LazyLock::new(|| {
65    // LCOV_EXCL_LINE defensive: compile-time-selector — constant selector string, parse cannot fail
66    Selector::parse("[rel=\"author\"]").expect("BUG: invalid CSS selector rel=author")
67});
68#[allow(clippy::expect_used)]
69static SEL_CSS_CLASS: LazyLock<Selector> = LazyLock::new(|| {
70    Selector::parse(".author, .byline, .post-author, .article-author")
71        // LCOV_EXCL_LINE defensive: compile-time-selector — constant selector string, parse cannot fail
72        .expect("BUG: invalid CSS selector author classes")
73});

75/// Extracts author information from HTML using a specific strategy.
76pub trait AuthorExtractor: Send + Sync {
77    /// Human-readable strategy name for logging.
78    fn name(&self) -> &str;
79
80    /// Try to extract an author from the HTML. Returns `None` if this
81    /// strategy finds nothing.
82    fn extract(&self, html: &str) -> Option<String>;
83}
84
85/// Strategy 1: JSON-LD structured data.
86///
87/// Handles `{"author": {"name": "..."}}`, `{"author": "..."}`,
88/// `{"author": [{"name": "..."}]}` and `@graph` wrappers.
89#[derive(Default)]
90pub struct JsonLdAuthorExtractor;
91
92impl AuthorExtractor for JsonLdAuthorExtractor {
93    fn name(&self) -> &str {
94        "json-ld"
95    }
96
97    fn extract(&self, html: &str) -> Option<String> {
98        let document = Html::parse_document(html);
99        document.select(&SEL_JSONLD).find_map(|element| {
100            let json: String = element.text().collect();
101            author_from_jsonld(&json)
102        })
103    }
104}
105
106/// Strategy 2: `<meta>` tags.
107#[derive(Default)]
108pub struct MetaTagAuthorExtractor;
109
110impl AuthorExtractor for MetaTagAuthorExtractor {
111    fn name(&self) -> &str {
112        "meta-tag"
113    }
114
115    fn extract(&self, html: &str) -> Option<String> {
116        let document = Html::parse_document(html);
117        document
118            .select(&SEL_META_AUTHOR)
119            .find_map(|element| element.value().attr("content").and_then(non_empty))
120    }
121}
122
123/// Strategy 3: microdata (`itemprop` / `rel`).
124#[derive(Default)]
125pub struct ItempropAuthorExtractor;
126
127impl AuthorExtractor for ItempropAuthorExtractor {
128    fn name(&self) -> &str {
129        "itemprop"
130    }
131
132    fn extract(&self, html: &str) -> Option<String> {
133        let document = Html::parse_document(html);
134
135        for element in document.select(&SEL_ITEMPROP_AUTHOR) {
136            let nested_name = element
137                .select(&SEL_ITEMPROP_NAME)
138                .next()
139                .and_then(|name_el| text_content(&name_el));
140            if let Some(name) = nested_name {
141                return Some(name);
142            }
143            if let Some(content) = element.value().attr("content").and_then(non_empty) {
144                return Some(content);
145            }
146            if let Some(text) = text_content(&element) {
147                return Some(text);
148            }
149        }
150
151        document
152            .select(&SEL_REL_AUTHOR)
153            .find_map(|element| text_content(&element))
154    }
155}

157/// Strategy 4: common CSS classes, with byline-prefix stripping.
158#[derive(Default)]
159pub struct CssClassAuthorExtractor;
160
161impl AuthorExtractor for CssClassAuthorExtractor {
162    fn name(&self) -> &str {
163        "css-class"
164    }
165
166    fn extract(&self, html: &str) -> Option<String> {
167        let document = Html::parse_document(html);
168        document.select(&SEL_CSS_CLASS).find_map(|element| {
169            let raw = text_content(&element)?;
170            let candidate = strip_byline_prefix(&raw);
171            let candidate = candidate.trim();
172            if candidate.is_empty() || candidate.len() > MAX_AUTHOR_LEN {
173                None
174            } else {
175                Some(candidate.to_string())
176            }
177        })
178    }
179}
180
181/// Try each extraction strategy in order, returning the first successful
182/// result. Falls back to the legible byline when no strategy matches.
183///
184/// # Arguments
185///
186/// * `html` - Full raw HTML of the page (must include `<head>` for meta/JSON-LD)
187/// * `legible_byline` - Byline produced by readability, used as last resort
188///
189/// # Returns
190///
191/// * `Some(author)` - The first non-empty author found
192/// * `None` - No strategy matched and no byline was supplied
193pub fn extract_author(html: &str, legible_byline: Option<&str>) -> Option<String> {
194    let extractors: Vec<Box<dyn AuthorExtractor>> = vec![
195        Box::new(JsonLdAuthorExtractor),
196        Box::new(MetaTagAuthorExtractor),
197        Box::new(ItempropAuthorExtractor),
198        Box::new(CssClassAuthorExtractor),
199    ];
200
201    for extractor in &extractors {
202        if let Some(author) = extractor.extract(html) {
203            let author = author.trim().to_string();
204            if !author.is_empty() {
205                tracing::debug!(strategy = extractor.name(), %author, "author extracted");
206                return Some(author);
207            }
208        }
209    }
210
211    legible_byline
212        .map(|b| b.trim().to_string())
213        .filter(|b| !b.is_empty())
214}

216/// Resolve an author name from a raw JSON-LD block.
217fn author_from_jsonld(json: &str) -> Option<String> {
218    let value: Value = serde_json::from_str(json).ok()?;
219    find_author(&value)
220}
221
222/// Recursively search a JSON-LD value for an `author` field.
223fn find_author(value: &Value) -> Option<String> {
224    match value {
225        Value::Object(map) => {
226            if let Some(name) = map.get("author").and_then(author_name) {
227                return Some(name);
228            }
229            map.get("@graph").and_then(find_author)
230        },
231        Value::Array(items) => items.iter().find_map(find_author),
232        _ => None,
233    }
234}
235
236/// Coerce an `author` JSON value (string, object, or array) into a name.
237fn author_name(value: &Value) -> Option<String> {
238    match value {
239        Value::String(s) => non_empty(s),
240        Value::Object(map) => map.get("name").and_then(Value::as_str).and_then(non_empty),
241        Value::Array(items) => items.iter().find_map(author_name),
242        _ => None,
243    }
244}
245
246/// Trim and return the string, or `None` if it is empty/whitespace.
247fn non_empty(s: &str) -> Option<String> {
248    let trimmed = s.trim();
249    if trimmed.is_empty() {
250        None
251    } else {
252        Some(trimmed.to_string())
253    }
254}
255
256/// Collapse an element's text nodes into a single trimmed string.
257fn text_content(element: &scraper::ElementRef<'_>) -> Option<String> {
258    let collapsed: String = element.text().collect::<Vec<_>>().join(" ");
259    let collapsed = collapsed.split_whitespace().collect::<Vec<_>>().join(" ");
260    non_empty(&collapsed)
261}
262
263/// Strip a leading byline prefix ("By ", "Written by ", "Por ") if present.
264fn strip_byline_prefix(s: &str) -> String {
265    let trimmed = s.trim();
266    let lower = trimmed.to_lowercase();
267    for prefix in BYLINE_PREFIXES {
268        let Some(rest) = lower.strip_prefix(prefix) else {
269            continue;
270        };
271        if rest.starts_with(' ') || rest.starts_with('\u{a0}') {
272            return trimmed[prefix.len()..].trim().to_string();
273        }
274    }
275    trimmed.to_string()
276}
