1//! DOM inspector implementations — infrastructure layer.
2//!
3//! Provides two implementations of [`DomInspectorPort`]:
4//! - [`NoOpInspector`] — Null Object pattern, zero overhead on happy path.
5//! - [`DefaultDomInspector`] — walks the DOM tree and computes Jaro-Winkler
6//!   similarity scores for selector suggestions.
7
8use std::cmp::Ordering;
9use std::collections::HashMap;
10
11use crate::domain::{DomInspectorPort, DomStructureReport, SelectorSuggestion};

13// ---------------------------------------------------------------------------
14// NoOpInspector
15// ---------------------------------------------------------------------------
16
17/// No-op DOM inspector — returns empty reports with zero overhead.
18///
19/// Null Object pattern: used when diagnostics are not needed (non-MCP paths
20/// or when an inspector is not configured in `McpState`).
21#[derive(Debug, Clone, Default)]
22pub struct NoOpInspector;
23
24impl DomInspectorPort for NoOpInspector {
25    fn inspect(&self, _document: &scraper::Html) -> DomStructureReport {
26        DomStructureReport::default()
27    }
28
29    fn suggest(
30        &self,
31        _document: &scraper::Html,
32        _failed_selector: &str,
33    ) -> Vec<SelectorSuggestion> {
34        Vec::new()
35    }
36}

38// ---------------------------------------------------------------------------
39// DefaultDomInspector
40// ---------------------------------------------------------------------------
41
42/// Default DOM inspector — walks the DOM tree and computes similarity scores.
43///
44/// Uses `scraper` for DOM traversal and `strsim::jaro_winkler` for similarity.
45/// Node-count cap prevents hangs on very large DOMs (default: 10,000).
46pub struct DefaultDomInspector {
47    node_count_cap: usize,
48    similarity_threshold: f64,
49}
50
51impl DefaultDomInspector {
52    /// Default node-count cap (10,000 elements).
53    pub const DEFAULT_NODE_CAP: usize = 10_000;
54
55    /// Default Jaro-Winkler similarity threshold (0.6).
56    pub const DEFAULT_THRESHOLD: f64 = 0.6;
57
58    /// Create a new `DefaultDomInspector` with default settings.
59    #[must_use]
60    pub fn new() -> Self {
61        Self {
62            node_count_cap: Self::DEFAULT_NODE_CAP,
63            similarity_threshold: Self::DEFAULT_THRESHOLD,
64        }
65    }
66
67    /// Set the maximum number of elements to inspect before truncating.
68    #[must_use]
69    pub fn with_node_cap(mut self, cap: usize) -> Self {
70        self.node_count_cap = cap;
71        self
72    }
73
74    /// Set the minimum Jaro-Winkler similarity score for suggestions.
75    #[must_use]
76    pub fn with_threshold(mut self, threshold: f64) -> Self {
77        self.similarity_threshold = threshold;
78        self
79    }
80
81    /// Collect candidate selectors from the DOM.
82    ///
83    /// Returns a deduplicated `Vec<String>` of bare tag names, `.class`
84    /// selectors, and `#id` selectors.
85    fn collect_candidates(document: &scraper::Html) -> Vec<String> {
86        let selector = match scraper::Selector::parse("*") {
87            Ok(s) => s,
88            Err(_) => return Vec::new(),
89        };
90
91        let mut seen: std::collections::HashSet<String> = std::collections::HashSet::new();
92        let mut candidates: Vec<String> = Vec::new();
93
94        for element_ref in document.select(&selector) {
95            let el = element_ref.value();
96
97            // Tag name candidate
98            let tag = el.name().to_owned();
99            if seen.insert(tag.clone()) {
100                candidates.push(tag);
101            }
102
103            // Class selector candidates
104            for class in el.classes() {
105                let class_sel = format!(".{class}");
106                if seen.insert(class_sel.clone()) {
107                    candidates.push(class_sel);
108                }
109            }
110
111            // ID selector candidate
112            if let Some(id) = el.id() {
113                let id_sel = format!("#{id}");
114                if seen.insert(id_sel.clone()) {
115                    candidates.push(id_sel);
116                }
117            }
118        }
119
120        candidates
121    }
122}

124impl Default for DefaultDomInspector {
125    fn default() -> Self {
126        Self::new()
127    }
128}

130impl DomInspectorPort for DefaultDomInspector {
131    fn inspect(&self, document: &scraper::Html) -> DomStructureReport {
132        let selector = match scraper::Selector::parse("*") {
133            Ok(s) => s,
134            Err(_) => return DomStructureReport::default(),
135        };
136
137        let mut element_count: usize = 0;
138        let mut truncated = false;
139        let mut max_depth: usize = 0;
140        let mut tag_counts: HashMap<String, usize> = HashMap::new();
141        let mut class_counts: HashMap<String, usize> = HashMap::new();
142        let mut id_counts: HashMap<String, usize> = HashMap::new();
143
144        for element_ref in document.select(&selector) {
145            if element_count >= self.node_count_cap {
146                truncated = true;
147                break;
148            }
149
150            let el = element_ref.value();
151
152            element_count += 1;
153
154            // Depth = number of ancestors (including the Document root node).
155            let depth = element_ref.ancestors().count();
156            if depth > max_depth {
157                max_depth = depth;
158            }
159
160            // Tag count
161            *tag_counts.entry(el.name().to_owned()).or_insert(0) += 1;
162
163            // Class counts
164            for class in el.classes() {
165                *class_counts.entry(class.to_string()).or_insert(0) += 1;
166            }
167
168            // ID count
169            if let Some(id) = el.id() {
170                *id_counts.entry(id.to_owned()).or_insert(0) += 1;
171            }
172        }
173
174        // Sort common_classes descending by frequency, then alphabetically for
175        // deterministic output when frequencies are tied. Top 10.
176        let mut common_classes: Vec<(String, usize)> = class_counts.into_iter().collect();
177        common_classes.sort_unstable_by(|a, b| b.1.cmp(&a.1).then_with(|| a.0.cmp(&b.0)));
178        common_classes.truncate(10);
179
180        // Sort common_ids descending by frequency, then alphabetically. Top 5.
181        let mut common_ids: Vec<(String, usize)> = id_counts.into_iter().collect();
182        common_ids.sort_unstable_by(|a, b| b.1.cmp(&a.1).then_with(|| a.0.cmp(&b.0)));
183        common_ids.truncate(5);
184
185        DomStructureReport {
186            element_count,
187            truncated,
188            tag_counts,
189            max_depth,
190            common_classes,
191            common_ids,
192        }
193    }
194
195    fn suggest(&self, document: &scraper::Html, failed_selector: &str) -> Vec<SelectorSuggestion> {
196        let candidates = Self::collect_candidates(document);
197
198        let mut suggestions: Vec<SelectorSuggestion> = candidates
199            .into_iter()
200            .map(|candidate| {
201                let score = strsim::jaro_winkler(failed_selector, &candidate);
202                SelectorSuggestion {
203                    selector: candidate,
204                    score,
205                }
206            })
207            .filter(|s| s.score >= self.similarity_threshold)
208            .collect();
209
210        // Sort by score descending. Use unwrap_or(Equal) for NaN safety.
211        suggestions
212            .sort_unstable_by(|a, b| b.score.partial_cmp(&a.score).unwrap_or(Ordering::Equal));
213
214        // Truncate to top 5.
215        suggestions.truncate(5);
216
217        suggestions
218    }
219}

254    #[test]
255    fn test_default_inspector_counts_tags_correctly() {
256        let html = r#"<html>
257  <head><title>Test</title></head>
258  <body>
259    <div class="container main">
260      <h1 id="header">Title</h1>
261      <p class="text">Paragraph</p>
262      <span>Span</span>
263    </div>
264  </body>
265</html>"#;
266        let document = scraper::Html::parse_document(html);
267        let inspector = DefaultDomInspector::new();
268        let report = inspector.inspect(&document);
269
270        // 8 elements: html, head, title, body, div, h1, p, span
271        assert_eq!(report.element_count, 8);
272        assert!(!report.truncated);
273
274        // Tag counts
275        assert_eq!(*report.tag_counts.get("html").unwrap_or(&0), 1);
276        assert_eq!(*report.tag_counts.get("head").unwrap_or(&0), 1);
277        assert_eq!(*report.tag_counts.get("title").unwrap_or(&0), 1);
278        assert_eq!(*report.tag_counts.get("body").unwrap_or(&0), 1);
279        assert_eq!(*report.tag_counts.get("div").unwrap_or(&0), 1);
280        assert_eq!(*report.tag_counts.get("h1").unwrap_or(&0), 1);
281        assert_eq!(*report.tag_counts.get("p").unwrap_or(&0), 1);
282        assert_eq!(*report.tag_counts.get("span").unwrap_or(&0), 1);
283
284        // Classes: container(1), main(1), text(1)
285        assert!(!report.common_classes.is_empty());
286        let class_names: Vec<&str> = report
287            .common_classes
288            .iter()
289            .map(|(n, _)| n.as_str())
290            .collect();
291        assert!(class_names.contains(&"container"));
292        assert!(class_names.contains(&"main"));
293        assert!(class_names.contains(&"text"));
294
295        // IDs: header(1)
296        assert!(!report.common_ids.is_empty());
297        assert_eq!(report.common_ids[0].0, "header");
298        assert_eq!(report.common_ids[0].1, 1);
299
300        // Max depth: html(1), head(2), title(3), body(2), div(3), h1(4), p(4), span(4)
301        assert_eq!(report.max_depth, 4);
302    }

337    #[test]
338    fn test_default_inspector_max_depth_calculation() {
339        let inspector = DefaultDomInspector::new();
340
341        // Shallow tree: <div><p>text</p></div>
342        let shallow_doc = scraper::Html::parse_fragment("<div><p>text</p></div>");
343        let shallow_report = inspector.inspect(&shallow_doc);
344
345        // Deep tree: <div><div><div><div><p>deep</p></div></div></div></div>
346        let deep_doc = scraper::Html::parse_fragment(
347            "<div><div><div><div><p>deep</p></div></div></div></div>",
348        );
349        let deep_report = inspector.inspect(&deep_doc);
350
351        // Deeper nesting must produce greater max_depth.
352        assert!(
353            deep_report.max_depth > shallow_report.max_depth,
354            "deep tree max_depth ({}) should exceed shallow tree max_depth ({})",
355            deep_report.max_depth,
356            shallow_report.max_depth
357        );
358        // Both should have non-zero depth (at least 1 ancestor: the root node).
359        assert!(shallow_report.max_depth >= 1);
360        assert!(deep_report.max_depth >= 1);
361    }
