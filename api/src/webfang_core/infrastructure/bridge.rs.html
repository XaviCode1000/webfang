38/// Tokio→Rayon crossing for CPU-bound ingestion work (frozen design decision #3).
39///
40/// Holds a dedicated [`RayonCpuPool`] (cloned cheaply — it wraps an `Arc`) and
41/// a [`ContentProcessor`] strategy for HTML-to-text conversion, plus exposes
42/// a generic [`dispatch`](CpuBridge::dispatch) that moves any CPU-bound
43/// closure off the event loop, and a typed
44/// [`dispatch_resource`](CpuBridge::dispatch_resource) that cleans
45/// `DownloadedResource` → `ProcessedResource` via the injected processor.
46///
47/// `CpuBridge` is `Send + Sync` (spec: "dispatch gateway MUST be Send + Sync"),
48/// so it can be shared across Tokio tasks via `Arc<CpuBridge>`.
49pub struct CpuBridge {
50    pool: RayonCpuPool,
51    processor: Arc<dyn ContentProcessor>,
52}
53

58        Self { pool, processor }
59    }

67    /// Dispatch an arbitrary CPU-bound closure onto the Rayon pool and return
68    /// a [`oneshot::Receiver`] holding `Result<R, ScraperError>`.
69    ///
70    /// The work runs under `tokio::task::spawn_blocking` + `pool.install`, so
71    /// the Tokio event loop stays unblocked and any nested `par_iter` routes to
72    /// the sized dedicated pool. CPU panics are caught via
73    /// `catch_unwind(AssertUnwindSafe(…))` (frozen user decision #1) and
74    /// mapped to [`ScraperError::Ingestion`] so Rayon threads stay alive.
75    ///
76    /// If the caller drops the receiver before the work finishes (Tokio task
77    /// abort), `tx.send()` fails and the bridge logs a `tracing::warn!` but
78    /// does NOT panic (Trap 2).
79    pub fn dispatch<F, R>(&self, work: F) -> oneshot::Receiver<Result<R, ScraperError>>
80    where
81        F: FnOnce() -> R + Send + 'static,
82        R: Send + 'static,
83    {
84        let (tx, rx) = oneshot::channel();
85        let pool = self.pool.clone();
86        // The Instrumented wrapper attaches the current tracing span to the
87        // blocking task. We intentionally drop it here because the task runs
88        // fire-and-forget via a oneshot channel — the JoinHandle is not awaited.
89        let handle = tokio::task::spawn_blocking(move || {
90            let caught = catch_unwind(AssertUnwindSafe(move || pool.install(work)));
91            // LCOV_EXCL_START defensive: cpu-pool-panic — a panic in Rayon work is a bug; the pool must not die with it
92            let mapped: Result<R, ScraperError> = caught.map_err(|panic| {
93                let msg = panic_message(&*panic);
94                ScraperError::ingestion(format!("CPU pool panic: {msg}"))
95            });
96            // LCOV_EXCL_STOP
97            // LCOV_EXCL_START defensive: oneshot-receiver-dropped — the Tokio task was aborted before consuming the result
98            if tx.send(mapped).is_err() {
99                warn!(
100                    reason = "receptor oneshot descartado",
101                    "canal CPU bridge descartado: tarea Tokio abortada antes de recibir el resultado"
102                );
103            }
104            // LCOV_EXCL_STOP
105        });
106        // Suppress clippy warning: this is fire-and-forget via oneshot channel.
107        // The span context is captured by in_current_span() before the handle
108        // is dropped — the spawned task still runs with the correct span.
109        #[allow(clippy::let_underscore_future)]
110        let _ = handle.in_current_span();
111        rx
112    }

114    /// Typed dispatch: clean a [`DownloadedResource`] into a
115    /// [`ProcessedResource`] on the Rayon pool.
116    ///
117    /// PR5 wires real `lol_html` boilerplate removal (via `clean_html_to_text`)
118    /// that strips `script`/`style`/`nav`/`footer`/`aside` chrome and extracts
119    /// visible text. The cleaner is infallible (`clean_html` falls back to the
120    /// raw HTML on a `lol_html` parse error), so the work closure returns
121    /// `ProcessedResource` directly and reuses [`dispatch`](Self::dispatch)'s single `Result`
122    /// wrap. Embeddings stay `None` here — ONNX inference is async and runs in
123    /// the orchestrator's async layer (Decision 5); the bridge is sync
124    /// CPU-bound text extraction only.
125    pub fn dispatch_resource(
126        &self,
127        payload: DownloadedResource,
128    ) -> oneshot::Receiver<Result<ProcessedResource, ScraperError>> {
129        let url = payload.url.clone();
130        let size = payload.size_bytes;
131        let processor = Arc::clone(&self.processor);
132        let processor_name = processor.name().to_owned();
133        self.dispatch(move || {
134            let html = String::from_utf8_lossy(&payload.bytes);
135            let text = processor.process(&html);
136            let chunk = ProcessedChunk {
137                content: text,
138                embedding: None,
139            };
140            let metadata = serde_json::json!({
141                "size_bytes": size,
142                "chunk_count": 1u64,
143                "cleaner": processor_name,
144            });
145            ProcessedResource {
146                resource_url: url,
147                chunks: vec![chunk],
148                metadata,
149            }
150        })
151    }
152}

154/// Extract a human-readable message from a captured panic payload.
155///
156/// Panics raised with `&str` / `String` (the common case, including
157/// `panic!` macros and `assert!`) yield their message; other payload types
158/// fall back to a Spanish placeholder.
159fn panic_message(payload: &(dyn std::any::Any + Send)) -> String {
160    if let Some(s) = payload.downcast_ref::<&'static str>() {
161        (*s).to_string()
162    } else if let Some(s) = payload.downcast_ref::<String>() {
163        s.clone()
164    } else {
165        "tipo de pánico desconocido (no es String ni &str)".to_string()
166    }
167}

287    async fn test_dispatch_concurrent_under_shared_pool() {
288        let bridge = std::sync::Arc::new(make_bridge(4));
289        let mut handles = Vec::new();
290        for i in 0u32..16 {
291            let b = std::sync::Arc::clone(&bridge);
292            handles.push(tokio::spawn(async move {
293                let rx = b.dispatch(move || i * i);
294                rx.await
295                    .expect("oneshot must not be closed")
296                    .expect("work must return Ok")
297            }));
298        }
299        let mut results = Vec::new();
300        for h in handles {
301            results.push(h.await.expect("join task must not panic"));
302        }
303        // Every i*i must be present exactly once — no lost/duplicated results.
304        results.sort_unstable();
305        let expected: Vec<u32> = (0u32..16).map(|i| i * i).collect();
306        assert_eq!(results, expected, "concurrent dispatch must be race-free");
307    }
