1//! Headless JSONL vector sink — dependency-free RAG export (core-slimming, T3-C).
2//!
3//! `StreamRepository` implements [`VectorRepository`] by emitting one JSON line
4//! per chunk that carries an embedding. It has **no** SQLite / `rusqlite` /
5//! `sys-info` dependency, so it compiles into the lightweight `webfang_core`
6//! binary and backs `--output-vectors <path|->` (spec R2 / S2.1).
7//!
8//! # Record shape
9//!
10//! Each emitted JSONL line is a [`VectorRecord`]:
11//!
12//! ```json
13//! {"url":"...","sha256_hex":"<64 hex>","title":null,
14//!  "chunk_text":"...","embedding":[0.01, … 384 floats …],
15//!  "metadata":null,"timestamp":"2026-07-11T12:00:00Z"}
16//! ```
17//!
18//! The embedding array is the **raw 384-dim** vector (no rounding, no base64) so
19//! downstream RAG pipelines can ingest it directly.
20//!
21//! # Key decisions (T3-C design)
22//!
23//! - **Q3 (ai OFF):** when a chunk has `embedding = None`, the record is *omitted*
24//!   (no line is written) — we never emit a null/zero vector.
25//! - **D2 (broken pipe):** a write / flush `io::Error` (incl. `WriteZero` from a
26//!   closed pipe) is returned as a fatal [`ScraperError::Io`], which propagates
27//!   out of `ElasticIngestion::run` and aborts the crawl.
28//! - **Concurrency:** writes are serialized through a `Mutex` so the JSONL stream
29//!   stays line-oriented even when [`crate::application::elastic_ingestion::ElasticIngestion`] processes URLs concurrently.
30

44/// A single JSONL vector record emitted by [`StreamRepository`].
45///
46/// Mirrors the fields surfaced by the elastic ingestion pipeline: the source
47/// URL, the content `sha256_hex` (content-hash dedup key), an optional title,
48/// the cleaned `chunk_text`, the raw `embedding` vector, arbitrary `metadata`,
49/// and an RFC3339 `timestamp`.
50#[derive(Debug, Clone, Serialize, Deserialize)]
51pub struct VectorRecord {
52    /// Source URL the chunk was extracted from.
53    pub url: String,
54    /// Content hash (SHA-256 hex) of the resource — the dedup key.
55    pub sha256_hex: String,
56    /// Best-effort title (first ≤200 chars of the first chunk line), or `null`.
57    pub title: Option<String>,
58    /// Cleaned chunk text.
59    pub chunk_text: String,
60    /// Raw embedding vector (e.g. 384 floats for all-MiniLM-L6-v2). Empty only
61    /// when the record would have been omitted (Q3).
62    pub embedding: Vec<f32>,
63    /// Optional arbitrary metadata (reserved for future use).
64    #[serde(default)]
65    pub metadata: Value,
66    /// RFC3339 timestamp of emission.
67    pub timestamp: String,
68}
69
70/// Headless vector sink that writes [`VectorRecord`] lines to a JSONL stream.
71///
72/// Construct with [`StreamRepository::new`]; `"-"` selects stdout (buffered),
73/// any other path selects a file.
74pub struct StreamRepository {
75    /// Serialized JSONL writer. `Box<dyn Write + Send>` so both `Stdout` and
76    /// `File` fit behind one type; the `Mutex` keeps the stream line-oriented
77    /// under concurrent ingestion.
78    writer: Mutex<std::io::BufWriter<Box<dyn Write + Send>>>,
79    /// Title cache keyed by resource URL, populated by [`VectorRepository::save_resource`]
80    /// and read by [`VectorRepository::save_chunk`] (the chunk call only receives
81    /// `resource_url`, not the title).
82    titles: Mutex<HashMap<String, String>>,
83    /// Injected clock for deterministic timestamps in tests.
84    clock: Arc<dyn UtcClock>,
85}
86
87impl StreamRepository {
88    /// Open the JSONL sink with the system clock.
89    ///
90    /// * `path == "-"` → buffered stdout.
91    /// * otherwise → a file created (truncated) at `path`.
92    ///
93    /// # Errors
94    ///
95    /// Returns [`ScraperError::Io`] if the file cannot be created.
96    pub fn new(path: &str) -> Result<Self, ScraperError> {
97        Self::with_clock(path, Arc::new(SystemUtcClock))
98    }
99
100    /// Open the JSONL sink with an injected clock.
101    ///
102    /// # Errors
103    ///
104    /// Returns [`ScraperError::Io`] if the file cannot be created.
105    pub fn with_clock(path: &str, clock: Arc<dyn UtcClock>) -> Result<Self, ScraperError> {
106        let boxed: Box<dyn Write + Send> = if path == "-" {
107            Box::new(std::io::stdout())
108        } else {
109            if let Some(parent) = std::path::Path::new(path).parent() {
110                std::fs::create_dir_all(parent).map_err(|e| {
111                    ScraperError::Io(std::io::Error::new(
112                        e.kind(),
113                        format!("no se pudo crear el directorio '{}': {e}", parent.display()),
114                    ))
115                })?;
116            }
117            Box::new(std::fs::File::create(path).map_err(|e| {
118                ScraperError::Io(std::io::Error::new(
119                    e.kind(),
120                    format!("no se pudo crear el archivo de vectores '{path}': {e}"),
121                ))
122            })?)
123        };
124        Ok(Self {
125            writer: Mutex::new(std::io::BufWriter::new(boxed)),
126            titles: Mutex::new(HashMap::new()),
127            clock,
128        })
129    }
130
131    /// Build a sink over an arbitrary writer.
132    ///
133    /// Used by tests to inject deterministic, failure-simulating writers
134    /// (e.g. a broken-pipe stub) without touching the filesystem or stdout.
135    /// The wrapping matches [`StreamRepository::new`] exactly.
136    #[cfg(test)]
137    pub(crate) fn from_writer(w: Box<dyn Write + Send>) -> Self {
138        Self {
139            writer: Mutex::new(std::io::BufWriter::new(w)),
140            titles: Mutex::new(HashMap::new()),
141            clock: Arc::new(SystemUtcClock),
142        }
143    }
144}

146impl VectorRepository for StreamRepository {
147    fn save_resource<'a>(
148        &'a self,
149        url: &'a str,
150        title: &'a str,
151        _content_hash: &'a str,
152        _size_bytes: u64,
153    ) -> Pin<Box<dyn Future<Output = Result<String, ScraperError>> + Send + 'a>> {
154        Box::pin(async move {
155            if !title.is_empty() {
156                // Invariant: a poisoned Mutex means another thread panicked while
157                // holding the lock — process state is undefined; panic is correct.
158                #[allow(clippy::expect_used)]
159                // LCOV_EXCL_LINE defensive: mutex-poisoning — a poisoned lock leaves process state undefined
160                let mut cache = self.titles.lock().expect("title cache poisoned");
161                cache.insert(url.to_string(), title.to_string());
162            }
163            Ok(url.to_string())
164        })
165    }
166
167    fn save_chunk<'a>(
168        &'a self,
169        id: &'a str,
170        resource_url: &'a str,
171        _chunk_index: i64,
172        content: &'a str,
173        embedding: Option<&'a [f32]>,
174    ) -> Pin<Box<dyn Future<Output = Result<(), ScraperError>> + Send + 'a>> {
175        Box::pin(async move {
176            // Q3: without an embedding there is nothing to vectorize — omit the
177            // record rather than emit a null/zero vector.
178            let embedding = match embedding {
179                Some(e) => e.to_vec(),
180                None => return Ok(()),
181            };
182
183            // The chunk id is formatted as "{sha256_hex}-{index}" by
184            // `ElasticIngestion::run`, so the hash is the segment before the
185            // first '-' (a SHA-256 hex string contains no '-').
186            let sha256_hex = id.split('-').next().unwrap_or(id).to_string();
187
188            // Invariant: poisoned Mutex — see save_resource.
189            #[allow(clippy::expect_used)]
190            let title = self
191                .titles
192                .lock()
193                // LCOV_EXCL_LINE defensive: mutex-poisoning — a poisoned lock leaves process state undefined
194                .expect("title cache poisoned")
195                .get(resource_url)
196                .cloned();
197
198            let record = VectorRecord {
199                url: resource_url.to_string(),
200                sha256_hex,
201                title,
202                chunk_text: content.to_string(),
203                embedding,
204                metadata: Value::Null,
205                timestamp: self.clock.now().to_rfc3339(),
206            };
207
208            let line = serde_json::to_string(&record)?;
209            // D2: a broken pipe / WriteZero must surface as a fatal Io error so
210            // the crawl aborts. `?` converts io::Error → ScraperError::Io.
211            // Invariant: poisoned RwLock — see save_resource.
212            #[allow(clippy::expect_used)]
213            // LCOV_EXCL_LINE defensive: mutex-poisoning — a poisoned lock leaves process state undefined
214            let mut writer = self.writer.lock().expect("vector stream poisoned");
215            writer.write_all(line.as_bytes())?;
216            writer.write_all(b"\n")?;
217            writer.flush()?;
218            Ok(())
219        })
220    }
221
222    fn resource_exists_by_hash<'a>(
223        &'a self,
224        _content_hash: &'a str,
225    ) -> Pin<Box<dyn Future<Output = Result<Option<String>, ScraperError>> + Send + 'a>> {
226        // No dedup for the stream sink — every chunk is emitted.
227        Box::pin(async move { Ok(None) })
228    }
229
230    fn get_vector<'a>(
231        &'a self,
232        _chunk_id: &'a str,
233    ) -> Pin<Box<dyn Future<Output = Result<Option<Vec<f32>>, ScraperError>> + Send + 'a>> {
234        Box::pin(async move { Ok(None) })
235    }
236}

242    #[test]
243    fn test_save_chunk_omits_record_without_embedding() {
244        // Write to a temp file so we can assert no line is produced.
245        let tmp = tempfile::NamedTempFile::new().expect("temp file");
246        let path = tmp.path().to_string_lossy().to_string();
247        let repo = StreamRepository::new(&path).expect("open stream");
248
249        futures::executor::block_on(async {
250            repo.save_resource("https://example.com", "", "deadbeef", 10)
251                .await
252                .expect("save_resource");
253            // No embedding → record omitted, write must succeed without a line.
254            repo.save_chunk("deadbeef-0", "https://example.com", 0, "hello", None)
255                .await
256                .expect("save_chunk");
257        });
258
259        let contents = std::fs::read_to_string(&path).expect("read stream");
260        assert!(
261            contents.trim().is_empty(),
262            "no line expected when embedding is None"
263        );
264    }

266    #[test]
267    fn test_save_chunk_emits_384_dim_embedding_and_hash() {
268        let tmp = tempfile::NamedTempFile::new().expect("temp file");
269        let path = tmp.path().to_string_lossy().to_string();
270        let repo = StreamRepository::new(&path).expect("open stream");
271
272        let embedding: Vec<f32> = (0..384).map(|i| i as f32 * 0.001).collect();
273        futures::executor::block_on(async {
274            repo.save_resource("https://example.com/p", "Page Title", "abc123def", 42)
275                .await
276                .expect("save_resource");
277            repo.save_chunk(
278                "abc123def-0",
279                "https://example.com/p",
280                0,
281                "cleaned chunk text",
282                Some(&embedding),
283            )
284            .await
285            .expect("save_chunk");
286        });
287
288        let contents = std::fs::read_to_string(&path).expect("read stream");
289        let line = contents.lines().next().expect("one JSONL line");
290        let record: VectorRecord = serde_json::from_str(line).expect("valid JSONL");
291
292        assert_eq!(record.sha256_hex, "abc123def");
293        assert_eq!(record.embedding.len(), 384, "explicit 384-dim embedding");
294        assert_eq!(record.title.as_deref(), Some("Page Title"));
295        assert_eq!(record.chunk_text, "cleaned chunk text");
296    }

298    // Collecting writer that buffers everything in memory so tests can inspect
299    // the exact JSONL bytes that `StreamRepository` emits.
300    /// D2 — a broken-pipe write error must surface as `Err(ScraperError::Io)`,
301    /// never as a panic. Uses a deterministic in-memory writer stub (no OS pipes).
302    #[test]
303    fn contract_broken_pipe_returns_err_not_panic() {
304        struct BrokenPipeWriter;
305
306        impl Write for BrokenPipeWriter {
307            fn write(&mut self, _buf: &[u8]) -> std::io::Result<usize> {
308                Err(std::io::Error::new(
309                    std::io::ErrorKind::BrokenPipe,
310                    "simulated broken pipe",
311                ))
312            }
313            fn flush(&mut self) -> std::io::Result<()> {
314                Ok(())
315            }
316        }
317
318        let repo = StreamRepository::from_writer(Box::new(BrokenPipeWriter));
319        let embedding: Vec<f32> = (0..384).map(|i| i as f32 * 0.001).collect();
320
321        let result = futures::executor::block_on(async {
322            repo.save_chunk(
323                "deadbeefcafe-0",
324                "https://example.com/p",
325                0,
326                "cleaned chunk text",
327                Some(&embedding),
328            )
329            .await
330        });
331
332        assert!(
333            result.is_err(),
334            "broken pipe must return Err, not panic/panic"
335        );
336        let err = result.expect_err("broken pipe error");
337        assert!(
338            matches!(err, ScraperError::Io(_)),
339            "broken pipe must map to ScraperError::Io, got: {err:?}"
340        );
341    }

391    /// Lowercase-hex SHA-256 integrity: what upstream provides (a 32-char
392    /// lowercase hex id) is written verbatim and stays lowercase.
393    #[test]
394    fn contract_sha256_hex_is_lowercase_and_preserved() {
395        let tmp = tempfile::NamedTempFile::new().expect("temp file");
396        let path = tmp.path().to_string_lossy().to_string();
397        let repo = StreamRepository::new(&path).expect("open stream");
398
399        let id = "abcdef0123456789abcdef0123456789-0";
400        futures::executor::block_on(async {
401            repo.save_resource(
402                "https://example.com/p",
403                "T",
404                "abcdef0123456789abcdef0123456789",
405                1,
406            )
407            .await
408            .expect("save_resource");
409            repo.save_chunk(
410                id,
411                "https://example.com/p",
412                0,
413                "cleaned chunk text",
414                Some(&vec![0.0f32; 384]),
415            )
416            .await
417            .expect("save_chunk");
418        });
419
420        let contents = std::fs::read_to_string(&path).expect("read stream");
421        let line = contents.lines().next().expect("one JSONL line");
422        let record: VectorRecord = serde_json::from_str(line).expect("valid JSONL");
423
424        assert_eq!(
425            record.sha256_hex, "abcdef0123456789abcdef0123456789",
426            "sha256_hex must be preserved verbatim from the id"
427        );
428
429        let is_lowercase_hex = record
430            .sha256_hex
431            .chars()
432            .all(|c| c.is_ascii_digit() || ('a'..='f').contains(&c))
433            && record.sha256_hex.len() == 32;
434        assert!(
435            is_lowercase_hex,
436            "sha256_hex must be 32 lowercase hex chars (no uppercase), got: {}",
437            record.sha256_hex
438        );
