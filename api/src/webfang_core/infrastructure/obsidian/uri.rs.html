1//! Obsidian URI protocol support.
2//!
3//! Opens notes directly in Obsidian using the `obsidian://` URI scheme.
4//!
5//! URI format: `obsidian://open?vault=<vault_name>&file=<file_path>`
6
7use std::path::Path;
8
9/// Comprehensive percent-encoding for Obsidian URI parameters.
10///
11/// Whitelist-based: only RFC 3986 unreserved characters (`A-Z a-z 0-9 - _ . ~`)
12/// plus `/` (Obsidian needs slashes unencoded in file paths) survive verbatim.
13/// Every other ASCII character — including all cmd.exe metacharacters
14/// (`| > < ^ ; ( ) & = # ? % + space`) — is percent-encoded so it can never be
15/// interpreted by a shell. This neutralizes shell metacharacters (Windows
16/// cmd.exe safety) while preserving `/` for Obsidian file paths. Non-ASCII
17/// characters are UTF-8 percent-encoded byte by byte.
18fn encode_obsidian_param(input: &str) -> String {
19    let mut out = String::with_capacity(input.len());
20    for ch in input.chars() {
21        match ch {
22            // RFC 3986 unreserved chars + '/' (Obsidian needs slashes unencoded in file paths).
23            'A'..='Z' | 'a'..='z' | '0'..='9' | '-' | '_' | '.' | '~' | '/' => out.push(ch),
24            // Every other ASCII char — including cmd.exe metacharacters | > < ^ ; ( ) & = # ? % +
25            // and space — is percent-encoded so it can never be interpreted by a shell.
26            c if c.is_ascii() => out.push_str(&format!("%{:02X}", c as u32)),
27            // Non-ASCII: UTF-8 percent-encode each byte.
28            c => {
29                let mut buf = [0u8; 4];
30                for &byte in c.encode_utf8(&mut buf).as_bytes() {
31                    out.push_str(&format!("%{byte:02X}"));
32                }
33            },
34        }
35    }
36    out
37}

39/// Build an Obsidian URI from vault name and file path.
40///
41/// # Arguments
42/// - `vault_name` — Name of the Obsidian vault (folder name, not full path)
43/// - `file_path` — Path to the note relative to the vault root (without extension)
44///
45/// # Returns
46/// URI string ready for opening
47pub fn build_obsidian_uri(vault_name: &str, file_path: &str) -> String {
48    format!(
49        "obsidian://open?vault={}&file={}",
50        encode_obsidian_param(vault_name),
51        encode_obsidian_param(file_path)
52    )
53}
54
55/// Validate Obsidian URI inputs, rejecting ASCII control characters.
56///
57/// Shell metacharacters are neutralized by `encode_obsidian_param` (percent-encoded),
58/// so they are safe in the URI. Control characters (newline, null byte, etc.) have no
59/// legitimate place in a vault name or note path and are rejected outright as a signal
60/// of malformed or hostile input.
61///
62/// # Errors
63/// Returns `Err` with a user-facing (Spanish) message if either input contains an
64/// ASCII control character.
65pub fn validate_obsidian_input(vault_name: &str, file_path: &str) -> Result<(), String> {
66    for (label, value) in [("vault_name", vault_name), ("file_path", file_path)] {
67        if value.chars().any(|c| c.is_ascii_control()) {
68            return Err(format!(
69                "{label} contiene caracteres de control no permitidos"
70            ));
71        }
72    }
73    Ok(())
74}

76/// Open a note in Obsidian using the URI protocol (fire-and-forget).
77///
78/// Uses `xdg-open` on Linux, `open` on macOS, `start` on Windows.
79/// Non-blocking: spawns the process and returns immediately.
80///
81/// # Arguments
82/// - `uri` — The obsidian:// URI to open
83///
84/// # Returns
85/// `Ok(())` on spawn, `Err(String)` if command fails to start
86pub fn open_in_obsidian(uri: &str) -> Result<(), String> {
87    // The URI is fully percent-encoded by `build_obsidian_uri` (no raw
88    // metacharacters or quotes can appear), and on Windows the empty `""`
89    // title prevents `start` from consuming the URI as a window title.
90    // Together these make `cmd /C start` safe on Windows.
91    let (cmd, args) = if cfg!(target_os = "windows") {
92        ("cmd", vec!["/C", "start", "", uri])
93    } else if cfg!(target_os = "macos") {
94        ("open", vec![uri])
95    } else {
96        // Linux: use xdg-open (standard on all Linux desktops)
97        ("xdg-open", vec![uri])
98    };
99
100    // Fire-and-forget: spawn and don't wait
101    std::process::Command::new(cmd)
102        .args(&args)
103        .spawn()
104        .map_err(|e| format!("failed to open URI: {e}"))?;
105
106    Ok(())
107}

109/// Extract vault name from a vault path (last directory component).
110///
111/// # Example
112/// `/home/user/Obsidian/MyVault` → `MyVault`
113#[must_use]
114pub fn extract_vault_name(vault_path: &Path) -> String {
115    vault_path
116        .file_name()
117        .map(|n| n.to_string_lossy().to_string())
118        .unwrap_or_else(|| "Unknown".to_string())
119}

121/// Open a note in Obsidian from vault path and relative file path.
122///
123/// Convenience function that combines `extract_vault_name`, `build_obsidian_uri`,
124/// and `open_in_obsidian`.
125///
126/// # Arguments
127/// - `vault_path` — Full path to the Obsidian vault
128/// - `file_path` — Path to the note relative to the vault root
129///
130/// # Returns
131/// `Ok(())` if URI was opened (or spawned), `Err(String)` on failure
132pub fn open_note(vault_path: &Path, file_path: &Path) -> Result<(), String> {
133    let vault_name = extract_vault_name(vault_path);
134
135    // Get relative path from vault root
136    let relative = if file_path.is_absolute() {
137        file_path.strip_prefix(vault_path).unwrap_or(file_path)
138    } else {
139        file_path
140    };
141
142    // Convert to string, normalize separators, remove .md extension
143    let file_str = relative
144        .to_string_lossy()
145        .replace('\\', "/")
146        .trim_end_matches(".md")
147        .to_string();
148
149    validate_obsidian_input(&vault_name, &file_str)?;
150
151    let uri = build_obsidian_uri(&vault_name, &file_str);
152    open_in_obsidian(&uri)
153}

160    fn test_build_obsidian_uri_simple() {
161        let uri = build_obsidian_uri("MyVault", "Inbox/example");

163    }

167        let uri = build_obsidian_uri("My Vault", "Inbox/notes");
168        assert!(uri.contains("vault=My%20Vault"));
169        assert!(uri.contains("file=Inbox/notes"));
170    }

172    #[test]
173    fn test_build_obsidian_uri_preserves_slashes() {
174        let uri = build_obsidian_uri("MyVault", "Folder/Subfolder/note");
175        assert!(uri.contains("file=Folder/Subfolder/note"));
176        assert!(!uri.contains("%2F"));
177    }

183        assert!(uri.contains("file=note%3D1"));

216    #[test]
217    fn test_encode_neutralizes_all_cmd_metacharacters() {
218        for meta in ['|', '>', '<', '^', ';', '(', ')', '&', '"', '\n', '\r'] {
219            let input = format!("a{meta}b");
220            let uri = build_obsidian_uri(&input, "note");
221            // Isolate the encoded vault value (between `vault=` and `&file=`).
222            // The whole URI structurally contains '&' and '=' as query separators,
223            // so asserting on the full string would false-positive on those
224            // legitimate characters even though the value is correctly encoded.
225            let vault_value = uri
226                .strip_prefix("obsidian://open?vault=")
227                .and_then(|rest| rest.split("&file=").next())
228                .unwrap_or_default();
229            assert!(
230                !vault_value.contains(meta),
231                "metacharacter {meta:?} leaked into vault value: {vault_value}"
232            );
233        }
234    }
