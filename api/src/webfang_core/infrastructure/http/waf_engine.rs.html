1//! WAF Detection Engine - Layer 7 Protection
2//!
3//! This module provides advanced WAF detection beyond the basic signature matching
4//! in http_client.rs. It includes:
5//! - Detection by Control Headers (x-datadome-response, cf-mitigated, etc.)
6//! - Entropy analysis for "Silent Challenge" detection
7//! - Efficient O(N) matching using Aho-Corasick for 60+ signatures
8//! - Context-aware, tiered, evidence-based inspection via [`WafInspector::inspect`]
9//!
10//! # Usage
11//!
12//! ```ignore
13//! use webfang_core::infrastructure::http::waf_engine::{InspectionContext, WafInspector};
14//!
15//! # let headers = wreq::header::HeaderMap::new();
16//! # let body = String::new();
17//! // Build the inspection context from the HTTP response (status + content-type
18//! // + headers); callers with only the body use the degraded default.
19//! let ctx = InspectionContext {
20//!     status: Some(200),
21//!     content_type: Some("text/html".into()),
22//!     headers,
23//!     ignore_waf: false,
24//! };
25//! let verdict = WafInspector::inspect(&body, &ctx);
26//! if verdict.is_blocked {
27//!     return Err(verdict.evidence_chain());
28//! }
29//! ```

40/// WAF signature tier — determines the blocking policy for a matched pattern.
41///
42/// Tier drives the verdict policy in [`WafInspector::inspect`]:
43/// - [`WafTier::Challenge`] blocks at ANY HTTP status, including 200.
44/// - [`WafTier::Fingerprint`] is evidence only; it blocks solely when correlated
45///   with a WAF-associated status code (403 / 429 / 503 / 520–529).
46#[derive(Debug, Clone, Copy, PartialEq, Eq)]
47pub enum WafTier {
48    /// Unambiguous challenge/captcha markers (widget tokens, challenge prose).
49    Challenge,
50    /// Fingerprint markers (bare vendor names, domains, cookies, control headers).
51    Fingerprint,
52}
53
54impl WafTier {
55    /// Spanish user-facing label for the evidence chain (REQ-WAF-08).
56    #[must_use]
57    pub const fn label_es(self) -> &'static str {
58        match self {
59            WafTier::Challenge => "desafío",
60            WafTier::Fingerprint => "huella",
61        }
62    }
63}

65/// Where a piece of WAF evidence was observed (FIX B).
66///
67/// Drives the 5xx body-vs-header carve-out in the verdict policy: on a 5xx
68/// response a bare vendor mention sourced from the BODY ([`EvidenceSource::Body`])
69/// is ubiquitous diagnostic noise and does not block, whereas the same Fingerprint
70/// tier sourced from a control HEADER ([`EvidenceSource::Header`], e.g.
71/// `cf-mitigated`) signals active mitigation and still blocks. Internal to the
72/// verdict — it is deliberately NOT part of the Spanish evidence chain.
73#[derive(Debug, Clone, Copy, PartialEq, Eq)]
74pub enum EvidenceSource {
75    /// Evidence matched in the response body (signatures or entropy).
76    Body,
77    /// Evidence matched in a response control header.
78    Header,
79}

128/// HTTP context for a WAF inspection (REQ-WAF-01).
129///
130/// [`Default`] is *degraded mode* — no HTTP context (status/content-type
131/// unknown, empty headers, `ignore_waf` false). Callers that only have the
132/// body (e.g. the MCP `detect_waf` tool) use degraded mode, where only
133/// [`WafTier::Challenge`] markers block and [`WafTier::Fingerprint`] evidence
134/// is reported as low-confidence and never blocks (REQ-WAF-05).
135#[derive(Debug, Clone, Default)]
136pub struct InspectionContext {
137    /// HTTP status code, if known.
138    pub status: Option<u16>,
139    /// Content-Type header value, if known.
140    pub content_type: Option<String>,
141    /// Response headers.
142    pub headers: HeaderMap,
143    /// Bypass WAF detection entirely (yields a clean verdict).
144    pub ignore_waf: bool,
145}

147impl InspectionContext {
148    /// Build a full context from an HTTP status and a lowercased-key header map,
149    /// as captured by the domain `HttpResponse` and infrastructure `FetchedPage`
150    /// types (REQ-WAF-01).
151    ///
152    /// The `content-type` entry is lifted into [`Self::content_type`] for the
153    /// REQ-WAF-02 gate, and the whole map is converted into a wreq
154    /// [`HeaderMap`] for control-header evidence (REQ-WAF-03). Invalid header
155    /// names/values are skipped — they cannot occur for headers already
156    /// validated by the downloaders, but the guard keeps this infallible.
157    /// `ignore_waf` short-circuits inspection to a clean verdict (REQ-WAF-07).
158    #[must_use]
159    pub fn from_lowercase_headers(
160        status: u16,
161        headers: &std::collections::HashMap<String, String>,
162        ignore_waf: bool,
163    ) -> Self {
164        let content_type = headers.get("content-type").cloned();
165        let mut map = HeaderMap::new();
166        for (name, value) in headers {
167            if let (Ok(name), Ok(value)) = (
168                wreq::header::HeaderName::from_bytes(name.as_bytes()),
169                wreq::header::HeaderValue::from_str(value),
170            ) {
171                map.insert(name, value);
172            }
173        }
174        Self {
175            status: Some(status),
176            content_type,
177            headers: map,
178            ignore_waf,
179        }
180    }
181}

183/// Control headers that indicate WAF processing (REQ-WAF-03).
184///
185/// Every control header is [`WafTier::Fingerprint`] evidence — it NEVER
186/// auto-blocks on mere presence, only when correlated with a WAF status code
187/// (correction B). The retained headers correlate with *active* processing or
188/// mitigation: a DataDome response, a Cloudflare mitigation flag, an Akamai
189/// edge-auth challenge — so status-correlated blocking is the intended #346
190/// semantics.
191///
192/// RES-01: `cf-ray` and `x-sucuri-id` were purged. They are ubiquitous trace
193/// headers — present on 100% of Cloudflare / Sucuri-proxied traffic, including
194/// every genuine transient origin failure — so a fingerprint that rides on all
195/// of a vendor's traffic carries zero challenge evidence. Treating them as
196/// Fingerprint evidence block-correlated every real 503 behind those vendors
197/// (instant `WafChallenge`, zero retries). They are noise, not signal.
198/// `x-wordpress` (not a real WAF header) and `x-cdn` (generic CDN header) were
199/// purged earlier as false-positive risks.

602/// WafInspector provides multi-layer WAF detection
603pub struct WafInspector;
604
605impl WafInspector {
606    /// Inspect a response body with full HTTP context and return a verdict.
607    ///
608    /// This is the single entry point for context-aware, evidence-based WAF
609    /// detection (REQ-WAF-01). It collects ALL evidence — boundary-filtered body
610    /// signatures (REQ-WAF-03/04), control headers (REQ-WAF-03), and entropy
611    /// signals (REQ-WAF-06) — then applies the verdict policy (REQ-WAF-05):
612    ///
613    /// - [`WafTier::Challenge`] (T1) blocks at ANY status, including 200.
614    /// - [`WafTier::Fingerprint`] (T2) blocks only with a correlated WAF status
615    ///   (403 / 429 / 503 / 520–529); in degraded mode (no status) T2 is reported
616    ///   low-confidence and never blocks.
617    ///
618    /// `ctx.ignore_waf` yields a clean verdict immediately (REQ-WAF-02 step 1),
619    /// and the content-type denylist (REQ-WAF-02) gates body scanning.
620    ///
621    /// Thread-safe: the AC automaton is immutable once compiled via `Lazy`.
622    #[must_use]
623    pub fn inspect(body: &str, ctx: &InspectionContext) -> WafVerdict {
624        // REQ-WAF-02 step 1: explicit bypass → clean verdict.
625        if ctx.ignore_waf {
626            return WafVerdict::clean();
627        }
628
629        let mut evidences = Vec::new();
630        let scan_body = should_scan_body(ctx);
631
632        // Body signatures (boundary-filtered), gated by content-type.
633        if scan_body {
634            evidences.extend(collect_body_evidence(body));
635        }
636
637        // Control headers are always inspected (independent of the body gate).
638        evidences.extend(collect_header_evidence(&ctx.headers));
639
640        // Entropy signals need full T2 awareness (rule (a) coexistence), so they
641        // run after body + header evidence is collected.
642        if scan_body {
643            let has_fingerprint = evidences.iter().any(|e| e.tier == WafTier::Fingerprint);
644            evidences.extend(entropy_evidence(body, ctx, has_fingerprint));
645        }
646
647        let is_blocked = decide(&evidences, ctx);
648
649        // Observability (REQ-WAF-08): the engine inspects silently — both the
650        // blocking and the informational path log at debug. The decision site
651        // (HTTP client / scraper service / crawler discovery) owns the WARN or
652        // structured error, so a block is not double-logged here.
653        if is_blocked {
654            tracing::debug!(
655                status = ?ctx.status,
656                evidences = evidences.len(),
657                "WAF/CAPTCHA challenge detected; blocking response"
658            );
659        } else if !evidences.is_empty() {
660            tracing::debug!(
661                status = ?ctx.status,
662                evidences = evidences.len(),
663                "WAF fingerprints observed but not blocking (informational)"
664            );
665        }
666
667        WafVerdict {
668            is_blocked,
669            evidences,
670        }
671    }

673    /// Get the list of supported WAF providers
674    #[must_use]
675    pub fn supported_providers() -> Vec<&'static str> {
676        // Extract unique provider names from signatures
677        let mut providers: Vec<&str> = Vec::new();
678        let mut seen: HashSet<&str> = HashSet::new();
679

692/// Collect body-signature evidence in a single Aho-Corasick pass, applying the
693/// boundary post-filter (REQ-WAF-04) to Fingerprint `[B]` matches.
694///
695/// The whole body is scanned: [`MAX_EVIDENCE_PER_BODY`] caps only the
696/// *Fingerprint* (T2) accumulation (RISK-01 DoS hardening), never detection. A
697/// [`WafTier::Challenge`] (T1) marker is a definitive verdict — the moment one
698/// survives the filter it clears any accumulated T2 noise and short-circuits the
699/// pass, so a T1 marker past 64 earlier T2 matches is still seen (the cap is not
700/// a detection limit). T2 matches keep accumulating (bounded by the cap) and the
701/// scan continues past the cap so a later T1 marker is still examined. A verdict
702/// needs at most one T1 marker or one correlated T2 marker, and the T2 cap is far
703/// beyond any real challenge page's signal count.
704fn collect_body_evidence(body: &str) -> Vec<WafEvidence> {
705    let mut evidences = Vec::new();
706    for mat in WAF_AC.find_iter(body) {
707        let sig = &WAF_BODY_SIGNATURES[mat.pattern()];
708        if !passes_boundary_filter(body, &mat, sig.2, sig.3) {
709            continue;
710        }
711        match sig.2 {
712            // T1 is a definitive verdict that noise density cannot hide: drop any
713            // accumulated T2 noise and short-circuit on the first T1 marker.
714            WafTier::Challenge => {
715                evidences.clear();
716                evidences.push(WafEvidence {
717                    provider: sig.1,
718                    tier: sig.2,
719                    matched_pattern: sig.0,
720                    source: EvidenceSource::Body,
721                });
722                break;
723            },
724            // T2 accumulation is capped (RISK-01); keep scanning past the cap so a
725            // later T1 marker is still examined.
726            WafTier::Fingerprint => {
727                if evidences.len() < MAX_EVIDENCE_PER_BODY {
728                    evidences.push(WafEvidence {
729                        provider: sig.1,
730                        tier: sig.2,
731                        matched_pattern: sig.0,
732                        source: EvidenceSource::Body,
733                    });
734                }
735            },
736        }
737    }
738    evidences
739}

741/// O(1) adjacent-byte boundary post-filter for Fingerprint-tier `[B]` matches
742/// (REQ-WAF-04).
743///
744/// Rejects a bare vendor-name match when the byte immediately before its start
745/// or after its end is ASCII alphanumeric or `_` — this is what stops
746/// `akamai_hash` from matching bare `akamai`. UTF-8 safe: any non-ASCII byte
747/// (>= 0x80) counts as a boundary. `[E]` (Exempt) patterns and all
748/// Challenge-tier patterns skip the filter entirely.
749#[inline]
750fn passes_boundary_filter(
751    body: &str,
752    mat: &aho_corasick::Match,
753    tier: WafTier,
754    boundary: BoundaryMode,
755) -> bool {
756    // Only Fingerprint-tier bare vendor names get the filter.
757    if tier != WafTier::Fingerprint || boundary != BoundaryMode::Bare {
758        return true;
759    }
760    let bytes = body.as_bytes();
761    if mat.start() > 0 {
762        let before = bytes[mat.start() - 1];
763        if before.is_ascii_alphanumeric() || before == b'_' {
764            return false;
765        }
766    }
767    if mat.end() < bytes.len() {
768        let after = bytes[mat.end()];
769        if after.is_ascii_alphanumeric() || after == b'_' {
770            return false;
771        }
772    }
773    true
774}

776/// Content-Type gate / denylist (REQ-WAF-02).
777///
778/// Decides whether the response body should be scanned for WAF signatures:
779/// 1. `application/xhtml+xml` carve-out → scan (it is HTML despite `+xml`).
780/// 2. Skip structured data (`+json`/`+xml` suffix, `application/json`,
781///    `application/xml`, `text/json`, `text/xml`) and binary assets
782///    (`image/*`, `font/*`, `application/wasm`, `application/javascript`).
783/// 3. `text/*`, missing content-type, or anything else not denied → scan
784///    (this is a denylist, not an allowlist).
785///
786/// The `ignore_waf` short-circuit (REQ-WAF-02 step 1) happens in
787/// [`WafInspector::inspect`] before this gate is consulted.
788fn should_scan_body(ctx: &InspectionContext) -> bool {
789    // Missing content-type → scan.
790    let Some(raw) = &ctx.content_type else {
791        return true;
792    };
793    // Strip parameters (e.g. "text/html; charset=utf-8" → "text/html").
794    let mime = raw
795        .split(';')
796        .next()
797        .unwrap_or("")
798        .trim()
799        .to_ascii_lowercase();
800    // Carve-out: XHTML is HTML — scan it (checked before the +xml skip).
801    if mime == "application/xhtml+xml" {
802        return true;
803    }
804    // Denylist: structured data formats are never HTML challenges.
805    let is_json = mime.ends_with("+json") || mime == "application/json" || mime == "text/json";
806    let is_xml = mime.ends_with("+xml") || mime == "application/xml" || mime == "text/xml";
807    if is_json || is_xml {
808        return false;
809    }
810    // Denylist: binary assets.
811    if mime.starts_with("image/") || mime.starts_with("font/") {
812        return false;
813    }
814    if mime == "application/wasm" || mime == "application/javascript" {
815        return false;
816    }
817    // text/* or anything else not denied → scan.
818    true
819}

841/// Verdict policy (REQ-WAF-05).
842///
843/// - [`WafTier::Challenge`] blocks at ANY status (including 200 and degraded).
844/// - [`WafTier::Fingerprint`] blocks only when the status correlates with a WAF
845///   response (403 / 429 / 503 / 520–529); degraded mode (no status) never
846///   blocks on Fingerprint evidence (reported low-confidence).
847///
848/// FIX B (body-vs-header granularity): on a 5xx response a Fingerprint match
849/// sourced from the BODY ([`EvidenceSource::Body`]) is ubiquitous diagnostic
850/// noise — the body analog of the purged `cf-ray` header — and does NOT block,
851/// so a genuine transient 5xx retries. Fingerprint evidence sourced from a
852/// control HEADER ([`EvidenceSource::Header`], e.g. `cf-mitigated`) signals
853/// active mitigation and still blocks on 5xx (RES-01). The distinction is by
854/// evidence source, not status — [`is_t2_blocking_status`] is unchanged.
855///
856/// Entropy challenges are emitted as [`WafTier::Challenge`] evidence only when
857/// their own policy (REQ-WAF-06) already decided to block, so they short-circuit
858/// here like any other Challenge marker.
859fn decide(evidences: &[WafEvidence], ctx: &InspectionContext) -> bool {
860    for ev in evidences {
861        match ev.tier {
862            WafTier::Challenge => return true,
863            WafTier::Fingerprint => {
864                if !is_t2_blocking_status(ctx.status) {
865                    continue;
866                }
867                // RES-01 extension: on 5xx a bare vendor mention in the BODY is
868                // ubiquitous diagnostic noise (the body analog of the purged
869                // cf-ray header) — not blocking. Header-sourced T2 (cf-mitigated
870                // etc.) signals active mitigation and still blocks.
871                if is_5xx(ctx.status) && ev.source == EvidenceSource::Body {
872                    continue;
873                }
874                return true;
875            },
876        }
877    }
878    false
879}

893/// Entropy-based challenge detection (REQ-WAF-06).
894///
895/// Returns evidence ONLY when a rule decides to block (so its presence in a
896/// verdict always means "blocked"):
897/// - Rule (a): body > 100KB AND entropy > 5.5 b/B → block if status != 200
898///   (unknown/degraded counts as != 200) OR a Fingerprint marker coexists.
899/// - Rule (b): body < 1500B AND > 5 `<script>` tags → block if non-2xx OR
900///   (200 + HTML content-type). The 200+HTML case is the "H3 fix" silent
901///   challenge and MUST be kept (`discovery.rs` depends on it).
902fn entropy_evidence(
903    body: &str,
904    ctx: &InspectionContext,
905    has_fingerprint: bool,
906) -> Option<WafEvidence> {
907    // Rule (a): large + high entropy → obfuscated WAF.
908    if body.len() > SUSPICIOUS_SIZE_THRESHOLD {
909        let entropy = calculate_entropy(body);
910        if entropy > ENTROPY_THRESHOLD {
911            if ctx.status != Some(200) || has_fingerprint {
912                return Some(WafEvidence {
913                    provider: "Obfuscated WAF",
914                    tier: WafTier::Challenge,
915                    matched_pattern: "high-entropy body (>100KB, >5.5 b/B)",
916                    source: EvidenceSource::Body,
917                });
918            }
919            // Informational detection (REQ-WAF-08): high entropy at status 200
920            // without a coexisting T2 marker is logged, not blocked.
921            tracing::debug!(
922                entropy,
923                "high-entropy body at status 200 without a T2 marker; not blocking"
924            );
925        }
926    }
927
928    // Rule (b): tiny body dense with <script> tags → silent challenge.
929    if body.len() < SILENT_CHALLENGE_MAX_BYTES {
930        let script_count = body.matches("<script").count();
931        if script_count > SILENT_CHALLENGE_MIN_SCRIPTS {
932            let is_2xx = matches!(ctx.status, Some(200..=299));
933            let is_html = ctx
934                .content_type
935                .as_deref()
936                .is_some_and(is_html_content_type);
937            // Unknown status (degraded) is treated as non-2xx (conservative).
938            if !is_2xx || (ctx.status == Some(200) && is_html) {
939                return Some(WafEvidence {
940                    provider: "Silent Challenge",
941                    tier: WafTier::Challenge,
942                    matched_pattern: "script-density (<1500B, >5 <script>)",
943                    source: EvidenceSource::Body,
944                });
945            }
946        }
947    }
948
949    None
950}

985/// Calculate Shannon entropy of a string
986///
987/// Used to detect obfuscated JavaScript in challenge pages, which often have
988/// high entropy due to minification and encoding.
989///
990/// # Arguments
991/// * `s` - The string to analyze
992///
993/// # Returns
994/// * Entropy value between 0.0 (uniform distribution) and ~8.0 (high entropy)
995#[inline]
996fn calculate_entropy(s: &str) -> f64 {
997    if s.is_empty() {
998        return 0.0;
999    }
1000
1001    let mut freq = [0u32; 256];
1002    let len = s.len() as f64;
1003
1004    for &b in s.as_bytes() {
1005        freq[b as usize] += 1;
1006    }
1007
1008    let mut entropy = 0.0;
1009    for &count in &freq {
1010        if count > 0 {
1011            let p = count as f64 / len;
1012            entropy -= p * p.log2();
1013        }
1014    }
1015
1016    entropy
1017}
