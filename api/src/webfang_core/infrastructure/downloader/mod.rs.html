1//! Downloader abstraction for page fetching.
2//!
3//! Provides the [`Downloader`] trait and supporting types for fetching pages.
4//! Implementations handle HTTP requests, cookie extraction, and connection pooling.
5//!
6//! # Architecture
7//!
8//! This module follows Clean Architecture: the trait defines the contract,
9//! and concrete implementations (like [`WreqDownloader`]) live in the same
10//! module but are swapped via dependency injection.

12//! [`WreqDownloader`]: wreq_downloader::WreqDownloader

32/// Downloader trait for fetching pages.
33///
34/// Implementations must be safe to share across threads (`Send + Sync`).
35/// Each implementation owns its connection pool and request configuration.
36///
37/// # Examples
38///
39/// ```ignore
40/// use webfang_core::infrastructure::downloader::wreq_downloader::WreqDownloader;
41/// use webfang_core::infrastructure::downloader::Downloader;
42///
43/// let downloader = WreqDownloader::new(30, 10, wreq_util::Profile::Chrome145, None).unwrap();
44/// let page = downloader.fetch(&"https://example.com".parse().unwrap()).await.unwrap();
45/// assert!(!page.html.is_empty());
46/// ```
47pub trait Downloader: Send + Sync {
48    /// Fetch a page from the given URL.
49    ///
50    /// Returns the fetched page with HTML content, HTTP status, headers, and cookies.
51    ///
52    /// Desugared to [`BoxFuture`] (instead of native `async fn`) so the trait is
53    /// dyn-compatible — `&dyn Downloader` can be passed for runtime dispatch and
54    /// test mocking. Matches the `VectorRepository` precedent.
55    ///
56    /// # Errors
57    ///
58    /// Returns [`DownloadError`] on network failure, timeout, or WAF detection.
59    fn fetch<'a>(&'a self, url: &'a Url) -> BoxFuture<'a, Result<FetchedPage, DownloadError>>;
60
61    /// Whether this downloader supports JavaScript rendering / interactions.
62    ///
63    /// Static HTTP downloaders return `false`. Headless browser implementations
64    /// return `true`.
65    fn supports_interactions(&self) -> bool;
66
67    /// Estimated memory cost of this downloader instance in bytes.
68    ///
69    /// Used by the scheduler to budget total memory across concurrent downloaders.
70    fn memory_cost(&self) -> usize;
71}

73/// A page fetched by a [`Downloader`].
74#[derive(Debug, Clone)]
75pub struct FetchedPage {
76    /// The final URL after redirects.
77    pub url: Url,
78    /// Raw HTML content of the page.
79    pub html: String,
80    /// HTTP status code.
81    pub status: u16,
82    /// Response headers (keys lowercased). Parity with domain `HttpResponse`.
83    ///
84    /// Used for content-type sniffing (binary detection) and filename derivation.
85    /// Downloaders with limited header access (e.g. subprocess-based) may leave
86    /// this empty.
87    pub headers: HashMap<String, String>,
88    /// Cookies set by the server during this request.
89    pub cookies: Vec<Cookie>,
90}
91
92/// An HTTP cookie extracted from a response.
93#[derive(Debug, Clone)]
94pub struct Cookie {
95    /// Cookie name.
96    pub name: String,
97    /// Cookie value.
98    pub value: String,
99    /// Domain the cookie applies to.
100    pub domain: String,
101    /// Path prefix the cookie applies to.
102    pub path: String,
103    /// Whether the cookie is HTTP-only (not accessible via JavaScript).
104    pub http_only: bool,
105    /// Whether the cookie requires HTTPS.
106    pub secure: bool,
107}

109/// Errors that can occur during page download.
110///
111/// Following **api-non-exhaustive**: can add variants without breaking changes.
112/// Following **err-thiserror-lib**: uses thiserror for structured error messages.
113#[derive(Debug, thiserror::Error)]
114#[non_exhaustive]
115pub enum DownloadError {
116    /// Network-level failure (DNS, connection, timeout).
117    ///
118    /// Carries the underlying error as `#[source]` (erased) so the root-cause
119    /// chain survives into `CrawlError`/`ScraperError` (D4).
120    #[error("network error: {0}")]
121    Network(#[source] Box<dyn std::error::Error + Send + Sync>),
122
123    /// HTTP error response (non-2xx status).
124    #[error("HTTP {status}: {message}")]
125    #[allow(missing_docs)] // enum variant fields can't have pub(crate) visibility
126    Http { status: u16, message: String },
127
128    /// WAF challenge detected (Cloudflare, reCAPTCHA, etc.).
129    #[error("WAF challenge detected: {0}")]
130    WafChallenge(String),
131
132    /// SPA detected — page requires JavaScript rendering.
133    #[error("SPA detected: {0}")]
134    SpaDetected(String),
135
136    /// URL is invalid or has an unsupported scheme.
137    #[error("invalid URL: {0}")]
138    InvalidUrl(String),
139
140    /// Request timed out.
141    #[error("request timed out after {0}s")]
142    Timeout(u64),
143
144    /// Internal error (should not happen in normal operation).
145    #[error("internal error: {0}")]
146    Internal(String),
147
148    /// System resources (RAM) insufficient to spawn a heavyweight downloader
149    /// layer. An *expected* operational condition under memory pressure —
150    /// distinct from [`Internal`](Self::Internal), which means a bug.
151    #[error("insufficient resources: {0}")]
152    ResourceExhausted(String),
153
154    /// Acquisition cancelled by the engine's cancellation token (#509) —
155    /// the caller was waiting for a resource permit when shutdown fired.
156    #[error("operation cancelled while waiting for resources")]
157    Cancelled,
158}
159
160impl From<DownloadError> for crate::domain::CrawlError {
161    fn from(err: DownloadError) -> Self {
162        crate::domain::CrawlError::Download(Box::new(err))
163    }
164}

166#[cfg(test)]
167mod tests {
168    use super::*;
169
170    #[test]
171    fn test_fetched_page_clone() {
172        let page = FetchedPage {
173            url: "https://example.com".parse().unwrap(),
174            html: "<html></html>".to_string(),
175            status: 200,
176            headers: HashMap::new(),
177            cookies: vec![],
178        };
179        let cloned = page.clone();
180        assert_eq!(page.url, cloned.url);
181        assert_eq!(page.html, cloned.html);
182    }
183
184    #[test]
185    fn test_cookie_struct() {
186        let cookie = Cookie {
187            name: "session".into(),
188            value: "abc123".into(),
189            domain: ".example.com".into(),
190            path: "/".into(),
191            http_only: true,
192            secure: true,
193        };
194        assert!(cookie.http_only);
195        assert!(cookie.secure);
196    }
197
198    #[test]
199    fn test_download_error_display() {
200        let err = DownloadError::Network(Box::new(std::io::Error::new(
201            std::io::ErrorKind::ConnectionRefused,
202            "connection refused",
203        )));
204        assert!(err.to_string().contains("connection refused"));
205
206        let err = DownloadError::Http {
207            status: 403,
208            message: "forbidden".into(),
209        };
210        assert!(err.to_string().contains("403"));
211
212        let err = DownloadError::WafChallenge("Cloudflare".into());
213        assert!(err.to_string().contains("Cloudflare"));
214
215        let err = DownloadError::SpaDetected("React SPA".into());
216        assert!(err.to_string().contains("React SPA"));
217
218        let err = DownloadError::Timeout(30);
219        assert!(err.to_string().contains("30"));
220    }
221
222    #[test]
223    fn test_download_error_into_crawl_error() {
224        let err = DownloadError::Network(Box::new(std::io::Error::new(
225            std::io::ErrorKind::ConnectionReset,
226            "reset",
227        )));
228        let crawl_err: crate::domain::CrawlError = err.into();
229        assert!(crawl_err.to_string().contains("download error"));
230        assert!(crawl_err.to_string().contains("reset"));
231    }
