1//! RAM-aware resource governor for heavyweight downloaders.
2//!
3//! Uses `sysinfo` to read available system memory and a [`tokio::sync::Semaphore`]
4//! to gate concurrent Chrome/headless-browser instances. The formula:
5//!
6//! ```text
7//! max_instances = (available_ram_bytes * 0.6) / CHROME_INSTANCE_COST
8//! ```
9//!
10//! Thresholds:
11//! - **80 %** RAM used → warning log, max_instances halved
12//! - **90 %** RAM used → all new Chrome permits denied
13//!
14//! # Sanitizer coverage
15//!
16//! - **Miri: inapplicable** — `sysinfo` calls `sysconf(SC_PHYS_PAGES)`, which
17//!   Miri cannot execute (unsupported operation). Tests are gated with
18//!   `#[cfg(not(miri))]` and document the reason.
19//! - **TSan: covered** — the semaphore-based concurrent gating is exercised
20//!   under ThreadSanitizer in CI (`sanitizers.yml`), where sysconf works.
21
22use std::sync::Arc;
23
24use sysinfo::System;
25use tokio::sync::Semaphore;
26use tokio_util::sync::CancellationToken;
27use tracing::{debug, warn};
28
29use super::DownloadError;
30
31/// Approximate RAM cost of one Chrome instance (200 MB).
32const CHROME_INSTANCE_COST: u64 = 200_000_000;
33
34/// Fraction of available RAM we're willing to allocate to Chrome instances.
35const RAM_BUDGET_FRACTION: f64 = 0.6;
36
37/// RAM usage percentage that triggers a warning and reduces capacity by half.
38const WARNING_THRESHOLD: u8 = 80;
39
40/// RAM usage percentage that denies all new Chrome instances.
41const CRITICAL_THRESHOLD: u8 = 90;
42
43/// Gates concurrent heavyweight downloader instances based on system RAM.
44///
45/// Holds a [`CancellationToken`] (#509) so a caller blocked in
46/// [`acquire`](Self::acquire) can be woken on engine shutdown instead of
47/// hanging on the semaphore forever. [`new`](Self::new) uses a fresh token
48/// that never fires (pre-#509 behavior); the crawl engine injects its own
49/// token via [`with_cancel_token`](Self::with_cancel_token).
50pub struct ResourceGovernor {
51    semaphore: Arc<Semaphore>,
52    cancel: CancellationToken,
53}
54
55impl ResourceGovernor {
56    /// Create a governor calibrated to current system RAM.
57    ///
58    /// The embedded cancellation token never fires; use
59    /// [`with_cancel_token`](Self::with_cancel_token) to wire shutdown.
60    pub fn new() -> Self {
61        Self::with_cancel_token(CancellationToken::new())
62    }
63
64    /// Create a governor calibrated to current system RAM whose
65    /// [`acquire`](Self::acquire) aborts when `cancel` fires (#509).
66    pub fn with_cancel_token(cancel: CancellationToken) -> Self {
67        Self::with_max_instances(Self::compute_max_instances(), cancel)
68    }
69
70    /// Create a governor with an explicit permit budget.
71    ///
72    /// Calibration seam behind [`new`](Self::new) /
73    /// [`with_cancel_token`](Self::with_cancel_token): lets constrained
74    /// environments (containers) and tests set the budget directly instead
75    /// of deriving it from system RAM.
76    pub fn with_max_instances(max_permits: usize, cancel: CancellationToken) -> Self {
77        let max_permits = max_permits.max(1);
78        debug!("ResourceGovernor: max_permits={max_permits}");
79
80        Self {
81            semaphore: Arc::new(Semaphore::new(max_permits)),
82            cancel,
83        }
84    }
85
86    /// Check whether resources are available and return a permit count, or an
87    /// error if the system is under too much memory pressure.
88    ///
89    /// The returned `usize` represents how many permits *could* be acquired
90    /// right now (0 when denied).
91    pub fn check_resources(&self) -> Result<usize, ResourceError> {
92        let usage = Self::ram_usage_percent();
93
94        if usage >= CRITICAL_THRESHOLD {
95            warn!("RAM usage {usage}% >= {CRITICAL_THRESHOLD}%: new Chrome instances denied");
96            return Err(ResourceError::RamTooHigh(usage));
97        }
98
99        let available = self.semaphore.available_permits();
100
101        if usage >= WARNING_THRESHOLD {
102            let reduced = available / 2;
103            warn!(
104                "RAM usage {usage}% >= {WARNING_THRESHOLD}%: available permits reduced {available} -> {reduced}"
105            );
106            return Ok(reduced);
107        }
108
109        Ok(available)
110    }
111
112    /// Acquire an owned semaphore permit, returning an error when resources
113    /// are exhausted.
114    ///
115    /// Returns a [`tokio::sync::OwnedSemaphorePermit`] (`'static`) so callers can hold
116    /// it across async boundaries without tying it to the governor's lifetime.
117    ///
118    /// The wait aborts with [`DownloadError::Cancelled`] if the governor's
119    /// cancellation token fires while blocked (#509). Dropping the pending
120    /// acquire consumes no permit, so a cancelled wait cannot leak one.
121    ///
122    /// # Errors
123    ///
124    /// Returns [`DownloadError::Cancelled`] on cancellation and
125    /// [`DownloadError::Internal`] if the semaphore was closed.
126    pub async fn acquire(&self) -> Result<tokio::sync::OwnedSemaphorePermit, DownloadError> {
127        let arc = Arc::clone(&self.semaphore);
128
129        tokio::select! {
130            result = arc.acquire_owned() => {
131                // LCOV_EXCL_LINE defensive: semaphore-closed — acquire_owned fails only when the governor is shut down
132                result.map_err(|_| {
133                    DownloadError::Internal("resource governor semaphore closed".to_string())
134                })
135            },
136            () = self.cancel.cancelled() => Err(DownloadError::Cancelled),
137        }
138    }
139
140    /// Current number of available permits.
141    pub fn available_permits(&self) -> usize {
142        self.semaphore.available_permits()
143    }
144
145    /// Compute `max_instances` from current system RAM.
146    fn compute_max_instances() -> usize {
147        let total = Self::total_ram_bytes();
148        let budget = (total as f64 * RAM_BUDGET_FRACTION) as u64;
149        let max = budget / CHROME_INSTANCE_COST;
150        max.max(1) as usize // at least 1 permit even on tiny machines
151    }
152
153    /// Total system RAM in bytes.
154    fn total_ram_bytes() -> u64 {
155        let mut sys = System::new();
156        sys.refresh_memory();
157        sys.total_memory()
158    }
159
160    /// RAM usage as a percentage (0–100).
161    pub fn ram_usage_percent() -> u8 {
162        let mut sys = System::new();
163        sys.refresh_memory();
164        let total = sys.total_memory();
165        if total == 0 {
166            return 0;
167        }
168        let used = sys.used_memory();
169        ((used * 100) / total) as u8
170    }
171}
172
173impl Default for ResourceGovernor {
174    fn default() -> Self {
175        Self::new()
176    }
177}
178
179/// Errors raised when system resources cannot support a new instance.
180#[derive(Debug, thiserror::Error)]
181pub enum ResourceError {
182    /// RAM usage exceeded the critical threshold.
183    #[error("RAM usage too high ({0}%): new Chrome instances denied")]
184    RamTooHigh(u8),
185
186    /// Generic resource exhaustion.
187    #[error("insufficient resources: {0}")]
188    Insufficient(String),
189}
190
191impl From<ResourceError> for DownloadError {
192    fn from(err: ResourceError) -> Self {
193        DownloadError::ResourceExhausted(err.to_string())
194    }
195}

251    // ========================================================================
252    // RAII permit tests (#509) — prove no semaphore permit leaks
253    // ========================================================================
254
255    #[tokio::test]
256    async fn acquire_releases_permit_on_drop() {
257        let gov = ResourceGovernor::with_max_instances(2, CancellationToken::new());
258        assert_eq!(gov.available_permits(), 2);
259
260        let permit = gov.acquire().await.expect("first acquire must succeed");
261        assert_eq!(gov.available_permits(), 1);
262
263        drop(permit);
264        assert_eq!(
265            gov.available_permits(),
266            2,
267            "dropping the permit must return it to the pool"
268        );
269    }

271    #[tokio::test]
272    async fn acquire_grants_permit_when_token_idle() {
273        let gov = ResourceGovernor::with_max_instances(1, CancellationToken::new());
274
275        let permit = tokio::time::timeout(Duration::from_secs(1), gov.acquire())
276            .await
277            .expect("acquire must not hang when permits are available");
278        assert!(permit.is_ok());
279    }

281    // ========================================================================
282    // Cancellation tests (#509)
283    // ========================================================================
284
285    #[tokio::test]
286    async fn acquire_returns_cancelled_when_blocked_and_token_fires() {
287        let cancel = CancellationToken::new();
288        let gov = std::sync::Arc::new(ResourceGovernor::with_max_instances(1, cancel.clone()));
289
290        // Take the single permit so the next acquire must block.
291        let held = gov.acquire().await.expect("single permit must be granted");
292        assert_eq!(gov.available_permits(), 0);
293
294        let waiter = {
295            let gov = std::sync::Arc::clone(&gov);
296            tokio::spawn(async move { gov.acquire().await })
297        };
298        tokio::time::sleep(Duration::from_millis(50)).await;
299        cancel.cancel();
300
301        let result = tokio::time::timeout(Duration::from_secs(1), waiter)
302            .await
303            .expect("cancelled acquire must return within the bound");
304        assert!(matches!(result, Ok(Err(DownloadError::Cancelled))));
305
306        // The cancelled wait must not have consumed a permit.
307        assert_eq!(gov.available_permits(), 0);
308
309        // RAII: releasing the held permit makes it visible again.
310        drop(held);
311        assert_eq!(gov.available_permits(), 1);
312    }
