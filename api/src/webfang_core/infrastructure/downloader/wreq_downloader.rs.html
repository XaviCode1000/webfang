1//! Wreq-based downloader implementation.
2//!
3//! Wraps a shared `wreq::Client` behind `Arc` for connection pooling.
4//! Extracts cookies from responses and returns [`FetchedPage`] with HTML + cookies.
5//!
6//! Following **own-arc-shared**: Uses `Arc<Client>` for thread-safe shared ownership
7//! of the connection pool. The client is created once and shared across all requests.
8

21/// Estimated memory cost of a wreq client instance in bytes.
22///
23/// This accounts for the connection pool, TLS session cache, and internal buffers.
24/// Value is approximate — real usage varies by pool size and active connections.
25const WREQ_MEMORY_COST: usize = 1_024 * 1_024; // ~1 MB
26
27/// Downloader implementation backed by `wreq` with connection pooling.
28///
29/// The internal `wreq::Client` is shared via `Arc` — all requests reuse the same
30/// connection pool, avoiding the per-request client creation anti-pattern.
31///
32/// # Examples
33///
34/// ```ignore
35/// use webfang_core::infrastructure::downloader::wreq_downloader::WreqDownloader;
36/// use webfang_core::infrastructure::downloader::Downloader;
37///
38/// let downloader = WreqDownloader::new(30, 10, wreq_util::Profile::Chrome145, None).unwrap();
39/// let page = downloader.fetch(&"https://example.com".parse().unwrap()).await.unwrap();
40/// assert_eq!(page.status, 200);
41/// ```
42pub struct WreqDownloader {
43    client: Arc<Client>,
44    timeout_secs: u64,
45    /// User-Agent pinned by the operator (`--user-agent`, #503).
46    ///
47    /// Applied once at client-build time so every request — first fetch and
48    /// all retries — carries it. When set, the 403 pool-rotation retry is
49    /// disabled: the operator asked to be identified exactly as configured.
50    pinned_ua: Option<String>,
51}

53impl WreqDownloader {
54    /// Create a new WreqDownloader with the given TLS/HTTP2 emulation profile.
55    ///
56    /// The client is built once and shared via `Arc` for connection pooling.
57    /// Pass [`Profile::Chrome145`] for the historical default fingerprint.
58    ///
59    /// # Arguments
60    ///
61    /// * `timeout_secs` - Request timeout in seconds
62    /// * `connect_timeout_secs` - Connection timeout in seconds
63    /// * `tls_emulation` - TLS/HTTP2 fingerprint profile applied to the client
64    /// * `user_agent` - Optional pinned User-Agent (#503). Applied at client
65    ///   build time via the builder's `user_agent` API, AFTER the emulation
66    ///   profile, so it wins over the profile-default UA on the wire. When
67    ///   set, the 403 pool-rotation retry is disabled.
68    ///
69    /// # Errors
70    ///
71    /// Returns [`DownloadError::Internal`] if the wreq client cannot be built.
72    pub fn new(
73        timeout_secs: u64,
74        connect_timeout_secs: u64,
75        tls_emulation: Profile,
76        user_agent: Option<String>,
77    ) -> Result<Self, DownloadError> {
78        let pool_size = std::cmp::max(6, num_cpus::get() - 1);
79
80        // `.emulation(profile)` installs the profile-default headers (including
81        // a browser UA) via `default_headers`; a pinned UA must be set AFTER
82        // it so `HeaderMap::insert` replaces the profile value (#503).
83        let builder = Client::builder().emulation(tls_emulation);
84        let builder = match user_agent.as_deref() {
85            Some(ua) => builder.user_agent(ua),
86            None => builder,
87        };
88        let client = builder
89            .timeout(Duration::from_secs(timeout_secs))
90            .connect_timeout(Duration::from_secs(connect_timeout_secs))
91            .pool_max_idle_per_host(pool_size)
92            .pool_idle_timeout(Duration::from_secs(60))
93            .gzip(true)
94            .brotli(true)
95            .cookie_store(true)
96            .redirect(wreq::redirect::Policy::limited(10))
97            .build()
98            // LCOV_EXCL_LINE defensive: wreq-client-build — client construction fails only on invalid TLS profile, an invariant
99            .map_err(|e| DownloadError::Internal(format!("failed to build wreq client: {e}")))?;
100
101        debug!(
102            pool_size = pool_size,
103            timeout_secs = timeout_secs,
104            connect_timeout_secs = connect_timeout_secs,
105            ua = user_agent.as_deref().unwrap_or("emulation-default"),
106            "WreqDownloader created"
107        );
108
109        Ok(Self {
110            client: Arc::new(client),
111            timeout_secs,
112            pinned_ua: user_agent,
113        })
114    }

127    /// Extract cookies from a wreq response.
128    fn extract_cookies(url: &Url, response: &wreq::Response) -> Vec<Cookie> {
129        let mut cookies = Vec::new();
130
131        // Extract cookies from the cookie store via the response cookies
132        for cookie in response.cookies() {
133            cookies.push(Cookie {
134                name: cookie.name().to_string(),
135                value: cookie.value().to_string(),
136                domain: cookie.domain().unwrap_or("").to_string(),
137                path: cookie.path().unwrap_or("/").to_string(),
138                http_only: cookie.http_only(),
139                secure: cookie.secure(),
140            });
141        }
142
143        // Also extract Set-Cookie headers for cookies not in the store
144        let set_cookie_headers = response.headers().get_all("set-cookie");
145        let existing_names: std::collections::HashSet<_> =
146            cookies.iter().map(|c| c.name.clone()).collect();
147
148        for header_value in set_cookie_headers {
149            if let Ok(value_str) = header_value.to_str() {
150                // Parse basic cookie fields from Set-Cookie header
151                if let Some(cookie) = parse_set_cookie(value_str, url) {
152                    if !existing_names.contains(&cookie.name) {
153                        cookies.push(cookie);
154                    }
155                }
156            }
157        }
158
159        cookies
160    }
161}

163/// Parse a Set-Cookie header value into a Cookie struct.
164fn parse_set_cookie(header: &str, url: &Url) -> Option<Cookie> {
165    let parts: Vec<&str> = header.split(';').collect();
166    if parts.is_empty() {
167        return None;
168    }
169
170    let name_value = parts[0].trim();
171    let pos = name_value.find('=')?;
172    let (name, value) = (
173        name_value[..pos].trim().to_string(),
174        name_value[pos + 1..].trim().to_string(),
175    );
176
177    if name.is_empty() {
178        return None;
179    }
180
181    let mut domain = url.host_str().unwrap_or("").to_string();
182    let mut path = "/".to_string();
183    let mut http_only = false;
184    let mut secure = false;
185
186    for part in &parts[1..] {
187        let part = part.trim().to_lowercase();
188        if let Some(val) = part.strip_prefix("domain=") {
189            domain = val.trim().to_string();
190        } else if let Some(val) = part.strip_prefix("path=") {
191            path = val.trim().to_string();
192        } else if part == "httponly" {
193            http_only = true;
194        } else if part == "secure" {
195            secure = true;
196        }
197    }
198
199    Some(Cookie {
200        name,
201        value,
202        domain,
203        path,
204        http_only,
205        secure,
206    })
207}

209/// Parse the `Retry-After` header (integer seconds) into milliseconds.
210///
211/// Returns the delay in ms, defaulting to 1000ms if the header is absent or unparseable.
212fn parse_retry_after_ms(response: &wreq::Response) -> u64 {
213    response
214        .headers()
215        .get("retry-after")
216        .and_then(|v| v.to_str().ok())
217        .and_then(|s| s.trim().parse::<u64>().ok())
218        .unwrap_or(1)
219        .saturating_mul(1000)
220}

222impl WreqDownloader {
223    /// Describe the effective User-Agent of a request for tracing (#503):
224    /// the request-level override (403 rotation), else the pinned value,
225    /// else the emulation-profile default.
226    fn effective_ua<'a>(&'a self, request_ua: Option<&'a str>) -> &'a str {
227        request_ua
228            .or(self.pinned_ua.as_deref())
229            .unwrap_or("emulation-default")
230    }
231
232    #[instrument(skip(self), fields(url = %url, ua = %self.effective_ua(user_agent)))]
233    async fn send_request(
234        &self,
235        url: &Url,
236        user_agent: Option<&str>,
237    ) -> Result<wreq::Response, DownloadError> {
238        let builder = self.client.get(url.as_str());
239        let builder = match user_agent {
240            Some(ua) => builder.header("User-Agent", ua),
241            None => builder,
242        };
243        builder.send().await.map_err(|e| {
244            if e.is_timeout() {
245                DownloadError::Timeout(self.timeout_secs)
246            } else {
247                DownloadError::Network(Box::new(e))
248            }
249        })
250    }
251
252    #[instrument(
253        skip(self),
254        fields(
255            url = %url,
256            // D5: stable identity of the shared pooled `Client` (Arc inner ptr).
257            // Constant across fetches => observable proof of connection-pool reuse
258            // (no silent re-handshake per request). See MAPA item 7.
259            client_id = %format!("{:p}", Arc::as_ptr(&self.client))
260        )
261    )]
262    async fn fetch_inner(&self, url: &Url) -> Result<FetchedPage, DownloadError> {
263        debug!("Fetching URL: {}", url);
264
265        let mut response = self.send_request(url, None).await?;
266        let mut status = response.status().as_u16();
267
268        // 403: one implicit retry with rotated User-Agent (mirrors HttpClient).
269        // Pinned UA (#503): the operator asked to be identified exactly as
270        // configured ("--user-agent QA-Bot/9.9" means QA-Bot or fail), so pool
271        // rotation is skipped and the 403 falls through to normal error handling.
272        if status == 403 && self.pinned_ua.is_none() {
273            warn!("403 Forbidden from {url} — retrying with rotated User-Agent");
274            let agents = UserAgentCache::fallback_agents();
275            let rotated_ua = agents.get(1).map(String::as_str);
276            response = self.send_request(url, rotated_ua).await?;
277            status = response.status().as_u16();
278        }
279
280        // 429: backoff retry up to 3 times, honouring Retry-After.
281        if status == 429 {
282            let delay_ms = parse_retry_after_ms(&response);
283            warn!("429 Rate Limited from {url} — retrying after {delay_ms}ms");
284            let mut succeeded = false;
285            for _ in 0..3 {
286                tokio::time::sleep(Duration::from_millis(delay_ms)).await;
287                response = self.send_request(url, None).await?;
288                status = response.status().as_u16();
289                if response.status().is_success() {
290                    succeeded = true;
291                    break;
292                }
293            }
294            if !succeeded {
295                return Err(DownloadError::Http {
296                    status: 429,
297                    message: "HTTP 429".to_string(),
298                });
299            }
300        }
301
302        // Remaining non-2xx → terminal error.
303        if !response.status().is_success() {
304            let message = format!("HTTP {status}");
305            return Err(DownloadError::Http { status, message });
306        }
307
308        let status = response.status().as_u16();
309
310        // Extract cookies before consuming the response body
311        let cookies = Self::extract_cookies(url, &response);
312
313        // Extract the final URL after redirects
314        let final_url = Url::parse(&response.uri().to_string())
315            .map_err(|e| DownloadError::InvalidUrl(e.to_string()))?;
316
317        // Capture response headers (lowercased keys) before consuming the body.
318        let headers = response
319            .headers()
320            .iter()
321            .map(|(name, value)| {
322                (
323                    name.as_str().to_ascii_lowercase(),
324                    value.to_str().unwrap_or_default().to_string(),
325                )
326            })
327            .collect();
328
329        let html = response
330            .text()
331            .await
332            .map_err(|e| DownloadError::Network(Box::new(e)))?;
333
334        debug!(
335            "Fetched {} ({} bytes, {} cookies)",
336            final_url,
337            html.len(),
338            cookies.len()
339        );
340
341        Ok(FetchedPage {
342            url: final_url,
343            html,
344            status,
345            headers,
346            cookies,
347        })
348    }
349}

623    /// Exact-match matcher for a single raw header value.
624    ///
625    /// Unlike `wiremock::matchers::header`, it does NOT comma-split the
626    /// received value: the fallback pool agents embed commas
627    /// ("(KHTML, like Gecko)"), which the built-in exact matcher splits into
628    /// two values and therefore never matches.
629    struct RawHeaderEquals {
630        name: &'static str,
631        value: String,
632    }
633
634    impl Match for RawHeaderEquals {
635        fn matches(&self, request: &wiremock::Request) -> bool {
636            request
637                .headers
638                .get_all(self.name)
639                .iter()
640                .any(|v| v.to_str().is_ok_and(|s| s == self.value))
641        }
642    }

644    /// Issue #503 policy: a pinned UA disables the 403 pool-agent rotation.
645    /// The first 403 surfaces as a terminal error (no retry under a rotated
646    /// identity), and the next fetch still carries the pinned UA.
647    #[tokio::test]
648    async fn pinned_user_agent_disables_403_rotation() {
649        const PINNED_UA: &str = "QA-Bot/9.9";
650        let mock_server = MockServer::start().await;
651        let pool_agent = UserAgentCache::fallback_agents().swap_remove(1);
652
653        // First hit: 403 for the pinned UA (consumed once, then falls through).
