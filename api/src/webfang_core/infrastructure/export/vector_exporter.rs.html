20/// Reserved character width of the `total_documents` header field.
21///
22/// The count is written right-aligned (space-padded) into this window so
23/// [`VectorExporter::close_json`] can patch it in place without shifting the
24/// rest of the file. Covers document counts up to 9,999,999,999.
25const COUNT_FIELD_WIDTH: usize = 10;
26
27/// Reserved character width of the `dimensions` header field.
28///
29/// Right-aligned like [`COUNT_FIELD_WIDTH`]; holds either the embedding
30/// dimension count or the literal `null`. Eight characters cover any
31/// realistic embedding size.
32const DIM_FIELD_WIDTH: usize = 8;
33
34/// Number of leading bytes scanned to locate and patch the metadata header.
35///
36/// The header line fits comfortably inside this window even with the reserved
37/// fields, so scanning a fixed prefix avoids reading whole files on close.
38const HEADER_SCAN_BYTES: usize = 512;

40/// Computes cosine similarity between two vectors
41///
42/// Returns a value between -1.0 and 1.0, where:
43/// - 1.0 means identical direction
44/// - 0.0 means orthogonal
45/// - -1.0 means opposite direction
46///
47/// Returns 0.0 for zero-magnitude vectors.
48///
49/// # Arguments
50/// * `a` - First vector
51/// * `b` - Second vector
52///
53/// # Errors
54/// Returns `DimensionMismatch` if vectors have different dimensions
55#[inline]
56pub fn cosine_similarity(a: &[f32], b: &[f32]) -> Result<f32, ExporterError> {
57    if a.len() != b.len() {
58        return Err(ExporterError::DimensionMismatch {
59            expected: b.len(),
60            actual: a.len(),
61        });
62    }
63
64    let dot_product: f32 = a.iter().zip(b.iter()).map(|(x, y)| x * y).sum();
65    let mag_a: f32 = a.iter().map(|x| x * x).sum::<f32>().sqrt();
66    let mag_b: f32 = b.iter().map(|x| x * x).sum::<f32>().sqrt();
67
68    if mag_a == 0.0 || mag_b == 0.0 {
69        return Ok(0.0);
70    }
71
72    Ok(dot_product / (mag_a * mag_b))
73}

107    /// Get a file writer with proper locking
108    ///
109    /// Creates directories if needed, acquires fs2 file lock,
110    /// and returns a BufWriter for efficient I/O.
111    ///
112    /// In append mode with an existing file, finds and truncates at the
113    /// closing `]` so the writer can append documents and re-close.
114    fn writer(&self) -> ExportResult<(File, BufWriter<File>)> {
115        // Create output directory if it doesn't exist
116        fs::create_dir_all(&self.config.output_dir)?;
117
118        let path = self.config.output_path();
119
120        let file = if self.config.append && path.exists() {
121            let mut f = OpenOptions::new().read(true).write(true).open(&path)?;
122
123            // Find the closing `]` and truncate there so we can re-append
124            let len = f.metadata()?.len();
125            if len > 0 {
126                let seek_start = len.saturating_sub(256);
127                f.seek(SeekFrom::Start(seek_start))?;
128                let mut buf = Vec::new();
129                f.read_to_end(&mut buf)?;
130
131                let content = String::from_utf8_lossy(&buf);
132                if let Some(last_bracket) = content.rfind(']') {
133                    let truncate_pos = seek_start + last_bracket as u64;
134                    f.set_len(truncate_pos)?;
135                    f.seek(SeekFrom::Start(truncate_pos))?;
136                }
137            }
138            f
139        } else {
140            OpenOptions::new()
141                .read(true)
142                .write(true)
143                .create(true)
144                .truncate(!self.config.append)
145                .open(&path)?
146        };
147
148        // Acquire exclusive lock
149        file.lock_exclusive()?;
150
151        let writer = BufWriter::new(file.try_clone()?);
152
153        Ok((file, writer))
154    }

156    /// Write metadata header to file
157    ///
158    /// For new files: writes the header template with reserved (right-aligned,
159    /// space-padded) `dimensions` and `total_documents` windows that
160    /// [`Self::close_json`] patches in place once the real values are known.
161    /// For append mode with existing documents: the file is already truncated
162    /// at the closing `]` by [`Self::writer`], so nothing is written here.
163    fn write_metadata_header(
164        &self,
165        writer: &mut BufWriter<File>,
166        file: &mut File,
167        is_first_doc: bool,
168    ) -> ExportResult<()> {
169        if !is_first_doc {
170            return Ok(());
171        }
172
173        if self.config.append && file.metadata()?.len() > 0 {
174            // File already truncated at `]` by writer() — just seek back to it
175            file.seek(SeekFrom::End(0))?;
176        } else {
177            // New file or overwrite mode — header with reserved patch windows
178            let timestamp = Utc::now().to_rfc3339();
179            let header = self.build_header_line(&timestamp);
180            write!(writer, "{header}")?;
181        }
182
183        Ok(())
184    }

186    /// Build the metadata header line with reserved in-place patch windows.
187    ///
188    /// `dimensions` and `total_documents` are right-aligned into fixed-width
189    /// windows ([`DIM_FIELD_WIDTH`] / [`COUNT_FIELD_WIDTH`]). Space padding is
190    /// legal JSON whitespace and right-alignment avoids leading zeros, so both
191    /// fields can be overwritten later without rewriting the whole file.
192    fn build_header_line(&self, created_at: &str) -> String {
193        // Mutex poisoning indicates a bug in the calling code, not a recoverable error.
194        #[allow(clippy::expect_used)]
195        let dimensions_json = self
196            .dimensions
197            .lock()
198            // LCOV_EXCL_LINE defensive: mutex-poisoning — poisoning indicates a bug in the calling code
199            .expect("lock poisoned")
200            .map(|d| d.to_string())
201            .unwrap_or_else(|| "null".to_string());
202        let count = 0_usize;
203        format!(
204            r#"{{"format_version": "1.0", "model_name": null, "dimensions": {dimensions_json:>DIM_FIELD_WIDTH$}, "total_documents": {count:>COUNT_FIELD_WIDTH$}, "created_at": "{created_at}", "documents": ["#
205        )
206    }

208    /// Serialize a document chunk to JSON
209    ///
210    /// Validates embedding dimensions if present.
211    /// Rejects NaN or Infinity values in embeddings — they produce invalid JSON.
212    fn serialize_document(&self, doc: &DocumentChunkValidated) -> ExportResult<String> {
213        // Validate embedding dimensions if present
214        if let Some(ref embeddings) = doc.embeddings {
215            // Mutex poisoning indicates a bug in the calling code, not a recoverable error.
216            #[allow(clippy::expect_used)]
217            // LCOV_EXCL_LINE defensive: mutex-poisoning — poisoning indicates a bug in the calling code
218            let mut dim_guard = self.dimensions.lock().expect("lock poisoned");
219            if let Some(exp) = *dim_guard {
220                if embeddings.len() != exp {
221                    // Dimension mismatch: the document is degraded (no embeddings in output).
222                    // error! because --output-vectors implies embeddings are the point.
223                    tracing::error!(
224                        expected_dimensions = exp,
225                        actual_dimensions = embeddings.len(),
226                        "Dimension mismatch — document serialized WITHOUT embeddings"
227                    );
228                    // Create a copy without embeddings
229                    let mut doc_without_embeddings = doc.clone();
230                    doc_without_embeddings.embeddings = None;
231                    return serde_json::to_string(&doc_without_embeddings)
232                        .map_err(|e| ExporterError::WriteError(e.to_string()));
233                }
234            } else {
235                // First document with embeddings - record dimensions
236                *dim_guard = Some(embeddings.len());
237            }
238
239            // Reject NaN/Infinity — serde_json serialises them as `null` silently
240            if embeddings.iter().any(|v| !v.is_finite()) {
241                return Err(ExporterError::WriteError(
242                    "embeddings contain NaN or Infinity".into(),
243                ));
244            }
245        }
246
247        // Serialize to JSON
248        let serialized = serde_json::to_string(doc)?;
249        Ok(serialized)
250    }

252    /// Read how many documents an existing export already holds.
253    ///
254    /// Called when appending to a non-empty file. Parses the reserved
255    /// `total_documents` window written by [`Self::close_json`]. Files whose
256    /// header predates the in-place counter fix (issue #502) fail the
257    /// sentinel check and are migrated on the fly by
258    /// [`Self::migrate_legacy_header`].
259    ///
260    /// Leaves the file offset at EOF so the caller can keep appending.
261    fn read_append_state(&self, file: &mut File) -> ExportResult<usize> {
262        let head = read_header_bytes(file)?;
263
264        let marker = b"\"total_documents\": ";
265        let value_start = find_subslice(&head, marker)
266            .map(|pos| pos + marker.len())
267            .ok_or_else(|| {
268                ExporterError::WriteError(
269                    "vector export header is missing the total_documents field".to_string(),
270                )
271            })?;
272
273        // The fixed-width layout guarantees a ',' exactly COUNT_FIELD_WIDTH
274        // bytes after the value start. Anything else means the file was
275        // written by the pre-fix exporter (or is corrupt) and needs migration.
276        let existing = if head.get(value_start + COUNT_FIELD_WIDTH) == Some(&b',') {
277            let count = parse_count_field(&head[value_start..value_start + COUNT_FIELD_WIDTH])?;
278            if count == 0 {
279                // Defensive: files written by this version always carry the
280                // true count. Fall back to counting document lines.
281                count_documents(file)?
282            } else {
283                count
284            }
285        } else {
286            self.migrate_legacy_header(file, &head)?
287        };
288
289        file.seek(SeekFrom::End(0))?;
290        Ok(existing)
291    }
292
293    /// Migrate a pre-fix header to the reserved-window layout.
294    ///
295    /// Legacy files stored a hardcoded `"total_documents": 0` and dimensions
296    /// learned too late to be written. The real document count is recovered
297    /// from the line structure, the header is rebuilt with the reserved
298    /// windows (preserving the original `created_at`), and the file head is
299    /// rewritten in place. Safe because the caller holds the exclusive fs2
300    /// lock for the whole export call.
301    fn migrate_legacy_header(&self, file: &mut File, head: &[u8]) -> ExportResult<usize> {
302        // serde_json never emits raw line breaks inside strings, so every
303        // 0x0A byte marks the end of exactly one document line.
304        let existing = count_documents(file)?;
305
306        let docs_marker = b"\"documents\": [";
307        let body_start = find_subslice(head, docs_marker)
308            .map(|pos| pos + docs_marker.len())
309            .ok_or_else(|| {
310                ExporterError::WriteError(
311                    "vector export header is missing the documents array".to_string(),
312                )
313            })?;
314        let created_at = extract_created_at(head).ok_or_else(|| {
315            ExporterError::WriteError("vector export header is missing created_at".to_string())
316        })?;
317
318        file.seek(SeekFrom::Start(body_start as u64))?;
319        let mut body = Vec::new();
320        file.read_to_end(&mut body)?;
321
322        let new_head = self.build_header_line(&created_at);
323        file.seek(SeekFrom::Start(0))?;
324        file.write_all(new_head.as_bytes())?;
325        file.write_all(&body)?;
326        let new_len = file.stream_position()?;
327        file.set_len(new_len)?;
328
329        tracing::warn!(
330            existing_documents = existing,
331            path = %self.config.output_path().display(),
332            "Migrating legacy vector export header to the fixed-count layout"
333        );
334
335        Ok(existing)
336    }
337
338    /// Close the JSON structure and patch the header counters in place.
339    ///
340    /// Writes the closing bracket, then overwrites the reserved
341    /// `total_documents` window (and the `dimensions` window once the
342    /// embedding size became known during this run) at their fixed header
343    /// offsets. The `file` handle shares its offset with the `BufWriter`'s
344    /// clone, so the writer is flushed before any seek.
345    fn close_json(
346        &self,
347        writer: &mut BufWriter<File>,
348        file: &mut File,
349        total_documents: usize,
350    ) -> ExportResult<()> {
351        writeln!(writer, "]}}")?;
352        writer.flush()?;
353
354        let head = read_header_bytes(file)?;
355
356        let count = format!("{total_documents:>COUNT_FIELD_WIDTH$}");
357        patch_header_field(
358            file,
359            &head,
360            b"\"total_documents\": ",
361            "total_documents",
362            &count,
363            COUNT_FIELD_WIDTH,
364        )?;
365
366        // Dimensions may have been learned mid-run by serialize_document —
367        // backfill the reserved window now that the final value is known.
368        // Mutex poisoning indicates a bug in the calling code, not a recoverable error.
369        #[allow(clippy::expect_used)]
370        // LCOV_EXCL_LINE defensive: mutex-poisoning — poisoning indicates a bug in the calling code
371        let known_dimensions = *self.dimensions.lock().expect("lock poisoned");
372        if let Some(dimensions) = known_dimensions {
373            let value = format!("{dimensions:>DIM_FIELD_WIDTH$}");
374            patch_header_field(
375                file,
376                &head,
377                b"\"dimensions\": ",
378                "dimensions",
379                &value,
380                DIM_FIELD_WIDTH,
381            )?;
382        }
383
384        file.seek(SeekFrom::End(0))?;
385        Ok(())
386    }
387}

389/// Read up to [`HEADER_SCAN_BYTES`] leading bytes from the file.
390///
391/// Seeks to the start first. Returns fewer bytes when the file is shorter
392/// than the scan window.
393fn read_header_bytes(file: &mut File) -> ExportResult<Vec<u8>> {
394    file.seek(SeekFrom::Start(0))?;
395    let mut head = vec![0_u8; HEADER_SCAN_BYTES];
396    let mut filled = 0;
397    while filled < head.len() {
398        let read = file.read(&mut head[filled..])?;
399        if read == 0 {
400            break;
401        }
402        filled += read;
403    }
404    head.truncate(filled);
405    Ok(head)
406}

408/// Find the byte offset of `needle` inside `haystack`.
409fn find_subslice(haystack: &[u8], needle: &[u8]) -> Option<usize> {
410    haystack
411        .windows(needle.len())
412        .position(|window| window == needle)
413}

415/// Parse the right-aligned `total_documents` window into a count.
416fn parse_count_field(field: &[u8]) -> ExportResult<usize> {
417    let text = std::str::from_utf8(field).map_err(|_| {
418        ExporterError::WriteError(
419            "vector export total_documents field is not valid UTF-8".to_string(),
420        )
421    })?;
422    text.trim().parse::<usize>().map_err(|_| {
423        ExporterError::WriteError(format!(
424            "vector export total_documents field holds an unparsable count: {text:?}"
425        ))
426    })
427}

429/// Count document lines by scanning the whole file for 0x0A bytes.
430///
431/// The header carries no newline and every document is written with a single
432/// `writeln!` (serde_json escapes line breaks inside strings), so in a file
433/// truncated at its closing `]` the newline total equals the document count.
434fn count_documents(file: &mut File) -> ExportResult<usize> {
435    file.seek(SeekFrom::Start(0))?;
436    let mut newlines = 0_usize;
437    let mut buffer = [0_u8; 8192];
438    loop {
439        let read = file.read(&mut buffer)?;
440        if read == 0 {
441            break;
442        }
443        newlines += buffer[..read].iter().filter(|&&byte| byte == b'\n').count();
444    }
445    Ok(newlines)
446}

448/// Extract the `created_at` value from a header scan, if present.
449fn extract_created_at(head: &[u8]) -> Option<String> {
450    let marker = b"\"created_at\": \"";
451    let start = find_subslice(head, marker)? + marker.len();
452    let end = head[start..].iter().position(|&byte| byte == b'"')? + start;
453    std::str::from_utf8(&head[start..end])
454        .ok()
455        .map(str::to_string)
456}

458/// Overwrite one reserved header window in place.
459///
460/// Verifies the sentinel byte (the ',' that must follow the fixed-width
461/// window) before writing, so any layout mismatch is an error instead of a
462/// silent corruption. Seeks `file` to the window but does not restore the
463/// offset afterwards.
464fn patch_header_field(
465    file: &mut File,
466    head: &[u8],
467    marker: &[u8],
468    field_name: &str,
469    value: &str,
470    width: usize,
471) -> ExportResult<()> {
472    if value.len() != width {
473        return Err(ExporterError::WriteError(format!(
474            "vector export {field_name} value does not fit the reserved {width}-byte window"
475        )));
476    }
477    let value_start = find_subslice(head, marker)
478        .map(|pos| pos + marker.len())
479        .ok_or_else(|| {
480            ExporterError::WriteError(format!(
481                "vector export header is missing the {field_name} field"
482            ))
483        })?;
484    if head.get(value_start + width) != Some(&b',') {
485        return Err(ExporterError::WriteError(format!(
486            "vector export header layout mismatch around {field_name}"
487        )));
488    }
489    file.seek(SeekFrom::Start(value_start as u64))?;
490    file.write_all(value.as_bytes())?;
491    Ok(())
492}

494impl Exporter for VectorExporter {
495    fn export(&self, document: DocumentChunkValidated) -> ExportResult<()> {
496        let (mut file, mut writer) = self.writer()?;
497        let is_first_doc =
498            !self.config.append || file.metadata().map(|m| m.len() == 0).unwrap_or(true);
499        let existing = if is_first_doc {
500            0
501        } else {
502            self.read_append_state(&mut file)?
503        };
504
505        self.write_metadata_header(&mut writer, &mut file, is_first_doc)?;
506
507        let serialized = self.serialize_document(&document)?;
508
509        if !is_first_doc {
510            write!(writer, ",")?;
511        }
512        writeln!(writer, "{serialized}")?;
513
514        self.close_json(&mut writer, &mut file, existing + 1)?;
515
516        // Release lock
517        fs2::FileExt::unlock(&file)?;
518
519        Ok(())
520    }
521
522    #[tracing::instrument(skip(self, documents), fields(exporter = "vector", documents = documents.len()))]
523    fn export_batch(&self, documents: &[DocumentChunkValidated]) -> ExportResult<()> {
524        if documents.is_empty() {
525            return Ok(());
526        }
527
528        let (mut file, mut writer) = self.writer()?;
529        let is_first_doc =
530            !self.config.append || file.metadata().map(|m| m.len() == 0).unwrap_or(true);
531        let existing = if is_first_doc {
532            0
533        } else {
534            self.read_append_state(&mut file)?
535        };
536
537        self.write_metadata_header(&mut writer, &mut file, is_first_doc)?;
538
539        let mut doc_count = 0;
540        for (i, doc) in documents.iter().enumerate() {
541            if i > 0 || !is_first_doc {
542                write!(writer, ",")?;
543            }
544
545            let serialized = self.serialize_document(doc)?;
546            writeln!(writer, "{serialized}")?;
547            doc_count += 1;
548        }
549
550        self.close_json(&mut writer, &mut file, existing + doc_count)?;
551
552        // Release lock
553        fs2::FileExt::unlock(&file)?;
554
555        Ok(())
556    }
557
558    fn config(&self) -> &ExporterConfig {
559        &self.config
560    }
