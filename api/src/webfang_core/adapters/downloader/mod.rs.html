1//! Asset Download Module
2//!
3//! Handles downloading of images and documents from URLs.
4//!
5//! # Architecture
6//!
7//! Following rust-skills best practices:
8//! - **True Streaming**: Writes chunks directly to disk, constant RAM (~8KB)
9//! - **Atomic Operations**: Temp file with UUID, atomic rename on success
10//! - **Init Once**: Directories pre-created in `new()`, zero runtime contention
11//! - **Configurable**: User-Agent externalized to config
12//! - **Cleanup**: Temp file removed on size limit exceeded
13//! - **Hash On-The-Fly**: SHA256 computed during streaming, no buffer needed
14

27/// Strategy for generating downloaded asset filenames.
28///
29/// # Variants
30///
31/// - `Hash` — SHA-256 hash of content (first 12 hex chars). Dedup-friendly.
32/// - `Slug` — Last path segment of the URL (e.g. `rust-book.pdf`).
33/// - `ContentDisposition` — `filename=` from `Content-Disposition` header, falls back to `Hash`.
34#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
35pub enum AssetNamingStrategy {
36    /// SHA-256 hash of content (first 12 hex chars). Dedup-friendly.
37    #[default]
38    Hash,
39    /// Last path segment of the URL (e.g. `rust-book.pdf`).
40    Slug,
41    /// `filename=` from `Content-Disposition` header, falls back to `Hash`.
42    ContentDisposition,
43}

45/// Result of a successful download
46#[derive(Debug)]
47pub struct DownloadedAsset {
48    /// Original URL
49    pub url: String,
50    /// Local file path where asset was saved
51    pub local_path: PathBuf,
52    /// MIME type detected from HTTP headers
53    pub mime_type: Option<String>,
54    /// File size in bytes
55    pub size: u64,
56    /// SHA256 hash of content (first 12 hex chars used in filename)
57    pub content_hash: String,
58}

128impl Downloader {
129    /// Create a new downloader with configuration.
130    ///
131    /// Pre-creates output directories once to avoid runtime contention.
132    ///
133    /// # Errors
134    ///
135    /// Returns `ScraperError::Io` if directory creation fails.
136    /// Returns `ScraperError::Config` if HTTP client build fails.
137    pub fn new(config: DownloadConfig) -> Result<Self> {
138        // Pre-create directories ONCE (init-once pattern)
139        let images_path = config.output_dir.join(&config.images_dir);
140        let documents_path = config.output_dir.join(&config.documents_dir);
141
142        std::fs::create_dir_all(&images_path).map_err(ScraperError::Io)?;
143        std::fs::create_dir_all(&documents_path).map_err(ScraperError::Io)?;
144
145        let client = Client::builder()
146            .emulation(config.h2_profile)
147            .timeout(Duration::from_secs(config.timeout_secs))
148            .user_agent(&config.user_agent)
149            .build()
150            .map_err(|e| ScraperError::Config(format!("failed to build http client: {e}")))?;
151
152        Ok(Self { client, config })
153    }
154
155    /// Download a single asset with true streaming to disk.
156    ///
157    /// # Architecture
158    ///
159    /// - Creates temp file with UUID
160    /// - Streams chunks directly to disk (constant RAM)
161    /// - Computes hash on-the-fly
162    /// - Atomic rename on success
163    /// - Cleanup temp file on failure
164    /// - Retry with exponential backoff on transient network errors
165    ///
166    /// # Errors
167    ///
168    /// Returns `ScraperError::Network` if HTTP request fails.
169    /// Returns `ScraperError::Io` if file operations fail.
170    /// Returns `ScraperError::Download` if file exceeds size limit.
171    pub async fn download(&self, url: &str) -> Result<DownloadedAsset> {
172        let mut last_err = None;
173
174        for attempt in 0..=self.config.max_retries {
175            if attempt > 0 {
176                let delay = compute_backoff_delay(
177                    attempt,
178                    self.config.backoff_base_ms,
179                    self.config.backoff_max_ms,
180                );
181                tracing::debug!(
182                    "retry {attempt}/{} for {} after {}ms",
183                    self.config.max_retries,
184                    url,
185                    delay.as_millis()
186                );
187                tokio::time::sleep(delay).await;
188            }
189
190            match self.download_once(url).await {
191                Ok(asset) => return Ok(asset),
192                Err(e) => {
193                    if e.classify() != ErrorClass::TransientRetriable
194                        || attempt == self.config.max_retries
195                    {
196                        return Err(e);
197                    }
198                    last_err = Some(e);
199                },
200            }
201        }
202
203        // Unreachable: loop always returns on last attempt, but required for type inference
204        // LCOV_EXCL_LINE defensive: unreachable-retry-exhaustion — the retry loop always returns on the last attempt
205        Err(last_err.unwrap_or_else(|| {
206            ScraperError::Internal("exhausted retries with no error captured".to_string())
207        }))
208    }
209
210    /// Single download attempt (no retry).
211    #[tracing::instrument(
212        skip(self),
213        fields(
214            url = %url,
215            // D5: stable identity of the shared pooled `Client`. Constant across
216            // all asset downloads within a run => observable proof of connection-pool
217            // reuse (no silent per-request handshake). See MAPA item 7.
218            client_id = %format!("{:p}", &self.client)
219        )
220    )]
221    async fn download_once(&self, url: &str) -> Result<DownloadedAsset> {
222        let response = self
223            .client
224            .get(url)
225            .send()
226            .await
227            .map_err(|e| ScraperError::Network(Box::new(e)))?;
228
229        // Fail-fast on 4xx (client errors). 5xx (server errors) are transient and will be retried.
230        let status = response.status();
231        if status.is_client_error() {
232            return Err(ScraperError::http(status.as_u16(), url));
233        }
234        if status.is_server_error() {
235            return Err(ScraperError::Http {
236                status: status.as_u16(),
237                url: url.to_string(),
238            });
239        }
240
241        let mime_type = response
242            .headers()
243            .get(wreq::header::CONTENT_TYPE)
244            .and_then(|v| v.to_str().ok())
245            .map(String::from);
246
247        // Extract Content-Disposition filename before consuming response
248        let content_disposition_filename = response
249            .headers()
250            .get(wreq::header::CONTENT_DISPOSITION)
251            .and_then(|v| v.to_str().ok())
252            .and_then(parse_content_disposition_header);
253
254        let asset_type = crate::adapters::detector::detect_from_url(url);
255        let subdir = if asset_type.is_image() {
256            &self.config.images_dir
257        } else {
258            &self.config.documents_dir
259        };
260
261        let subdir_path = self.config.output_dir.join(subdir);
262
263        // Create temp file with UUID (atomic operation pattern)
264        let temp_path = subdir_path.join(format!("{}.tmp", Uuid::new_v4()));
265        let mut file = fs::File::create(&temp_path)
266            .await
267            .map_err(ScraperError::Io)?;
268
269        // Stream to disk with real-time size check
270        let mut stream = into_stream(response);
271        let mut downloaded: u64 = 0;
272        let mut hasher = Sha256::new();
273
274        while let Some(chunk_result) = stream.next().await {
275            let chunk = chunk_result.map_err(|e| ScraperError::Network(Box::new(e)))?;
276            if chunk.is_empty() {
277                continue;
278            }
279
280            let chunk_len = chunk.len() as u64;
281            // LCOV_EXCL_LINE defensive: integer-overflow — u64 overflow requires an unrealistic download size
282            downloaded = downloaded.checked_add(chunk_len).ok_or_else(|| {
283                ScraperError::Internal("integer overflow in download size".to_string())
284            })?;
285
286            // Check limit in real-time
287            if downloaded > self.config.max_file_size {
288                let _ = fs::remove_file(&temp_path).await;
289                return Err(ScraperError::PayloadTooLarge);
290            }
291
292            // Write chunk to disk IMMEDIATELY (true streaming)
293            file.write_all(&chunk).await.map_err(ScraperError::Io)?;
294            hasher.update(&chunk);
295        }
296
297        // Sync to ensure data is on disk
298        file.sync_all().await.map_err(ScraperError::Io)?;
299        drop(file); // Close file before rename
300
301        // Calculate hash and generate final filename
302        let content_hash = format!("{:x}", hasher.finalize());
303        let filename = self.generate_filename(
304            url,
305            &content_hash,
306            mime_type.as_deref(),
307            content_disposition_filename.as_deref(),
308        );
309        let mut final_path = subdir_path.join(&filename);
310
311        // Slug/ContentDisposition naming can collide when different URLs produce
312        // the same filename. Append a short hash suffix to disambiguate.
313        if final_path.exists() {
314            let hash_suffix = &content_hash[..8];
315            let stem = final_path
316                .file_stem()
317                .and_then(|s| s.to_str())
318                .unwrap_or("file");
319            let ext = final_path
320                .extension()
321                .and_then(|s| s.to_str())
322                .unwrap_or("");
323            if ext.is_empty() {
324                final_path = subdir_path.join(format!("{stem}-{hash_suffix}"));
325            } else {
326                final_path = subdir_path.join(format!("{stem}-{hash_suffix}.{ext}"));
327            }
328        }
329
330        // Atomic rename
331        fs::rename(&temp_path, &final_path)
332            .await
333            .map_err(ScraperError::Io)?;
334
335        tracing::info!(
336            client_id = %format!("{:p}", &self.client),
337            "downloaded: {} -> {:?}",
338            url,
339            final_path
340        );
341
342        Ok(DownloadedAsset {
343            url: url.to_string(),
344            local_path: final_path,
345            mime_type,
346            size: downloaded,
347            content_hash: content_hash[..12].to_string(),
348        })
349    }

351    /// Download multiple assets with configurable concurrency control.
352    ///
353    /// Filters URLs against `include_patterns` / `exclude_patterns` before downloading.
354    /// Returns partial results — individual failures don't abort the batch.
355    pub async fn download_batch(&self, urls: &[String]) -> Vec<Result<DownloadedAsset>> {
356        if urls.is_empty() {
357            return Vec::new();
358        }
359
360        let filtered: Vec<String> = urls
361            .iter()
362            .filter(|url| {
363                url_matches_filters(
364                    url,
365                    &self.config.include_patterns,
366                    &self.config.exclude_patterns,
367                )
368            })
369            .cloned()
370            .collect();
371
372        if filtered.is_empty() {
373            return Vec::new();
374        }
375
376        let concurrency = self.config.concurrency_limit;
377        let mut results = Vec::with_capacity(filtered.len());
378        let mut futs = Vec::with_capacity(filtered.len());
379        for url in &filtered {
380            futs.push(self.download(url));
381        }
382        let stream = futures::stream::iter(futs).buffer_unordered(concurrency);
383        results.extend(stream.collect::<Vec<_>>().await);
384        results
385    }

387    /// Generate filename according to the configured naming strategy.
388    fn generate_filename(
389        &self,
390        url: &str,
391        content_hash: &str,
392        mime_type: Option<&str>,
393        content_disposition_filename: Option<&str>,
394    ) -> String {
395        let extension =
396            mime_type_to_extension(mime_type.unwrap_or("")).unwrap_or_else(|| "bin".into());
397
398        let base_name = match self.config.asset_naming {
399            AssetNamingStrategy::Hash => {
400                // Use first 12 characters of hash (96 bits of entropy)
401                format!("{}.{}", &content_hash[..12], extension)
402            },
403            AssetNamingStrategy::Slug => {
404                let slug = derive_slug_from_url(url);
405                let name = sanitize_filename(&slug);
406                if name.is_empty() {
407                    // Fallback to hash if slug is empty
408                    format!("{}.{}", &content_hash[..12], extension)
409                } else {
410                    // Preserve original extension from slug if present, otherwise use MIME
411                    let slug_ext = name.rsplit('.').next().unwrap_or("");
412                    if !slug_ext.is_empty() && slug_ext != name {
413                        sanitize_filename(&name)
414                    } else {
415                        format!("{name}.{extension}")
416                    }
417                }
418            },
419            AssetNamingStrategy::ContentDisposition => {
420                if let Some(name) = content_disposition_filename {
421                    let sanitized = sanitize_filename(name);
422                    if !sanitized.is_empty() {
423                        sanitized
424                    } else {
425                        format!("{}.{}", &content_hash[..12], extension)
426                    }
427                } else {
428                    format!("{}.{}", &content_hash[..12], extension)
429                }
430            },
431        };
432
433        base_name
434    }

437impl crate::domain::ports::AssetDownloaderPort for Downloader {
438    fn download_batch(
439        &self,
440        urls: &[String],
441    ) -> std::pin::Pin<
442        Box<
443            dyn std::future::Future<
444                    Output = crate::error::Result<Vec<crate::domain::entities::DownloadedAsset>>,
445                > + Send
446                + '_,
447        >,
448    > {
449        let urls = urls.to_vec();
450        Box::pin(async move {
451            let results = self.download_batch(&urls).await;
452            let assets = results
453                .into_iter()
454                .filter_map(|r| r.ok())
455                .map(|a| {
456                    let asset_type = crate::adapters::detector::detect_from_url(&a.url);
457                    let asset_type_str = match asset_type {
458                        crate::adapters::detector::AssetType::Image => "image",
459                        crate::adapters::detector::AssetType::Document => "document",
460                        crate::adapters::detector::AssetType::Unknown => "unknown",
461                    };
462                    crate::domain::entities::DownloadedAsset {
463                        url: a.url,
464                        local_path: a.local_path.to_string_lossy().into_owned(),
465                        asset_type: asset_type_str.to_string(),
466                        size: a.size,
467                    }
468                })
469                .collect();
470            Ok(assets)
471        })
472    }

480/// Check if a URL matches include/exclude filters.
481///
482/// If `include_patterns` is empty, all URLs pass the include check.
483/// `exclude_patterns` are always applied (deny wins).
484fn url_matches_filters(url: &str, includes: &[String], excludes: &[String]) -> bool {
485    if excludes.iter().any(|p| pattern_matches_asset(url, p)) {
486        return false;
487    }
488    if includes.is_empty() {
489        return true;
490    }
491    includes.iter().any(|p| pattern_matches_asset(url, p))
492}

494/// Match a URL against a pattern, supporting both extension globs (`*.pdf`)
495/// and the standard host/path glob from `domain::pattern_matching`.
496fn pattern_matches_asset(url: &str, pattern: &str) -> bool {
497    let p = pattern.trim();
498    // Extension glob: *.ext (but NOT host globs like *.example.com which contain a dot after the prefix)
499    if let Some(ext) = p.strip_prefix("*.") {
500        if !ext.is_empty() && !ext.contains('.') {
501            if let Ok(u) = url::Url::parse(url) {
502                let last = u.path().rsplit('/').next().unwrap_or("");
503                let low = last.to_ascii_lowercase();
504                let ext_low = ext.to_ascii_lowercase();
505                return low.ends_with(&format!(".{ext_low}"));
506            }
507        }
508    }
509    crate::domain::matches_pattern(url, pattern)
510}

512/// Compute exponential backoff delay with jitter.
513fn compute_backoff_delay(attempt: u32, base_ms: u64, max_ms: u64) -> Duration {
514    use rand::Rng;
515    use std::cmp::min;
516
517    // Exponential: base * 2^(attempt-1), clamped to max. Shift is capped at 62
518    // to avoid panic from shifting u64 by >= 64 bits (attempt=1 is shift=0).
519    let shift = (attempt - 1).min(62);
520    let delay_ms = min(base_ms.saturating_mul(1u64 << shift), max_ms);
521    // Add jitter: 75%-125% of delay, then clamp final result to max_ms
522    let jitter = delay_ms / 4;
523    let offset = rand::rng().random_range(0..=jitter.saturating_mul(2));
524    let final_ms = min(
525        delay_ms.saturating_sub(jitter).saturating_add(offset),
526        max_ms,
527    );
528    Duration::from_millis(final_ms)
529}

531/// Derive a slug from the last path segment of a URL.
532fn derive_slug_from_url(url: &str) -> String {
533    url::Url::parse(url)
534        .ok()
535        .and_then(|u| {
536            let path = u.path();
537            path.rsplit('/')
538                .next()
539                .filter(|s| !s.is_empty() && *s != "/")
540                .map(String::from)
541        })
542        .unwrap_or_default()
543}

545/// UTF-8 safe percent-decoding for Content-Disposition filenames.
546/// Uses the `percent-encoding` crate for correct handling of truncated
547/// sequences, multi-byte chars, and all edge cases in RFC 5987/3986.
548fn percent_decode_utf8(input: &str) -> String {
549    percent_encoding::percent_decode_str(input)
550        .decode_utf8_lossy()
551        .into_owned()
552}

554/// Sanitize a filename: remove path separators and null bytes.
555fn sanitize_filename(name: &str) -> String {
556    name.chars()
557        .filter(|c| *c != '/' && *c != '\\' && *c != '\0')
558        .collect()
559}

561/// Parse `filename=` from a Content-Disposition header value.
562///
563/// RFC 6266 / 5987: parameter names are case-insensitive. We lowercase
564/// the header value before matching to handle `Filename=`, `FILENAME=`, etc.
565fn parse_content_disposition_header(value: &str) -> Option<String> {
566    let lower = value.to_ascii_lowercase();
567
568    // Try filename*=utf-8''encoded first (RFC 5987)
569    if let Some(start) = lower.find("filename*=utf-8''") {
570        let encoded = &value[start + "filename*=utf-8''".len()..];
571        let name: String = encoded
572            .chars()
573            .take_while(|c| *c != ';' && *c != ' ')
574            .collect();
575        let decoded = percent_decode_utf8(&name);
576        return Some(decoded);
577    }
578
579    // Try filename="name" or filename=name (case-insensitive)
580    let after = lower.find("filename=")?;
581    let rest = &value[after + "filename=".len()..];
582    let name = if let Some(inner) = rest.strip_prefix('"') {
583        // Quoted: filename="name.pdf"
584        let end = inner.find('"')?;
585        &inner[..end]
586    } else {
587        // Unquoted: filename=name.pdf
588        rest.split(';').next().unwrap_or(rest).trim()
589    };
590
591    if name.is_empty() {
592        None
593    } else {
594        Some(name.to_string())
595    }
596}

598/// MIME type to file extension mapping
599fn mime_type_to_extension(mime: &str) -> Option<String> {
600    let mime = mime.trim();
601    match mime {
602        "image/jpeg" | "image/jpg" => Some("jpg".to_string()),
603        "image/png" => Some("png".to_string()),
604        "image/gif" => Some("gif".to_string()),
605        "image/webp" => Some("webp".to_string()),
606        "image/svg+xml" => Some("svg".to_string()),
607        "image/bmp" => Some("bmp".to_string()),
608        "image/tiff" => Some("tiff".to_string()),
609        "image/x-icon" => Some("ico".to_string()),
610        "application/pdf" => Some("pdf".to_string()),
611        "application/msword" => Some("doc".to_string()),
612        "application/vnd.openxmlformats-officedocument.wordprocessingml.document" => {
613            Some("docx".to_string())
614        },
615        "application/vnd.ms-excel" => Some("xls".to_string()),
616        "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet" => {
617            Some("xlsx".to_string())
618        },
619        "application/vnd.ms-powerpoint" => Some("ppt".to_string()),
620        "application/vnd.openxmlformats-officedocument.presentationml.presentation" => {
621            Some("pptx".to_string())
622        },
623        "text/csv" => Some("csv".to_string()),
624        "application/vnd.oasis.opendocument.text" => Some("odt".to_string()),
625        "application/vnd.oasis.opendocument.spreadsheet" => Some("ods".to_string()),
626        "application/epub+zip" => Some("epub".to_string()),
627        "application/rtf" => Some("rtf".to_string()),
628        "text/plain" => Some("txt".to_string()),
629        "application/json" => Some("json".to_string()),
630        "application/xml" | "text/xml" => Some("xml".to_string()),
631        _ => None,
632    }

741    fn test_generate_filename_hash_strategy() {
742        let temp_dir = TempDir::new().unwrap();
743        let config = DownloadConfig {
744            output_dir: temp_dir.path().to_path_buf(),
745            ..Default::default()
746        };
747        let downloader = Downloader::new(config).unwrap();
748
749        let filename = downloader.generate_filename(
750            "https://example.com/img.png",
751            "abc123def456789",
752            Some("image/png"),
753            None,
754        );
755        assert!(
756            filename.ends_with(".png"),
757            "Expected .png but got: {filename}"
758        );
759        assert!(
760            filename.starts_with("abc123def456"),
761            "Filename should start with first 12 chars of hash"
762        );
763
764        let filename =
765            downloader.generate_filename("https://example.com/file", "xyz789abc123456", None, None);
766        assert!(
767            filename.ends_with(".bin"),
768            "Expected .bin but got: {filename}"
769        );
770    }
