1//! Configuration for HTTP client behavior
2//!
3//! Controls headers, retry behavior, and cookie handling.
4//! Use `Default` for sensible production defaults.
5//!
6//! This type is owned by the **domain** layer. The application HTTP client
7//! consumes it; infrastructure maps domain values (e.g. the TLS profile)
8//! onto a concrete client when building the network stack.
9
10use thiserror::Error;

12/// Configuration for HTTP client behavior
13///
14/// Controls headers, retry behavior, and cookie handling.
15/// Use `Default` for sensible production defaults.
16#[derive(Debug, Clone)]
17pub struct HttpClientConfig {
18    /// Accept-Language header value
19    pub accept_language: String,
20    /// Accept header value
21    pub accept: String,
22    /// Referer header value
23    pub referer: String,
24    /// Cache-Control header value
25    pub cache_control: String,
26    /// Maximum number of retry attempts
27    pub max_retries: u32,
28    /// Base delay for exponential backoff (in milliseconds)
29    pub backoff_base_ms: u64,
30    /// Maximum delay for exponential backoff (in milliseconds)
31    pub backoff_max_ms: u64,
32    /// Enable cookie jar
33    pub enable_cookies: bool,
34    /// Request timeout in seconds
35    pub timeout_secs: u64,
36    /// Connection timeout in seconds
37    pub connect_timeout_secs: u64,
38    /// Rate limit: requests per minute (None for no limit)
39    pub rate_limit_rpm: Option<u32>,
40    /// TLS fingerprint emulation preset.
41    ///
42    /// This is the **single source of truth** for the TLS/H2 fingerprint: the
43    /// network stack applies it directly when building the client. Construct it
44    /// from a user-supplied name with [`HttpClientConfig::profile_from_name`].
45    pub tls_emulation: wreq_util::Profile,
46    /// Custom User-Agent override
47    pub user_agent: Option<String>,
48    /// Bypass WAF/CAPTCHA detection entirely (REQ-WAF-07).
49    ///
50    /// When true, the HTTP client builds its WAF `InspectionContext` with
51    /// `ignore_waf` set, short-circuiting inspection to a clean verdict so
52    /// challenge markers never block a response.
53    pub ignore_waf: bool,
54}

56impl Default for HttpClientConfig {
57    fn default() -> Self {
58        Self {
59            accept_language: "en-US,en;q=0.9".into(),
60            accept: "text/html,application/xhtml+xml,application/xml;q=0.9,*/*;q=0.8".into(),
61            referer: "https://www.google.com/".into(),
62            cache_control: "no-cache".into(),
63            max_retries: 3,
64            backoff_base_ms: 1000,
65            backoff_max_ms: 10000,
66            enable_cookies: true,
67            timeout_secs: 30,
68            connect_timeout_secs: 10,
69            rate_limit_rpm: None,
70            tls_emulation: wreq_util::Profile::Chrome145,
71            user_agent: None,
72            ignore_waf: false,
73        }
74    }
75}
76
77impl HttpClientConfig {
78    /// Parse an H2/TLS profile name into a [`wreq_util::Profile`].
79    ///
80    /// This is the fallible boundary that converts a user-facing profile name
81    /// (e.g. the `--h2-profile` CLI value) into the typed
82    /// [`Self::tls_emulation`] preset. It delegates to the domain resolver
83    /// [`crate::domain::profile::profile_from_name`], which accepts the full
84    /// [`wreq_util::Profile`] catalog. Unknown names are an error — there is
85    /// **no** silent fallback to a default profile.
86    ///
87    /// # Errors
88    ///
89    /// Returns [`UnknownProfileError`] if `name` is not a recognized profile.
90    pub fn profile_from_name(name: &str) -> Result<wreq_util::Profile, UnknownProfileError> {
91        crate::domain::profile::profile_from_name(name).ok_or_else(|| UnknownProfileError {
92            name: name.to_owned(),
93        })
94    }
95}
96
97/// Error returned when an H2/TLS profile name is not recognized.
98///
99/// The user-facing [`Display`](std::fmt::Display) message is in Spanish and
100/// lists the full dynamic catalog of valid options (every
101/// [`wreq_util::Profile`] variant), per the project's user-facing-error
102/// convention.
103#[derive(Clone, Debug, Error, PartialEq, Eq)]
104#[error("Perfil TLS desconocido: '{name}'. Opciones válidas: {}", crate::domain::profile::valid_profile_names().join(", "))]
105pub struct UnknownProfileError {
106    /// The unrecognized profile name supplied by the user.
107    pub name: String,
108}

119        assert_eq!(
120            config.accept,
121            "text/html,application/xhtml+xml,application/xml;q=0.9,*/*;q=0.8"
122        );
123        assert_eq!(config.referer, "https://www.google.com/");
124        assert_eq!(config.cache_control, "no-cache");
125        assert_eq!(config.max_retries, 3);
126        assert_eq!(config.backoff_base_ms, 1000);
127        assert_eq!(config.backoff_max_ms, 10000);
128        assert!(config.enable_cookies);
129        assert_eq!(config.timeout_secs, 30);
130        assert_eq!(config.connect_timeout_secs, 10);
131        assert_eq!(config.rate_limit_rpm, None);
