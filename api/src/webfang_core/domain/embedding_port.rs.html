1//! Embedding port — async domain trait for text vectorization.
2//!
3//! Defines the contract for generating embedding vectors from text.
4//! Concrete implementations live in the `webfang_ai` crate (wrapping
5//! `InferencePool`). The trait uses `BoxFuture` for dyn-compatibility,
6//! matching the pattern established in [`super::semantic_inspector`] and
7//! [`super::repository::VectorRepository`].
8//!
9//! # Design decisions
10//!
11//! - **Not sealed**: unlike `SemanticCleaner`, this trait is open for
12//!   testing (mock embeddings) and future alternative backends.
13//! - **Always compiled**: the trait definition has no `#[cfg(feature = "ai")]`
14//!   guard. The Container stores `Option<Arc<dyn EmbeddingPort>>` which is
15//!   `None` when the `ai` feature is disabled.
16//! - **Separate from `SemanticCleaner`**: the cleaner is a sealed, HTML-only
17//!   pipeline. This port exposes the raw embedding primitive needed by vault
18//!   search (issue #386) and any future semantic feature.
19
20use std::future::Future;
21use std::pin::Pin;
22
23use crate::error::SemanticError;
24
25/// A boxed future for dyn-compatible async traits.
26type BoxFuture<'a, T> = Pin<Box<dyn Future<Output = T> + Send + 'a>>;
27
28/// Domain trait for text embedding generation.
29///
30/// Implementations wrap an ONNX inference pool to produce fixed-dimension
31/// embedding vectors. The default model (Granite-97M) produces 384-dimensional
32/// vectors; both supported models output 384d via Matryoshka truncation.
33///
34/// # Errors
35///
36/// Returns [`SemanticError`] when:
37/// - [`SemanticError::Inference`]: ONNX model execution failed
38/// - [`SemanticError::Tokenize`]: input text tokenization failed
39/// - [`SemanticError::ModelLoad`]: model not available
40pub trait EmbeddingPort: Send + Sync {
41    /// Generate an embedding vector for a single text.
42    ///
43    /// Returns a fixed-dimension vector (384d for both Granite models).
44    ///
45    /// # Errors
46    ///
47    /// Returns [`SemanticError::Inference`] if the model execution fails,
48    /// or [`SemanticError::Tokenize`] if the text cannot be tokenized.
49    fn embed<'a>(&'a self, text: &'a str) -> BoxFuture<'a, Result<Vec<f32>, SemanticError>>;
50
51    /// Generate embedding vectors for multiple texts in a batch.
52    ///
53    /// The default implementation calls [`embed`](Self::embed) sequentially.
54    /// Implementations may override this to use batched ONNX inference for
55    /// better throughput.
56    ///
57    /// Returns one vector per input text, in the same order.
58    ///
59    /// # Errors
60    ///
61    /// Returns [`SemanticError`] if any individual embedding fails.
62    fn embed_batch<'a>(
63        &'a self,
64        texts: &'a [String],
65    ) -> BoxFuture<'a, Result<Vec<Vec<f32>>, SemanticError>> {
66        Box::pin(async move {
67            let mut results = Vec::with_capacity(texts.len());
68            for text in texts {
69                results.push(self.embed(text).await?);
70            }
71            Ok(results)
72        })
73    }
74
75    /// The dimensionality of embedding vectors produced by this port.
76    ///
77    /// Both supported models (Granite-97M and Granite-311M) return 384.
78    fn embedding_dim(&self) -> usize;
79}

81#[cfg(test)]
82mod tests {
83    use super::*;
84
85    /// A deterministic mock that returns unit vectors based on text length.
86    struct MockEmbedder {
87        dim: usize,
88    }
89
90    impl EmbeddingPort for MockEmbedder {
91        fn embed<'a>(&'a self, text: &'a str) -> BoxFuture<'a, Result<Vec<f32>, SemanticError>> {
92            Box::pin(async move {
93                let mut v = vec![0.0f32; self.dim];
94                // Deterministic: use text length mod dim as the hot index.
95                let idx = text.len() % self.dim;
96                v[idx] = 1.0;
97                Ok(v)
98            })
99        }
100
101        fn embedding_dim(&self) -> usize {
102            self.dim
103        }
104    }
105
106    #[tokio::test]
107    async fn test_mock_embedder_produces_correct_dim() {
108        let embedder = MockEmbedder { dim: 384 };
109        let v = embedder.embed("hello world").await.unwrap();
110        assert_eq!(v.len(), 384);
111        assert_eq!(embedder.embedding_dim(), 384);
112    }
113
114    #[tokio::test]
115    async fn test_mock_embedder_deterministic() {
116        let embedder = MockEmbedder { dim: 384 };
117        let v1 = embedder.embed("test").await.unwrap();
118        let v2 = embedder.embed("test").await.unwrap();
119        assert_eq!(v1, v2, "same input must produce same embedding");
120    }
121
122    #[tokio::test]
123    async fn test_embed_batch_default_impl() {
124        let embedder = MockEmbedder { dim: 4 };
125        let texts = vec!["a".to_string(), "bb".to_string(), "ccc".to_string()];
126        let results = embedder.embed_batch(&texts).await.unwrap();
127        assert_eq!(results.len(), 3);
128        for (i, v) in results.iter().enumerate() {
129            assert_eq!(v.len(), 4, "batch item {i} must have correct dim");
130        }
131    }
132
133    #[tokio::test]
134    async fn test_embed_batch_empty_input() {
135        let embedder = MockEmbedder { dim: 384 };
136        let texts: Vec<String> = vec![];
137        let results = embedder.embed_batch(&texts).await.unwrap();
138        assert!(results.is_empty(), "empty input must produce empty output");
139    }
140
141    #[test]
142    fn test_embedding_port_is_object_safe() {
143        fn assert_dyn_compatible(_: &dyn EmbeddingPort) {}
144        let embedder = MockEmbedder { dim: 384 };
145        assert_dyn_compatible(&embedder);
146    }
