1//! Semantic Cleaner trait definition — Domain layer
2//!
3//! Defines the contract for AI-powered semantic cleaning operations.
4//! Following Clean Architecture: this trait is pure domain logic with no
5//! infrastructure dependencies.
6//!
7//! # Design Decisions
8//!
9//! - **Sealed trait pattern** (`api-sealed-trait`): Prevents external implementations
10//!   that could violate invariants. Only infrastructure/ai module can implement.
11//! - **Async interface** (boxed futures): Async methods are manually desugared to
12//!   `Pin<Box<dyn Future<Output = T> + Send>>` to preserve dyn-compatibility
13//!   (`Box<dyn SemanticCleaner>`) without depending on the `async-trait` macro.
14//! - **Explicit error types**: Uses `SemanticError` for type-safe error handling
15//!   (`err-thiserror-lib`, `err-context-chain`)
16//! - **Borrowed input**: Accepts `&str` not `&String` (`own-borrow-over-clone`)
17//! - **Owned output**: Returns `Vec<DocumentChunk>` for ownership transfer
18
19use std::future::Future;
20use std::pin::Pin;
21
22use crate::domain::DocumentChunk;
23use crate::error::SemanticError;

25/// Sealed trait to prevent external implementations
26///
27/// Following `api-sealed-trait` rust-skill: This module is sealed so that
28/// only crates within webfang can implement `SemanticCleaner`.
29/// This prevents users from creating invalid implementations that could
30/// violate memory safety or caching invariants.
31pub mod private {
32    /// Marker trait for the sealed pattern — only implementable within this crate.
33    pub trait Sealed {}
34}
35
36/// Semantic Cleaner — AI-powered content cleaning interface
37///
38/// This trait defines the contract for cleaning HTML content using AI models.
39/// Implementations use ONNX models (sentence-transformers) to:
40/// - Split content into semantic chunks
41/// - Validate chunk sizes (token limits)
42/// - Prepare content for embedding generation
43///
44/// # Examples
45///
46/// ```no_run
47/// # use webfang_core::domain::semantic_cleaner::SemanticCleaner;
48/// # async fn example(cleaner: &dyn SemanticCleaner) -> Result<(), Box<dyn std::error::Error>> {
49/// let html = "<html><body><p>Hello World</p></body></html>";
50/// let chunks = cleaner.clean(html).await?;
51/// println!("Generated {} chunks", chunks.len());
52/// # Ok(())
53/// # }
54/// ```
55///
56/// # Errors
57///
58/// Returns [`SemanticError`] in these cases:
59/// - [`SemanticError::ModelLoad`]: Failed to load ONNX model from cache
60/// - [`SemanticError::Tokenize`]: Tokenization failed (invalid input)
61/// - [`SemanticError::Inference`]: ONNX inference failed
62/// - [`SemanticError::ChunkTooLarge`]: Content exceeds model's token limit
63/// - [`SemanticError::Download`]: Model download failed (if auto-download enabled)
64///
65/// # Implementation Notes
66///
67/// - Implementations MUST be thread-safe (`Send + Sync`)
68/// - Implementations MUST cache models to avoid reloading
69/// - Implementations MUST use memory-mapped files for large models (`mem-zero-copy`)
70pub trait SemanticCleaner: private::Sealed + Send + Sync {
71    /// Clean HTML content and split into semantic chunks
72    ///
73    /// This is the main entry point for semantic cleaning. It:
74    /// 1. Strips HTML tags and extracts text
75    /// 2. Splits text into semantic chunks (paragraphs, sections)
76    /// 3. Validates chunk sizes against model token limits
77    /// 4. Returns chunks ready for embedding generation
78    ///
79    /// # Arguments
80    ///
81    /// * `html` - Raw HTML content to clean (borrowed, `&str` not `&String`)
82    ///
83    /// # Returns
84    ///
85    /// * `Ok(Vec<DocumentChunk>)` - Successfully cleaned chunks
86    /// * `Err(SemanticError)` - Cleaning failed
87    ///
88    /// # Errors
89    ///
90    /// Returns [`SemanticError::ChunkTooLarge`] if any chunk exceeds the model's
91    /// token limit (32768 tokens for IBM Granite-97M).
92    ///
93    /// Returns [`SemanticError::Tokenize`] if the input contains invalid UTF-8
94    /// or special characters that break tokenization.
95    ///
96    /// # Examples
97    ///
98    /// ```no_run
99    /// # use webfang_core::domain::semantic_cleaner::SemanticCleaner;
100    /// # async fn example(cleaner: &dyn SemanticCleaner) -> Result<(), Box<dyn std::error::Error>> {
101    /// let html = "<article><h1>Title</h1><p>Content here...</p></article>";
102    /// let chunks = cleaner.clean(html).await?;
103    ///
104    /// for chunk in &chunks {
105    ///     println!("Chunk {}: {} chars", chunk.id, chunk.content.len());
106    /// }
107    /// # Ok(())
108    /// # }
109    /// ```
110    ///
111    /// # Performance
112    ///
113    /// - **First call**: May trigger model download (~90MB) + load (~100-500ms)
114    /// - **Subsequent calls**: Cache hit, ~10-50ms per page
115    /// - **Memory**: Uses memory-mapped files, ~90MB virtual memory (not RSS)
116    ///
117    /// # Note
118    ///
119    /// Desugared to a boxed future (`Pin<Box<dyn Future + Send>>`) instead of
120    /// `async fn` so the trait stays dyn-compatible (`Box<dyn SemanticCleaner>`).
121    /// The single lifetime `'a` ties `&self` and `html` to the returned future,
122    /// matching the `async-trait` macro's default desugaring.
123    fn clean<'a>(
124        &'a self,
125        html: &'a str,
126    ) -> Pin<Box<dyn Future<Output = Result<Vec<DocumentChunk>, SemanticError>> + Send + 'a>>;
127
128    /// Get the model's maximum token limit
129    ///
130    /// This is useful for validating content before processing.
131    ///
132    /// # Returns
133    ///
134    /// Maximum number of tokens the model accepts per chunk.
135    /// For `all-MiniLM-L6-v2`, this is 512 tokens.
136    ///
137    /// # Examples
138    ///
139    /// ```no_run
140    /// # use webfang_core::domain::semantic_cleaner::SemanticCleaner;
141    /// fn example(cleaner: &dyn SemanticCleaner) {
142    ///     let max_tokens = cleaner.max_tokens();
143    ///     println!("Model accepts up to {} tokens per chunk", max_tokens);
144    /// }
145    /// ```
146    fn max_tokens(&self) -> usize;
147
148    /// Check if the model is ready for inference
149    ///
150    /// This method verifies that:
151    /// - Model file exists in cache
152    /// - Model file passes SHA256 validation
153    /// - Model can be loaded into memory
154    ///
155    /// # Returns
156    ///
157    /// * `true` - Model is ready
158    /// * `false` - Model needs download or reload
159    ///
160    /// # Examples
161    ///
162    /// ```no_run
163    /// # use webfang_core::domain::semantic_cleaner::SemanticCleaner;
164    /// fn example(cleaner: &dyn SemanticCleaner) {
165    ///     if cleaner.is_ready() {
166    ///         println!("Model ready for inference");
167    ///     } else {
168    ///         println!("Model needs download or reload");
169    ///     }
170    /// }
171    /// ```
172    fn is_ready(&self) -> bool;
173}
