1//! Domain trait for vector persistence (dependency inversion).
2//!
3//! Defines the persistence contract used by the elastic ingestion pipeline.
4//! The infrastructure layer
5//! ([`crate::infrastructure::persistence::sqlite::SqliteVectorRepository`])
6//! implements this trait; the application layer depends on the trait — not the
7//! concrete repo — so SQLite can be swapped or mocked without touching
8//! orchestration logic.
9//!
10//! # Dyn-compatibility (A1 desugar — core-slimming)
11//!
12//! The four trait methods use manual `async fn` desugaring to
13//! `Pin<Box<dyn Future<Output = …> + Send + '_>>` (BoxFuture) instead of native
14//! `async fn` in traits. This makes the trait **dyn-compatible** so
15//! `Arc<dyn VectorRepository + Send + Sync>` can be used for runtime dispatch
16//! (spec S3.4), without adding the `async_trait` crate (frozen decision #1).
17//! A blanket impl for `Arc<T>` lets `ElasticIngestion<R: VectorRepository>`
18//! accept `Arc<dyn VectorRepository + Send + Sync>` as `R`.
19
20use std::future::Future;
21use std::pin::Pin;

23use crate::error::ScraperError;

25/// Domain trait for vector persistence (dependency inversion).
26///
27/// Implementations store crawl resources (with content-hash deduplication) and
28/// semantic chunks whose embeddings are serialized as raw little-endian `f32`
29/// BLOBs (frozen design decision #7).
30///
31/// All database failures surface as [`ScraperError::Persistence`] (frozen
32/// decision #4: no separate `StorageError` enum), matching the pattern
33/// established by PR1.
34///
35/// The methods are desugared to `Pin<Box<dyn Future<…> + Send + '_>>` so the
36/// trait is dyn-compatible (A1, spec S3.4) without the `async_trait` crate.
37pub trait VectorRepository: Send + Sync {
38    /// Save a resource with its content hash. Returns the resource URL.
39    ///
40    /// If a resource with the same `content_hash` already exists, this
41    /// short-circuits (dedup, frozen decision #3) and returns the **existing**
42    /// URL without inserting a duplicate row — saving the heavier chunk inserts.
43    ///
44    /// # Errors
45    ///
46    /// Returns [`ScraperError::Persistence`] on any database failure.
47    fn save_resource<'a>(
48        &'a self,
49        url: &'a str,
50        title: &'a str,
51        content_hash: &'a str,
52        size_bytes: u64,
53    ) -> Pin<Box<dyn Future<Output = Result<String, ScraperError>> + Send + 'a>>;
54
55    /// Save a chunk, optionally with its embedding vector.
56    ///
57    /// When `embedding` is `Some`, it is serialized to a little-endian `f32`
58    /// BLOB; when `None`, the `embedding_vector` column is stored as SQL `NULL`.
59    ///
60    /// # Errors
61    ///
62    /// Returns [`ScraperError::Persistence`] on any database failure (e.g. a
63    /// foreign-key violation if `resource_url` was never saved first).
64    fn save_chunk<'a>(
65        &'a self,
66        id: &'a str,
67        resource_url: &'a str,
68        chunk_index: i64,
69        content: &'a str,
70        embedding: Option<&'a [f32]>,
71    ) -> Pin<Box<dyn Future<Output = Result<(), ScraperError>> + Send + 'a>>;
72
73    /// Check whether a resource with this `content_hash` already exists.
74    ///
75    /// Returns `Ok(Some(url))` with the existing resource's URL if found, or
76    /// `Ok(None)` if no resource has that hash.
77    ///
78    /// # Errors
79    ///
80    /// Returns [`ScraperError::Persistence`] on any database failure.
81    fn resource_exists_by_hash<'a>(
82        &'a self,
83        content_hash: &'a str,
84    ) -> Pin<Box<dyn Future<Output = Result<Option<String>, ScraperError>> + Send + 'a>>;
85
86    /// Get the embedding vector for a chunk.
87    ///
88    /// Returns `Ok(Some(vec))` if the chunk exists and has an embedding,
89    /// `Ok(None)` if the chunk is missing or has a `NULL` embedding, or an
90    /// error if the stored BLOB is corrupt (length not a multiple of 4).
91    ///
92    /// # Errors
93    ///
94    /// Returns [`ScraperError::Persistence`] on a corrupt BLOB or database
95    /// failure.
96    #[allow(clippy::type_complexity)]
97    fn get_vector<'a>(
98        &'a self,
99        chunk_id: &'a str,
100    ) -> Pin<Box<dyn Future<Output = Result<Option<Vec<f32>>, ScraperError>> + Send + 'a>>;
101}

103/// Blanket impl so `Arc<dyn VectorRepository + Send + Sync>` satisfies
104/// `R: VectorRepository + Send + Sync` in
105/// [`crate::application::elastic_ingestion::ElasticIngestion<R>`] (spec S3.4).
106///
107/// Delegates each method through the `Arc` deref to the inner repository. This
108/// is the bridge that lets the Container store
109/// `Option<Arc<ElasticIngestion<Arc<dyn VectorRepository + Send + Sync>>>>`
110/// for runtime repo dispatch (SQLite when `persistence` is ON, StreamRepository
111/// when OFF).
112impl<T: VectorRepository + ?Sized> VectorRepository for std::sync::Arc<T> {
113    fn save_resource<'a>(
114        &'a self,
115        url: &'a str,
116        title: &'a str,
117        content_hash: &'a str,
118        size_bytes: u64,
119    ) -> Pin<Box<dyn Future<Output = Result<String, ScraperError>> + Send + 'a>> {
120        (**self).save_resource(url, title, content_hash, size_bytes)
121    }
122
123    fn save_chunk<'a>(
124        &'a self,
125        id: &'a str,
126        resource_url: &'a str,
127        chunk_index: i64,
128        content: &'a str,
129        embedding: Option<&'a [f32]>,
130    ) -> Pin<Box<dyn Future<Output = Result<(), ScraperError>> + Send + 'a>> {
131        (**self).save_chunk(id, resource_url, chunk_index, content, embedding)
132    }
133
134    fn resource_exists_by_hash<'a>(
135        &'a self,
136        content_hash: &'a str,
137    ) -> Pin<Box<dyn Future<Output = Result<Option<String>, ScraperError>> + Send + 'a>> {
138        (**self).resource_exists_by_hash(content_hash)
139    }
140
141    fn get_vector<'a>(
142        &'a self,
143        chunk_id: &'a str,
144    ) -> Pin<Box<dyn Future<Output = Result<Option<Vec<f32>>, ScraperError>> + Send + 'a>> {
145        (**self).get_vector(chunk_id)
146    }
147}

149/// Erased vector repository for runtime dispatch.
150///
151/// Lets the Container hold either the SQLite-backed [`crate::infrastructure::persistence::sqlite::SqliteVectorRepository`]
152/// (under the `persistence` feature) or the dependency-free `StreamRepository`
153/// JSONL sink behind a single type, enabling `Arc<dyn VectorRepository + Send + Sync>`
154/// to satisfy the `R: VectorRepository + Send + Sync` bound on
155/// [`crate::application::elastic_ingestion::ElasticIngestion`].
156pub type DynVectorRepository = std::sync::Arc<dyn VectorRepository + Send + Sync>;
